    strata={key:{'count':v['count'],'source_groups':v['source_groups'],'semantic':ratio(v['semantic_numerator'],v['count']),**({'format':ratio(v['format_numerator'],v['format_denominator'])} if v['format_denominator'] else {}),**{k:v[k] for k in ('initial_semantic_numerator','end_to_end_semantic_numerator','end_to_end_success_numerator') if k in v}} for key,v in metrics['stratified_final_reply'].items() if '/intent=' in key or '/query_class=' in key or '/query_position=' in key}
    failures=[]
    for key,value in strata.items():
        threshold=constants['THRESHOLDS']['text_semantic_per_intent'] if key.startswith('text/intent=') else constants['THRESHOLDS']['voice_per_intent'] if key.startswith('voice_') else constants['THRESHOLDS']['vision_per_class_relation'] if key.startswith('vision_') else None
        if threshold is not None and value['semantic']['rate']<threshold:failures.append({'criterion':key,'observed':value['semantic'],'minimum_rate':threshold})
    for task in ('tool_missing','tool_unsupported','tool_unavailable'):
        if tasks[task]['semantic']['rate']<constants['THRESHOLDS']['unsupported_clarification']:failures.append({'criterion':task,'observed':tasks[task]['semantic'],'minimum_rate':constants['THRESHOLDS']['unsupported_clarification']})
    for task,value in tasks.items():
        if 'format' in value and value['format']['rate']<constants['THRESHOLDS']['format']:failures.append({'criterion':task+'/format','observed':value['format'],'minimum_rate':constants['THRESHOLDS']['format']})
    tool=metrics['per_task_final_reply']['tool_call'];chain={k:tool[k] for k in ('tool_requested','tool_valid','tool_parameters','tool_executed','tool_final','tool_roundtrip')}
    if chain['tool_roundtrip']['rate']<constants['THRESHOLDS']['tool_roundtrip']:failures.append({'criterion':'tool_call/true_roundtrip','observed':chain['tool_roundtrip'],'minimum_rate':constants['THRESHOLDS']['tool_roundtrip']})
    if tasks['ocr']['exact']['rate']<constants['THRESHOLDS']['ocr_exact']:failures.append({'criterion':'ocr/exact','observed':tasks['ocr']['exact'],'minimum_rate':constants['THRESHOLDS']['ocr_exact']})
    if tasks['ocr']['final_cer']>constants['THRESHOLDS']['ocr_cer_max']:failures.append({'criterion':'ocr/CER','observed':tasks['ocr']['final_cer'],'maximum_rate':constants['THRESHOLDS']['ocr_cer_max']})
