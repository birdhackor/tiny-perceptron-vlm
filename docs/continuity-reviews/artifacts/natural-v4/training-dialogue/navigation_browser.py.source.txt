import json,hashlib,shutil
from pathlib import Path
from datetime import datetime,timezone
from playwright.sync_api import sync_playwright
public=Path('docs/continuity-reviews/artifacts/natural-v4/training-dialogue')
records=[]
with sync_playwright() as p:
 b=p.chromium.launch(headless=True,executable_path='/usr/bin/chromium',args=['--no-sandbox'])
 page=b.new_page(viewport={'width':1280,'height':1000})
 for origin,selector,label in [
  ('5.1.html','article a[href$="4.7.html"]','5.1-prerequisite-4.7'),
  ('5.1.html','article a[href$="training.html#T.4"]','5.1-training-T.4'),
  ('6.5.html','article a[href$="first-steps.html#W.4"]','6.5-log-W.4'),
  ('7.1.html','article a[href$="first-steps.html#W.2"]','7.1-python-W.2'),
  ('chapter-07.html','article a[href$="7.17.html"]','chapter-07-posttrain-route'),
  ('7.17.html','a.md-footer__link--next','7.17-next-7.18'),
  ('7.18.html','a.md-footer__link--next','7.18-next-chapter08'),
  ('chapter-08.html','a.md-footer__link--next','chapter08-next-8.1')
 ]:
  page.goto('http://127.0.0.1:8788/'+origin,wait_until='networkidle')
  link=page.locator(selector).first
  before={'url':page.url,'text':link.inner_text(),'href':link.get_attribute('href')}
  link.click()
  page.wait_for_load_state('networkidle')
  fragment=page.url.split('#',1)[1] if '#' in page.url else None
  target=page.locator('[id="'+fragment+'"]') if fragment else page.locator('article h1').first
  target.scroll_into_view_if_needed()
  page.screenshot(path=str(public/(label+'.png')))
  record={'timestamp_utc':datetime.now(timezone.utc).isoformat(),'action':'actual click','origin':before,'destination_url':page.url,'title':page.title(),'visible_heading':target.inner_text(),'anchor_exists':target.count()==1,'screenshot':str(public/(label+'.png'))}
  if fragment=='T.4':
   record['section_text']=target.evaluate('''(e)=>{let s=[];for(let n=e;n&&!(n!==e&&n.tagName==='H2');n=n.nextElementSibling)s.push(n.innerText);return s.join('\\n\\n');}''')
   print(record['section_text'])
  records.append(record)
  print(json.dumps({k:v for k,v in record.items() if k!='section_text'},ensure_ascii=False),flush=True)
  if fragment=='W.2':
   img=page.locator('article img[src$="foundations_tensor_axes.svg"]')
   img.scroll_into_view_if_needed()
   img.screenshot(path=str(public/'foundations_tensor_axes.svg.browser.png'))
   shutil.copyfile('course/figures/foundations_tensor_axes.svg',public/'foundations_tensor_axes.svg')
  page.wait_for_timeout(100)
 print('Browser',b.version)
 b.close()
(public/'navigation-receipts.json').write_text(json.dumps(records,ensure_ascii=False,indent=2)+'\n')
