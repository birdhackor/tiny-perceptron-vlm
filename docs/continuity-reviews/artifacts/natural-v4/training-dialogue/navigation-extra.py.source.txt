import json
from pathlib import Path
from datetime import datetime,timezone
from playwright.sync_api import sync_playwright
pub=Path('docs/continuity-reviews/artifacts/natural-v4/training-dialogue')
rpath=pub/'navigation-extra-receipts.json'
records=[]
with sync_playwright() as p:
 b=p.chromium.launch(headless=True,executable_path='/usr/bin/chromium',args=['--no-sandbox'])
 page=b.new_page(viewport={'width':1280,'height':1000})
 for origin,selector,name in [
  ('7.5.html','article a[href$="1.10.html"]','7.5-chain-prerequisite'),
  ('7.7.html','article a[href$="3.6.html"]','7.7-causal-prerequisite'),
  ('7.18.html','.md-sidebar--primary a[href$="7.12.html"]','posttrain-return-7.12')
 ]:
  page.goto('http://127.0.0.1:8788/'+origin,wait_until='networkidle')
  link=page.locator(selector).first
  before={'url':page.url,'text':link.inner_text(),'href':link.get_attribute('href')}
  link.click()
  page.wait_for_load_state('networkidle')
  heading=page.locator('article h1').first
  heading.scroll_into_view_if_needed()
  r={'timestamp_utc':datetime.now(timezone.utc).isoformat(),'action':'actual click','origin':before,'destination_url':page.url,'title':page.title(),'visible_heading':heading.inner_text()}
  records.append(r)
  print(json.dumps(r,ensure_ascii=False),flush=True)
 b.close()
rpath.write_text(json.dumps(records,ensure_ascii=False,indent=2)+'\n')
