import json
from pathlib import Path
from playwright.sync_api import sync_playwright

out = Path('docs/continuity-reviews/artifacts/natural-v4/natural-public')
records = []
with sync_playwright() as p:
    browser = p.chromium.launch(executable_path='/usr/bin/chromium', args=['--no-sandbox'], headless=True)
    page = browser.new_page(viewport={'width': 1440, 'height': 1000})
    def record(name):
        article = page.locator('article').first
        value = {'name':name,'url':page.url,'title':page.title(),'headings':article.locator('h1,h2,h3').all_text_contents(),'article_text':article.inner_text(),'links':article.locator('a[href]').evaluate_all('els => els.map(e=>({text:e.textContent.trim(),href:e.href}))')}
        records.append(value)
        print(name,page.url,value['headings'])
        (out/'browser-current.json').write_text(json.dumps({'base':'http://127.0.0.1:8789/','browser':browser.version,'records':records},ensure_ascii=False,indent=2)+'\n')
    page.goto('http://127.0.0.1:8789/',wait_until='networkidle');record('entry')
    page.locator('article').get_by_role('link',name='第 20 章：做一位能看圖、讀字與聽問題的助理',exact=True).click();page.wait_for_load_state('networkidle');record('chapter20')
    page.locator('article').get_by_role('link',name='20.2 不重新訓練，怎麼先開啟成品？',exact=False).first.click();page.wait_for_load_state('networkidle');record('20.2')
    page.locator('article').get_by_role('link',name='學生操作指引',exact=True).first.click();page.wait_for_load_state('networkidle');record('STUDENT')
    page.locator('article').get_by_role('link',name='資料說明',exact=True).click();page.wait_for_load_state('networkidle');record('DATA')
    page.locator('article').get_by_role('link',name='訓練指引',exact=True).last.click();page.wait_for_load_state('networkidle');record('TRAINING')
    page.locator('article').get_by_role('link',name='20.13',exact=True).last.click();page.wait_for_load_state('networkidle');record('capability-card')
    page.goto('http://127.0.0.1:8789/16.1.html',wait_until='networkidle');record('16.1')
    link=page.locator('article').get_by_role('link',name='T.8的效率量測方法',exact=True)
    print('clicked efficiency destination',link.get_attribute('href'))
    link.click();page.wait_for_load_state('networkidle');record('efficiency-target')
    target=page.locator('[id="選讀效率實驗的量測條件"]');target.scroll_into_view_if_needed()
    page.screenshot(path=str(out/'current-efficiency-target.png'))
    print('anchor heading',target.inner_text(),'visible',target.is_visible())
    browser.close()
