from playwright.sync_api import sync_playwright
from pathlib import Path
import json
out=Path("docs/continuity-reviews/artifacts/natural-v4/foundations-entries")
records=[]
with sync_playwright() as p:
    browser=p.chromium.launch(executable_path="/usr/bin/chromium",headless=True,args=["--no-sandbox"])
    page=browser.new_page(viewport={"width":1440,"height":1050})
    def click(name):
        source=page.url
        link=page.locator("article").last.get_by_role("link",name=name,exact=True)
        href=link.get_attribute("href")
        link.click()
        page.wait_for_load_state("networkidle")
        records.append({"action":"click","source_url":source,"link_text":name,"href":href,"destination_url":page.url,"destination_h1":page.locator("article h1").all_text_contents()})
    page.goto("http://127.0.0.1:8788/",wait_until="networkidle")
    click("先看基礎暖身")
    click("1.1 正文")
    page.locator("article pre").nth(1).scroll_into_view_if_needed()
    page.screenshot(path=str(out/"browser-1.1-output.png"))
    records.append({"label":"visible-CPU-output","url":page.url,"text":page.locator("article pre").nth(1).inner_text(),"screenshot":str(out/"browser-1.1-output.png")})
    page.locator("article").last.get_by_role("link",name="下載本節 .ipynb",exact=True).scroll_into_view_if_needed()
    page.screenshot(path=str(out/"browser-1.1-notebook-entry.png"))
    records.append({"label":"notebook-entry","url":page.url,"links":page.locator("article").last.locator("a").evaluate_all("es=>es.map(e=>({text:e.innerText,href:e.href})).filter(e=>/Colab|ipynb|W.1/.test(e.text))"),"screenshot":str(out/"browser-1.1-notebook-entry.png")})
    for chapter in ["01","02","03","04","05"]:
        page.goto("http://127.0.0.1:8788/chapter-"+chapter+".html",wait_until="networkidle")
        article=page.locator("article").last
        records.append({"label":"chapter-introduction-"+chapter,"url":page.url,"heading":article.locator("h1").inner_text(),"first_paragraph":article.locator("p").first.inner_text(),"first_lesson":article.locator("a").evaluate_all("(es, suffix)=>es.map(e=>({text:e.innerText,href:e.href})).find(e=>e.href.endsWith(suffix))", str(int(chapter))+".1.html")})
    for fragment in ["T.4","T.8"]:
        page.goto("http://127.0.0.1:8788/training.html#"+fragment,wait_until="networkidle")
        records.append({"label":"training-section-"+fragment,"url":page.url,"section":page.evaluate("id=>{const h=document.getElementById(id);const nodes=[h.innerText];for(let e=h.nextElementSibling;e&&!/^H[12]$/.test(e.tagName);e=e.nextElementSibling)nodes.push(e.innerText);return nodes;}",fragment)})
    browser.close()
print(json.dumps({"records":records},ensure_ascii=False,indent=2))
