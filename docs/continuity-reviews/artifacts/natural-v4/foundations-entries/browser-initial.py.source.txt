from playwright.sync_api import sync_playwright
import json
from pathlib import Path
out=Path("docs/continuity-reviews/artifacts/natural-v4/foundations-entries")
with sync_playwright() as p:
    browser=p.chromium.launch(executable_path="/usr/bin/chromium",headless=True,args=["--no-sandbox"])
    page=browser.new_page(viewport={"width":1440,"height":1050},device_scale_factor=1)
    response=page.goto("http://127.0.0.1:8788/",wait_until="networkidle",timeout=45000)
    print(json.dumps({"url":page.url,"http_status":response.status,"title":page.title(),"headings":page.locator("main h1,main h2").all_text_contents(),"links":page.locator("main a").evaluate_all("els => els.map(e => ({text:e.textContent.trim(),href:e.href}))")},ensure_ascii=False,indent=2))
    page.screenshot(path=str(out/"browser-entry.png"),full_page=False)
    browser.close()
