from playwright.sync_api import sync_playwright
from pathlib import Path
import json,platform,subprocess
out=Path("docs/continuity-reviews/artifacts/natural-v4/foundations-entries")
records=[]
with sync_playwright() as p:
    browser=p.chromium.launch(executable_path="/usr/bin/chromium",headless=True,args=["--no-sandbox"])
    page=browser.new_page(viewport={"width":1440,"height":1050},device_scale_factor=1)
    def snapshot(label, screenshot=None):
        article=page.locator("article").last
        r={"label":label,"url":page.url,"title":page.title(),"headings":article.locator("h1,h2,h3").all_text_contents(),"links":article.locator("a").evaluate_all("els=>els.map(e=>({text:e.textContent.trim(),href:e.href})).filter(e=>e.text!=='¶')"),"body_excerpt":article.inner_text()[:1700],"fragment_target":page.evaluate("() => {const id=decodeURIComponent(location.hash.slice(1));const e=id?document.getElementById(id):null;return {id,exists:!!e,text:e?.innerText||null,top:e?.getBoundingClientRect().top||null};}")}
        if screenshot:
            page.screenshot(path=str(out/screenshot),full_page=False)
            r["screenshot"]=str(out/screenshot)
        records.append(r)
    def click_text(text):
        source=page.url
        link=page.locator("article").last.get_by_role("link",name=text,exact=True)
        href=link.get_attribute("href")
        link.click()
        page.wait_for_load_state("networkidle")
        records.append({"action":"click","source_url":source,"link_text":text,"href":href,"destination_url":page.url})
    page.goto("http://127.0.0.1:8788/",wait_until="networkidle")
    click_text("閱讀指南")
    snapshot("guide", "browser-guide.png")
    click_text("第一節：文字怎麼變成數字？")
    snapshot("guide-first-lesson", "browser-1.1.png")
    click_text("W.2")
    snapshot("lesson-prerequisite-W.2", "browser-warmup-W.2.png")
    page.goto("http://127.0.0.1:8788/first-steps.html#W.1",wait_until="networkidle")
    snapshot("warmup-W.1", "browser-warmup-W.1.png")
    page.goto("http://127.0.0.1:8788/1.1.html",wait_until="networkidle")
    article=page.locator("article").last
    records.append({"label":"1.1-execution-and-notebook-links","url":page.url,"visible_output_elements":article.locator("details").all_text_contents(),"last_body":article.inner_text()[-2800:]})
    page.goto("http://127.0.0.1:8788/4.7.html",wait_until="networkidle")
    click_text("T.4的實跑")
    snapshot("4.7-to-T.4", "browser-training-T.4.png")
    page.goto("http://127.0.0.1:8788/training.html#T.8",wait_until="networkidle")
    snapshot("training-T.8", "browser-training-T.8.png")
    browser.close()
print(json.dumps({"environment":{"python":platform.python_version(),"chromium":subprocess.check_output(["/usr/bin/chromium","--version"],text=True).strip()},"records":records},ensure_ascii=False,indent=2))
