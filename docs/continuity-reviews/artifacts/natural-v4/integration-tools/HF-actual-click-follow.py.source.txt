from pathlib import Path
import json
from playwright.sync_api import sync_playwright

OUT = Path('docs/continuity-reviews/artifacts/natural-v4/integration-tools')
with sync_playwright() as p:
    b = p.chromium.launch(executable_path='/usr/bin/chromium',args=['--no-sandbox'])
    page = b.new_page(viewport={'width':1440,'height':1080})
    page.goto('http://127.0.0.1:8789/19.11.html',wait_until='networkidle')
    link = page.locator('article a').filter(has_text='Hugging Face公開模型庫').first
    href = link.get_attribute('href')
    link.click()
    page.wait_for_function('document.title.includes("tiny-perceptron-course-models")',timeout=20000)
    page.wait_for_function('document.body.innerText.includes("capstone_deployment")',timeout=20000)
    text = page.locator('body').inner_text()
    data = {'origin':'http://127.0.0.1:8789/19.11.html','actual_clicked_href':href,
            'destination':page.url,'visible_title':page.title(),'visible_text':text,
            'semantic_assessment':'Pinned integration-v2 tree, with capstone_deployment and capstone_student group folders; no weight downloads or release audit.'}
    (OUT/'HF-actual-click-follow-receipt.json').write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n')
    print(json.dumps(data,ensure_ascii=False),flush=True)
    page.screenshot(path=str(OUT/'HF-actual-click-follow.png'))
    b.close()
