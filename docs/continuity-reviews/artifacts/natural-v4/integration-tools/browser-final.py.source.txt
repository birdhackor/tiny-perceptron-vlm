from pathlib import Path
from urllib.parse import unquote, urlparse
import hashlib
import json
from playwright.sync_api import sync_playwright

BASE = 'http://127.0.0.1:8789/'
OUT = Path('docs/continuity-reviews/artifacts/natural-v4/integration-tools')
records = []

def save():
    (OUT / 'preview8789-browser-receipt.json').write_text(json.dumps(records, ensure_ascii=False, indent=2) + '\n')

def record(page, label, screenshot=None):
    text = page.locator('article').inner_text()
    data = {'label': label, 'url': page.url, 'title': page.title(),
            'visible_heading': page.locator('article h1').inner_text(),
            'visible_excerpt': text[:1800],
            'article_text_sha256': hashlib.sha256(text.encode()).hexdigest()}
    anchor = unquote(urlparse(page.url).fragment)
    if anchor:
        el = page.locator('[id="' + anchor + '"]')
        data['anchor_id'] = anchor
        data['anchor_exists'] = el.count() == 1
        if el.count() == 1:
            data['anchor_visible_heading'] = el.inner_text()
            data['anchor_top_px'] = el.evaluate('el=>el.getBoundingClientRect().top')
            data['section_visible_excerpt'] = el.evaluate('el=>{let s="";for(let n=el.nextElementSibling;n && !["H2","H3"].includes(n.tagName);n=n.nextElementSibling){s+=n.innerText+"\\n";}return s;}')[:4500]
    if screenshot:
        page.screenshot(path=str(OUT / screenshot))
        data['screenshot'] = str(OUT / screenshot)
    records.append(data)
    save()
    print(label, page.url, data.get('anchor_visible_heading', data['visible_heading']), flush=True)

def click(page, link, label, screenshot=None):
    origin = page.url
    text = link.first.inner_text().strip()
    href = link.first.get_attribute('href')
    link.first.click()
    page.wait_for_load_state('networkidle')
    record(page, label, screenshot)
    records[-1].update(clicked_from=origin, clicked_text=text, clicked_href=href)
    save()
