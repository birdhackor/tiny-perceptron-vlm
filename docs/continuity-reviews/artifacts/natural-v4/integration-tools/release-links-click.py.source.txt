from pathlib import Path
import json
from playwright.sync_api import sync_playwright

OUT = Path('docs/continuity-reviews/artifacts/natural-v4/integration-tools')
records = []
with sync_playwright() as p:
    browser = p.chromium.launch(executable_path='/usr/bin/chromium', args=['--no-sandbox'])
    for label in ['Hugging Face公開模型庫', '公開清單', '訓練器原始碼']:
        page = browser.new_page(viewport={'width':1440,'height':1080})
        page.goto('http://127.0.0.1:8789/19.11.html', wait_until='networkidle')
        link = page.locator('article a').filter(has_text=label).first
        href = link.get_attribute('href')
        target = link.get_attribute('target')
        if target == '_blank':
            with page.expect_popup() as popup:
                link.click()
            destination = popup.value
        else:
            link.click()
            destination = page
        destination.wait_for_load_state('networkidle')
        body = destination.locator('body').inner_text()
        records.append({'label':label,'clicked_from':'http://127.0.0.1:8789/19.11.html',
                        'actual_clicked_href':href,'target_attribute':target,'actual_destination':destination.url,
                        'visible_title':destination.title(),
                        'semantic_terms':{term:term in body for term in ['course-integration-v2','capstone_deployment','capstone_student','33c6898','joint','capstone-public.json','resume','capstone.py']}})
        print(json.dumps(records[-1],ensure_ascii=False),flush=True)
        if destination != page:
            destination.close()
        page.close()
    browser.close()
(OUT/'release-links-click-receipt.json').write_text(json.dumps(records,ensure_ascii=False,indent=2)+'\n')
