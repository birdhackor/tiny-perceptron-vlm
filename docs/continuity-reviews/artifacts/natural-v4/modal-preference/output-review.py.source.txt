from pathlib import Path
import json
from playwright.sync_api import sync_playwright

ROOT = Path('/workspace/tiny-perceptron-vlm')
OUT = ROOT / 'docs/continuity-reviews/artifacts/natural-v4/modal-preference'
BASE = 'http://127.0.0.1:8788/'
receipts = json.loads((OUT / 'browser-receipts.json').read_text())
with sync_playwright() as pw:
    browser = pw.chromium.launch(executable_path='/usr/bin/chromium', headless=True, args=['--no-sandbox'])
    page = browser.new_page(viewport={'width': 1280, 'height': 900}, device_scale_factor=1)
    for slug in ['11.5', '12.9', '13.15']:
        page.goto(BASE + slug + '.html', wait_until='domcontentloaded')
        widgets = page.locator('article div.output')
        observed = widgets.evaluate_all('(xs)=>xs.map(x=>x.innerText)')
        assert observed, (slug, 'no executed output widget')
        widget = widgets.first
        widget.scroll_into_view_if_needed()
        shot = OUT / 'navigation' / ('output-' + slug + '.png')
        widget.screenshot(path=str(shot))
        receipts['outputs'].append({'url': page.url, 'observed_widget_text': observed, 'screenshot': str(shot.relative_to(ROOT)), 'scope': 'Observed existing root-built executed preview; this reviewer did not rerun the teaching kernel.'})
        print('Observed output', slug, observed)
    browser.close()
(OUT / 'browser-receipts.json').write_text(json.dumps(receipts, ensure_ascii=False, indent=2) + '\n')
print('Preserved', len(receipts['navigation']), 'navigation clicks and', len(receipts['outputs']), 'output pages')
