figures = []
for i, name in enumerate(view_order, 1):
    source = ROOT / 'course/figures' / (name + '.svg')
    original = OUT / 'svg-originals' / source.name
    render = OUT / 'figure-renders' / (name + '.png')

scopes = json.loads((TMP / 'scopes.json').read_text())
coverage = []
for file, sections in scopes.items():
    raw = (ROOT / file).read_bytes()
    content = raw.decode()
    matches = list(re.finditer(r'^## ([A-Z]?\d*\.[0-9]+|[A-Z]\.[0-9]+)\b[^\n]*', content, re.M))
    selected = []
    for i, match in enumerate(matches):
        identifier = match.group(1)
        if sections is None or identifier in sections:
            stop = matches[i+1].start() if i+1 < len(matches) else len(content)
            selected.append({'section': identifier, 'heading': match.group(0), 'sha256': sha(content[match.start():stop].encode())})
    assert sections is None or set(sections) == {s['section'] for s in selected}, (file, sections, selected)
    intro_read = file in ['course/chapters/11.md','course/chapters/12.md','course/chapters/13.md','course/chapters/10.md','course/chapters/14.md','course/chapters/20.md']
    coverage.append({'path': file, 'current_wholefile_sha256': sha(raw), 'scope': 'complete assigned file, including intro, all sections/code/tables/formulas/exercises' if sections is None else 'complete listed prerequisite or boundary sections; no claim of reading unlisted sections', 'introduction_read': intro_read, 'introduction_sha256': sha(content[:matches[0].start()].encode()) if intro_read and matches else None, 'complete_sections_read': selected})
