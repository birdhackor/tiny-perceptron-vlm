const fs = require('fs');
const path = require('path');
const crypto = require('crypto');
const {chromium} = require('/opt/codex/runtimes/cua/lib/node_modules/playwright-core');
const root='/workspace/tiny-perceptron-vlm';
const out=path.join(root,'docs/continuity-reviews/artifacts/natural-v4/style-safety/browser');
const base='http://127.0.0.1:8788/';
const records=[];
const sha=bytes=>crypto.createHash('sha256').update(bytes).digest('hex');
let browser;
async function main(){
  browser=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox']});
  const context=await browser.newContext({viewport:{width:1400,height:1000},deviceScaleFactor:1});
  const page=await context.newPage();
  const consoleErrors=[];
  page.on('pageerror',error=>consoleErrors.push(String(error)));
  async function capture(label,detail,screen=false,excerpt=false){
    const article=page.locator('article');
    const text=await article.innerText();
    const textFile=path.join(out,label+'.text.txt');
    fs.writeFileSync(textFile,excerpt?text.slice(0,1200):text);
    const record={label,url:page.url(),title:await page.title(),detail,headings:await article.locator('h1,h2').allTextContents(),links:await article.locator('a').evaluateAll(a=>a.map(a=>({text:a.innerText.trim(),href:a.href}))),code_and_output:excerpt?[]:await article.locator('pre').allTextContents(),text_artifact:path.relative(root,textFile),text_sha256:sha(fs.readFileSync(textFile))};
    if(page.url().includes('#')){
      record.anchor=await page.evaluate(()=>{const id=decodeURIComponent(location.hash.slice(1));const e=document.getElementById(id);return{id,exists:!!e,text:e?.innerText,bounds:e?{top:e.getBoundingClientRect().top,bottom:e.getBoundingClientRect().bottom}:null}});
    }
    if(screen){
      const png=path.join(out,label+'.png');
      await page.screenshot({path:png});
      record.screenshot=path.relative(root,png);record.screenshot_sha256=sha(fs.readFileSync(png));
    }
    records.push(record);
    fs.writeFileSync(path.join(out,'path-records.json'),JSON.stringify({records,console_errors:consoleErrors,limits:'Selected links and displayed outputs only; no fresh Torch execution.'},null,2)+'\n');
  }
  async function go(route,label,detail,screen=false,excerpt=false){const response=await page.goto(base+route);await page.waitForLoadState('networkidle');await capture(label,{...detail,http_status:response?.status() ?? null},screen,excerpt)}
  async function click(selector,label,detail,screen=false,excerpt=false){
    const a=page.locator(selector).first();
    const before={url:page.url(),text:(await a.innerText()).trim(),href:await a.getAttribute('href')};
    await a.click();await page.waitForLoadState('networkidle');await capture(label,{...detail,clicked:before},screen,excerpt);
  }
  await go('course.html','route-guide',{purpose:'Identify actual chapter 8 route promise'},false,true);
  await click('article a[href$="chapter-08.html"]','guide-to-chapter08',{purpose:'Actual reading-guide chapter 8 link; inspect introduction'},false);
  await go('7.18.html','preceding-end',{purpose:'Full preceding last-section browser presentation'});
  await click('.md-footer__link--next','boundary07-08',{purpose:'Actual preceding ending to chapter 8 introduction'});
  await go('8.3.html','style-supervision',{purpose:'Check formal training-scope caveats and actual CPU outputs'});
  await click('article a[href$="7.3.html"]','style-to-supervision-prereq',{purpose:'Prerequisite promise: which answer positions are taught'});
  await go('8.8.html','lora-output',{purpose:'Actual CPU output 56, initial equality and A/B gradient distinction'},false);
  const output=page.locator('article pre').filter({hasText:'可訓練參數 56'}).first();
  await output.scrollIntoViewIfNeeded();await page.screenshot({path:path.join(out,'lora-output-screen.png')});
  await click('article a[href$="first-steps.html#W.5"]','lora-to-matrix-prereq',{purpose:'Warm-up matrix / weighted-recipe prerequisite anchor'},false,true);
  await go('8.3.html','style-before-training-link',{purpose:'Use actual T.5 link'});
  await click('article a[href$="training.html#T.5"]','style-to-training-T5',{purpose:'T.5 recipe separates content/style and toy/GPU/public-adapter scope'},false,true);
  await go('training.html#T.4','training-T4',{purpose:'Check actual visible T.4 topic heading and working anchor'},true,true);
  await go('training.html#T.8','training-T8',{purpose:'Contract-requested heading/anchor inspection only; not an assigned substantive prerequisite'},false,true);
  await go('8.13.html','style-ending',{purpose:'Full chapter 8 ending including exercise'});
  await click('.md-footer__link--next','boundary08-09',{purpose:'Actual chapter 8 ending to chapter 9 introduction'});
  await go('9.1.html','safety-start',{purpose:'Permission and preference-label distinction'});
  await click('article a[href*="release-licenses"]','safety-license-destination',{purpose:'Actual data source / release scope link'},false,true);
  await go('9.7.html','injection-source',{purpose:'Quoted-content versus trusted task prerequisite link'});
  await click('article a[href$="7.1.html"]','safety-to-roles-prereq',{purpose:'Actual role/content representation prerequisite'});
  await go('9.10.html','calibration-output',{purpose:'Actual bin outputs, synthetic execution versus trained classifiers'},false);
  const fig=page.locator('article img[src*="audio_calibration_test"]');
  await fig.scrollIntoViewIfNeeded();await page.screenshot({path:path.join(out,'calibration-screen.png')});
  await click('.md-footer__link--next','boundary09-10',{purpose:'Actual chapter 9 ending to chapter 10 introduction'});
  await go('10.7.html','modal-alignment-output',{purpose:'Actual CPU six-position output and -100/answer alignment'},false);
  await page.locator('article pre').filter({hasText:'下一格目標 [[-100'}).first().scrollIntoViewIfNeeded();
  await page.screenshot({path:path.join(out,'modal-alignment-screen.png')});
  await click('article a[href$="7.4.html"]','modal-to-shift-prereq',{purpose:'Expand first, shift once; assistant predicts first answer'});
  await go('10.8.html','pure-text-output',{purpose:'Actual equality output separate from trained retention'});
  await go('10.11.html','vision-ending',{purpose:'Full last section distinguishes retrieval from generation'});
  await click('.md-footer__link--next','boundary10-11',{purpose:'Actual chapter 10 ending to chapter 11 introduction'});
  await click('article a[href$="11.1.html"]','chapter11-first-section',{purpose:'Full next first section connects width compatibility to semantic checks'},true);
  await click('article a[href$="10.6.html"]','next-intro-to-projector-prereq',{purpose:'Next-boundary explicit projector prerequisite'});
  fs.writeFileSync(path.join(out,'path-records.json'),JSON.stringify({records,console_errors:consoleErrors,limits:'Selected actual browser links and present outputs; no fresh Torch execution, external paper/data/model fetch or whole-site audit.'},null,2)+'\n');
  console.log(JSON.stringify({visited_records:records.length,urls:records.map(r=>r.url),console_errors:consoleErrors},null,2));
}
main().catch(e=>{console.error(e.stack);process.exitCode=1}).finally(async()=>{if(browser)await browser.close()});
