const fs=require('fs');
const crypto=require('crypto');
const {request}=require('/opt/codex/runtimes/cua/lib/node_modules/playwright-core');
const root='/workspace/tiny-perceptron-vlm';
const out=root+'/docs/continuity-reviews/artifacts/natural-v4/style-safety';
const sha=b=>crypto.createHash('sha256').update(b).digest('hex');
(async()=>{
  const context=await request.newContext();
  const renders=JSON.parse(fs.readFileSync(out+'/figure-renders.json','utf8'));
  const records=[];
  for(const figure of renders){
    const url='http://127.0.0.1:8788/figures/'+figure.source.split('/').pop();
    const response=await context.get(url);
    const bytes=await response.body();
    records.push({source:figure.source,url,status:response.status(),served_sha256:sha(bytes),matches_personally_viewed_current_source:sha(bytes)===figure.source_sha256});
  }
  await context.dispose();
  fs.writeFileSync(out+'/browser/served-figures.json',JSON.stringify(records,null,2)+'\n');
  console.log(JSON.stringify(records,null,2));
  if(records.some(r=>r.status!==200||!r.matches_personally_viewed_current_source))process.exitCode=1;
})().catch(e=>{console.error(e.stack);process.exitCode=1});
