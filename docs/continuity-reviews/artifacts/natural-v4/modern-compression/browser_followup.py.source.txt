import json
from pathlib import Path

from playwright.sync_api import sync_playwright


DEST = Path('docs/continuity-reviews/artifacts/natural-v4/modern-compression')
receipts = []

with sync_playwright() as p:
    browser = p.chromium.launch(executable_path='/usr/bin/chromium', args=['--no-sandbox'])
    page = browser.new_page(viewport={'width':1360, 'height':980}, device_scale_factor=1)
    page.goto('http://127.0.0.1:8788/16.1.html', wait_until='networkidle')
    link = page.locator('article a').filter(has_text='T.8的效率量測方法')
    start = page.url
    clicked = {'text':link.inner_text(), 'href':link.get_attribute('href')}
    link.click()
    page.wait_for_load_state('networkidle')
    heading = page.locator('[id="T.8"]')
    intended = page.locator('article strong').filter(has_text='選讀：效率實驗的量測條件。')
    target = intended.evaluate('(e)=>({tag:e.tagName,id:e.id,parentTag:e.parentElement.tagName,parentId:e.parentElement.id,text:e.parentElement.innerText})')
    target['initial_rect'] = intended.bounding_box()
    toc = page.locator('.md-sidebar--secondary').inner_text()
    receipts.append({'from':start,'clicked':clicked,'url':page.url,'landing_heading':heading.inner_text(), 'landing_heading_rect':heading.bounding_box(),'intended_topic':target,'visible_toc':toc})
    intended.scroll_into_view_if_needed()
    page.screenshot(path=str(DEST / 'browser-T8-efficiency-body.png'))

    page.goto('http://127.0.0.1:8788/18.13.html', wait_until='networkidle')
    for label in ['模態展開後的位置', '回答第一token的預測位置']:
        source = page.url
        link = page.locator('article a').filter(has_text=label)
        clicked = {'text':link.inner_text(),'href':link.get_attribute('href')}
        link.click()
        page.wait_for_load_state('networkidle')
        receipts.append({'from':source,'clicked':clicked,'url':page.url,'heading':page.locator('article h1').inner_text(),'article_body':page.locator('article').inner_text()})
        page.goto('http://127.0.0.1:8788/18.13.html',wait_until='networkidle')
    browser.close()

(DEST / 'browser-followup-raw.json').write_text(json.dumps(receipts,ensure_ascii=False,indent=2)+'\n')
print(json.dumps(receipts[0],ensure_ascii=False,indent=2))
