def normalize(value):
    value = value.strip().removeprefix("./")
    if value.startswith(str(ROOT) + "/"):
        value = value[len(str(ROOT)) + 1:]
    value = value.split("#", 1)[0]
    value = re.sub(r":\d+(?:[-–]\d+)?$", "", value)
    if not re.fullmatch(r"[A-Za-z0-9_.+/@-]+", value):
        return None
    if not (value.startswith(PREFIXES) or value in ROOT_FILES):
        return None
    p = (ROOT / value).resolve()
    if not p.is_relative_to(ROOT):
        return None
    return p.relative_to(ROOT).as_posix()


def add(value, source, locator, kind, strict=False):
    path = normalize(value)
    if not path:
        return
    p = ROOT / path
    if p.suffix in (".pt", ".pth", ".ckpt", ".safetensors") and not strict:
        excluded_weights.append({"path": path, "source": source, "locator": locator, "kind": kind, "exists": p.is_file(), "reason": "Runtime/historical weight observation; excluded from Git staging requirements by task policy."})
        return
    reason = json.dumps({"source": source, "locator": locator, "kind": kind}, sort_keys=True, ensure_ascii=False)
    if p.is_file():
        if path not in reasons:
            pending.append(path)
        reasons[path].add(reason)
    elif p.is_dir():
        directories.append({"source": source, "path": path, "locator": locator, "kind": kind, "directory_not_expanded": True})
    else:
        row = {"source": source, "path": path, "locator": locator, "kind": kind}
        (missing if strict else observations).append(row)


def strings(value, trail=()):
    if isinstance(value, dict):
        for k, v in value.items():
            if isinstance(k, str):
                yield k, trail + ("<key>",)
            yield from strings(v, trail + (str(k),))
    elif isinstance(value, list):
        for i, v in enumerate(value):
            yield from strings(v, trail + (str(i),))
    elif isinstance(value, str):
        yield value, trail


def explicit_strings(data, source):
    for value, trail in strings(data):
        loc = ".".join(trail)
        exact = normalize(value)
        if exact and "\n" not in value and not re.search(r"\s", value):
            # Historical git object locators and model/runtime observations do
            # not imply a second working-tree source or a Git-staged weight.
            add(exact, source, loc, "structured_repository_reference")
            continue
        if trail and re.fullmatch(r"[A-Za-z0-9_.+@-]+", value) and trail[-1] in ("path", "file", "name", "snapshot_path", "receipt", "previous_report_snapshot"):
            candidate = (ROOT / source).parent / value
            if candidate.is_file():
                add(candidate.relative_to(ROOT).as_posix(), source, loc, "receipt_relative_file_reference")
        for match in EXPLICIT.finditer(value):
            before = value[max(0, match.start() - 180):match.start()]
            if re.search(r"https?://[^\s]*$", before) or re.search(r"[0-9a-f]{40}:$", before):
                continue
            add(match.group(), source, loc, "explicit_repository_token")


def eval_expr(node, env):
    if isinstance(node, ast.Constant):
        return node.value if isinstance(node.value, (str, int)) else None
    if isinstance(node, ast.Name):
        return env.get(node.id)
    if isinstance(node, (ast.List, ast.Tuple)):
        return [eval_expr(x, env) for x in node.elts]
    if isinstance(node, ast.JoinedStr):
        parts = []
        for x in node.values:
            value = eval_expr(x.value if isinstance(x, ast.FormattedValue) else x, env)
            if not isinstance(value, (str, int)):
                return None
            parts.append(str(value))
        return "".join(parts)
    if isinstance(node, ast.Call):
        if isinstance(node.func, ast.Name) and node.func.id == "Path" and node.args:
            value = eval_expr(node.args[0], env)
            return Path(value) if isinstance(value, (str, Path)) else None
        if isinstance(node.func, ast.Attribute):
            value = eval_expr(node.func.value, env)
            if isinstance(value, Path) and node.func.attr in ("resolve", "absolute"):
                return value.resolve()
        return None
    if isinstance(node, ast.Attribute):
        value = eval_expr(node.value, env)
        if isinstance(value, Path):
            if node.attr == "parent":
                return value.parent
            if node.attr == "parents":
                return list(value.parents)
    if isinstance(node, ast.Subscript):
        value, index = eval_expr(node.value, env), eval_expr(node.slice, env)
        if isinstance(value, list) and isinstance(index, int) and -len(value) <= index < len(value):
            return value[index]
    if isinstance(node, ast.BinOp):
        left, right = eval_expr(node.left, env), eval_expr(node.right, env)
        if isinstance(node.op, ast.Div) and isinstance(left, (str, Path)) and isinstance(right, str):
            return Path(left) / right
        if isinstance(node.op, ast.Add) and isinstance(left, str) and isinstance(right, str):
            return left + right
    return None


def replay_script(path):
    p = ROOT / path
    text = p.read_text(errors="replace")
    try:
        tree = ast.parse(text)
    except SyntaxError as exc:
        parse_errors.append({"path": path, "type": "python_syntax", "error": str(exc)})
        return
    env = {"__file__": p}
    helpers = {}
    for fn in (x for x in tree.body if isinstance(x, ast.FunctionDef)):
        reads = [x.func.value for x in ast.walk(fn) if isinstance(x, ast.Call) and isinstance(x.func, ast.Attribute) and x.func.attr in ("read_text", "read_bytes")]
        if reads:
            helpers[fn.name] = ([x.arg for x in fn.args.args], reads)
    nodes = sorted(ast.walk(tree), key=lambda n: (getattr(n, "lineno", 0), getattr(n, "col_offset", 0)))
    for node in nodes:
        if isinstance(node, ast.Assign):
            val = eval_expr(node.value, env)
            for target in node.targets:
                if isinstance(target, ast.Name) and val is not None:
                    env[target.id] = val
        if isinstance(node, ast.ImportFrom) and node.module:
            module = node.module.replace(".", "/")
            if node.level:
                candidate = p.parent / (module + ".py")
            else:
                candidate = ROOT / (module + ".py")
            if candidate.is_file() and candidate.is_relative_to(ROOT):
                add(candidate.relative_to(ROOT).as_posix(), path, "line:" + str(node.lineno), "repository_python_import")
        if not isinstance(node, ast.Call):
            continue
        name = node.func.id if isinstance(node.func, ast.Name) else node.func.attr if isinstance(node.func, ast.Attribute) else ""
        # File readers plus custom common replay reading helpers. Outputs are
        # excluded here; report/receipt references still preserve saved outputs.
        inputs = []
        local_env = env
        if isinstance(node.func, ast.Name) and name in helpers and node.args:
            args, expressions = helpers[name]
            local_env = env | {key: eval_expr(value, env) for key, value in zip(args, node.args)}
            inputs.extend(expressions)
        elif isinstance(node.func, ast.Attribute) and name in ("read_text", "read_bytes"):
            inputs.append(node.func.value)
        elif isinstance(node.func, ast.Attribute) and name == "open":
            receiver = eval_expr(node.func.value, env)
            if isinstance(receiver, Path):
                mode = eval_expr(node.args[0], env) if node.args else "r"
                if not isinstance(mode, str) or not any(c in mode for c in "wax"):
                    inputs.append(node.func.value)
            elif isinstance(node.func.value, ast.Name) and node.func.value.id in ("tarfile", "gzip", "bz2", "lzma") and node.args:
                inputs.append(node.args[0])
            else:
                inputs.append(node.func.value)
        elif name in ("read", "sha", "audit_file", "audit_evaluation", "load", "loadtxt", "read_csv", "read_json", "load_state_dict", "load_capstone", "load_quantized_capstone") and node.args:
            inputs.append(node.args[0])
        elif name == "open" and node.args:
            mode = eval_expr(node.args[1], env) if len(node.args) > 1 else "r"
            if not isinstance(mode, str) or not any(c in mode for c in "wax"):
                inputs.append(node.args[0])
        for expr in inputs:
            val = eval_expr(expr, local_env)
            rendered = ast.get_source_segment(text, expr) or ast.dump(expr)
            if not isinstance(val, (Path, str)):
                # Parameter reads/dynamic loops are explicitly outside coverage.
                if isinstance(expr, (ast.BinOp, ast.JoinedStr)):
                    potential.append({"script": path, "line": node.lineno, "expression": rendered, "status": "dynamic_expression_unresolved"})
                continue
            candidate = Path(val)
            if not candidate.is_absolute():
                if normalize(str(candidate)):
                    candidate = ROOT / candidate
                elif (p.parent / candidate).is_file():
                    candidate = p.parent / candidate
                else:
                    candidate = ROOT / candidate
            candidate = candidate.resolve()
            if not candidate.is_relative_to(ROOT):
                replay_probes.append({"script": path, "line": node.lineno, "expression": rendered, "path": str(candidate), "status": "external_runtime_path_not_for_git"})
                continue
            relative = candidate.relative_to(ROOT).as_posix()
            row = {"script": path, "line": node.lineno, "expression": rendered, "path": relative, "exists": candidate.is_file()}
            replay_probes.append(row)
            if candidate.is_file():
                add(relative, path, "line:" + str(node.lineno), "literal_replay_input")
            else:
                potential.append(row | {"status": "literal_read_path_absent_at_audit"})

                    inspect(x, trail + (str(i),))
