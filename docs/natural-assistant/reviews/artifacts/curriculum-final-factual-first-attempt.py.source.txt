def expand_references(value):
    identifiers = set()
    for match in re.finditer(r"(?<![\w.])([A-Z]|\d{1,2})\.(\d+)(?:\s*[––-]\s*(?:([A-Z]|\d{1,2})\.)?(\d+))?", value):
        chapter, first, last_chapter, last = match.groups()
        identifiers.add(f"{chapter}.{int(first)}")
        if last:
            assert last_chapter in (None, chapter), match[0]
            assert int(last) >= int(first), match[0]
            identifiers.update(f"{chapter}.{number}" for number in range(int(first), int(last) + 1))
    return identifiers

    table_rows.append(row)

links = re.findall(r"\[[^\]]+\]\(([^)]+)\)", text)
local_links, remote_links = [], []
for target in links:
    parts = urlsplit(target)
    if parts.scheme in ("https", "http"):
        remote_links.append(target)
        continue
    path = (curriculum.parent / unquote(parts.path)).resolve()
    record = {"href": target, "path": str(path.relative_to(ROOT)), "exists": path.is_file(), "fragment": parts.fragment}
    if parts.fragment:
        record["fragment_exists"] = bool(re.search(r"^#{1,6}\s+" + re.escape(parts.fragment) + r"(?:\s|$)", path.read_text(), re.M)) if path.is_file() else False
    record["sha256"] = sha(path) if path.is_file() else None
    local_links.append(record)
assert all(row["exists"] and row.get("fragment_exists", True) for row in local_links), local_links


def check_pinned_remote(url):
    parsed = urlsplit(url)
    match = re.fullmatch(r"/([^/]+/[^/]+)/blob/([a-f0-9]{40})/(.+)", parsed.path)
    assert parsed.scheme == "https" and parsed.netloc == "github.com" and match, url
    repo, pin, path = match.groups()
    raw_url = f"https://raw.githubusercontent.com/{repo}/{pin}/{path}"
    started = time.monotonic()
    try:
        request = urllib.request.Request(raw_url, method="HEAD", headers={"User-Agent": "independent-curriculum-reference-audit", "Accept-Encoding": "identity"})
        with urllib.request.urlopen(request, timeout=30) as response:
            return {"href": url, "fixed_revision": pin, "raw_file_check": raw_url, "method": "HEAD", "status": response.status, "elapsed_seconds": time.monotonic() - started, "check_scope": "Fixed code/document path availability only; no historical content rereview or training execution"}
    except Exception as error:
        return {"href": url, "fixed_revision": pin, "raw_file_check": raw_url, "method": "HEAD", "error": f"{type(error).__name__}: {error}", "elapsed_seconds": time.monotonic() - started}
