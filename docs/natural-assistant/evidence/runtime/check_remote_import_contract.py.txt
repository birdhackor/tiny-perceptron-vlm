"""Real Modal 1.6 bootstrap-context import checks, without authentication/calls."""
import hashlib,json,subprocess,sys
from pathlib import Path
program=r'''
import importlib,importlib.metadata,json,os,sys
from modal._runtime.task_lifecycle_manager import _TaskLifecycleManager
from modal_proto import api_pb2
import modal
mode=sys.argv[1]
os.environ.pop('NATURAL_MODAL_PHASE',None)
os.environ.pop('NATURAL_MANIFEST',None)
if mode=='local-execute':
    os.environ['NATURAL_MODAL_PHASE']='execute'
elif mode.startswith('remote'):
    # Instantiate the actual SDK execution-context manager as its bootstrap
    # does. Client=None is never used: this check makes no remote/API call.
    _TaskLifecycleManager('offline-task','offline-function',api_pb2.Function(),None,None)
    if mode=='remote-custom-manifest':
        os.environ['NATURAL_MANIFEST']='docs/natural-assistant/custom.json'
sys.path.insert(0,'scripts')
m=importlib.import_module('modal_natural')
expected=['reserve_remote','finish_remote','prepare_remote','external_prepare_remote','gpu_remote','backup_remote','release_remote','execute_stage']
assert all(hasattr(m,name) for name in expected)
assert modal.is_local()==(not mode.startswith('remote'))
assert (m.natural_image is m.control_image)==(mode!='local-execute')
assert m.manifest_path.as_posix()==('docs/natural-assistant/custom.json' if mode=='remote-custom-manifest' else 'docs/natural-assistant/manifest.json')
assert importlib.metadata.version('modal')=='1.6.0'
print(json.dumps({'mode':mode,'modal_version':importlib.metadata.version('modal'),'is_local':modal.is_local(),'phase_env_present':'NATURAL_MODAL_PHASE' in os.environ,'all_entrypoints_present':True,'cuda_recipe_constructed':m.natural_image is not m.control_image,'manifest':m.manifest_path.as_posix(),'remote_calls':0}))
'''
results=[]
for mode in ['local-control','local-execute','remote-no-phase-env','remote-custom-manifest']:
    result=subprocess.run([sys.executable,'-c',program,mode],check=True,capture_output=True,text=True)
    results.append(json.loads(result.stdout))
output={'scope':'Real installed Modal 1.6.0 SDK startup import contract; actual SDK lifecycle context with no client, authentication, deployment, image build or remote call','program_sha256':hashlib.sha256(Path('scripts/modal_natural.py').read_bytes()).hexdigest(),'cases':results}
Path('outputs/natural-extension/research/gpu-route/remote-import-contract.json').write_text(json.dumps(output,indent=2)+'\n')
print(json.dumps(output,indent=2))
