"""Inventory genuine completed factual owners; do not stage or alter reports."""

import argparse
import hashlib
import json
from datetime import UTC, datetime
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
HERE = Path(__file__).resolve().parent
ARTIFACTS = ROOT / "docs/technical-reviews/artifacts/natural-v4-factual"
ALLOWED = {".json", ".md", ".py", ".txt", ".svg", ".png", ".csv", ".tsv", ".log", ".diff", ".sh", ".snapshot", ".bytes"}
RAW_DIRECTORIES = {"workspace", "data", "dataset", "datasets", "model", "models", "cache", "venv", "node_modules", "__pycache__"}


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def record(path):
    return {"path": path.relative_to(ROOT).as_posix(), "bytes": path.stat().st_size, "sha256": sha(path)}


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    if args.output.exists():
        parser.error("Preserve the earlier inventory; choose a new output path")
    journal_path = HERE / "factual-current-dispatch-plan.json"
    journal = json.loads(journal_path.read_bytes())
    dispatches = {row["lesson_id"]: row for row in journal["dispatches"]}
    completed = journal["completions"]
    if len({row["lesson_id"] for row in completed}) != len(completed):
        raise ValueError("Repeated completion requires explicit current-recheck handling")
    selected, excluded, owners = {}, [], []
    for completion in completed:
        lesson = completion["lesson_id"]
        dispatch = dispatches[lesson]
        path = ROOT / completion["report_path"]
        report = json.loads(path.read_bytes())
        assert completion["actual_observed_owner_final"] is True, lesson
        assert completion["actual_official_checker_exit_code"] == 0, lesson
        assert completion["owner_verdict"] == report["verdict"] == "pass", lesson
        assert completion["actual_owner"] == dispatch["actual_owner"] == report["reviewer_task"], lesson
        assert report["reviewer_context"] == "fresh", lesson
        assert sha(path) == completion["report_sha256"], lesson
        selected[path] = record(path)
        owner_directory = ARTIFACTS / lesson
        assert owner_directory.is_dir(), lesson
        for original in sorted(owner_directory.rglob("*")):
            if not original.is_file():
                continue
            parts = original.relative_to(owner_directory).parts[:-1]
            if (
                any(part.lower() in RAW_DIRECTORIES for part in parts)
                or original.suffix.lower() not in ALLOWED
                or original.stat().st_size > 8 * 1024 * 1024
            ):
                excluded.append(record(original))
                continue
            selected[original] = record(original)
        for artifact in report.get("artifacts", []):
            if not artifact.get("path"):
                continue
            original = ROOT / artifact["path"]
            assert original.is_file() and sha(original) == artifact["sha256"], artifact["path"]
            if original.is_relative_to(owner_directory):
                assert original in selected, "A required local artifact needs explicit publication review: " + str(original)
        owners.append({"lesson_id": lesson, "actual_owner": completion["actual_owner"], "canonical_report": record(path)})
    result = {
        "schema_version": 1,
        "record_kind": "actual_completed_factual_small_evidence_inventory_not_review_or_stage",
        "created_at": datetime.now(UTC).isoformat(),
        "scope": "Only observed completed numbered owners with exact current canonical reports. Unfinished owners and supplemental whole-file checks are not included. This inventories provenance; it does not certify truth, create judgments, or edit/stage files. Full external papers/HTML/PDF and raw workspaces remain outside ordinary Git.",
        "journal_sha256_at_capture": sha(journal_path),
        "completed_owners": owners,
        "selected_files": [selected[path] for path in sorted(selected)],
        "excluded_files": excluded,
        "counts": {"completed_owners": len(owners), "selected_files": len(selected), "selected_bytes": sum(row["bytes"] for row in selected.values()), "excluded_files": len(excluded)},
    }
    args.output.parent.mkdir(parents=True, exist_ok=True)
    args.output.write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
    print(json.dumps(result["counts"]))


if __name__ == "__main__":
    main()
