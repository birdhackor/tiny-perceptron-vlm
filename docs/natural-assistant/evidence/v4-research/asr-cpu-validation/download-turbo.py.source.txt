import hashlib,json,pathlib,time
from huggingface_hub import snapshot_download
root=pathlib.Path('outputs/natural-v4/asr-cpu-validation');cache=pathlib.Path('outputs/natural-v4/student-base-cache/hf')
started=time.monotonic();p=pathlib.Path(snapshot_download('openai/whisper-large-v3-turbo',revision='41f01f3fe87f28c78e2fbf8b568835947dd65ed9',cache_dir=cache,token=False,allow_patterns=['*.json','*.safetensors','*.txt','*.model','LICENSE*','README.md'],max_workers=4))
files=[]
for file in sorted(p.rglob('*')):
 if not file.is_file():continue
 digest=hashlib.sha256()
 with file.open('rb') as stream:
  for block in iter(lambda:stream.read(1024*1024),b''):digest.update(block)
 files.append({'path':file.relative_to(p).as_posix(),'bytes':file.stat().st_size,'sha256':digest.hexdigest()})
receipt={'repo':'openai/whisper-large-v3-turbo','revision':'41f01f3fe87f28c78e2fbf8b568835947dd65ed9','snapshot':str(p),'download_and_hash_seconds':time.monotonic()-started,'files':files,'anonymous':True,'scope':'Local CPU preparation, not a Modal/GPU job or inference result'}
(root/'turbo-download-receipt.json').write_text(json.dumps(receipt,indent=2)+'\n');print(json.dumps({'files':len(files),'bytes':sum(f['bytes'] for f in files),'seconds':receipt['download_and_hash_seconds']}))
