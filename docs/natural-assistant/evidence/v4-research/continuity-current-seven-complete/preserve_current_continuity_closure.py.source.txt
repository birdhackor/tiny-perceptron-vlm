"""Preserve completed genuine owners and their bounded original evidence, not a review."""
from pathlib import Path
from datetime import UTC, datetime
import hashlib, json, subprocess

root=Path.cwd(); base=root/'outputs/natural-v4/review-plan'
closure=base/'whole-continuity-current7-and292-and13-closure.json'
data=json.loads(closure.read_bytes())
assert data['whole_continuity_phase_complete'] and data['actual_fresh_owner_groups']==7
assert data['actual_numbered_section_coverage_count']==292
packet=root/'docs/natural-assistant/evidence/v4-research/continuity-current-seven-complete'
assert not packet.exists(); packet.mkdir()
paths=set(); originals=[]
for row in data['actual_owner_reports']:
    report=root/row['report_path']
    assert hashlib.sha256(report.read_bytes()).hexdigest()==row['report_sha256']
    paths.add(row['report_path'])
    for name, expected in row['current_document_sha256'].items():
        assert hashlib.sha256((root/name).read_bytes()).hexdigest()==expected
    for name, expected in row['figure_sha256'].items():
        assert hashlib.sha256((root/name).read_bytes()).hexdigest()==expected
folder=root/'docs/continuity-reviews/artifacts/natural-v4'
allowed={'.txt','.json','.svg','.png'}
for original in folder.rglob('*'):
    if not original.is_file(): continue
    assert original.suffix in allowed and original.stat().st_size<8*1024*1024,original
    assert not any(part.lower() in {'cache','dataset','datasets','models','weights','workspace','__pycache__'} for part in original.relative_to(folder).parts)
    paths.add(str(original.relative_to(root)))
# Derivative screenshot/render attribution supplements the canonical SVG metadata.
notice=folder/'NOTICE.txt'
notice.write_text('Course-owned text, code and diagrams retain the repository license.\n\nCopies and rendered PNGs of natural-v4-training-cat.svg and screenshots containing that figure reproduce DOCCI train_01827. Image creator: Jason Baldridge and family. Licensor: Google LLC. Original: https://storage.googleapis.com/docci/thumbnails/train_01827.jpg?generation=1713028329088595 . License: CC BY 4.0, https://creativecommons.org/licenses/by/4.0/ . Canonical course/figures/natural-v4-training-cat.svg preserves the original image SHA and attribution. These derivatives add course annotations, scaling and browser rendering; they do not change the underlying photograph.\n')
paths.add(str(notice.relative_to(root)))
control=[closure,base/'continuity-current-dispatch-journal.json',Path(__file__)]
for directory in ['continuity-current-completions','continuity-current-assignments']:
    control.extend(p for p in (base/directory).rglob('*') if p.is_file())
for original in control:
    assert original.stat().st_size<8*1024*1024 and original.suffix in {'.json','.py','.txt'}
    destination=packet/original.relative_to(base)
    if original.suffix=='.py': destination=destination.with_name(destination.name+'.source.txt')
    destination.parent.mkdir(parents=True,exist_ok=True); destination.write_bytes(original.read_bytes())
    originals.append({'original':str(original.relative_to(root)),'preserved':str(destination.relative_to(root)),'sha256':hashlib.sha256(original.read_bytes()).hexdigest(),'bytes':original.stat().st_size})
(packet/'README.md').write_text('# Completed current continuity review\n\nSeven genuine fresh owners read their complete assigned documents, necessary preceding material, actual CPU outputs and figures. Their combined scope covers all 292 numbered sections and 38 full files, including the thirteen supplemental documents. Original judgments, issue histories, final reports and provenance receipts are preserved unchanged. This packet verifies preservation and current source binding; it does not substitute for the owners’ reading or approve reading-time estimates or deployment.\n')
(packet/'preserved-originals.json').write_text(json.dumps(originals,indent=2)+'\n')
paths.update(str(p.relative_to(root)) for p in packet.rglob('*') if p.is_file())
paths=sorted(paths)
run=subprocess.run(['git','add','--pathspec-from-file=-','--pathspec-file-nul'],input=('\0'.join(paths)+'\0').encode(),capture_output=True)
assert run.returncode==0,run.stderr.decode()
proc=subprocess.Popen(['git','cat-file','--batch'],stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
errors=[];total=0
for name in paths:
    expected=(root/name).read_bytes(); total+=len(expected)
    proc.stdin.write((':'+name+'\n').encode());proc.stdin.flush();header=proc.stdout.readline().decode().split()
    if header[-1]=='missing':errors.append(name);continue
    content=proc.stdout.read(int(header[-1]));assert proc.stdout.read(1)==b'\n'
    if content!=expected:errors.append(name)
proc.stdin.close();proc.wait();assert proc.returncode==0
result={'kind':'actual_continuity_evidence_Git_INDEX_byte_check_not_review','captured_at':datetime.now(UTC).isoformat(),'actual_git_add_exit_code':run.returncode,'paths_checked':len(paths),'bytes_checked':total,'errors':errors,'paths':paths}
p=packet/'staged-index-verification.json';p.write_text(json.dumps(result,indent=2)+'\n')
subprocess.run(['git','add','--',str(p.relative_to(root))],check=True)
print(json.dumps({k:v for k,v in result.items() if k!='paths'}));assert not errors
