"""Close an already completed selected-public trial without loading models."""

import hashlib
import importlib.util
import json
import shutil
import subprocess
import sys
from datetime import UTC, datetime
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
PREFLIGHT = ROOT / "outputs/natural-v4/release-preflight"
FETCH = PREFLIGHT / "anonymous-selected-fetch-001"
UI = PREFLIGHT / "selected-public-ui-run-001"
RELEASE = PREFLIGHT / "anonymous-selected-release"
EVIDENCE = ROOT / "docs/natural-assistant/evidence/v4-research/student-selected-public-ui"
EXPECTED_MANIFEST = "1758ceb9f8859108fa3ad315fb8f6a61e0c8cef40ec49b0bc41361d9c70e5ccf"


def digest(path):
    content = path.read_bytes()
    return {"bytes": len(content), "sha256": hashlib.sha256(content).hexdigest()}


def read(path):
    return json.loads(path.read_text())


def write_json(path, value):
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n")


def copy(source, relative):
    destination = EVIDENCE / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(source, destination)
    assert digest(source) == digest(destination)
    return {"source": str(source.relative_to(ROOT)), "checkout_copy": relative, **digest(source)}

    product.verify_release(manifest, RELEASE)
    product.check_runtime(manifest)
    for source in ui["sources_before"]:
        assert digest(ROOT / source["path"]) == {k: source[k] for k in ("bytes", "sha256")}
    for record in ui["inputs"].values():
        assert digest(ROOT / record["path"]) == {k: record[k] for k in ("bytes", "sha256")}

    http = read(FETCH / "fetch.http.json")["requests"]
    assert len(http) == 6
    assert all(not request["authentication_headers"] for request in http)
    assert len([r for r in http if r["method"] == "GET" and r["status"] == 200]) == 2
    assert all(not read(FETCH / f"{phase}.http.json")["requests"] for phase in ("list", "verify"))
    core = [e for e in events if e["kind"] == "load_core_begin"]
    asr = [e for e in events if e["kind"] == "load_asr_begin"]

    transcripts = [e["result"] for e in events if e["kind"] == "transcribe_end"]
    begins = [e for e in events if e["kind"] == "generate_begin"]
    assert len(core) == len(asr) == 1 and len(generated) == 4 and len(transcripts) == 1

    assert options["adapter"] == "None" and options["device"] == "cpu"
    assert options["dtype"] == "float32" and options["local_files_only"] == "True"
    assert options["model_revision"] == manifest["base_model"]["revision"]
    assert options["asr_revision"] == manifest["asr_model"]["revision"]
    reset = [e["state"] for e in events if e["kind"] == "operation_end" and e["route"] == "/api/reset"]
    assert reset == [{"history_messages": 0, "assets": 0, "transcriptions": 0, "asset_files": []}]
    resources = ui["server_cpu_resources"]
    assert not Path(f"/proc/{resources['pid']}").exists(), "Actual model server must have exited"
    uploads = {asset["path"] for e in events if e["kind"] == "operation_end" and e["state"]
               for asset in e["state"]["asset_files"]}

                source_records.append({"raw_path": str(path.relative_to(ROOT)), **digest(path),
                                       "availability": "local outputs; requires separately published proof archive outside checkout"})
    write_json(EVIDENCE / "source-index.json", {
        "schema_version": 1,

    })

                    "core_load_seconds": next(e["seconds"] for e in events if e["kind"] == "load_core_end"),
                    "asr_load_seconds": next(e["seconds"] for e in events if e["kind"] == "load_asr_end"),

            "audio": {"dataset": "AISHELL/AISHELL-1", "revision": "bbe295d530192a4cd41644b711c9aecd087df653",

                      "indexed_raw_files": len(source_records), "checkout_copies": len(copies),
