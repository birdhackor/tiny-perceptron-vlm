"""Capture original bounded corrective command output before evaluating its exit."""
import argparse,hashlib,json,subprocess
from datetime import datetime,UTC
from pathlib import Path
ROOT=Path(__file__).resolve().parents[3]
def main():
 p=argparse.ArgumentParser();p.add_argument("label");p.add_argument("command",nargs=argparse.REMAINDER);a=p.parse_args()
 assert a.command
 out=ROOT/"outputs/natural-v4/review-plan/answer-mask-corrective-review-001/native";out.mkdir(parents=True,exist_ok=True)
 stem=out/a.label
 assert not stem.with_suffix(".receipt.json").exists(),"Original native receipts must remain immutable."
 r=subprocess.run(a.command,cwd=ROOT,capture_output=True)
 stem.with_suffix(".stdout.txt").write_bytes(r.stdout);stem.with_suffix(".stderr.txt").write_bytes(r.stderr)
 receipt={"captured_at":datetime.now(UTC).isoformat(),"actual_command":a.command,"actual_cwd":str(ROOT),"actual_exit_code":r.returncode,"actual_stdout":stem.with_suffix(".stdout.txt").relative_to(ROOT).as_posix(),"actual_stderr":stem.with_suffix(".stderr.txt").relative_to(ROOT).as_posix(),"stdout_sha256":hashlib.sha256(r.stdout).hexdigest(),"stderr_sha256":hashlib.sha256(r.stderr).hexdigest(),"executed_wrapper_sha256":hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}
 stem.with_suffix(".receipt.json").write_text(json.dumps(receipt,ensure_ascii=False,indent=2)+"\n")
 print(json.dumps(receipt,ensure_ascii=False));print(r.stdout.decode(errors="replace"));print(r.stderr.decode(errors="replace"))
 raise SystemExit(r.returncode)
if __name__=="__main__":main()
