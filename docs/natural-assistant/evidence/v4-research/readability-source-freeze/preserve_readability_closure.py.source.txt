"""Copy only completed readability coordination evidence, not active factual work."""
import hashlib
import json
from datetime import UTC,datetime
from pathlib import Path

root=Path.cwd()
source=root/'outputs/natural-v4/review-plan'
dest=root/'docs/natural-assistant/evidence/v4-research/readability-source-freeze'
assert not dest.exists()
closure=json.loads((source/'whole-readability-current-closure.json').read_bytes())
assert closure['whole_readability_phase_complete'] is True
fixed_files=[
 'whole-readability-current-closure.json','numbered-reader-checker-at-whole-readability-closure.txt',
 'root-reader-identity-and-intro-check.json','root-accepted-nonblocking-notes.json',
 'reader-contract.md','readability-queue.json','root-current-section-inventory.json',
 'current-version-impact.json','factual-routing-scope-expansion-68-necessary-figures.json',
 'factual-reader-contract.md','root-small-evidence-policy-notes.json',
 'root-attribution-supplement.json','root-existing-technical-reference-index-inspection.json',
 'reader-selective-stage-001.json','reader-selective-index-check-001.json',
 'build-small-evidence-manifest.py','stage_selected_reader_evidence.py',
 'preserve_readability_closure.py',
]
paths=[source/name for name in fixed_files if (source/name).is_file()]
for name in ['stable-prefix','cleanup','chapter20-stable','stable20-tail','final-publication','supplemental-final-publication']:
 base=source/name
 if base.exists():
  paths.extend(p for p in base.rglob('*') if p.is_file() and p.suffix in {'.json','.md','.txt','.py'})
paths=sorted(set(paths))
rows=[]
for path in paths:
 assert path.stat().st_size<8*1024*1024,path
 relative=path.relative_to(source)
 target=dest/relative
 if target.suffix=='.py':
  target=target.with_name(target.name+'.source.txt')
 target.parent.mkdir(parents=True,exist_ok=True)
 content=path.read_bytes()
 target.write_bytes(content)
 assert target.read_bytes()==content
 rows.append({'original_path':path.relative_to(root).as_posix(),'preserved_path':target.relative_to(root).as_posix(),'bytes':len(content),'sha256':hashlib.sha256(content).hexdigest()})
index={'record_kind':'byte_exact_completed_readability_coordination_preservation_not_new_review','created_at':datetime.now(UTC).isoformat(),'source_version':closure['source_version'],'files':rows,'scope':'Actual independent dispatches, genuine returned verdicts, complete/current section and whole-guide source routing, preserved original failure/revision histories, and native official-check outputs. The closure explicitly distinguishes 57 fresh numbered readers, 235 genuine unchanged baseline reports, and 6 whole-file owners covering 8 files. No factual/continuity/reading-time completion is claimed. Active factual work is excluded.'}
(dest/'preservation-index.json').write_text(json.dumps(index,ensure_ascii=False,indent=2)+'\n')
(dest/'README.md').write_text('# Completed readability at the source freeze\n\nThese files preserve actual coordinator dispatches, returned owner results and raw native checker output without changing their bytes. The closure records 57 genuinely new independent numbered readers and 235 unchanged current baseline reports; it does not claim 292 fresh new reads. Six additional independent owners read eight full canonical guides/entry files and actually revisited the corrected executed web preview. Original genuine revisions and accepted nonblocking notes remain explicit. The factual phase starts afterward and is not approved by this archive. `preservation-index.json` maps original runtime paths to the byte-exact copies. Python helper copies use inert `.source.txt` names.\n')
print(json.dumps({'files':len(rows),'bytes':sum(r['bytes'] for r in rows),'directory':str(dest.relative_to(root))}))
