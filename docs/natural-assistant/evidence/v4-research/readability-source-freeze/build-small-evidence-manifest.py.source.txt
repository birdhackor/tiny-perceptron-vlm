"""Index small actual reader evidence; never stage, delete or edit evidence."""

import hashlib
import json
import re
import subprocess
from collections import Counter, deque
from datetime import UTC, datetime
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
BASE = Path(__file__).resolve().parent
ARTIFACT_ROOTS = sorted((ROOT / "docs/reader-reviews/artifacts").glob("natural-v4-*"))
ALLOWED = {".json", ".md", ".py", ".txt", ".svg", ".png", ".csv", ".tsv", ".log", ".diff", ".sh", ".snapshot", ".bytes"}
RAW_WORDS = {"workspace", "data", "dataset", "datasets", "model", "models", "cache", "venv", "node_modules", "__pycache__"}
MAX_SMALL_BYTES = 8 * 1024 * 1024


def relative(path):
    return path.relative_to(ROOT).as_posix()


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def exclusion(path):
    parts = path.relative_to(ROOT).parts
    # Exclude actual raw/workspace directories. A drawing named cache.png is
    # a teaching illustration, not a directory containing a model cache.
    # docs/reader-reviews/artifacts/<round>/<assigned lesson or guide>/...
    # The supplemental guide named "data" is an owner scope, not raw media.
    directories = {part.lower() for part in parts[5:-1]}
    if directories & RAW_WORDS:
        return "execution_workspace_or_data_model_cache"
    if path.suffix.lower() not in ALLOWED:
        return "raw_data_archive_binary_or_full_external_document_extension"
    if path.stat().st_size > MAX_SMALL_BYTES:
        return "over_small_evidence_size_limit"
    if re.search(r"(?i)(full[-_ ]?(paper|document|html)|downloaded[-_ ]?(paper|html|pdf)|official[-_ ]?full)", path.name):
        return "possible_full_third_party_document_requires_manual_review"
    return None


def strings(value):
    if isinstance(value, str):
        yield value
    elif isinstance(value, list):
        for item in value:
            yield from strings(item)
    elif isinstance(value, dict):
        for key, item in value.items():
            yield from strings(key)
            yield from strings(item)


def artifact_path(text):
    if text.startswith(str(ROOT) + "/"):
        candidate = Path(text)
    elif text.startswith("docs/reader-reviews/artifacts/"):
        candidate = ROOT / text
    else:
        return None
    candidate = candidate.resolve()
    if not any(candidate == base or candidate.is_relative_to(base) for base in ARTIFACT_ROOTS):
        return None
    return candidate if candidate.exists() else None


files = sorted(path for base in ARTIFACT_ROOTS for path in base.rglob("*") if path.is_file())
excluded, eligible = [], {}
for path in files:
    reason = exclusion(path)
    if reason:
        excluded.append({"path": relative(path), "bytes": path.stat().st_size, "reason": reason})
    else:
        eligible[path] = {"path": relative(path), "bytes": path.stat().st_size, "sha256": digest(path)}

reports = []
for journal_path in BASE.glob("*/dispatch-journal.json"):
    journal = json.loads(journal_path.read_bytes())
    for dispatch in journal["dispatches"]:
        report_path = dispatch.get("report_path")
        if report_path is None:
            lesson_id = dispatch.get("lesson_id")
            if lesson_id is None:
                continue
            report_path = f"docs/reader-reviews/{lesson_id}.json"
        paths = {report_path, dispatch.get("current_observed_actual_owner_report_path", report_path)}
        for report_path in paths:
            path = ROOT / report_path
            if not path.exists():
                continue
            value = json.loads(path.read_bytes())
            owner = dispatch.get("reviewer_task") or dispatch.get("actual_owner")
            declared_owner = value.get("reviewer_task") or value.get("canonical_reviewer_task") or value.get("reviewer_identity")
            if declared_owner == owner:
                reports.append(path)

queue = deque(reports)
seen_json, referenced = set(), set()
while queue:
    path = queue.popleft()
    if path in seen_json:
        continue
    seen_json.add(path)
    if "prior-report-unread" in path.name:
        # Keep its bytes, but do not use the older review to expand this scope.
        continue
    try:
        value = json.loads(path.read_bytes())
    except (OSError, ValueError):
        continue
    for text in strings(value):
        target = artifact_path(text)
        if target is None:
            continue
        candidates = [target] if target.is_file() else [file for file in target.rglob("*") if file.is_file()]
        for file in candidates:
            if file not in eligible:
                continue
            referenced.add(file)
            if file.suffix == ".json":
                queue.append(file)

workspace_dirs = sorted({relative(path) for base in ARTIFACT_ROOTS for path in base.rglob("workspace") if path.is_dir()})
ignored = {}
for path in workspace_dirs:
    result = subprocess.run(["git", "check-ignore", "-q", "--", path], cwd=ROOT, check=False)
    ignored[path] = result.returncode == 0

selected = [eligible[path] for path in sorted(referenced)]
manifest = {
    "schema_version": 1,
    "record_kind": "small_actual_report_referenced_reader_evidence_stage_candidates_not_review_or_stage",
    "captured_at": datetime.now(UTC).isoformat(),
    "scope_note": "Snapshot while reviews may still be in progress. No files staged/deleted/moved and no report/verdict/hash rewritten. Inclusion is provenance inventory, not approval or PASS. Refresh before final selective staging and verify candidate bytes. No full third-party papers/HTML/PDF, datasets, model/cache/archive/audio workspaces are included.",
    "max_small_file_bytes": MAX_SMALL_BYTES,
    "actual_owned_current_reports": [{"path": relative(path), "bytes": path.stat().st_size, "sha256": digest(path)} for path in sorted(set(reports))],
    "report_referenced_small_paths": selected,
    "other_small_preservation_candidates_require_manual_selection": [eligible[path] for path in sorted(set(eligible) - referenced)],
    "excluded_unstaged_raw_paths": excluded,
    "raw_workspace_scope": [{"path": path, "git_ignored_at_capture": ignored[path], "availability": "Actual execution workspace retained at this original local path; external source/provenance belongs to the genuine owner receipts. It is not included for Git staging. If not already ignored, root can add an appropriate ignore/LFS/retrieval policy without changing original evidence."} for path in workspace_dirs],
    "counts": {"all_artifact_files": len(files), "all_artifact_bytes": sum(path.stat().st_size for path in files), "report_referenced_small_files": len(selected), "report_referenced_small_bytes": sum(item["bytes"] for item in selected), "other_small_files": len(eligible) - len(referenced), "excluded_files": len(excluded), "excluded_bytes": sum(item["bytes"] for item in excluded)},
    "excluded_reason_counts": dict(Counter(item["reason"] for item in excluded)),
}
path = BASE / "review-report-referenced-small-paths.json"
path.write_text(json.dumps(manifest, ensure_ascii=False, indent=2) + "\n")
print(json.dumps({"manifest": relative(path), **manifest["counts"]}))
