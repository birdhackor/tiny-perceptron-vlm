from pathlib import Path
from datetime import datetime, UTC
import subprocess,json,sys,time
base=Path(sys.argv[1]);base.mkdir(exist_ok=False)
commands=[['.venv/bin/python','scripts/build_course.py','--check'],['.venv/bin/python','scripts/check_course_reviews.py'],['.venv/bin/python','scripts/check_technical_reviews.py'],['.venv/bin/python','docs/review-tools/check_review_round.py'],['.venv/bin/ruff','check','.'],['.venv/bin/ruff','format','--check','.']]
rows=[]
for i,cmd in enumerate(commands):
 start=datetime.now(UTC).isoformat();t=time.monotonic()
 with (base/f'{i}.stdout.txt').open('wb') as out,(base/f'{i}.stderr.txt').open('wb') as err:
  r=subprocess.run(cmd,stdout=out,stderr=err)
 row={'argv':cmd,'started_at':start,'exit_code':r.returncode,'elapsed_seconds':time.monotonic()-t};rows.append(row);print(json.dumps(row),flush=True)
 (base/'receipt.json').write_text(json.dumps({'kind':'actual final native source gates, not independent content review','head':subprocess.check_output(['git','rev-parse','HEAD'],text=True).strip(),'commands':rows},indent=2)+'\n')
 if r.returncode:raise SystemExit(r.returncode)
