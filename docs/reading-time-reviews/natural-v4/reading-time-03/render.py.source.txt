from pathlib import Path
import hashlib
import json
import subprocess

root = Path('/workspace/tiny-perceptron-vlm')
own = root / 'docs/reading-time-reviews/natural-v4/reading-time-03'
assignment = json.loads((root / 'outputs/natural-v4/review-plan/reading-time-current-assignments/reading-time-03.json').read_text())
figures = {}
for page in assignment['pages']:
    figures.update(page['figures_sha256'])
(own / 'figures').mkdir(exist_ok=True)
receipts = []
for source, expected in figures.items():
    path = root / source
    actual = hashlib.sha256(path.read_bytes()).hexdigest()
    assert actual == expected, (source, actual)
    dest = own / 'figures' / (path.stem + '.png')
    command = ['inkscape', str(path), '--export-type=png', '--export-width=1400', '--export-filename=' + str(dest)]
    result = subprocess.run(command, capture_output=True, text=True)
    print(json.dumps({'command':command, 'exit_status':result.returncode, 'stdout':result.stdout, 'stderr':result.stderr}, ensure_ascii=False))
    result.check_returncode()
    receipts.append({'source':source, 'source_sha256':actual, 'render':str(dest.relative_to(root)), 'render_sha256':hashlib.sha256(dest.read_bytes()).hexdigest()})
(own / 'figure-render-receipts.json').write_text(json.dumps(receipts,ensure_ascii=False,indent=2)+'\n')
