import hashlib,json,pathlib,subprocess,xml.etree.ElementTree as E
ROOT=pathlib.Path('/workspace/tiny-perceptron-vlm')
OUT=ROOT/'docs/reading-time-reviews/natural-v4/reading-time-06'
a=json.loads((ROOT/'outputs/natural-v4/review-plan/reading-time-current-assignments/reading-time-06.json').read_text())
figures={p:h for row in a['pages'] for p,h in row['figures_sha256'].items()}
(OUT/'figures').mkdir(exist_ok=True)
receipts=[]
notice=['Rendered PNG derivatives for personal reading, made with Inkscape. Original SVGs remain unchanged.']
for path,expected in figures.items():
 source=ROOT/path
 actual=hashlib.sha256(source.read_bytes()).hexdigest()
 assert actual==expected,(path,actual,expected)
 dest=OUT/'figures'/(source.stem+'.png')
 args=['inkscape',str(source),'--export-type=png','--export-width=1200','--export-filename='+str(dest)]
 result=subprocess.run(args,capture_output=True,text=True)
 print(json.dumps({'command':args,'exit_code':result.returncode,'stdout':result.stdout,'stderr':result.stderr},ensure_ascii=False),flush=True)
 if result.returncode: raise RuntimeError(path)
 metadata=[]
 root=E.parse(source).getroot()
 for elem in root:
  if elem.tag.rsplit('}',1)[-1] in ('metadata','title','desc'): metadata.append(E.tostring(elem,encoding='unicode'))
 receipts.append({'source':path,'source_sha256':actual,'png':str(dest.relative_to(ROOT)),'png_sha256':hashlib.sha256(dest.read_bytes()).hexdigest(),'pages':[x['page_id'] for x in a['pages'] if path in x['figures_sha256']],'source_metadata':metadata})
 if 'docci' in source.read_text().lower():
  notice.append('\nSource SVG: '+path+'\nSource SHA256: '+actual+'\nDerivative: '+str(dest.relative_to(ROOT))+'\n'+ '\n'.join(metadata))
(OUT/'figure-render-receipts.json').write_text(json.dumps(receipts,ensure_ascii=False,indent=2)+'\n')
(OUT/'NOTICE.txt').write_text('\n'.join(notice)+'\n')
