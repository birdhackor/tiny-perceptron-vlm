root = Path('/workspace/tiny-perceptron-vlm')
base = root / 'docs/reading-time-reviews/natural-v4/reading-time-04'
receipts = base / 'receipts'
task = '/root/v4_review_coordinator/readingtime_v4_group_04'

judgments = [json.loads(x) for x in (receipts/'manual-judgments.jsonl').read_text().splitlines()]
assert len(judgments) == 51
assert [j['page_id'] for j in judgments] == a['canonical_page_ids']
browser = json.loads((receipts/'browser-extraction.json').read_text())['records']
assert [p['page_id'] for p in browser] == a['canonical_page_ids']
assert all(p['http_status'] == 200 for p in browser)
assert all(i['complete'] and i['width'] > 0 for p in browser for i in p['images'])
assert sum(len(p['cpu_output_blocks']) for p in browser) == 48

assert [p['page_id'] for p in current['pages']] == a['canonical_page_ids']
figures = json.loads((receipts/'figure-rendering.json').read_text())
assert len(figures) == 14

pages = []
scopes = []
for assignment, judgment, render, binding in zip(a['pages'], judgments, browser, current['pages']):
    for field in ['minutes_min','minutes_max']:
        assert type(judgment[field]) is int and judgment[field] > 0
    assert judgment['minutes_max'] >= judgment['minutes_min']
    assert judgment['reason'] and judgment['scope']
    pages.append({k:judgment[k] for k in ['page_id','minutes_min','minutes_max','reason']} | {'reviewer_task':task,'source_sha256':binding['source_sha256'],'figures_sha256':binding['figures_sha256']})
    scopes.append({'page_id':judgment['page_id'],'source':assignment['source'],'selector':assignment['selector'],'snapshot':assignment['snapshot'],'source_sha256':binding['source_sha256'],'figures_sha256':binding['figures_sha256'],'source_reading':'complete','rendered_article_reading':'complete','scope':judgment['scope'],'render_url':render['url'],'render_receipt':render['render_receipt'],'render_sha256':render['render_sha256'],'cpu_output_blocks_personally_read':len(render['cpu_output_blocks']),'figures_personally_viewed':list(binding['figures_sha256'])})
metadata = {'schema_version':1,'estimate_kind':'ai_estimate','pages':pages}
metadata_path = base/'metadata.json'
metadata_path.write_text(json.dumps(metadata, ensure_ascii=False, indent=2)+'\n')

artifacts.append({'path':str(metadata_path.relative_to(root)),'sha256':hashlib.sha256(metadata_path.read_bytes()).hexdigest()})
