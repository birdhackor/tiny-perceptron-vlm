import json, hashlib, sys
from pathlib import Path
root = Path('/workspace/tiny-perceptron-vlm')
sys.path.insert(0, str(root))
from scripts.reading_time import chapter_introduction, lesson_slices, source_sha256, figure_hashes
out = root / 'docs/reading-time-reviews/natural-v4/reading-time-04/receipts'
assignment_path = root / 'outputs/natural-v4/review-plan/reading-time-current-assignments/reading-time-04.json'
a = json.loads(assignment_path.read_text())
assert hashlib.sha256((root/a['inventory']).read_bytes()).hexdigest() == a['inventory_sha256']
current = []
for p in a['pages']:
    source_path = root / p['source']
    text = source_path.read_text(encoding='utf-8')
    body = chapter_introduction(text) if p['kind'] == 'chapter_introduction' else lesson_slices(text)[p['page_id']]
    current_hash = source_sha256(body)
    figures = figure_hashes(root, source_path, body)
    assert current_hash == p['source_sha256'], p['page_id']
    assert body.encode('utf-8') == (root/p['snapshot']).read_bytes(), p['page_id']
    assert figures == p['figures_sha256'], p['page_id']
    current.append({'page_id':p['page_id'],'source':p['source'],'selector':p['selector'],'source_sha256':current_hash,'figures_sha256':figures,'current_slice_matches_snapshot':True})
    print(p['page_id'], 'current source and snapshot and figures match', flush=True)
(out/'current-bindings.json').write_text(json.dumps({'assignment_sha256':hashlib.sha256(assignment_path.read_bytes()).hexdigest(),'source_version':a['source_version'],'inventory':a['inventory'],'inventory_sha256':a['inventory_sha256'],'pages':current}, ensure_ascii=False, indent=2)+'\n')
