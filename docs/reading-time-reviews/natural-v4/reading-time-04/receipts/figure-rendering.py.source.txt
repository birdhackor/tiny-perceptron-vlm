import json, hashlib, subprocess
from pathlib import Path
root = Path('/workspace/tiny-perceptron-vlm')
out = root / 'docs/reading-time-reviews/natural-v4/reading-time-04/receipts'
a = json.loads((root / 'outputs/natural-v4/review-plan/reading-time-current-assignments/reading-time-04.json').read_text())
figures = {}
for p in a['pages']:
    figures.update(p['figures_sha256'])
records = []
for name, expected in figures.items():
    src = root / name
    actual = hashlib.sha256(src.read_bytes()).hexdigest()
    assert actual == expected, name
    target = out / (src.stem + '.png')
    command = ['inkscape', str(src), '--export-type=png', '--export-width=1200', '--export-filename='+str(target)]
    result = subprocess.run(command, capture_output=True, text=True)
    print(json.dumps({'command':command,'returncode':result.returncode,'stdout':result.stdout,'stderr':result.stderr}), flush=True)
    assert result.returncode == 0, name
    records.append({'source':name,'source_sha256':actual,'render':str(target.relative_to(root)),'render_sha256':hashlib.sha256(target.read_bytes()).hexdigest()})
(out / 'figure-rendering.json').write_text(json.dumps(records, indent=2)+'\n')
