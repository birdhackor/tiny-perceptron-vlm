import json, hashlib, sys
from pathlib import Path
root = Path('/workspace/tiny-perceptron-vlm')
sys.path.insert(0,str(root))
from scripts.reading_time import load_estimates
base = root/'docs/reading-time-reviews/natural-v4/reading-time-04'
a = json.loads((root/'outputs/natural-v4/review-plan/reading-time-current-assignments/reading-time-04.json').read_text())
loaded = load_estimates(base/'metadata.json', a, require_complete=True)
assert list(loaded) == a['canonical_page_ids']
r = json.loads((base/'report.json').read_text())
assert r['personally_read_page_ids'] == a['canonical_page_ids']
assert len(r['per_page_reading_scope']) == 51
for artifact in r['artifacts']:
    assert hashlib.sha256((root/artifact['path']).read_bytes()).hexdigest() == artifact['sha256'], artifact['path']
assert set(r['personally_viewed_figure_sha256']) == {name for p in a['pages'] for name in p['figures_sha256']}
assert all(s['source_reading']=='complete' and s['rendered_article_reading']=='complete' for s in r['per_page_reading_scope'])
print('Official load_estimates accepted all 51 assigned records with require_complete=True.')
print('Report scope covers all 51 assigned pages; artifacts match; all 14 unique referenced figures are present.')
print('This validates schema and bytes only; personal reading and manual judgment are documented independently.')
