import json, platform, sys
import torch
print(json.dumps({"python":sys.version,"torch":torch.__version__,"torch_commit":torch.version.git_version,"device":"cpu","platform":platform.platform(),"threads":torch.get_num_threads()},ensure_ascii=False))
steps = [(2, 2, 4), (5, -1, 4)]
truth = 2 + 2
valid = [a + b == claimed for a, b, claimed in steps]
linked = [steps[i][0] == steps[i - 1][2] for i in range(1, len(steps))]
print("等式算對", valid)
print("接上前步", linked)
print("最後答案對", steps[-1][2] == truth)
assert valid == [True, True]
assert linked == [False]
assert steps[-1][2] == truth
print("沿用舊起點", steps[1][0], "新上一步claimed", steps[0][2])
