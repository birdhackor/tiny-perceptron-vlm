assert torch.get_num_threads() == 1

labels = torch.tensor([[1, 2, 3, 4], [1, -100, -100, -100]])
result = {"environment": {"python": sys.version, "torch": torch.__version__, "torch_git_version": torch.version.git_version, "device": "cpu", "threads": str(torch.get_num_threads()), "OMP_NUM_THREADS": "1", "MKL_NUM_THREADS": "1"}, "cases": {}}
# 原例的實際 logits、標籤及 loss 分母；不執行參數更新。
def case(name, targets):
    logits = torch.zeros(2, 4, 5, requires_grad=True)
    total, count = loss_sum(logits, targets)
    mean = total / count
    before = logits.detach().clone()
    mean.backward()
    valid = targets != -100
    ignored_grad = logits.grad[~valid]
    expected_grad = logits.detach().softmax(-1)
    expected_grad[valid] -= F.one_hot(targets[valid], 5).float()
    expected_grad[~valid] = 0
    expected_grad /= int(count)
    assert torch.equal(logits.detach(), before)
    assert torch.allclose(logits.grad, expected_grad, atol=2e-8, rtol=0)
    assert torch.equal(ignored_grad, torch.zeros_like(ignored_grad))
    assert abs(total.item() - int(count) * math.log(5)) < 2e-6
    assert abs(mean.item() - math.log(5)) < 1e-6
    result["cases"][name] = {"count": count.item(), "total": total.item(), "mean": mean.item(), "shapes": {"logits": list(logits.shape), "count": list(count.shape), "total": list(total.shape), "mean": list(mean.shape)}, "numel": {"count": count.numel(), "total": total.numel(), "mean": mean.numel()}, "item_types": {"count": type(count.item()).__name__, "total": type(total.item()).__name__, "mean": type(mean.item()).__name__}, "requires_grad": {"logits": logits.requires_grad, "total": total.requires_grad, "mean": mean.requires_grad, "count": count.requires_grad}, "gradient_at_first_valid_position": logits.grad[0, 0].tolist(), "ignored_gradient_max": ignored_grad.abs().max().item(), "unchanged_logits_after_backward": torch.equal(logits.detach(), before)}
case("original_5", labels)
changed = labels.clone(); changed[1, 1] = 2
case("exercise_6", changed)
changed[1, 1] = -100
case("exercise_restored_5", changed)
# 不同分數核對平均梯度恆等式及兩種不同加權目標。
z = (torch.arange(40, dtype=torch.float64).reshape(2, 4, 5) / 11).requires_grad_()
valid = labels != -100
per_token = F.cross_entropy(z.reshape(-1, 5), labels.reshape(-1), ignore_index=-100, reduction="none").reshape(2, 4)
mean_tokens = per_token[valid].mean()
g_all = torch.autograd.grad(mean_tokens, z, retain_graph=True)[0]
g_each = [torch.autograd.grad(per_token[row,col], z, retain_graph=True)[0] for row,col in valid.nonzero().tolist()]
g_separate = torch.stack(g_each).mean(0)
row_means = [per_token[row][valid[row]].mean() for row in range(2)]
mean_rows = torch.stack(row_means).mean()
g_rows = torch.autograd.grad(mean_rows, z, retain_graph=True)[0]
row_fractions = (g_rows[0].abs().sum()/g_all[0].abs().sum(), g_rows[1].abs().sum()/g_all[1].abs().sum())
assert torch.allclose(g_all, g_separate, atol=1e-15, rtol=0)
assert abs(row_fractions[0].item() - 0.625) < 1e-12
assert abs(row_fractions[1].item() - 2.5) < 1e-12
s0,n0 = loss_sum(z[:1], labels[:1]); s1,n1 = loss_sum(z[1:], labels[1:])
g_accum = torch.autograd.grad((s0+s1)/(n0+n1), z)[0]
assert torch.allclose(g_accum,g_all,atol=1e-15,rtol=0)
result["unequal_objectives"] = {"mean_per_token": mean_tokens.item(), "mean_per_example": mean_rows.item(), "weights_per_token": [0.2]*5, "weights_per_example_then_mean": [0.125]*4+[0.5], "gradient_linearity_max_difference": (g_all-g_separate).abs().max().item(), "microbatch_global_denominator_max_difference": (g_all-g_accum).abs().max().item(), "global_count": (n0+n1).item()}
