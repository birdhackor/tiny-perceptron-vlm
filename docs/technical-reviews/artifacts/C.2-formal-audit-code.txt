def independently_check(candidate, row):
    ids = candidate["generated_ids"]
    eos = bool(ids and ids[-1] == 2)
    content = ids[:-1] if eos else ids
    illegal = [token for token in content if token < 8]
    decoded = bytes(token - 8 for token in content if token >= 8).decode("utf-8", errors="replace")
    assert decoded == candidate["generated"]
    assert eos == candidate["eos"]
    assert illegal == candidate["invalid_special_tokens"]
    assert len(ids) == candidate["generated_tokens"]
    text = decoded.strip()
    matched = re.fullmatch(
        r"(-?[0-9]+)\+(-?[0-9]+)=(-?[0-9]+);(-?[0-9]+)\+(-?[0-9]+)=(-?[0-9]+);answer=(-?[0-9]+)",
        text,
    )
    if matched and not illegal:
        a, b, subtotal, previous, c, total, final = map(int, matched.groups())
        first_equation = a + b == subtotal
        second_equation = previous + c == total
        equations = first_equation and second_equation
        linked = previous == subtotal and total == final
        operands = (a, b, c) == (row["a"], row["b"], row["c"])
        parsed = True
        final_answer = final
    else:
        tail = re.search(r";answer=(-?[0-9]+)$", text) if not illegal else None
        final_answer = int(tail[1]) if tail else None
        parsed = equations = linked = operands = False
        first_equation = second_equation = None
    final_correct = final_answer == row["truth"]
    flags = {
        "final_answer": final_answer,
        "parsed": parsed,
        "equations_valid": equations,
        "linked": linked,
        "task_operands_valid": operands,
        "final_correct": final_correct,
        "fully_verified": equations and linked and operands and final_correct,
    }
    for key, value in flags.items():
        assert candidate[key] == value, (row["question"], decoded, key, candidate[key], value)
    replay = _verify_reasoning(candidate, row, "steps")
    for key, value in flags.items():
        assert replay[key] == value
    return flags, eos, illegal, [first_equation, second_equation]


budgets = report["results"]["comparison"]["steps"]["budgets"]
assert [budget["candidate_count"] for budget in budgets] == [1, 2, 4, 8]
summary = []
audit_rows = []
questions = None
cited_bad_samples = []
malformed_samples = []
for budget in budgets:
    count = budget["candidate_count"]
    samples = budget["samples"]
    assert len(samples) == 24
    current_questions = [sample["question"] for sample in samples]
    assert len(set(current_questions)) == 24
    if questions is None:
        questions = current_questions
    assert current_questions == questions
    totals = dict(candidates=0, parsed=0, equations_valid=0, linked=0, task_operands_valid=0,
                  final_correct=0, fully_verified=0, final_correct_but_steps_invalid=0,
                  eos=0, illegal_control_candidates=0)
    for sample_index, row in enumerate(samples):
        assert row["truth"] == row["a"] + row["b"] + row["c"]
        assert len(row["candidates"]) == count
        assert row["generation"]["candidate_count"] == count
        assert row["generation"]["temperature"] == 0.7
        assert row["generation"]["max_new_tokens"] == 48
        for candidate_index, candidate in enumerate(row["candidates"]):
            flags, eos, illegal, separate_equations = independently_check(candidate, row)
            totals["candidates"] += 1
            for key in ("parsed", "equations_valid", "linked", "task_operands_valid", "final_correct", "fully_verified"):
                totals[key] += int(flags[key])
            totals["final_correct_but_steps_invalid"] += int(flags["final_correct"] and not flags["fully_verified"])
            totals["eos"] += int(eos)
            totals["illegal_control_candidates"] += int(bool(illegal))
            compact = {
                "budget": count, "sample_index": sample_index, "candidate_index": candidate_index,
                "question": row["question"], "original_operands": [row["a"], row["b"], row["c"]],
                "truth": row["truth"], "generated": candidate["generated"],
                "generated_ids": candidate["generated_ids"], "eos": eos,
                "invalid_special_tokens": illegal, "recomputed": flags,
                "separate_equations": separate_equations,
            }
            audit_rows.append(compact)
            if candidate["generated"] == "5+0=5;5+3=10;answer=10":
                cited_bad_samples.append(compact)
            if not flags["parsed"]:
                malformed_samples.append(compact)
    for metric, key in (("parsed_candidates", "parsed"), ("candidate_final_accuracy", "final_correct"),
                        ("fully_verified_candidates", "fully_verified"),
                        ("final_correct_but_steps_invalid", "final_correct_but_steps_invalid")):
        assert budget[metric]["numerator"] == totals[key]
        assert budget[metric]["denominator"] == totals["candidates"]
    summary.append({"budget": count, "questions": len(samples), **totals})

combined = {key: sum(row[key] for row in summary) for key in totals}
