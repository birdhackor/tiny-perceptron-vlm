def execute(label, snippet, expected):
    out = io.StringIO()
    with contextlib.redirect_stdout(out):
        exec(compile(snippet, label, "exec"), {})
    actual = out.getvalue()
    assert actual == expected
    return {"code": snippet, "stdout": actual, "expected": expected, "pass": True}

strict = []
for prediction in ([1, 2, 3], [1, 2, 3, 4, 5]):
    try:
        sum(a == b for a, b in zip([1, 2, 3, 4], prediction, strict=True))
    except ValueError as error:
        strict.append({"lengths": [4, len(prediction)], "error": str(error)})
    else:
        raise AssertionError("zip(strict=True) did not reject unequal lengths")
result["strict_length_checks"] = strict
result["bool_sum"] = sum([True, True, False, True])
result["list_equality"] = {"same": [1, 2, 3, 4] == [1, 2, 3, 4], "reordered": [1, 2, 3, 4] == [1, 2, 4, 3], "different_length": [1, 2, 3, 4] == [1, 2, 3]}
assert result["bool_sum"] == 3
assert result["list_equality"] == {"same": True, "reordered": False, "different_length": False}
result["same_accuracy_different_nll"] = [
    {"p_correct": p, "p_other": 1 - p, "argmax_correct": p > 1 - p, "nll": -math.log(p)}
    for p in (0.6, 0.9)
]
assert result["same_accuracy_different_nll"][0]["nll"] > result["same_accuracy_different_nll"][1]["nll"]
