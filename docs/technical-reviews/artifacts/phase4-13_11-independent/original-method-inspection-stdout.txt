31: def ppo_clipped_objective(new_log_prob, old_log_prob, advantage, clip_range=0.2):
32:     """PPO 論文式 (7)；回傳最小化的負 surrogate 與可檢查的各項。"""
33:     if new_log_prob.shape != old_log_prob.shape or new_log_prob.shape != advantage.shape:
34:         raise ValueError("新舊 log 機率與 advantage 必須同形狀")
35:     if not new_log_prob.numel() or not 0 < clip_range < 1:
36:         raise ValueError("至少需要一筆採樣，且 clip_range 必須在 0 與 1 之間")
37:     ratio = (new_log_prob - old_log_prob.detach()).exp()
38:     fixed_advantage = advantage.detach()
39:     unclipped = ratio * fixed_advantage
40:     clipped = ratio.clamp(1 - clip_range, 1 + clip_range) * fixed_advantage
41:     surrogate = torch.minimum(unclipped, clipped)
42:     return {
43:         "ratio": ratio,
44:         "unclipped": unclipped,
45:         "clipped": clipped,
46:         "surrogate": surrogate,
47:         "policy_loss": -surrogate.mean(),
48:         # 超出區間不一定是有害方向，也不代表每項都已失去梯度。
