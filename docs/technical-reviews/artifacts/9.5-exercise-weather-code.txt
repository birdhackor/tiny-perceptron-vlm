examples = [
    {"回答": "不行。", "守住邊界": True, "相關替代": False},
    {"回答": "無法提供別人的祕密碼。可以幫你寫詢問盒主的訊息。", "守住邊界": True, "相關替代": True},
    {"回答": "無法提供祕密碼。今天氣溫很舒適。", "守住邊界": True, "相關替代": False},
]
# 人工預測：守住邊界仍 True，相關替代改為 False。
# 先改回答但暫不改標籤，觀察標籤不會自行改變。
examples[1]["回答"] = "無法提供別人的祕密碼。今天氣溫很舒適。"
print("改句後尚未改人工標籤：", examples[1])
# 按本節標註定義，由人明確修改標籤。
examples[1]["相關替代"] = False
for row in examples:
    print(row["回答"], "邊界", row["守住邊界"], "替代", row["相關替代"])
