import contextlib
import hashlib
import io
import json
import platform
from pathlib import Path
import torch

ad = Path('docs/technical-reviews/artifacts')
fence = (ad / '7.13-recheck-original-code.txt').read_text()
old_assert = 'assert scores["格式符合"] and not scores["1+1內容正確"]'
new_assert = 'assert scores["格式符合"] and scores["1+1內容正確"]'
assert 'type(parsed["answer"]) is int' in fence
assert old_assert in fence

def execute(source):
    namespace = {}
    stream = io.StringIO()
    failed = False
    with contextlib.redirect_stdout(stream):
        try:
            exec(compile(source, '<7.13-current-fence>', 'exec'), namespace)
        except AssertionError:
            failed = True
    return {'parsed': namespace['parsed'], 'value_type': type(next(iter(namespace['parsed'].values()))).__name__, 'scores':namespace['scores'], 'assert_failed':failed, 'stdout':stream.getvalue()}

reply3 = "reply = '{\"answer\": 3}'"
reply2 = "reply = '{\"answer\": 2}'"
code2 = fence.replace(reply3, reply2)
code2_corrected = code2.replace(old_assert, new_assert)
code_result = code2_corrected.replace(reply2, "reply = '{\"result\": 2}'")
rows = {'current_original':execute(fence), 'exercise_2_old_assert':execute(code2), 'exercise_2_corrected_assert':execute(code2_corrected), 'exercise_result_2':execute(code_result)}
assert rows['current_original']['scores'] == {'格式符合':True, '1+1內容正確':False}
assert not rows['current_original']['assert_failed']
assert rows['exercise_2_old_assert']['scores'] == {'格式符合':True, '1+1內容正確':True}
assert rows['exercise_2_old_assert']['assert_failed']
assert not rows['exercise_2_corrected_assert']['assert_failed']
assert rows['exercise_result_2']['scores'] == {'格式符合':False, '1+1內容正確':False}
assert rows['exercise_result_2']['assert_failed']
assert rows['exercise_result_2']['parsed'].get('answer') is None
