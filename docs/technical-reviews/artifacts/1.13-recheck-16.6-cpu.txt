def average_values(values, split):
    x = torch.tensor(values)
    parts = (x[:split], x[split:])
    whole = x.mean()
    means_mean = torch.stack([part.mean() for part in parts]).mean()
    shared = sum(part.sum() / len(x) for part in parts)
    return {"counts": [len(part) for part in parts], "losses": values,
            "whole_mean": whole.item(), "means_mean": means_mean.item(),
            "sum_div_shared_N": shared.item()}


same_values = average_values([1.0] * 11, 1)
changed_values = average_values([2.0] + [1.0] * 10, 1)
equal_counts = average_values([1.0, 2.0, 3.0, 4.0], 2)
assert same_values["whole_mean"] == same_values["means_mean"] == 1.0
assert abs(changed_values["whole_mean"] - 12 / 11) < 1e-12
assert changed_values["means_mean"] == 1.5
assert abs(changed_values["whole_mean"] - changed_values["sum_div_shared_N"]) < 1e-12
assert equal_counts["whole_mean"] == equal_counts["means_mean"] == 2.5

# 11個位置在w=0的loss皆1，但第一份的局部導數2，第二份各位置局部導數4。
# 值相同不保證導數相同。每次新前向、同一w，只在小份間累積.grad。
a = torch.tensor([1.0] + [2.0] * 10)
whole_w = torch.tensor(0.0, requires_grad=True)
whole_loss = (1 + a * whole_w).square().mean()
whole_loss.backward()
correct_w = torch.tensor(0.0, requires_grad=True)
wrong_w = torch.tensor(0.0, requires_grad=True)
correct_loss_value = 0.0
wrong_loss_value = 0.0
for part in (a[:1], a[1:]):
    correct_loss = (1 + part * correct_w).square().sum() / len(a)
    wrong_loss = (1 + part * wrong_w).square().mean() / 2
    correct_loss_value += correct_loss.item()
    wrong_loss_value += wrong_loss.item()
    correct_loss.backward()
    wrong_loss.backward()
gradient = {
    "counts": [1, 10], "fixed_w": 0.0,
    "per_position_loss_at_w": [1.0] * 11,
    "per_position_derivative_at_w": [2.0] + [4.0] * 10,
    "whole_loss": whole_loss.item(), "means_mean_loss": wrong_loss_value,
    "sum_div_shared_N_loss": correct_loss_value,
    "whole_gradient": whole_w.grad.item(),
    "means_mean_gradient": wrong_w.grad.item(),
    "sum_div_shared_N_gradient": correct_w.grad.item(),
}
assert whole_loss.item() == wrong_loss_value == 1.0
assert abs(whole_w.grad.item() - float(Fraction(42, 11))) < 1e-12
assert wrong_w.grad.item() == 3.0
assert abs(correct_w.grad.item() - whole_w.grad.item()) < 1e-12
