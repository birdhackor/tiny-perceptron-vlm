import io
import sys
import torch
from torch import nn
from tiny_perceptron.model import TinyLM, ModelConfig
from scripts.course_experiments.architecture import _copy_matching

torch.set_num_threads(1)
print("ENV", sys.version.split()[0], torch.__version__, "cpu", torch.get_num_threads())
E = torch.tensor([[1., 0.], [0., 1.], [1., 1.]])
h = torch.tensor([[2., 1.]])
print("matmul", (h @ E.T).tolist(), "weights", E.numel())
assert (h @ E.T).tolist() == [[2., 1., 3.]]
torch.manual_seed(42)
base = TinyLM(ModelConfig(width=64, layers=2, heads=4, max_length=128))
torch.manual_seed(42)
tied = TinyLM(ModelConfig(width=64, layers=2, heads=4, max_length=128, tied=True))
copied = _copy_matching(base, tied)
counts = [base.description()["parameters"], tied.description()["parameters"]]
print("parameters", counts, "difference", counts[0]-counts[1], "VxD", 264*64)
assert counts == [141568, 124672]
assert tied.embedding.weight is tied.output.weight
assert "embedding.weight" in copied and "output.weight" not in copied
assert torch.equal(tied.embedding.weight, base.embedding.weight)
print("initial_tables", "embedding reused", "output skipped", torch.equal(base.output.weight, tied.output.weight))
# Only a structural forward probe; no model training or dataset download.
x = torch.tensor([[1, 2, 3]])
with torch.no_grad():
    p = base(x)["logits"].softmax(-1)
    q = tied(x)["logits"].softmax(-1)
print("different_initial_distribution", not torch.equal(p,q))
assert not torch.equal(p,q)
# Serialization storage sharing and module Parameter identity are distinct checks.
buf = io.BytesIO()
torch.save(tied.state_dict(), buf)
buf.seek(0)
state = torch.load(buf, map_location="cpu", weights_only=True)
print("loaded_state_storage_shared", state["embedding.weight"].data_ptr() == state["output.weight"].data_ptr())
fresh_tied = TinyLM(ModelConfig(width=64, layers=2, heads=4, max_length=128, tied=True))
fresh_tied.load_state_dict(state)
fresh_separate = TinyLM(ModelConfig(width=64, layers=2, heads=4, max_length=128, tied=False))
fresh_separate.load_state_dict(state)
print("loaded_modules_identity", fresh_tied.output.weight is fresh_tied.embedding.weight, fresh_separate.output.weight is fresh_separate.embedding.weight)
print("separate_loaded_equal_values", torch.equal(fresh_separate.output.weight, fresh_separate.embedding.weight))
assert fresh_tied.output.weight is fresh_tied.embedding.weight
assert fresh_separate.output.weight is not fresh_separate.embedding.weight
print("unique_parameters", len(list(tied.parameters())), "registered_names", len(list(tied.named_parameters(remove_duplicate=False))))
