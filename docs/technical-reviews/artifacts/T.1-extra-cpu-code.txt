def run_code(section, code):
    stdout = io.StringIO()
    with contextlib.redirect_stdout(stdout):
        exec(compile(code, section, "exec"), {"__name__": "__main__"})
    return {"section": section, "code": code, "stdout": stdout.getvalue()}

images = []
for line in Path("data/training/vision-initial/train.jsonl").read_text().splitlines():
    row = json.loads(line)
    path = Path("data/training/vision-initial") / row["image"]
    sha = hashlib.sha256(path.read_bytes()).hexdigest()
    assert sha == row["sha256"]
    with Image.open(path) as image:

        assert image.size == (28, 28) and image.mode == "L"
    images.append({"source_row": row["source_row"], "label": row["label"], "family": path.name,
                   "sha256": sha, "size": [28, 28], "mode": "L"})

with tempfile.TemporaryDirectory(prefix="T.1-unpack-contract-") as temporary:
    root = Path(temporary)
    archive = root / "test.tar.gz"
    data = b'{"text":"toy"}\n'
    with tarfile.open(archive, "w:gz") as package:
        item = tarfile.TarInfo("training/examples.jsonl")
        item.size = len(data)
        package.addfile(item, io.BytesIO(data))
    asset = {"id": "reviewer-generated-contract-test", "archive_bytes": archive.stat().st_size,
             "archive_sha256": hashlib.sha256(archive.read_bytes()).hexdigest(),
             "files": [{"path": "training/examples.jsonl", "bytes": len(data), "sha256": hashlib.sha256(data).hexdigest()}]}
    destination = root / "data"
    first = unpack_asset(archive, asset, destination)
    second = unpack_asset(archive, asset, destination)
    bad = dict(asset, archive_sha256="0" * 64)
    try:
        unpack_asset(archive, bad, root / "bad")
    except ValueError as error:
        hash_error = str(error)
    else:
        raise AssertionError("mismatched hash must be rejected")
    result["unpack_contract"] = {"first": first, "second": second, "bad_hash_error": hash_error,
                                  "source": "self-generated 15-byte toy JSONL; not a download or training corpus"}

with tempfile.TemporaryDirectory(prefix="T.1-wav-header-") as temporary:
    first, second = Path(temporary) / "a.wav", Path(temporary) / "b.wav"
    values = np.array([0, .125, -.125], dtype="float32")
    sf.write(first, values, 16000, subtype="FLOAT")
    time.sleep(1.1)
    sf.write(second, values, 16000, subtype="FLOAT")
    a, b = first.read_bytes(), second.read_bytes()
    def chunks(raw):
        parsed = {}
        offset = 12
        while offset + 8 <= len(raw):
            name = raw[offset:offset+4].decode("ascii")
            length = int.from_bytes(raw[offset+4:offset+8], "little")
            parsed[name] = raw[offset+8:offset+8+length]
            offset += 8 + length + length % 2
        return parsed
    ca, cb = chunks(a), chunks(b)
    assert ca["data"] == cb["data"] and a != b
    differing = [name for name in ca if ca[name] != cb[name]]
    result["wav_byte_replay_limit"] = {"first_sha256": hashlib.sha256(a).hexdigest(),
      "second_sha256": hashlib.sha256(b).hexdigest(), "audio_data_chunk_equal": True,
      "differing_chunks": differing, "first_peak_hex": ca["PEAK"].hex(), "second_peak_hex": cb["PEAK"].hex(),
      "observation": "同一浮點陣列寫两次，當次只有 PEAK chunk bytes 不同。未持有正式凍結 WAV 原檔，故不能推定 CPU 衍生檔與正式產物逐位元相同；T.1 點數/時長主張不要求整檔重寫 SHA 相同。"}
