import torch, json, sys, hashlib
from pathlib import Path
from tiny_perceptron.simple import BigramLM, ContextMLP
torch.set_num_threads(1)
model = BigramLM(4)
with torch.no_grad():
    model.table.weight.copy_(torch.arange(16,dtype=torch.float32).reshape(4,4))
contexts = torch.tensor([[1,2,3],[0,0,3],[3,2,1]])
out = model(contexts)
assert torch.equal(out[0],out[1])
assert out.tolist() == [[12.,13.,14.,15.],[12.,13.,14.,15.],[4.,5.,6.,7.]]
mlp = ContextMLP(4,context=3,width=2)
assert list(mlp(torch.tensor([[1,2,3],[3,2,1]])).shape) == [2,4]
try:
    mlp(torch.tensor([[1,2,3,1,2]]))
except ValueError as e:
    fixed_window_error=str(e)
else:
    raise AssertionError("C5 should be rejected by ContextMLP configured for C3")
print(json.dumps({"bigram_contexts":contexts.tolist(),"bigram_outputs":out.detach().tolist(),"context_mlp_C3_output_shape":[2,4],"context_mlp_C5_error":fixed_window_error,"context_dtype":str(contexts.dtype),"weight_dtype":str(model.table.weight.dtype),"repository_file_sha256":hashlib.sha256(Path("tiny_perceptron/simple.py").read_bytes()).hexdigest(),"environment":{"python":sys.version,"torch":torch.__version__,"device":str(out.device),"torch_threads":str(torch.get_num_threads())}},ensure_ascii=False,indent=2))
