def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def section(path, identifier):
    raw = path.read_bytes()
    headings = list(re.finditer(rb"^## .*$", raw, re.M))
    index = next(i for i,h in enumerate(headings) if h[0].startswith(("## "+identifier+" ").encode()))
    return raw[headings[index].start():headings[index+1].start() if index+1 < len(headings) else len(raw)]


def preserve(path, raw):
    if path.exists():
        assert path.read_bytes() == raw, "不得覆寫既有原始版本"
    else:
        path.write_bytes(raw)
