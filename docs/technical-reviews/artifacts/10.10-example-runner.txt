import sys, re, json, platform, hashlib
from pathlib import Path
import torch
from torch.nn import functional as F

torch.set_num_threads(1)
def audit(event, args):
    if event.startswith("socket.") or event in ("subprocess.Popen", "os.system"):
        raise RuntimeError("offline bounded CPU audit: " + event)
sys.addaudithook(audit)
raw = Path("docs/technical-reviews/artifacts/10.10-section.txt").read_bytes()
assert hashlib.sha256(raw).hexdigest() == "4b15c536bbdf978bf90061890ee0352e88f1b4a2e3940a1e072db0aaaa061a48"
code = re.findall(r"```python\n(.*?)```", raw.decode(), re.S)[0]
if sys.argv[1] == "exercise":
    code = code.replace("targets = torch.arange(2)", "targets = torch.tensor([1, 0])")
namespace = {}
print("environment", json.dumps({"python": platform.python_version(), "torch": str(torch.__version__), "torch_git_version": torch.version.git_version, "device": "cpu", "threads": str(torch.get_num_threads())}))
print("case", sys.argv[1])
print("code", code)
exec(compile(code, "10.10-" + sys.argv[1], "exec"), namespace)
s = namespace["scores"]
t = namespace["targets"]
l = namespace["loss"]
print("result", json.dumps({"scores_after_backward": s.detach().tolist(), "targets": t.tolist(), "target_dtype": str(t.dtype), "loss": l.item(), "grad": s.grad.tolist(), "probs": (s.detach() / 0.5).softmax(-1).tolist()}))
assert s.device.type == "cpu"
assert torch.equal(s.detach(), torch.tensor([[2.0, 0.0], [0.0, 2.0]]))
if sys.argv[1] == "original":
    assert abs(l.item() - 0.01814992791780978) < 1e-6
    assert torch.allclose(s.grad, torch.tensor([[-0.01798621, 0.01798621], [0.01798621, -0.01798621]]), atol=1e-6)
else:
    assert abs(l.item() - 4.01814992791781) < 1e-6
    assert s.grad[0,1] < 0 and s.grad[1,0] < 0
    assert s.grad[0,0] > 0 and s.grad[1,1] > 0
print("assertions passed")
