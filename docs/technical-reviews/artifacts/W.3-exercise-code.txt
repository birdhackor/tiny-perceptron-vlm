import json, os, platform, sys
import torch

def guard(event, args):
    if event in {"socket.connect", "socket.getaddrinfo", "socket.sendto", "subprocess.Popen", "os.system"}:
        raise RuntimeError("bounded offline CPU execution blocked " + event)
sys.addaudithook(guard)
torch.set_default_device("cpu")
scores = torch.tensor([[60.0, 70.0, 80.0], [80.0, 90.0, 100.0]])
before = scores.clone()
scores[0, 2] = 110.0
row = scores.mean(dim=1)
column = scores.mean(dim=0)
assert row.tolist() == [80.0, 90.0]
assert column.tolist() == [70.0, 80.0, 105.0]
assert torch.equal(before[1], scores[1])
assert torch.equal(before[:, :2], scores[:, :2])
print(json.dumps({"mode":"exercise", "environment":{"python":platform.python_version(), "torch":str(torch.__version__), "torch_commit":torch.version.git_version,"device":str(scores.device),"OMP_NUM_THREADS":os.environ["OMP_NUM_THREADS"],"MKL_NUM_THREADS":os.environ["MKL_NUM_THREADS"],"threads":str(torch.get_num_threads())},"input":before.tolist(),"changed_input":scores.tolist(),"mean_dim1":row.tolist(),"mean_dim0":column.tolist(),"changed_row_indices":[0],"changed_column_indices":[2],"assertions":"passed"},ensure_ascii=False,indent=2))
