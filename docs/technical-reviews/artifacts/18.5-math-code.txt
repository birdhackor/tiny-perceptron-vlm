import math, json, re
from pathlib import Path
import torch
rows=[]
svg=Path("course/figures/architecture_distillation_temperature.svg").read_text()
for T in (1.,2.,4.):
    weights=[math.exp(x/T) for x in (4.,1.,0.)]
    probs=[w/sum(weights) for w in weights]
    widths=[520*p for p in probs]
    groups=re.findall(r'<g clip-path="url\(#bar-[124]\)">(.*?)</g>',svg,re.S)
    printed_widths=[float(x) for x in re.findall(r'<rect[^>]*width="([0-9.]+)"',groups[(1.,2.,4.).index(T)])]
    max_error=max(abs(a-b) for a,b in zip(widths,printed_widths))
    assert max_error < 0.0001
    assert probs[0]>probs[1]>probs[2]
    rows.append({"T":T,"probabilities_float64":probs,"ratio_4_over_3":math.exp(3/T),"percent_labels":[round(100*p,2) for p in probs],"expected_svg_widths":widths,"observed_svg_widths":printed_widths,"max_width_error":max_error,"sum":sum(probs)})
print(json.dumps({"e":math.e,"rows":rows,"uniform_limit":"for finite fixed logits and N=3, exp(z_i/T)->1 and p_i->1/3","post_division_sums":{str(T):1/T for T in (1.,2.,4.)}},ensure_ascii=False,indent=2))
