# Transparent matrix-composition witness, without an activation.
x = torch.tensor([[1.0, 2.0]], dtype=torch.float64)
w1 = torch.tensor([[1.0, 2.0], [3.0, 4.0]], dtype=torch.float64)
w2 = torch.tensor([[2.0, 1.0], [0.0, 3.0]], dtype=torch.float64)
assert torch.equal((x @ w1) @ w2, x @ (w1 @ w2))
print("matrix_composition", ((x @ w1) @ w2).tolist())

for values in ([-1.0, 0.0, 2.0], [-2.0, -1.0, 0.0, 1.0, 2.0], [-4.0, -2.0, 0.0, 2.0, 4.0]):
    t = torch.tensor(values, dtype=torch.float64)
    phi = [0.5 * (1.0 + math.erf(v / math.sqrt(2.0))) for v in values]
    exact = torch.tensor([v * p for v, p in zip(values, phi)], dtype=torch.float64)
    actual = F.gelu(t)
    assert torch.allclose(actual, exact, atol=2e-15, rtol=2e-15)
    assert actual.shape == F.relu(t).square().shape == t.shape
    print("formula", json.dumps({"x": values, "Phi": phi, "gelu": actual.tolist(), "relu": F.relu(t).tolist(), "relu2": F.relu(t).square().tolist(), "max_abs_error": (actual-exact).abs().max().item()}))

ints = torch.tensor([-4, -2, 0, 2, 4])

print("inferred_dtypes", str(ints.dtype), str(floats.dtype))

    F.gelu(ints)
except (RuntimeError, NotImplementedError) as exc:
    print("integer_gelu_rejected", type(exc).__name__, str(exc))
else:
    raise AssertionError("GELU unexpectedly accepted int64")
