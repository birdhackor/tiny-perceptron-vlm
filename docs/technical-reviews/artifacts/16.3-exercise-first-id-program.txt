import torch
from tiny_perceptron.model import TinyLM, ModelConfig
torch.manual_seed(0)
model = TinyLM(ModelConfig(width=8)).eval()
old_ids = torch.tensor([[1, 2, 3, 4]])
changed_ids = torch.tensor([[6, 2, 3, 4]])
with torch.no_grad():
    old_prefix = model(old_ids[:, :3])
    rebuilt_prefix = model(changed_ids[:, :3])
    full = model(changed_ids)["logits"][:, -1]
    cached = model(changed_ids[:, 3:], cache=rebuilt_prefix["cache"])["logits"][:, 0]
    stale = model(changed_ids[:, 3:], cache=old_prefix["cache"])["logits"][:, 0]
print("重建prefix最大差異", (full-cached).abs().max().item())
print("錯用舊prefix最大差異", (full-stale).abs().max().item())
assert torch.allclose(full, cached, atol=1e-6)
assert not torch.allclose(full, stale, atol=1e-6)
