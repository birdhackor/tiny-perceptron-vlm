from pathlib import Path
import json
import platform
import torch
from tiny_perceptron.model import TinyLM, ModelConfig

torch.manual_seed(0)
lm = TinyLM(ModelConfig(width=8)).eval()
ids = torch.tensor([1, 20, 30])
with torch.no_grad():
    original = lm(ids[None])["logits"]
    changed_positions = lm(ids[None], positions=torch.tensor([1, 2, 3]))["logits"]
    added_role_ids = torch.cat((ids.new_tensor([4]), ids))
    added_role = lm(added_role_ids[None])["logits"][:, 1:]
result = {
    "command": "OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 timeout 40 .venv/bin/python docs/technical-reviews/artifacts/10.8-path-controls-code.txt",
    "environment": {"python": platform.python_version(), "torch": torch.__version__, "device": "cpu", "omp_num_threads": "1", "mkl_num_threads": "1"},
    "scope": "額外的短 CPU 故障示範；直接改 TinyLM 的位置或前綴，不修改多模態包裝、教材或任何權重。",
    "ids": ids.tolist(),
    "original_shape": list(original.shape),
    "changed_positions": [1, 2, 3],
    "positions_scores_equal": torch.equal(original, changed_positions),
    "positions_max_abs_difference": float((original - changed_positions).abs().max()),
    "added_role_ids": added_role_ids.tolist(),
    "added_role_scores_equal_at_corresponding_input_positions": torch.equal(original, added_role),
    "added_role_max_abs_difference": float((original - added_role).abs().max()),
    "exit_code": 0,
}
assert result["positions_scores_equal"] is False
assert result["added_role_scores_equal_at_corresponding_input_positions"] is False
Path("docs/technical-reviews/artifacts/10.8-path-controls.json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
print(json.dumps(result, ensure_ascii=False))
