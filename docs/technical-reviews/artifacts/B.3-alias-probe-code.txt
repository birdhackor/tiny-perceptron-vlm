import json
import platform
from types import SimpleNamespace
import torch
from scripts.course_experiments.applications import _sample, _prompt_ids, _TOOLS_SYSTEM
from tiny_perceptron.data import ByteTokenizer

class OneEOSStub(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.config = SimpleNamespace(max_length=384)
    def forward(self, current, cache=None):
        logits = torch.full((*current.shape, 264), -100.0)
        logits[..., ByteTokenizer.eos_id] = 100.0
        return {'logits': logits, 'cache': None}

ctx = SimpleNamespace(device='cpu')
messages = [{'role':'system','content':_TOOLS_SYSTEM}, {'role':'user','content':'CALC:add(9,9)'}]
initial_ids = _prompt_ids(messages, ByteTokenizer())
g = _sample(OneEOSStub(), messages, ctx, tokens=1)
assert g['messages'] is messages
assert g['input_ids'] == initial_ids
messages += [{'role':'assistant','content':'{"name":"add","arguments":{"a":9,"b":9}}'}, {'role':'user','content':'TOOL_RESULT:18'}]
assert len(g['messages']) == 4
assert g['input_ids'] == initial_ids
assert g['input_ids'] != _prompt_ids(g['messages'], ByteTokenizer())
print(json.dumps({'environment':{'python':platform.python_version(),'torch':str(torch.__version__),'device':'cpu','OMP_NUM_THREADS':'1','MKL_NUM_THREADS':'1'}, 'scope':'單次 CPU forward 的 EOS stub 僅核 _sample 傳回物件身份及 += 造成清單別名變更；不是正式模型品質或重生成證據。', 'generation_messages_is_original_list': g['messages'] is messages, 'generation_messages_length_after_in_place_append': len(g['messages']), 'input_ids_remain_original_two_message_encoding': g['input_ids'] == initial_ids, 'later_messages_encoding_differs_from_first_input_ids':g['input_ids'] != _prompt_ids(g['messages'],ByteTokenizer()), 'stub_generated_ids':g['samples'][0]['generated_ids']},ensure_ascii=False,indent=2))
