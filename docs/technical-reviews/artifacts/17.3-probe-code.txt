import torch,json,sys
w=torch.tensor([[0.7,0.2],[-0.7,1.2]])
r=(w/0.5).round()*0.5
before=[w.clone(),r.clone()]
m=(w-r).abs().mean();v=m.item();py_round=round(v,4)
print('SCALAR',json.dumps({'mean_shape':list(m.shape),'mean_numel':m.numel(),'item_type':type(v).__name__,'python_round':py_round,'python_display':str(py_round),'unchanged_w':torch.equal(w,before[0]),'unchanged_restored':torch.equal(r,before[1]),'rounded_dtype':str(w.round(decimals=4).dtype),'original_dtype':str(w.dtype),'rounded_same_storage':w.round(decimals=4).data_ptr()==w.data_ptr()}))
for scale in [1,10]:
 x=torch.tensor([[1.,1.],[1.,-1.]])*scale
 d=(x@w.T)-(x@r.T); symbolic=x@(w-r).T
 print('DISTRIBUTIVITY',json.dumps({'input_scale':scale,'separately_subtracted':d.tolist(),'single_matmul':symbolic.tolist(),'max_fp32_residual':(d-symbolic).abs().max().item(),'allclose_default':torch.allclose(d,symbolic),'allclose_atol_2e-6':torch.allclose(d,symbolic,atol=2e-6,rtol=1e-5)}))
print('RELU',torch.relu(torch.tensor([-2.,-1.,0.,1.,2.])).tolist())
print('ENV',json.dumps({'python':sys.version,'torch':torch.__version__,'torch_git':torch.version.git_version,'device':'cpu','threads':torch.get_num_threads()}))
