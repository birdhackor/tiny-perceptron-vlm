import json
import math
import platform
import torch

# 僅驗證新增文字的加權公式；不是正式模型訓練或原例重跑。
batches = [(2, 1.0), (6, 2.4)]
aux_sum = 0.0
input_tokens = 0
for effective_inputs, auxiliary in batches:
    aux_sum += auxiliary * effective_inputs
    input_tokens += effective_inputs
weighted_mean = aux_sum / input_tokens
weights = [count / input_tokens for count, _ in batches]
unweighted_mean = sum(aux for _, aux in batches) / len(batches)
assert input_tokens == 8
assert weights == [0.25, 0.75]
assert math.isclose(weighted_mean, 2.05, rel_tol=0.0, abs_tol=1e-15)
assert weights[1] > weights[0]
assert not math.isclose(weighted_mean, unweighted_mean, rel_tol=0.0, abs_tol=1e-15)
print(json.dumps({
    'environment': {'python': platform.python_version(), 'torch': torch.__version__, 'torch_commit': torch.version.git_version, 'device': 'cpu', 'threads': str(torch.get_num_threads())},
    'batches': [{'effective_input_count': n, 'auxiliary': a} for n, a in batches],
    'weighted_numerator': aux_sum,
    'effective_input_denominator': input_tokens,
    'weights': weights,
    'weighted_mean': weighted_mean,
    'unweighted_mean': unweighted_mean,
    'formula': 'sum(n_b * a_b) / sum(n_b)',
    'scope': '本人新增的有限算術核對；只支持加權公式及較大有效輸入batch占較大權重。沒有重播正式39256輸入的逐批值，這些明細本來就未保存。',
}, ensure_ascii=False, indent=2))
