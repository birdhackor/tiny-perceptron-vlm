from pathlib import Path
import re,json,hashlib,platform
root=Path("docs/technical-reviews/artifacts")
sha=lambda b:hashlib.sha256(b).hexdigest()
prior=json.loads((root/"R.3-before-chapter15-second-recheck-report.json").read_text())
old=(root/"R.3-chapter15-current-source.txt").read_bytes()
new=Path("course/chapters/15.md").read_bytes()
assert sha(old)==next(s for s in prior["sources"] if s["id"]=="s_ch_15")["sha256"]=="55dda9f5851aa43ded13bda2bad646c50d013e845c841f0e7ba7bd9b608e56f3"
assert sha(new)=="a1143d3d402f6c46693595f2ebfe06cdf9cf58f141d05bf941dad790de533d9a"
assert new==(root/"R.3-chapter15-second-current-source.txt").read_bytes()
def sections(raw):
    heads=list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+",raw))
    return {h[1].decode():{"raw":raw[h.start():heads[i+1].start() if i+1<len(heads) else len(raw)],"first_line":raw[:h.start()].count(b"\n")+1} for i,h in enumerate(heads)},raw[:heads[0].start()]
a,oldintro=sections(old);b,newintro=sections(new)
assert a.keys()==b.keys()
changed=[sid for sid in a if a[sid]["raw"]!=b[sid]["raw"]]
assert changed==["15.9"]
assert sha(b["15.9"]["raw"])=="218f17638b38cf8e9a89da8067ebdf05c5cc6d5548d9605803ad8cca96b881ca"
assert oldintro==newintro
for sid in ["15.1","15.4","15.8","15.11"]:assert a[sid]["raw"]==b[sid]["raw"]
assert re.findall(rb"```[^\n]*\n[\s\S]*?```",old)==re.findall(rb"```[^\n]*\n[\s\S]*?```",new)
assert re.findall(r"(?m)^練習[^\n]*",old.decode())==re.findall(r"(?m)^練習[^\n]*",new.decode())
assert re.findall(rb"!\[[^\]]*\]\(([^)]+\.svg)\)",old)==re.findall(rb"!\[[^\]]*\]\(([^)]+\.svg)\)",new)
previous=json.loads((root/"R.3-chapters15-16-recheck-output.json").read_text())
previous15=next(c for c in previous["chapter_identity_and_required_scope_checks"] if c["chapter"]=="15")
scopes15=[]
for scope in previous15["required_ranges"]:
    text="\n".join(old.decode().splitlines()[scope["current_first_line"]-1:scope["current_last_line"]])+"\n"
    assert sha(text.encode())==scope["current_required_scope_sha256"]
    assert text.encode() in new
    start=new.index(text.encode())
    scopes15.append({"old_first_line":scope["current_first_line"],"old_last_line":scope["current_last_line"],"current_first_line":new[:start].count(b"\n")+1,"current_last_line":new[:start].count(b"\n")+text.count("\n"),"scope_sha256_unchanged":sha(text.encode()),"required_support_bytes_identical":True})
current16=Path("course/chapters/16.md").read_bytes()
frozen16=(root/"R.3-chapter16-second-current-source.txt").read_bytes()
assert current16==frozen16 and sha(current16)==next(s for s in prior["sources"] if s["id"]=="s_ch_16")["sha256"]=="05794da02b7d951f3ab9bd2d305e9f93497a277b5e4a971f857e22ff8dd8f6b5"
previous16=json.loads((root/"R.3-chapter16-second-recheck-output.json").read_text())
scopes16=[]
for scope in previous16["required_scopes"]:
    text="\n".join(current16.decode().splitlines()[scope["current_first_line"]-1:scope["current_last_line"]])+"\n"
    assert sha(text.encode())==scope["scope_sha256_unchanged"]
    scopes16.append({"current_first_line":scope["current_first_line"],"current_last_line":scope["current_last_line"],"scope_sha256_unchanged":sha(text.encode()),"matches_prior_actual_read_support":True})
assert sha(sections(current16)[1])==previous16["intro_raw_sha256_unchanged"]
primary=sections(Path("course/README.md").read_bytes())[0]["R.3"]["raw"]
assert sha(primary)==prior["source_sha256"] and primary==(root/"R.3-source.txt").read_bytes()
for artifact in prior["artifacts"]:assert sha(Path(artifact["path"]).read_bytes())==artifact["sha256"],artifact["id"]
for path,digest in prior["figure_sha256"].items():assert sha(Path(path).read_bytes())==digest,path
