import json, sys, platform
from pathlib import Path
import torch
from tiny_perceptron.data import ByteTokenizer, render_chat

torch.set_num_threads(1)
torch.set_num_interop_threads(1)
answer, expected = sys.argv[1], int(sys.argv[2])
original = Path("docs/technical-reviews/artifacts/7.3-original-code.txt").read_text()
modified = original.replace('"content": "答"', '"content": ' + repr(answer))
assert modified != original
namespace = {}
exec(compile(modified, "7.3-original-fence-exercise", "exec"), namespace)
x, y = namespace["x"], namespace["y"]
tok = ByteTokenizer()
assert y[y != -100].tolist() == tok.encode(answer) + [tok.eos_id]
assert (y != -100).sum().item() == expected
assert x[2:5].tolist() == tok.encode("問")
assert x.min().item() >= 0 and x.max().item() < tok.vocab_size
print("核對", json.dumps({"answer":answer,"bytes":list(answer.encode("utf-8")),"x_length":len(x),"effective_targets":y[y!=-100].tolist(),"effective_count":expected,"user_bytes_preserved":x[2:5].tolist(),"python":platform.python_version(),"torch":torch.__version__,"torch_git_version":torch.version.git_version,"device":str(x.device),"intra_threads":torch.get_num_threads(),"interop_threads":torch.get_num_interop_threads()},ensure_ascii=False))
