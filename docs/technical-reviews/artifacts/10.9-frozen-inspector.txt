import hashlib
import json
import platform
from pathlib import Path
import torch
p=Path("docs/course-experiments/results/contrastive.json")
d=json.loads(p.read_text())
r=d["results"]
code={}
for name in ["scripts/course_experiments/modalities.py","tiny_perceptron/multimodal.py","tiny_perceptron/data.py","tiny_perceptron/training.py"]:
    actual=hashlib.sha256(Path(name).read_bytes()).hexdigest()
    expected=d["code_sha256"][name]
    assert actual==expected
    code[name]={"current_sha256":actual,"frozen_result_sha256":expected,"matches":True}
variants={}
for name,v in r["variants"].items():
    train=v["training"]
    assert len(train["history"])==train["steps"]==250
    assert sum(x["effective_targets"] for x in train["history"])==train["effective_targets"]==1500
    assert all(x["effective_targets"]==6 for x in train["history"])
    stats={}
    for phase in ["before","validation","test"]:
        e=v[phase]
        im=e["image_samples"];tx=e["text_samples"]
        assert e["image_queries"]==len(im)==6
        assert e["text_queries"]==len(tx)==6
        assert sum(x["correct"] for x in im)/len(im)==e["image_to_text_accuracy"]
        assert sum(x["correct"] for x in tx)/len(tx)==e["text_to_image_accuracy"]
        stats[phase]={"image_queries":len(im),"text_queries":len(tx),"image_to_text_correct":sum(x["correct"] for x in im),"text_to_image_correct":sum(x["correct"] for x in tx),"image_to_text_accuracy":e["image_to_text_accuracy"],"text_to_image_accuracy":e["text_to_image_accuracy"]}
    variants[name]={"training":{k:v for k,v in train.items() if k!="history"},"history_denominator_check":{"updates":len(train["history"]),"targets_per_update":6,"summed_targets":1500},"evaluation":stats}
print(json.dumps({"environment":{"python":platform.python_version(),"torch":torch.__version__,"device":"cpu inspection of frozen CUDA result; no formal run"},"original_path":str(p),"original_sha256":hashlib.sha256(p.read_bytes()).hexdigest(),"candidate_original_url_from_report_not_fetched":f"https://github.com/birdhackor/tiny-perceptron-vlm/blob/{d['hf']['result_revision']}/{p}","run_id":d["modal"]["run_id"],"source_revision":d["revision"],"recorded_environment":{k:d[k] for k in ["device","seed","torch_version","python_version","gpu","step_scale","evidence_status","unfinished_schedules","status"]},"config":r["config"],"scope":r["scope"],"data":r["data"],"variants":variants,"code_pin_matches":code,"inspection_note":"只解析當前凍結正式報告、重算分母及核 SHA；未訓練、未下載權重、未執行 GPU。正文未提出成績主張，此保存僅限定正式入口與學習敘述的實驗範圍。"},ensure_ascii=False,indent=2))
