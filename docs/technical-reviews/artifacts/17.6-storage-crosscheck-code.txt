import torch
from tiny_perceptron.quantization import quantize_symmetric, pack_int4, unpack_int4
w=torch.linspace(-1,1,33)
q,scale=quantize_symmetric(w,4)
packed=pack_int4(q)
restored=unpack_int4(packed,q.shape)
print("STORAGE", {"q_dtype":str(q.dtype),"q_bytes":q.numel()*q.element_size(),"packed_dtype":str(packed.dtype),"packed_bytes":packed.numel()*packed.element_size(),"scale_bytes":scale.numel()*scale.element_size(),"packed_shape":list(packed.shape),"original_shape":list(q.shape),"roundtrip_equal":bool(torch.equal(q,restored)),"q_unique":int(q.unique().numel())})
assert torch.equal(q,restored)
assert packed.numel()*packed.element_size()==17
assert q.numel()*q.element_size()==33
assert scale.numel()*scale.element_size()==4
