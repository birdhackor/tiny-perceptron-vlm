import hashlib
import json
import os
import platform
import re
import subprocess
from pathlib import Path

import torch

root = Path.cwd()
commands = []
def run(argv):
    p = subprocess.run([str(root / ".venv/bin/python"), *argv], capture_output=True, text=True, timeout=30, env={**os.environ, "OMP_NUM_THREADS": "1", "MKL_NUM_THREADS": "1"})
    row = {"command": ".venv/bin/python " + " ".join(argv), "exit_code": p.returncode, "stdout": p.stdout, "stderr": p.stderr}
    commands.append(row)
    assert p.returncode == 0, row
    return p.stdout

run(["scripts/prepare_data.py", "--kind", "preference", "--output", "/tmp/T.7-own-generated"])
for script in ("scripts/train.py", "scripts/infer.py", "scripts/fetch_training_assets.py"):
    run([script, "--help"])
run(["-m", "scripts.course_experiments.run", "--help"])
for model in ("model.pt", "beta1.pt", "format-model.pt"):
    prompt = "4+2=?" if model == "model.pt" else "2+2=?"
    raw = run(["scripts/infer.py", "checkpoints/course/dpo/" + model, "--chat", "--prompt", prompt, "--tokens", "32", "--temperature", "0", "--device", "cpu", "--json"])
    value = json.loads(raw)
    assert value["answer"] == ("3" if model == "format-model.pt" else "8")
    assert value["eos"] is True

scope = {"reviewer_task": "/root/technical_round_coordinator/technical_t_7_fresh", "environment": {"python": platform.python_version(), "torch": torch.__version__, "device": "cpu", "OMP_NUM_THREADS": "1", "MKL_NUM_THREADS": "1"}, "scope": "T.7没有Python短fence。資料產生器僅另設/tmp輸出，CLI以--help核契約；既有pinned公開權重CPU推論是明記替代checkpoint/device的重播。未執行探索配方200步、正式CUDA排程或fetch資料命令。探索配方style.pt/preferred.pt在本地缺失，未冒稱已執行其前後推論。", "commands": commands}
