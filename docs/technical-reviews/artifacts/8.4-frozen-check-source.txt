"""離線重算凍結資料與評分；不載入正式模型、不生成正式回答、不更新權重。"""
import ast
import hashlib
import json
import random
import sys
from pathlib import Path

import torch

from tiny_perceptron.model import generate

root = Path.cwd()
base = root / "outputs/technical-sources/8.4-fresh"
namespace = {"hashlib": hashlib, "json": json, "random": random}
selected = {
    "formal-scripts--course_experiments--common.py": ["split_records", "records_sha256"],
    "formal-scripts--course_experiments--behavior.py": ["_conversation", "_style_record", "_style_metrics"],
    "formal-text.py": ["arithmetic_records"],
}
read_sources = []
for name, functions in selected.items():
    data = (base / name).read_bytes()
    tree = ast.parse(data)
    nodes = [node for node in tree.body if isinstance(node, ast.FunctionDef) and node.name in functions]
    assert {node.name for node in nodes} == set(functions)
    exec(compile(ast.Module(body=nodes, type_ignores=[]), name, "exec"), namespace)
    read_sources.append({"file": name, "sha256": hashlib.sha256(data).hexdigest(), "functions": functions})

result_path = root / "docs/course-experiments/results/style.json"
document = json.loads(result_path.read_text())
result = document["results"]
arithmetic = namespace["split_records"](namespace["arithmetic_records"](), seed=document["seed"])
parts = {
    split: [namespace["_style_record"](row, style, True) for row in rows for style in ("concise", "vivid", "json")]
    for split, rows in arithmetic.items()
}
dates = []
for day in range(1, 25):
    date = f"2026-10-{day:02d}"
    dates.append(namespace["_conversation"](f"task=date;date={date};confirm", "已確認" + date + "。", "date-" + date, style="clarification"))
    dates.append(namespace["_conversation"](f"task=date;id={day};date=?;confirm", "請提供日期。", "date-" + date, style="clarification"))
date_parts = namespace["split_records"](dates, seed=document["seed"])
parts = {split: rows + date_parts[split] for split, rows in parts.items()}
checks = {}
for split, rows in parts.items():
    raw = "".join(json.dumps(row, ensure_ascii=False) + "\n" for row in rows).encode()
    observed_hash = hashlib.sha256(raw).hexdigest()
    assert observed_hash == result["data"][split]["sha256"]
    checks[split] = {
        "records": len(rows),
        "arithmetic_questions": len(arithmetic[split]),
        "date_records": len(date_parts[split]),
        "jsonl_sha256": observed_hash,
        "all_three_styles_for_every_arithmetic_question": all(
            {row["style"] for row in rows if row.get("a") == question["a"] and row.get("b") == question["b"]} == {"concise", "vivid", "json"}
            for question in arithmetic[split]
        ),
    }
    assert checks[split]["all_three_styles_for_every_arithmetic_question"]
    if split == "train":
        assert namespace["records_sha256"](rows) == result["training"]["records_sha256"]
        continue
    report = json.loads(json.dumps(result["after"][split]))
    assert len(report["samples"]) == len(rows)
    invalid_outputs = []
    for row, sample in zip(rows, report["samples"], strict=True):
        assert sample["messages"] == row["messages"][:-1]
        assert sample["expected"] == row["messages"][-1]["content"]
        ids = sample["generated_ids"]
        assert ids[-1] == 2 and ids.count(2) == 1 and sample["eos"]
        invalid = [value for value in ids[:-1] if not 8 <= value < 264]
        invalid_outputs.extend(invalid)
        assert not invalid
        assert bytes(value - 8 for value in ids[:-1]).decode("utf-8", errors="strict") == sample["generated"]
    checked = namespace["_style_metrics"](report, rows)
    assert checked["rubric"] == result["after"][split]["rubric"]
    checks[split]["recalculated_rubric"] = checked["rubric"]
    checks[split]["invalid_special_tokens_before_eos"] = invalid_outputs
    checks[split]["eos_count"] = sum(sample["eos"] for sample in report["samples"])
    checks[split]["visible_samples"] = [
        {"prompt": sample["messages"][0]["content"], "expected": sample["expected"], "generated": sample["generated"], "eos": sample["eos"]}
        for sample in report["samples"]
    ]

release = json.loads((root / "docs/course-experiments/public-releases/style.json").read_text())
card = root / "checkpoints/course/style/README.md"
evaluation = root / "checkpoints/course/style/evaluation.json"
files = release["release"]["public_manifest"]["files"]
binding = []
for path in [card, evaluation, root / "checkpoints/course/style/model.pt"]:
    raw = path.read_bytes()
    sha = hashlib.sha256(raw).hexdigest()
    published = next(item for item in files if item["output"] == path.name)
    assert sha == published["sha256"] and len(raw) == published["bytes"]
    binding.append({"local_input": str(path.relative_to(root)), "sha256": sha, "bytes": len(raw), "immutable_url": "https://huggingface.co/" + release["release"]["repo"] + "/resolve/" + release["release"]["revision"] + "/" + published["path"]})
evaluation_data = json.loads(evaluation.read_text())
assert evaluation_data["conditional"]["training"]["steps"] == result["training"]["steps"] == 1000
for split in ["validation", "test"]:
    assert evaluation_data["conditional"]["evaluation"][split]["rubric"] == checks[split]["recalculated_rubric"]

class GreedyToy(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.config = type("Config", (), {"max_length": 10})()
        self.contexts = []

    def forward(self, ids, cache=None):
        self.contexts.append(ids.tolist())
        logits = torch.zeros(ids.shape[0], ids.shape[1], 264)
        logits[:, -1, 59 if len(self.contexts) == 1 else 2] = 9.0
        return {"logits": logits, "cache": None}

toy = GreedyToy()
greedy = generate(toy, torch.tensor([[1, 3]]), max_new_tokens=3, temperature=0.0, eos_id=2)
assert greedy.tolist() == [[1, 3, 59, 2]]
assert toy.contexts == [[[1, 3]], [[1, 3, 59]]]
assert toy.training

all_date_prompts = set()
def inspect_dates(value):
    if isinstance(value, dict):
        for key, item in value.items():
            if key == "content" and isinstance(item, str) and item.startswith("task=date;"):
                all_date_prompts.add(item)
            inspect_dates(item)
    elif isinstance(value, list):
        for item in value:
            inspect_dates(item)
inspect_dates(document)
training_date_prompts = {row["messages"][0]["content"] for row in date_parts["train"]}
assert len(all_date_prompts) == 10
assert len(training_date_prompts) == 38
assert not all_date_prompts.intersection(training_date_prompts)
