import torch
import torch.nn.functional as F

x = torch.tensor([[2.0, 2.0, 2.0], [1.0, 2.0, 3.0]])
eps = 1e-5
before_ln = F.layer_norm(x, (3,), eps=eps)
before_rms = x / torch.sqrt(x.square().mean(-1, keepdim=True)+eps)
ln_same = torch.allclose(before_ln, ns["ln"], atol=1e-6, rtol=1e-6)
rms_same = torch.allclose(before_rms, ns["rms"], atol=1e-6, rtol=1e-6)
assert ln_same and not rms_same
zero = torch.zeros(2,3)
zero_ln = F.layer_norm(zero,(3,),eps=eps)
zero_rms = zero / torch.sqrt(zero.square().mean(-1,keepdim=True)+eps)
assert torch.isfinite(zero_ln).all() and torch.isfinite(zero_rms).all()
