import json

records = []
for question, fact in [("1+1=?", "2"), ("2+2=?", "4")]:
    targets = {
        "簡短": fact,
        "生動": fact + "；把兩組物件合起來數就知道了。",
        "JSON": json.dumps({"answer": int(fact)}),
    }
    for style in ["簡短", "生動", "JSON"]:
        records.append({"提問": f"風格={style}；{question}", "回答": targets[style]})
    assert len(set(targets.values())) == 3
    assert json.loads(targets["JSON"]) == {"answer": int(fact)}
print("筆數", len(records))
for row in records:
    print(row["提問"], "→", row["回答"])
naive = ["2" if s == "簡短" else "2；把兩組物件合起來數就知道了。" for s in ["簡短", "生動", "JSON"]]
print("只新增名字、沿用原條件判斷：生動與JSON目標相同", naive[1] == naive[2])
assert len(records) == 6
