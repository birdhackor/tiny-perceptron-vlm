from urllib.request import urlopen
from bs4 import BeautifulSoup
from pathlib import Path
import json, hashlib
from datetime import datetime, UTC
items=[
('builtins','https://docs.python.org/3.13/library/functions.html',['enumerate','sorted','len','print']),
('types','https://docs.python.org/3.13/library/stdtypes.html',['text-sequence-type-str','set-types-set-frozenset','mapping-types-dict','str.join','common-sequence-operations']),
('comparisons','https://docs.python.org/3.13/reference/expressions.html',['value-comparisons','displays-for-lists-sets-and-dictionaries']),
('assert','https://docs.python.org/3.13/reference/simple_stmts.html',['the-assert-statement']),
('tutorial','https://docs.python.org/3.13/tutorial/datastructures.html',['list-comprehensions']),
('unicode','https://docs.python.org/3.13/howto/unicode.html',['definitions','encodings']),
('hf','https://huggingface.co/learn/llm-course/en/chapter2/4',['main-content'])]
records=[]
for label,url,ids in items:
 raw=urlopen(url,timeout=25).read(); soup=BeautifulSoup(raw,'html.parser'); passages=[]
 for selector in ids:
  el=soup.find(id=selector)
  if el is None and label=='hf': el=soup.find('main')
  if el is None: raise RuntimeError((label,selector,'not found'))
  if el.name=='dt':
   el=el.find_parent('dl'); nodes=[el]
  elif label=='types' and el.name=='section':
   nodes=el.find_all(['p','table','pre'],recursive=False)
  else:
   if el.name not in ('section','main','article'): el=el.parent
   nodes=[el]
  passages.append({'locator':'#'+selector,'text':'\n'.join(n.get_text(' ',strip=True) for n in nodes)})
 record={'id':label,'url':url,'retrieved_at':datetime.now(UTC).isoformat(),'page_title':soup.title.get_text(' ',strip=True),'raw_html_sha256':hashlib.sha256(raw).hexdigest(),'passages':passages}
 records.append(record)
 print(json.dumps(record,ensure_ascii=False))
Path('docs/technical-reviews/artifacts/1.1-official-docs.json').write_text(json.dumps(records,ensure_ascii=False,indent=2)+'\n')
