import hashlib
import json
import math
import platform
from pathlib import Path

source_path = Path('docs/course-experiments/results/moe.json')
source_bytes = source_path.read_bytes()
r = json.loads(source_bytes)
v = r['results']['variants']['top1_aux0']
route = v['validation_routing']
layer = route['layers'][1]
counts = layer['dispatch_counts']
denominator = sum(counts)
fractions = [c / denominator for c in counts]
terms = [-f * math.log(f) if f > 0 else 0.0 for f in fractions]
h = sum(terms)
normalized = h / math.log(len(counts))
percentages = [round(100 * f, 2) for f in fractions]
assert counts == [8536, 25630, 4875, 215]
assert route['effective_input_tokens'] == layer['dispatch_denominator'] == denominator == 39256
assert route['padding_excluded'] is True
assert v['model']['config']['top_k'] == 1
assert v['model']['config']['experts'] == 4
assert v['model']['config']['layers'] == 2
assert v['training']['auxiliary_weight'] == 0
assert percentages == [21.74, 65.29, 12.42, 0.55]
assert round(normalized, 5) == 0.64755
assert abs(h - layer['load_entropy_nats']) < 1e-14
assert abs(normalized - layer['normalized_load_entropy']) < 1e-14
assert max(abs(x-y) for x,y in zip(fractions, layer['load_fraction'])) < 1e-15
code_pins = {}
for name in ('tiny_perceptron/modern.py', 'tiny_perceptron/data.py', 'tiny_perceptron/model.py', 'scripts/course_experiments/common.py'):
    actual = hashlib.sha256(Path(name).read_bytes()).hexdigest()
    expected = r['code_sha256'][name]
    assert actual == expected
    code_pins[name] = {'current_sha256': actual, 'frozen_sha256': expected, 'matches': True}
arch = Path('/tmp/15.8-fresh-frozen-architecture.py').read_bytes()
arch_sha = hashlib.sha256(arch).hexdigest()
assert arch_sha == r['code_sha256']['scripts/course_experiments/architecture.py']
print(json.dumps({
    'environment': {'python': platform.python_version(), 'torch': '不載入；只解析凍結 JSON 與 Python math', 'device': 'cpu'},
    'source_path': source_path.as_posix(), 'source_sha256': hashlib.sha256(source_bytes).hexdigest(),
    'frozen_runtime': {k: r[k] for k in ('revision','seed','device','gpu','python_version','torch_version','evidence_status','status')},
    'dataset': r['results']['dataset'],
    'variant': 'top1_aux0',
    'model_config': v['model']['config'],
    'training_scope': {k: v['training'][k] for k in ('requested_steps','steps','optimizer_updates','skipped_updates','effective_tokens','auxiliary_weight','batch_size','status')},
    'validation_routing': route,
    'recomputed_layer2': {'counts': counts,'denominator': denominator,'fractions':fractions,'percentages':percentages,'entropy_terms':terms,'entropy_nats':h,'normalized_load_entropy':normalized,'printed5':round(normalized,5)},
    'code_pins': code_pins,
    'frozen_architecture': {'url': 'https://raw.githubusercontent.com/birdhackor/tiny-perceptron-vlm/'+r['revision']+'/scripts/course_experiments/architecture.py', 'sha256': arch_sha, 'matches_frozen_report': True},
    'scope': '親讀原始凍結報告後離線重算第二層負載；未載入權重、未下載語料、未重訓、未重跑 CUDA 實驗。',
}, ensure_ascii=False, indent=2))
