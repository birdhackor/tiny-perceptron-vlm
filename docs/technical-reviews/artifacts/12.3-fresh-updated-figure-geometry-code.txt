import hashlib
import json
import math
import os
import platform
from pathlib import Path
from xml.etree import ElementTree as ET

path = Path("course/figures/multimodal_audio_axes.svg")
raw = path.read_bytes()
assert hashlib.sha256(raw).hexdigest() == "a36139901176e9b8356a785a19138b1ee0b6ae8a61064f5d8eb71496cb0f8820"
root = ET.fromstring(raw)
ns = {"svg": "http://www.w3.org/2000/svg"}
rects = [x for x in root.findall("svg:rect", ns) if x.get("y") == "483"]
centers = [float(x.get("x")) + float(x.get("width")) / 2 for x in rects]
texts = [x for x in root.findall("svg:text", ns) if x.get("y") == "556"]
tick_x = [float(x.get("x")) for x in texts]
tick_seconds = [float(x.text) for x in texts]
assert centers == tick_x == [175.0, 260.0, 345.0, 430.0, 515.0]
assert tick_seconds == [j * 160 / 16000 for j in range(5)]
short_marks = [x.get("d") for x in root.findall("svg:path", ns) if x.get("d", "").endswith(" 537")]
assert short_marks == [f"M{int(x)} 531L{int(x)} 537" for x in centers]
points = root.find("svg:polyline", ns).get("points").split()
errors = []
for pair in points:
    x, y = map(float, pair.split(","))
    t = (x - 60) / 495 * 0.008
    errors.append(abs(y - (165 - 45 * math.sin(2 * math.pi * 440 * t))))
assert max(errors) < 0.04
result = {
    "source_svg_sha256": hashlib.sha256(raw).hexdigest(),
    "column_centers_x": centers,
    "tick_x": tick_x,
    "tick_seconds": tick_seconds,
    "short_marks": short_marks,
    "center_formula": "j × 160 / 16000 seconds, j=0..4; center=True",
    "polyline_points": len(points),
    "polyline_max_pixel_error": max(errors),
    "polyline_tolerance_pixels": 0.04,
    "scope": "新增中心刻度的座標核對與未變波形公式；不是功率品質或速度量測。",
    "environment": {
        "python": platform.python_version(),
        "device": "cpu",
        "libraries": "Python standard library: hashlib, json, math, xml.etree.ElementTree",
        "OMP_NUM_THREADS": os.environ["OMP_NUM_THREADS"],
        "MKL_NUM_THREADS": os.environ["MKL_NUM_THREADS"],
    },
}
Path("docs/technical-reviews/artifacts/12.3-fresh-updated-figure-geometry-result.json").write_text(
    json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8"
)
print(json.dumps(result, ensure_ascii=False))
