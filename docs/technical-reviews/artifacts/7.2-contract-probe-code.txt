import json, sys, platform, hashlib
from pathlib import Path
import torch
from tiny_perceptron.data import ByteTokenizer, render_chat

tok = ByteTokenizer()
question = "1+1=?"
prompt = [tok.bos_id, tok.user_id] + tok.encode(question) + [tok.eos_id, tok.assistant_id]
x, y = render_chat([{"role": "user", "content": question}, {"role": "assistant", "content": "2"}], tok)
assert x[:len(prompt)].tolist() == prompt
assert y[len(prompt)-1].item() == tok.encode("2")[0] == 58
assert y[len(prompt)-2].item() == -100
assert tok.encode("使用者") != [tok.user_id]
messages = [{"role":"user","content":"1+1=?"},{"role":"assistant","content":"2"},{"role":"user","content":"2+2=?"}]
multi_prompt = [tok.bos_id]
roles = {"user": tok.user_id, "assistant": tok.assistant_id}
for message in messages:
    multi_prompt += [roles[message["role"]]] + tok.encode(message["content"]) + [tok.eos_id]
multi_prompt += [tok.assistant_id]
full_x, full_y = render_chat(messages + [{"role":"assistant","content":"4"}], tok)
assert full_x[:len(multi_prompt)].tolist() == multi_prompt
assert full_y[len(multi_prompt)-1].item() == tok.encode("4")[0]
print(json.dumps({"prompt":prompt,"question_utf8_bytes":list(question.encode("utf-8")),"prompt_length":len(prompt),"answer_byte_id":58,"single_turn_training_x":x.tolist(),"single_turn_training_y":y.tolist(),"first_answer_target_position":len(prompt)-1,"missing_assistant_last_position_target":y[len(prompt)-2].item(),"literal_role_ids":tok.encode("使用者"),"multi_turn_prompt":multi_prompt,"multi_turn_first_answer_target":full_y[len(multi_prompt)-1].item(),"environment":{"python":sys.version,"torch":torch.__version__,"torch_git_version":torch.version.git_version,"device":"cpu","OMP_NUM_THREADS":"1","MKL_NUM_THREADS":"1"},"scope":"僅格式與監督對齊，沒有訓練與自由生成結果；EOS 位置的 assistant 目標被忽略，不能宣稱缺標記後模型必然產生哪個 token。"},ensure_ascii=False,indent=2))
