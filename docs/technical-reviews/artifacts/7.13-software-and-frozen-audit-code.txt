import hashlib
import json
import platform
import tempfile
from pathlib import Path
from types import SimpleNamespace
import torch
from scripts.course_experiments.common import Context
from tiny_perceptron.data import ByteTokenizer
from tiny_perceptron.model import ModelConfig, TinyLM, generate
from tiny_perceptron.training import load_checkpoint, save_checkpoint

torch.set_num_threads(1)
repo = Path.cwd()
def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

tok = ByteTokenizer()
text = '甲A'
encoded = tok.encode(text)
assert tok.decode(encoded) == text
assert encoded == [b + 8 for b in text.encode('utf-8')]
class FixedScores(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.config = SimpleNamespace(max_length=16)
    def forward(self, ids, cache=None):
        scores = torch.tensor([0., -1., -2., 1., 2., 3.])
        return {'logits': scores.expand(ids.shape[0], ids.shape[1], -1), 'cache': None}
model = FixedScores()
output = generate(model, torch.tensor([[1]]), max_new_tokens=3, temperature=0)
assert output.tolist() == [[1, 5, 5, 5]]
probabilities = {str(t): (torch.tensor([0., 1., 2.], dtype=torch.float64) / t).softmax(-1).tolist() for t in (0.5, 1., 2.)}
assert probabilities['0.5'][-1] > probabilities['1.0'][-1] > probabilities['2.0'][-1]
with tempfile.TemporaryDirectory(prefix='7.13-checkpoint-') as work:
    torch.manual_seed(7)
    small = TinyLM(ModelConfig(width=8, layers=1, heads=1, max_length=16))
    path = Path(work) / 'test.pt'
    save_checkpoint(path, small, step=0, metadata={'review_scope': 'untrained save/load only'})
    loaded, payload = load_checkpoint(path, 'cpu')
    states_equal = all(torch.equal(value, loaded.state_dict()[key]) for key,value in small.state_dict().items())
    assert states_equal and payload['step'] == 0

report_path = repo/'docs/course-experiments/results/sft.json'
report = json.loads(report_path.read_text())
result = report['results']
release_path = repo/'docs/course-experiments/public-releases/sft.json'
release = json.loads(release_path.read_text())
public = release['release']['public_manifest']
files = {entry['output']: entry for entry in public['files']}
card_path = repo/'checkpoints/course/sft/README.md'
card_hash = sha(card_path)
assert card_hash == files['README.md']['sha256']
local_model = repo/'checkpoints/course/sft/model.pt'
model_hash = sha(local_model)
assert model_hash == files['model.pt']['sha256']
weights = torch.load(local_model, map_location='cpu', weights_only=True)
assert weights['config']['width'] == 64 and weights['config']['layers'] == 2 and weights['config']['heads'] == 1
assert Context('cpu', Path('/tmp'), repo/'checkpoints/course', Path('/tmp')).dependency('sft') == local_model
assert result['training']['steps'] == 900 and result['training']['checkpoint'] == 'model.pt'
rows = []
for row,sample in enumerate(result['after']['test']['samples']):
    ids = sample['generated_ids']
    content = ids[:ids.index(tok.eos_id)] if tok.eos_id in ids else ids
    expected_ids = tok.encode(sample['expected'])
    exact = content == expected_ids
    eos = tok.eos_id in ids
    assert exact == sample['exact'] and eos == sample['eos']
    rows.append({'row':row, 'question':sample['messages'], 'expected':sample['expected'], 'expected_ids':expected_ids, 'generated':sample['generated'], 'generated_ids':ids, 'recomputed_exact':exact, 'recomputed_eos':eos})
correct = sum(row['recomputed_exact'] for row in rows)
assert correct == result['after']['test']['matches'] == 5
assert len(rows) == result['after']['test']['records'] == 10
assert result['after']['test']['effective_tokens'] == sum(len(tok.encode(s['expected'])) + 1 for s in result['after']['test']['samples']) == 69
assert correct / len(rows) == result['after']['test']['exact_match'] == 0.5
pretrained = result['pretrain_then_sft']
assert pretrained['pretraining']['steps'] == 250 and pretrained['sft']['steps'] == 900
assert pretrained['sft']['checkpoint'] == 'pretrain-sft.pt'
assert sum(s['exact'] for s in pretrained['after_sft']['test']['samples']) == 7
artifact_model = next(a for a in report['artifacts'] if a['path']=='model.pt')

 'software':{'byte_tokenizer_input':text,'byte_tokenizer_ids':encoded,'roundtrip':tok.decode(encoded),'temperature_zero_ids':output.tolist(),'softmax_probabilities':probabilities,'untrained_checkpoint_save_load_equal':states_equal,'checkpoint_updates':0},
 'frozen_evidence':{

  'pretrain_then_sft':{'pretrain_steps':pretrained['pretraining']['steps'],'sft_steps':pretrained['sft']['steps'],'checkpoint':pretrained['sft']['checkpoint'],'test_matches':pretrained['after_sft']['test']['matches'],'test_records':pretrained['after_sft']['test']['records']},

}
