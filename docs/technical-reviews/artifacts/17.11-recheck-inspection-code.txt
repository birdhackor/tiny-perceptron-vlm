import pathlib, json, hashlib, re, torch, sys, os
root = pathlib.Path("docs/technical-reviews/artifacts")
oldreport = json.loads((root / "17.11-prior-report-b12e02137b5da00d8a5e666df49ee2d7f57ac160f4f85e7bc9a1892ac79de5b1.json").read_bytes())
print("ENVIRONMENT", json.dumps({"python":sys.version,"torch":torch.__version__,"torch_git":torch.version.git_version,"device":"cpu","OMP_NUM_THREADS":os.environ.get("OMP_NUM_THREADS"),"MKL_NUM_THREADS":os.environ.get("MKL_NUM_THREADS")},ensure_ascii=False))
for artifact in oldreport["artifacts"]:
    assert hashlib.sha256(pathlib.Path(artifact["path"]).read_bytes()).hexdigest() == artifact["sha256"]
for source in oldreport["sources"]:
    if source["kind"] == "repository_code":
        assert hashlib.sha256(pathlib.Path(source["path"]).read_bytes()).hexdigest() == source["sha256"]
old = (root / "17.11-source-raw.txt").read_bytes()

assert new == old.replace("而且特徵每次輸入都不同".encode(), "而且特徵取決於輸入，可能隨輸入改變".encode(), 1)
assert re.search(rb"```python\n(.*?)```", old, re.S).group(1) == re.search(rb"```python\n(.*?)```", new, re.S).group(1) == (root / "17.11-original-code.txt").read_bytes()
linear = pathlib.Path(".venv/lib/python3.13/site-packages/torch/nn/modules/linear.py")
assert hashlib.sha256(linear.read_bytes()).hexdigest() == "1961ad1a1f650e104b3c7f78d19af2d21bf977e7af6c3fdf2420aad563c9e202"
w = torch.tensor([[0.7, 0.2], [-0.7, 1.2]])

assert not torch.equal(a, b) and torch.equal(a, repeat)
