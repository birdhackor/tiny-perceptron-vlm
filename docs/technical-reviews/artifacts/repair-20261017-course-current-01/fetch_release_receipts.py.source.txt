import datetime,hashlib,json,urllib.request
from course_io import W,note
base='https://huggingface.co/birdhackor/tiny-perceptron-course-models/resolve/'
jobs=[(f'public-{p}-training',base+'2c6a1062f3e66a602cf83eda6c703456ec18bca0/lesson-products/v1-20261016/'+p+'/training-receipt.json','GET') for p in ['b','c']]
jobs += [(f'public-v2-{a}-weight-head',base+'979cdfacc588ad0536f1c64fff96f264571cf054/selftrained/v2/'+a+'-joint/model.safetensors','HEAD') for a in ['moe','dense']]
out=[]
for name,url,method in jobs:
    r={'name':name,'url':url,'method':method,'started_at':datetime.datetime.now(datetime.timezone.utc).isoformat()}
    try:
        with urllib.request.urlopen(urllib.request.Request(url,method=method,headers={'User-Agent':'Mozilla/5.0'}),timeout=20) as h:
            r.update(status=h.status,headers={k:h.headers.get(k) for k in ['content-length','etag','x-linked-etag','x-repo-commit']})
            if method=='GET':
                b=h.read(100000)
                if h.read(1):raise ValueError('Bound exceeded')
                json.loads(b);(W/'primary'/f'{name}.json').write_bytes(b)
                r.update(bytes=len(b),sha256=hashlib.sha256(b).hexdigest())
            else:r.update(body_bytes_read=0)
    except Exception as e:r['error']=type(e).__name__+': '+str(e)
    out.append(r);print(json.dumps(r,ensure_ascii=False))
(W/'release-receipts-fetch.json').write_text(json.dumps(out,ensure_ascii=False,indent=2)+'\n')
note('release_receipts_fetch',receipt='release-receipts-fetch.json',scope='B/C small original training receipts; v2 weight HEAD only. No weight content downloaded, no model inference.')
