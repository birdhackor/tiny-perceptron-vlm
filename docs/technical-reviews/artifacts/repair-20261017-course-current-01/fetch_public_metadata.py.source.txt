from pathlib import Path
import datetime, hashlib, json, urllib.request
from course_io import W, note

repo = 'https://huggingface.co/birdhackor/tiny-perceptron-course-models/resolve/'
v2 = '979cdfacc588ad0536f1c64fff96f264571cf054'
products = '2c6a1062f3e66a602cf83eda6c703456ec18bca0'
requests = [
    ('public-v2-moe-config', repo+v2+'/selftrained/v2/moe-joint/model-config.json'),
    ('public-v2-dense-config', repo+v2+'/selftrained/v2/dense-joint/model-config.json'),
    ('public-b-manifest', repo+products+'/lesson-products/v1-20261016/b/artifact-manifest.json'),
    ('public-c-manifest', repo+products+'/lesson-products/v1-20261016/c/artifact-manifest.json'),
]
receipts=[]
for name,url in requests:
    receipt={'name':name,'url':url,'started_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'method':'unauthenticated bounded HTTP GET; metadata only, no model weights'}
    try:
        with urllib.request.urlopen(urllib.request.Request(url,headers={'User-Agent':'Mozilla/5.0'}),timeout=20) as response:
            data=response.read(100000)
            if response.read(1): raise ValueError('metadata exceeded 100000 byte bound')
            value=json.loads(data)
            (W/'primary'/f'{name}.json').write_bytes(data)
            receipt.update(status=response.status,bytes=len(data),sha256=hashlib.sha256(data).hexdigest(),value=value)
    except Exception as e:
        receipt.update(error=type(e).__name__+': '+str(e))
    receipts.append(receipt)
    print(json.dumps(receipt,ensure_ascii=False))
(W/'public-metadata-fetch.json').write_text(json.dumps(receipts,ensure_ascii=False,indent=2)+'\n')
note('public_metadata_fetch',receipt='public-metadata-fetch.json',requests=len(receipts),scope='Original immutable public metadata fetched; no weight download or inference. Content requires next semantic read.')
