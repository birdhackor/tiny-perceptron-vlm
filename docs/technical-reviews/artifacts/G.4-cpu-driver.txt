import contextlib
import hashlib
import io
import json
import platform
import re
import sys
from datetime import UTC, datetime
from pathlib import Path

ROOT = Path.cwd()
sys.path.insert(0, str(ROOT))
import torch

torch.set_num_threads(1)
torch.set_default_device("cpu")
sections = [
    ("15", "15.1"), ("15", "15.4"), ("16", "16.3"), ("16", "16.9"),
    ("17", "17.2"), ("18", "18.1"), ("09", "9.1"), ("09", "9.4"),
    ("0A", "A.2"), ("0A", "A.3"), ("0B", "B.1"), ("0B", "B.2"),
]

def section(path, lesson):
    raw = path.read_bytes()
    headings = list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+", raw))
    i = next(i for i, h in enumerate(headings) if h[1].decode() == lesson)
    end = headings[i + 1].start() if i + 1 < len(headings) else len(raw)
    return raw[headings[i].start():end]

def variants(lesson, code):
    if lesson == "15.1":
        return [("重複三遍", code.replace("(x, torch.cat([x, x], dim=0))", "(torch.cat([x, x, x], dim=0),)"))]
    if lesson == "15.4":
        changed = code.replace("[[0.1, 0.7, 0.2]", "[[0.1, 0.2, 0.7]")
        return [("第一列換排序", changed), ("全部三位", changed.replace("(2, 1)", "(3,)"))]
    if lesson == "16.3":
        return [("只換最後ID為5", code.replace("[[1, 2, 3, 4]]", "[[1, 2, 3, 5]]")),
                ("第一ID換5並重建prefix", code.replace("[[1, 2, 3, 4]]", "[[5, 2, 3, 4]]"))]
    if lesson == "16.9":
        return [("每塊一項", code.replace("range(0, 3, 2)", "range(0, 3, 1)").replace("start + 2", "start + 1"))]
    if lesson == "17.2":
        return [("scale=0.25", code.replace("scale = 0.5", "scale = 0.25"))]
    if lesson == "18.1":
        return [("只保存回答", code.replace('print("回答", answer, "分布", prob.tolist())', 'print("回答", answer)'))]
    if lesson == "9.1":
        return [("回答0安全=True", code.replace('"回答0安全": False', '"回答0安全": True'))]
    if lesson == "9.4":
        return [("第一筆改有效自己的公開測試場景", code.replace('{"owner": "他人", "permission": False}', '{"owner": "自己", "permission": True}'))]
    if lesson == "A.2":
        return [("地址改紅街9號", code.replace("青街8號", "紅街9號"))]
    if lesson == "A.3":
        return [("地址未公布仍命中", code.replace("書店地址是青街8號", "書店地址尚未公布"))]
    if lesson == "B.1":
        mismatch = code.replace('"user": "123乘45"', '"user": "123乘46"')
        return [("只改問題造成參數不一致", mismatch), ("b改46恢復一致", mismatch.replace('"b": 45', '"b": 46'))]
    if lesson == "B.2":
        return [("true改數字2", code.replace('"a":true', '"a":2'))]
    raise ValueError(lesson)

records = []
for chapter, lesson in sections:
    path = ROOT / "course" / "chapters" / (chapter + ".md")
    raw = section(path, lesson)
    fences = re.findall(rb"(?ms)^```python\n(.*?)^```", raw)
    assert len(fences) == 1, (lesson, len(fences))
    code = fences[0].decode()
    for role, label, active in [("original_example", "原文Python fence", code)] + [("exercise", label, modified) for label, modified in variants(lesson, code)]:
        stdout = io.StringIO()
        scope = {"__name__": "__main__"}
        with contextlib.redirect_stdout(stdout), torch.no_grad():
            exec(compile(active, f"{path}#{lesson}:{label}", "exec"), scope)
        records.append({"source": str(path.relative_to(ROOT)) + "#" + lesson,
                        "section_sha256": hashlib.sha256(raw).hexdigest(),
                        "original_fence_sha256": hashlib.sha256(fences[0]).hexdigest(),
                        "role": role, "label": label, "executed_code": active,
                        "executed_code_sha256": hashlib.sha256(active.encode()).hexdigest(),
                        "exit_code": 0, "stdout": stdout.getvalue()})
environment = {"python": platform.python_version(), "torch": str(torch.__version__),
               "torch_git_version": str(torch.version.git_version), "device": "cpu",
               "cuda_available": str(torch.cuda.is_available()), "num_threads": str(torch.get_num_threads()),
               "execution_scope": "既有短CPU教學fence與其練習；未執行正式實驗bash、模型訓練或GPU。"}
result = {"executed_at_utc": datetime.now(UTC).isoformat(), "environment": environment,
          "primary": {"source": "course/glossary.md#G.4", "python_fences": 0, "executable_exercise": False,
                      "exercise_note": "原節只有選問題及解釋方法的文字練習；選模型檔案太大，17.2的float到int8例子作起點，資訊會近似且仍需scale，沒有主節執行可冒稱。"},
          "records": records}
(ROOT / "docs/technical-reviews/artifacts/G.4-cpu-results.json").write_text(json.dumps(result, ensure_ascii=False, indent=2) + "\n")
print(json.dumps({"environment": environment, "examples": sum(r["role"] == "original_example" for r in records),
                  "exercises": sum(r["role"] == "exercise" for r in records),
                  "results": [{k: r[k] for k in ("source", "role", "label", "stdout")} for r in records]}, ensure_ascii=False, indent=2))
