import json
import platform
import sys
import torch

def offline_guard(event, arguments):
    if event.startswith('socket.') or event in ('subprocess.Popen', 'os.system'):
        raise RuntimeError('Offline bounded review rejects network or child process')

sys.addaudithook(offline_guard)
torch.set_num_threads(1)
torch.set_default_device('cpu')
print('ENVIRONMENT', json.dumps({'python': platform.python_version(), 'torch': str(torch.__version__), 'device': 'cpu', 'cuda_build': str(torch.version.cuda)}, ensure_ascii=False))
case = sys.argv[1]
short = '答案是4。'
if case == 'repeat2':
    long = short * 2
elif case == 'explanation':
    long = '答案是4；兩組各兩個物件合起來共有四個。'
elif case == 'wrong_explanation':
    long = '答案是5；兩組各兩個物件合起來共有五個。'
elif case == 'missing_period':
    long = '答案是4；兩組各兩個物件合起來共有四個'
elif case == 'comma_variation':
    long = '答案是4。答案，是4。'
elif case == 'paraphrase':
    long = '答案是4。結果是4。'
else:
    raise ValueError(case)
candidates = [short, long]
lengths = [len(text) for text in candidates]
chosen = max(candidates, key=len)
print('CASE', case)
print('字元數', lengths)
print('長度裁判選長版', chosen == long)
print('不同句子數', len(set(long.split('。')[:-1])))
print('SPLIT', long.split('。'))
