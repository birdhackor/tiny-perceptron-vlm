def sha(data):
    return hashlib.sha256(data).hexdigest()


def section(path, identifier):
    raw = (ROOT / path).read_bytes()
    headings = list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+", raw))
    index = next(i for i, h in enumerate(headings) if h[1].decode() == identifier)
    end = headings[index + 1].start() if index + 1 < len(headings) else len(raw)
    return raw[headings[index].start():end]


def run(label, code=None, args=None, cwd=ROOT, scope=""):
    command = [str(PYTHON), "-c", code] if code is not None else [str(PYTHON), *args]
    process = subprocess.run(command, cwd=cwd, env=ENV, capture_output=True, text=True, timeout=45)
    return {"label": label, "command": command, "cwd": str(cwd), "code": code, "code_sha256": sha(code.encode()) if code is not None else None, "returncode": process.returncode, "stdout": process.stdout, "stderr": process.stderr, "environment": environment, "scope": scope}
