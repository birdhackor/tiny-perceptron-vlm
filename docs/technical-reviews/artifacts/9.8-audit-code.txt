tok = ByteTokenizer()
sample_audits = []


def audit_samples(label, report):
    samples = report["samples"]
    matches = 0
    ended = 0
    for sample in samples:
        ids = sample["generated_ids"]
        eos = tok.eos_id in ids
        raw = ids[:ids.index(tok.eos_id)] if eos else ids
        exact = raw == tok.encode(sample["expected"])
        assert exact == sample["exact"] and eos == sample["eos"]
        assert tok.decode(raw) == sample["generated"]
        matches += exact
        ended += eos
    assert matches == report["matches"]
    assert len(samples) == report["records"] == report["examples"]
    assert matches / len(samples) == report["exact_match"]
    assert ended / len(samples) == report["eos_rate"]
    sample_audits.append({"path": label, "records": len(samples), "matches": matches, "eos": ended, "effective_tokens": report["effective_tokens"]})
