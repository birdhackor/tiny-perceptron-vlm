def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()
def persist(name, value):
    (out / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
def parameter_hash(state):
    h = hashlib.sha256()
    for name, tensor in state.items():
        h.update(name.encode())
        h.update(tensor.detach().cpu().contiguous().numpy().tobytes())
    return h.hexdigest()

    states = []
    for method in ['ce', 'teacher_hard']:
        filename = f'sft-w{width}-{method}.pt'
        path = directory / filename
        assert sha(path) == manifest[filename]['sha256']
        payload = torch.load(path, map_location='cpu', weights_only=True)
        states.append(payload['model'])
        h = parameter_hash(payload['model'])
        assert h == compared['final_sha256']
        proof['weights'][filename] = {'sha256': sha(path), 'parameter_hash': h,
            'tensor_count': len(payload['model']), 'config': payload['config'],
            'url': f"https://huggingface.co/{public['repo']}/resolve/{public['weights_revision']}/course/course-v1/distillation/{filename}"}

    proof['tasks']['attributes'][f'width{width}_CE_hard_equality'] = {'training': compared, 'scores': scores}

messages = [{'role': 'user', 'content': '2+2=?'}, {'role': 'assistant', 'content': '4'}]
x, y = render_chat(messages)
logits = torch.zeros((1, len(y), 264), requires_grad=True)
loss = F.cross_entropy(logits.reshape(-1, 264), y, ignore_index=-100, reduction='sum') / (y != -100).sum()
loss.backward()
assert y[y != -100].tolist() == [60, 2]
assert logits.grad[0, y == -100].abs().sum() == 0

      'loss', loss.item(), 'masked_direct_gradient', logits.grad[0, y == -100].abs().sum().item())
