import json
import math
import platform

import torch

from tiny_perceptron.model import masked_loss
from tiny_perceptron.training import learning_rate

assert torch.version.cuda is None
torch.set_num_threads(1)
torch.set_default_device("cpu")
print("environment", json.dumps({"python": platform.python_version(), "torch": str(torch.__version__), "torch_git_version": str(torch.version.git_version), "device": "cpu", "dtype": str(torch.get_default_dtype()), "threads": str(torch.get_num_threads())}))

# 原例僅把起點 1.0 改成練習要求的 4.0。
import torch

w = torch.tensor(4.0)
gradient = 2 * (w - 3)
print("起點代價", (w - 3).square().item())
for lr in [0.01, 0.1, 2.0]:
    new = w - lr * gradient
    print("步幅", lr, "新位置", new.item(), "新代價", (new - 3).square().item())

exercise = []
for lr, expected_position, expected_cost in [(0.01, 3.98, 0.9604), (0.1, 3.8, 0.64), (2.0, 0.0, 9.0)]:
    new = w - lr * gradient
    cost = (new - 3).square().item()
    assert math.isclose(new.item(), expected_position, abs_tol=1e-6)
    assert math.isclose(cost, expected_cost, abs_tol=1e-6)
    exercise.append({"lr": lr, "position": new.item(), "cost": cost})
assert w.item() == 4.0
assert gradient.item() == 2.0
print("exercise", json.dumps(exercise))

ratios = {}
for start in [1.0, 4.0]:
    value = torch.tensor(start)
    grad = 2 * (value - 3)
    ratios[str(start)] = [((value - lr * grad - 3).square() / (value - 3).square()).item() for lr in [0.01, 0.1, 2.0]]
    assert (value - 0.5 * grad).item() == 3.0
    for ratio, expected in zip(ratios[str(start)], [0.9604, 0.64, 9.0]):
        assert math.isclose(ratio, expected, abs_tol=1e-6)
print("ratios", json.dumps(ratios))
print("lr_0.5_both_starts", True)

# 只用單一平方函數做一個 SGD 更新，核對參數名單與 step 診斷。
p = torch.nn.Parameter(torch.tensor(1.0))
q = torch.nn.Parameter(torch.tensor(1.0))
wrong = torch.optim.SGD([q], lr=0.1)
(p - 3).square().backward()
before = p.item()
wrong.step()
assert p.item() == before and q.item() == 1.0
right = torch.optim.SGD([p], lr=0.1)
assert p.item() == before
right.step()
assert math.isclose(p.item(), 1.4, abs_tol=1e-6)
print("parameter_step_probe", json.dumps({"before": before, "wrong_optimizer_unchanged": True, "without_step_unchanged": True, "after_right_step": p.item()}))

try:
    masked_loss(torch.zeros(1, 3, 5), torch.full((1, 3), -100, dtype=torch.long))
except ValueError as error:
    print("all_ignored", str(error))
else:
    raise AssertionError("空答案必須拒絕")

zero = torch.tensor(0.0)
assert torch.isnan(zero / zero).item()
assert not torch.isfinite(torch.tensor(1.0) / zero).item()
x = torch.tensor([1.0, 1.0], requires_grad=True)
div = torch.tensor([0.0, 1.0])
y = x / div
y[div != 0].sum().backward()
assert torch.isnan(x.grad[0]).item() and x.grad[1].item() == 1.0
print("nonfinite_probe", json.dumps({"zero_over_zero_is_nan": True, "one_over_zero_is_nonfinite": True, "masked_zero_division_gradient": ["nan", x.grad[1].item()]}))

rates = [learning_rate(step, 40, peak=0.001) for step in [0, 1, 9, 10, 20, 39]]
assert rates[0] < rates[1] < rates[2] == rates[3]
assert rates[3] > rates[4] > rates[5]
print("schedule_probe", json.dumps({"steps": [0, 1, 9, 10, 20, 39], "rates": rates}))
print("all_checks_passed", True)
