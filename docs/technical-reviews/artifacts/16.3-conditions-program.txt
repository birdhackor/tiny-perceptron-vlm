import torch
from tiny_perceptron.model import TinyLM, ModelConfig
from tiny_perceptron.data import ByteTokenizer
torch.manual_seed(0)
model=TinyLM(ModelConfig(width=8)).eval()
ids=torch.tensor([[1,2,3,4]])
with torch.no_grad():
    prefix=model(ids[:,:3]); full=model(ids)["logits"][:,-1]
    wrong=model(ids[:,3:],cache=prefix["cache"],positions=torch.tensor([0]))["logits"][:,0]
    print("錯誤offset最大差異",(full-wrong).abs().max().item())
    assert not torch.allclose(full,wrong,atol=1e-6)
    longer=model(ids)
    for a,b in zip(prefix["cache"][0],longer["cache"][0]):
        assert torch.allclose(a,b[:,:,:3],atol=1e-6)
    print("舊位置KV不受新位置影響",True)
    padded=torch.tensor([[1,2,3],[4,5,0]]); valid=padded.ne(0)
    padded_prefix=model(padded,valid=valid)
    try:
        model(torch.tensor([[4],[6]]),cache=padded_prefix["cache"],valid=torch.ones(2,4,dtype=torch.bool))
    except ValueError as e:
        print("padding cache限制",str(e))
    else:
        raise AssertionError("應拒絕帶valid的cache路徑")
    print("dtype",str(full.dtype),"bytes per value",full.element_size())
    assert full.dtype==torch.float32 and full.element_size()==4
tok=ByteTokenizer();print("含BOS提示位置",len([tok.bos_id]+tok.encode("Once upon a time, a girl")))
a=torch.tensor([1.,1.0000001192092896]); b=a.flip(0)
print("近分數最大差異",(a-b).abs().max().item(),"argmax",a.argmax().item(),b.argmax().item())
assert torch.allclose(a,b,atol=1e-6) and a.argmax()!=b.argmax()
