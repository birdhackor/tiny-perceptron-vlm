from pathlib import Path
import json, hashlib, html, sys, urllib.request, platform, threading, functools
from http.server import ThreadingHTTPServer, SimpleHTTPRequestHandler
from html.parser import HTMLParser
import torch
from scripts.export_course import checked_notebook

site=Path(sys.argv[1]); root=Path.cwd()
index=json.loads((root/'course/lesson-index.json').read_bytes())
info=json.loads((site/'build-info.json').read_bytes())
class Links(HTMLParser):
    def __init__(self): super().__init__(); self.links=[]
    def handle_starttag(self,tag,attrs):
        attrs=dict(attrs)
        if tag=='a' and 'href' in attrs: self.links.append(attrs['href'])
records=[]; failures=[]; execution_cells=0; stream_outputs=0; matched_stream_outputs=0
for item in index:
    original=root/item['notebook']; download=site/item['notebook']; page=site/(item['id']+'.html')
    executed=root/'outputs/notebooks-reader-round'/Path(item['notebook']).relative_to('notebooks')
    notebook=json.loads(original.read_bytes())
    run=checked_notebook(notebook,executed)
    text=page.read_text(); links=Links(); links.feed(text)
    visible=[c for c in run['cells'] if c['cell_type']=='code' and not c.get('metadata',{}).get('course_setup') and 'course_figure' not in c.get('metadata',{})]
    record={'id':item['id'],'original_sha256':hashlib.sha256(original.read_bytes()).hexdigest(),'executed_sha256':hashlib.sha256(executed.read_bytes()).hexdigest(),'page_sha256':hashlib.sha256(page.read_bytes()).hexdigest(),'download_equal':original.read_bytes()==download.read_bytes(),'colab_same_revision':f'https://colab.research.google.com/github/birdhackor/tiny-perceptron-vlm/blob/{info["revision"]}/{item["notebook"]}' in links.links,'download_link':item['notebook'] in links.links,'outputs_label_count':text.count('實際執行結果 · CPU'),'warm_nav':any('first-steps.html' in link for link in links.links)}
    execution_cells+=sum(c['cell_type']=='code' for c in run['cells'])
    record['checked_current_cell_sources']=True
    record['stream_outputs']=0; record['matched_stream_outputs']=0
    for cell in visible:
        for output in cell.get('outputs',[]):
            if output['output_type']=='stream':
                stream_outputs+=1; record['stream_outputs']+=1
                joined=''.join(output['text']) if isinstance(output['text'],list) else output['text']
                matched='<pre>'+html.escape(joined)+'</pre>' in text
                matched_stream_outputs+=int(matched);record['matched_stream_outputs']+=int(matched)
    if not all(record[key] for key in ['download_equal','colab_same_revision','download_link','warm_nav']): failures.append(item['id'])
    if record['stream_outputs']!=record['matched_stream_outputs']: failures.append(item['id']+': stream mismatch')
    records.append(record)
all_pages=[]
for page in site.glob('*.html'):
    links=Links();links.feed(page.read_text())
    all_pages.append({'name':page.name,'warm_nav':any('first-steps.html' in link for link in links.links)})

class QuietHandler(SimpleHTTPRequestHandler):
    def log_message(self,*args): pass

thread=threading.Thread(target=server.serve_forever,daemon=True);thread.start()

    with urllib.request.urlopen(http_url,timeout=10) as response:
        response_bytes=response.read()
    http={'url':http_url,'status':response.status,'sha256':hashlib.sha256(response_bytes).hexdigest(),'matches_file':response_bytes==(site/'1.1.html').read_bytes()}
finally:
    server.shutdown();server.server_close();thread.join()

print(json.dumps({k:v for k,v in result.items() if k!='records'},ensure_ascii=False,indent=2))
