import torch, sys, json, os
from torch import nn
torch.set_num_threads(1)
assert torch.version.cuda is None
code = open("docs/technical-reviews/artifacts/2.1-original-code.txt").read()
ns = {"__name__": "__main__"}
exec(compile(code, "2.1-original-code.txt", "exec"), ns)
e = ns["embedding"]
original = e.weight.detach().clone()
x = ns["x"]
coeff = torch.tensor([1.0, 2.0, 3.0])
u = (x * coeff).sum()
loss = u.square()
loss.backward()
expected_grad = torch.tensor([[0.,0.,0.],[16.,32.,48.],[8.,16.,24.],[0.,0.,0.],[0.,0.,0.]])
assert torch.equal(e.weight.grad, expected_grad)
print("gradient", e.weight.grad.tolist())
with torch.no_grad():
    e.weight -= 0.001 * e.weight.grad
assert torch.equal(e.weight[[0,3,4]], original[[0,3,4]])
assert not torch.equal(e.weight[[1,2]], original[[1,2]])
print("single_update", e.weight.tolist())
# API / default initializer check is not model training.
torch.manual_seed(17)
default = nn.Embedding(5,3)
torch.manual_seed(17)
expected_default = torch.empty(5,3).normal_()
assert torch.equal(default.weight, expected_default)
print("default_seed_17", default.weight.tolist())
print("default_matches_normal_", True)
print("default_abs_max", default.weight.abs().max().item())
print("parameter", e.weight.requires_grad, str(ns["ids"].dtype), str(x.dtype), str(x.device))
print("environment", json.dumps({"python":sys.version,"torch":str(torch.__version__),"torch_git_version":str(torch.version.git_version),"device":"cpu","omp_threads":os.environ.get("OMP_NUM_THREADS"),"mkl_threads":os.environ.get("MKL_NUM_THREADS")}, ensure_ascii=False))
