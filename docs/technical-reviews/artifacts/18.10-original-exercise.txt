runs = [
    {
        "method": "SFT",
        "student_params": 6104,
        "steps": 100,
        "tokens": 1000,
        "student_seconds": 10,
        "teacher_seconds": 0,
        "correct": 3,
        "total": 4,
    },
    {
        "method": "distill",
        "student_params": 6104,
        "steps": 100,
        "tokens": 1000,
        "student_seconds": 10,
        "teacher_seconds": 0,
        "correct": 4,
        "total": 4,
    },
]
for run in runs:
    accuracy = run["correct"] / run["total"]
    seconds = run["student_seconds"] + run["teacher_seconds"]
    print(
        run["method"],
        "參數",
        run["student_params"],
        "步數",
        run["steps"],
        "token",
        run["tokens"],
        "正確率",
        accuracy,
        "總秒數",
        seconds,
    )
