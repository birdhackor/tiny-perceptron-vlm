import sys
from pathlib import Path
sys.path.insert(0, str(Path.cwd()))
def offline_guard(event, arguments):
    if event in ("socket.connect", "socket.getaddrinfo", "socket.bind", "socket.sendto", "subprocess.Popen", "os.system", "os.exec", "os.posix_spawn"):
        raise PermissionError("review exercise: offline CPU guard")
sys.addaudithook(offline_guard)
import torch
from tiny_perceptron.quantization import quantize_symmetric

w = torch.tensor([[0.01, 0.03, 0.05, 0.08], [10.0, 30.0, 50.0, 80.0]])
for per_row in (False, True):
    q, scale = quantize_symmetric(w, bits=4, per_channel=per_row)
    restored = q.float() * scale
    error = (w - restored).abs().mean(-1)
    print("每列獨立", per_row, "整數", q.tolist())
    print("每列MAE", error.round(decimals=4).tolist(), "scale格數", scale.numel())

print("environment", sys.version, torch.__version__, torch.version.git_version, str(w.device))
for per_row in (False, True):
    q, scale = quantize_symmetric(w, bits=4, per_channel=per_row)
    restored = q.float() * scale
    print("details", per_row, "scale", scale.tolist(), "scale_shape", list(scale.shape), "scale_dtype", str(scale.dtype), "scale_bytes", scale.numel() * scale.element_size(), "q_dtype", str(q.dtype), "restored", restored.tolist(), "error_shape", list((w-restored).abs().mean(-1).shape))
    assert list(scale.shape) == ([2,1] if per_row else [])
    assert scale.numel() * scale.element_size() == (8 if per_row else 4)
    assert scale.dtype == torch.float32 and q.dtype == torch.int8
    assert q.tolist() == ([[1,3,4,7],[1,3,4,7]] if per_row else [[0,0,0,0],[1,3,4,7]])
    expected=torch.tensor([0.0025,2.5] if per_row else [0.0425,2.5])
    assert torch.allclose((w-restored).abs().mean(-1),expected,atol=1e-6,rtol=0)
assert torch.version.cuda is None and str(w.device) == "cpu"
