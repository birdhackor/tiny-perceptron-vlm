import json
import math
import platform
import torch

# 只核對新增定義的係數作用，不建立或更新模型。
task_error = 0.5
auxiliary = 2.4
cases = []
for coefficient in (0.0, 0.01):
    auxiliary_contribution = coefficient * auxiliary
    total = task_error + auxiliary_contribution
    cases.append({'coefficient': coefficient, 'task_error': task_error, 'auxiliary': auxiliary, 'auxiliary_contribution': auxiliary_contribution, 'total': total})
assert cases[0]['auxiliary_contribution'] == 0.0
assert cases[0]['total'] == task_error
assert math.isclose(cases[1]['total'], 0.524, rel_tol=0.0, abs_tol=1e-15)
print(json.dumps({
    'environment': {'python': platform.python_version(), 'torch': torch.__version__, 'torch_commit': torch.version.git_version, 'device': 'cpu', 'threads': str(torch.get_num_threads())},
    'formula': 'task_error + coefficient * auxiliary',
    'cases': cases,
    'scope': '有限算術probe，只支持係數如何乘輔助項，以及係數0的輔助貢獻為0。不是正式CUDA訓練，也沒有說係數0就不再計算或記錄輔助項。',
}, ensure_ascii=False, indent=2))
