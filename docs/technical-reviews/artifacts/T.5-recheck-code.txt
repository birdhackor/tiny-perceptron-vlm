sha = lambda raw: hashlib.sha256(raw).hexdigest()


def save_json(name, value):
    path = ART / name
    assert not path.exists(), path
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


def section(path, identifier):
    raw = path.read_bytes()
    headings = list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+", raw))
    index = next(i for i, heading in enumerate(headings) if heading[1].decode() == identifier)
    end = headings[index + 1].start() if index + 1 < len(headings) else len(raw)
    return raw[headings[index].start():end]


def fences(raw):
    return re.findall(rb"(?ms)^```[^\r\n]*\r?\n.*?^```[^\r\n]*(?:\r?\n|$)", raw)

def text_keys(value, prefix=""):
    found = []
    if isinstance(value, dict):
        for key, item in value.items():
            path = f"{prefix}.{key}" if prefix else key
            if key in PRIVATE_TEXT_KEYS:
                found.append(path)
            found.extend(text_keys(item, path))
    elif isinstance(value, list):
        for index, item in enumerate(value):
            found.extend(text_keys(item, f"{prefix}[{index}]"))
    return found
