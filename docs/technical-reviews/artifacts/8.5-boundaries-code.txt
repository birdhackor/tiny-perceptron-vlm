import json
texts = [' {"answer":4}', '\n\t{"answer":4}\r\n', '答案是：{"answer":4}', '{"answer":4} 完成', '4', '[]', '{"answer":4.0}', '{"answer":4', '{"answer":4,"note":"完成"}']
for text in texts:
    try:
        value = json.loads(text)
        valid = isinstance(value, dict) and set(value) == {"answer"}
        correct = valid and value["answer"] == 4
        print(repr(text), "first_char", repr(text[0]), "type", type(value).__name__, "格式", valid, "內容", correct)
    except json.JSONDecodeError as e:
        print(repr(text), type(e).__name__, "pos", e.pos, "格式", False, "內容", False)
