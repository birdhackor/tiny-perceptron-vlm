from pathlib import Path
import sys, json, tempfile

# Avoid dill's implicit filesystem probe while discovering the temp directory.
tempfile.tempdir = "/tmp"

# Review guard: no network, subprocess, or filesystem writes by lesson code.
def guard(event, args):
    if event.startswith("socket.") or event in {"subprocess.Popen", "os.system", "os.fork", "os.posix_spawn"}:
        raise RuntimeError("Review forbids network or subprocess: " + event)
    if event == "open":
        # dill probes Python file types through /dev/null during torch import.
        if args[0] == "/dev/null":
            return
        mode, flags = args[1], args[2]
        if (isinstance(mode, str) and any(c in mode for c in "wax+")) or (isinstance(flags, int) and flags & 3):
            raise RuntimeError("Review forbids filesystem writes")
sys.addaudithook(guard)
data = json.loads(Path("docs/technical-reviews/artifacts/W.1-notebook-code-cells.json").read_text())
exercise = sys.argv[1] == "exercise"
namespace = {}
for cell in data["code_cells"]:
    source = cell["source"]
    if exercise:
        source = source.replace('text = "貓看狗，狗看貓。"', 'text = "鳥看狗，狗看鳥。"', 1)
    exec(compile(source, "Notebook code cell " + str(cell["index"]), "exec"), namespace)
assert namespace["restored"] == ("鳥看狗，狗看鳥。" if exercise else "貓看狗，狗看貓。")
print("Completed local fresh-process Notebook code cells", [c["index"] for c in data["code_cells"]])
print("No Colab runtime or Jupyter server was started.")
