from pathlib import Path
import json, os, platform, shlex, subprocess, sys
import torch
out = Path("docs/technical-reviews/artifacts")
settings = {"OMP_NUM_THREADS": "1", "MKL_NUM_THREADS": "1", "CUDA_VISIBLE_DEVICES": "", "HF_HUB_OFFLINE": "1", "HF_DATASETS_OFFLINE": "1", "TRANSFORMERS_OFFLINE": "1", "PYTHONDONTWRITEBYTECODE": "1"}
environment = {"python": platform.python_version(), "python_build": sys.version, "python_executable": sys.executable, "torch": str(torch.__version__), "torch_git_version": str(torch.version.git_version), "device": "cpu", "cuda_build": str(torch.version.cuda), "cuda_available": str(torch.cuda.is_available()), "settings": settings, "cwd": str(Path.cwd())}
assert torch.version.cuda is None and not torch.cuda.is_available()
for label, filename in [("example", "A.3-example-code.txt"), ("exercise", "A.3-exercise-code.txt"), ("api", "A.3-api-code.txt"), ("formal", "A.3-formal-replay-code.txt")]:
    codepath = out / filename
    snippet = f"from pathlib import Path; exec(compile(Path({str(codepath)!r}).read_bytes(), {str(codepath)!r}, 'exec'))"
    command = [".venv/bin/python", "-c", snippet]
    child = subprocess.run(command, cwd=Path.cwd(), env={**os.environ, **settings, "PYTHONPATH": str(Path.cwd())}, capture_output=True, text=True, timeout=20)
    record = {"command": " ".join(f"{k}={shlex.quote(v)}" for k,v in settings.items()) + " " + shlex.join(command), "timeout_seconds": 20, "returncode": child.returncode, "environment": environment, "code_path": str(codepath), "stdout": child.stdout, "stderr": child.stderr, "result": f"當次實際 child exit {child.returncode}"}
    (out / f"A.3-{label}-run.json").write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n")
    print(label, child.returncode, child.stdout)
    assert child.returncode == 0
