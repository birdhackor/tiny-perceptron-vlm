import sys, json, platform, io, contextlib
import torch
log = io.StringIO()
with contextlib.redirect_stdout(log):
    families = {
        "train": ["盒子沒有數量資訊，請問有幾球？"],
        "test": ["盒子還沒打開，你能確定裡面有幾顆嗎？"],
    }
    expected = "資訊不足，請提供數量或可看清的圖片。"
    families["test"].append("只看外觀能知道盒內球數嗎")
    for split, prompts in families.items():
        for prompt in prompts:
            print(split, prompt, "→", expected)
    print("訓練問句與測試問句相同", families["train"][0] == families["test"][0])

assert len(families["train"]) == 1 and len(families["test"]) == 2
assert families["test"][-1] not in families["train"]
assert expected == "資訊不足，請提供數量或可看清的圖片。"
print(json.dumps({"environment":{"python":platform.python_version(),"torch":torch.__version__,"device":"cpu","OMP_NUM_THREADS":"1","MKL_NUM_THREADS":"1"},"stdout":log.getvalue(),"train_records":len(families["train"]),"test_records":len(families["test"]),"new_test_not_in_train":families["test"][-1] not in families["train"],"scope":"人工目標練習，無模型推論或更新"},ensure_ascii=False,indent=2))
