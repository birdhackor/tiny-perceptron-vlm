import json
import platform
import torch

answers = [
    '{"answer":5,"explanation":"5"}',
    '{"answer":5,"explanation":"5，像把空盒與裝五塊積木的盒子合起來，共有五塊。"}',
]
print('environment', platform.python_version(), str(torch.__version__), 'cpu')
for text in answers:
    parsed = json.loads(text)
    print({'answer_correct': type(parsed['answer']) is int and parsed['answer'] == 0 + 5, 'json_fields_correct': set(parsed) == {'answer', 'explanation'}, 'metaphor_present': '像把空盒' in parsed['explanation']})
print('manual_metaphor_check: empty box represents 0; five blocks represents 5; combining gives 5. Only the second explanation gives this mapping.')
