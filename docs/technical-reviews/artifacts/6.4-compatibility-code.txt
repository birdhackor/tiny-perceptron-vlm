import json
import platform
import torch
from tiny_perceptron.model import TinyLM, ModelConfig

assert torch.version.cuda is None
assert torch.get_num_threads() == 1
model = TinyLM(ModelConfig(vocab_size=20, width=8))
old_shape = list(model.embedding.weight.shape)
model.config.vocab_size = 25
assert list(model.embedding.weight.shape) == [20, 8]
assert list(model.output.weight.shape) == [20, 8]
with torch.no_grad():
    unchanged_logits = list(model(torch.tensor([[1, 2, 3]]))['logits'].shape)
    try:
        model(torch.tensor([[24]]))
    except IndexError as exc:
        out_of_range = type(exc).__name__
    else:
        raise AssertionError('config 欄位改寫不應擴大既有 embedding')
    rebuilt = TinyLM(ModelConfig(vocab_size=25, width=8))
    rebuilt_shape = list(rebuilt(torch.tensor([[1, 2, 3]]))['logits'].shape)
assert unchanged_logits == [1, 3, 20]
assert rebuilt_shape == [1, 3, 25]
print(json.dumps({'python':platform.python_version(),'torch':str(torch.__version__),'torch_git_version':torch.version.git_version,'device':'cpu','config_after_edit':25,'old_embedding_shape':old_shape,'embedding_after_edit':list(model.embedding.weight.shape),'logits_after_edit':unchanged_logits,'id24_after_edit':out_of_range,'rebuilt_logits':rebuilt_shape,'scope':'只核配置欄位與已建立表格的關係；不訓練、不評估正式模型品質。'},ensure_ascii=False,indent=2))
