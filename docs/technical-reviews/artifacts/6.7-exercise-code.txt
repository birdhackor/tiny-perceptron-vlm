import contextlib, io, json, sys
from pathlib import Path
code = Path("docs/technical-reviews/artifacts/6.7-original-code.txt").read_text()
results = []
for boundary, expected in [(3, "貓"), (4, "貓�")]:
    variant = code.replace("raw[:1]", f"raw[:{boundary}]")
    capture = io.StringIO()
    namespace = {}
    with contextlib.redirect_stdout(capture):
        exec(compile(variant, f"6.7-exercise-boundary-{boundary}", "exec"), namespace)
    raw = namespace["raw"]
    observed = raw[:boundary].decode("utf-8", errors="replace")
    assert observed == expected
    assert raw == "貓🙂".encode()
    assert raw.decode() == "貓🙂"
    results.append({"slice":f"raw[:{boundary}]", "bytes":list(raw[:boundary]), "expected":expected, "observed":observed, "complete_decoded":raw.decode(), "stdout":capture.getvalue(), "assertions":"passed"})
print(json.dumps({"environment":{"python":sys.version,"device":"cpu","OMP_NUM_THREADS":"1","MKL_NUM_THREADS":"1"},"results":results},ensure_ascii=False,indent=2))
