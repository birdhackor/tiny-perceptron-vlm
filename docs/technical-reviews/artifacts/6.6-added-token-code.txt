from tokenizers import Tokenizer
from pathlib import Path
import hashlib
p=Path("checkpoints/course/tokenizer/tokenizer-bpe512.json")
t=Tokenizer.from_file(str(p))
text="a<assistant>b"
e=t.encode(text, add_special_tokens=False)
assert e.tokens == ["a", "<assistant>", "b"]
assert e.ids[1] == 4 and len(e.ids) == 3
assert t.decode(e.ids, skip_special_tokens=False) == text
print({"text":text,"tokens":e.tokens,"ids":e.ids,"restored":t.decode(e.ids,skip_special_tokens=False),"tokenizer_sha256":hashlib.sha256(p.read_bytes()).hexdigest()})
