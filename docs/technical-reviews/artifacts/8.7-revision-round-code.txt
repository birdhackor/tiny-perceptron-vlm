import json,platform,torch
print("environment",json.dumps({"python":platform.python_version(),"torch":torch.__version__,"torch_git_version":torch.version.git_version,"device":"cpu","threads":torch.get_num_threads()}))
print("inline", "round(1.25,1)",round(1.25,1), "input_exact_ratio",1.25.as_integer_ratio())
assert round(1.25,1)==1.2
print("own_original_counterexamples", "round(0.5)",round(0.5),"round(1.5)",round(1.5))
assert round(0.5)==0 and round(1.5)==2
print("ties_last_decimal_even", "round(1.75,1)",round(1.75,1))
assert round(1.75,1)==1.8
print("ordinary_nearest", "round(1.375,1)",round(1.375,1))
assert round(1.375,1)==1.4
