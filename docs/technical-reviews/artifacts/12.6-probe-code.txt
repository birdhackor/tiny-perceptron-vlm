import sys, platform, json, math, torch
from tiny_perceptron.multimodal import mel_filter_bank, tone
print(json.dumps({"python":sys.version.split()[0],"torch":torch.__version__,"device":"cpu","platform":platform.platform(),"torch_threads":torch.get_num_threads()}))
vals=[2595*math.log10(1+f/700) for f in [0,700,2100]]
print("mel_formula",json.dumps(vals))
print("log10",json.dumps([math.log10(x) for x in [1,10,100,2,4]]))
assert max(abs(a-b) for a,b in zip(vals,[0,781.1728387480312,1562.3456774960624]))<1e-10
w=torch.tensor([[1.,.5,0.],[0.,.5,1.]])
p=torch.tensor([4.,2.,1.])
print("toy",(w@p).tolist(),"contributions",(w*p).tolist())
assert (w@p).tolist()==[5.,2.]
for bands in [16,32]:
 b=mel_filter_bank(bands=bands)
 hz=700*(10**(torch.linspace(0,2595*math.log10(1+8000/700),bands+2,dtype=torch.float64)/2595)-1)
 delta=hz.diff()
 print("bank",bands,"shape",list(b.shape),"row_sums",b.sum(1).tolist(),"centres_hz",hz[1:-1].tolist(),"boundary_gaps_hz",delta.tolist(),"column_sum_minmax",[b.sum(0).min().item(),b.sum(0).max().item()],"overlap_bins",int((torch.count_nonzero(b,dim=0)>1).sum()),"rank",int(torch.linalg.matrix_rank(b)))
 assert b.shape==(bands,201) and bool((b>=0).all())
 assert bool((delta[1:]>delta[:-1]).all())
 assert not torch.allclose(b.sum(1),torch.ones(bands))
 try: b@torch.ones(3,201)
 except RuntimeError as e: print("wrong_axes",bands,type(e).__name__,str(e))
 else: raise AssertionError("shape mismatch did not fail")
wave=tone(440.,seconds=.2)
power=torch.stft(wave,400,160,window=torch.hann_window(400),return_complex=True).abs().square()
print("prerequisite_12.5",list(power.shape),"peak_bin",int(power.mean(-1).argmax()),"peak_hz",int(power.mean(-1).argmax())*16000/400)
assert power.shape==(201,21) and int(power.mean(-1).argmax())==11
recipes=torch.tensor([[2.,1.,0.],[1.,2.,1.]])
weights=torch.tensor([[10.,0.],[20.,1.],[30.,4.]])
print("prerequisite_W.5_matmul",(recipes@weights).tolist())
assert (recipes@weights).tolist()==[[40.,1.],[80.,6.]]
