def preference_audit(value):
    rows = value["samples"]
    count = len(rows)
    absolute = sum(r["policy_chosen_logp"] > r["policy_rejected_logp"] for r in rows)
    relative = sum(r["policy_chosen_logp"] - r["policy_rejected_logp"] - r["reference_margin"] > 0 for r in rows)
    for r in rows:
        assert abs(r["policy_margin"] - (r["policy_chosen_logp"] - r["policy_rejected_logp"])) < 1e-9
        assert abs(r["relative_margin"] - (r["policy_margin"] - r["reference_margin"])) < 1e-9
    assert (count, absolute, relative) == (value["records"], value["chosen_higher_absolute_probability"], value["relative_preference_improved"])
    return {"records": count, "chosen_higher_absolute_probability": absolute, "relative_preference_improved": relative, "mean_relative_margin": value["mean_relative_margin"]}

def arithmetic_audit(value):
    rows = value["samples"]
    for row in rows:
        raw = row["generated_ids"]
        eos = 2 in raw
        body = raw[:raw.index(2)] if eos else raw
        assert row["exact"] == (body == ByteTokenizer().encode(row["expected"]))
        assert row["eos"] == eos
    matches = sum(r["exact"] for r in rows)
    ended = sum(r["eos"] for r in rows)
    assert matches == value["matches"] and len(rows) == value["records"]
    assert matches / len(rows) == value["exact_match"] and ended / len(rows) == value["eos_rate"]
    return {"records": len(rows), "matches": matches, "eos_count": ended, "samples": [{"prompt": r["messages"][0]["content"], "expected": r["expected"], "generated": r["generated"], "generated_ids": r["generated_ids"], "eos": r["eos"]} for r in rows]}

def function_digests(text):
    tree = ast.parse(text)
    return {n.name: hashlib.sha256(ast.get_source_segment(text, n).encode()).hexdigest() for n in tree.body if isinstance(n, (ast.FunctionDef, ast.AsyncFunctionDef))}
