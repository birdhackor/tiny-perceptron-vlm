raw = Path('course/chapters/05.md').read_bytes()
headings = list(re.finditer(rb'(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+', raw))
def extract(lesson):
    p = next(i for i, h in enumerate(headings) if h[1].decode() == lesson)
    end = headings[p+1].start() if p+1 < len(headings) else len(raw)
    return raw[headings[p].start():end]

# Independent two-domain cap and formula oracle; no parameter updates.
configurations = 0
rates_checked = 0
max_error = 0.0
examples = []
for total in range(1,129):
    cap = 1 if total <= 3 else total // 4
    if total <= 3:
        assert cap == 1
    else:
        assert cap <= total / 4
    for requested in (0,1,2,3,5,10,30,1000):
        effective = min(requested,cap)
        assert 0 <= effective <= cap
        for i in range(total):
            if i < effective:
                expected = 0.001*(i+1)/effective
            else:
                fraction = min(1.0,(i-effective)/max(1,total-effective))
                expected = 0.0001 + 0.00045*(1+math.cos(math.pi*fraction))
            actual = learning_rate(i,total,peak=0.001,warmup=requested)
            error = abs(actual-expected)
            assert error <= 1e-15
            max_error = max(max_error,error)
            rates_checked += 1
        configurations += 1
        if total in (1,2,3,4,5,40) and requested in (0,1,2,30):
            examples.append({'total':total,'requested_warmup':requested,'cap':cap,'effective_warmup':effective,'first_rate':learning_rate(0,total,peak=0.001,warmup=requested)})
