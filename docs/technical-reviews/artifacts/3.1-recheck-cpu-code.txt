from fractions import Fraction
import json
import sys
import torch

torch.set_num_threads(1)

# 本次新增的一欄算例：比的是正數加權項，不是模型最終預測。
exact_terms = [Fraction(9, 10) * 1, Fraction(1, 10) * 100]
assert exact_terms == [Fraction(9, 10), Fraction(10)]
assert exact_terms[1] > exact_terms[0]

results = []
for dtype in [torch.float32, torch.float64]:
    weights = torch.tensor([0.9, 0.1], dtype=dtype)
    values = torch.tensor([[1.0], [100.0]], dtype=dtype)
    terms = weights[:, None] * values
    mixed = weights @ values
    expected = torch.tensor([[0.9], [10.0]], dtype=dtype)
    assert weights[0] > weights[1]
    assert terms[1, 0] > terms[0, 0]
    assert torch.allclose(terms, expected, atol=1e-8, rtol=1e-5)
    assert torch.allclose(mixed, torch.tensor([10.9], dtype=dtype))
    results.append({
        'dtype': str(dtype), 'weights': weights.tolist(), 'values': values.tolist(),
        'weighted_terms': terms.tolist(), 'mixed_output': mixed.tolist(),
        'larger_coefficient_first': bool(weights[0] > weights[1]),
        'larger_positive_term_second': bool(terms[1, 0] > terms[0, 0])
    })

print(json.dumps({
    'environment': {'python': sys.version, 'torch': torch.__version__, 'torch_commit': torch.version.git_version, 'device': 'cpu', 'num_threads': str(torch.get_num_threads())},
    'exact_hand_derivation': {'terms': [str(t) for t in exact_terms], 'sum': str(sum(exact_terms))},
    'executed': results,
    'scope': '只驗證給定正數 value 的單欄數值反例；係數為非負且和一。這是可能性反例，不對任意帶符號向量或模型最終影響宣稱通用大小排序。'
}, ensure_ascii=False, indent=2))
