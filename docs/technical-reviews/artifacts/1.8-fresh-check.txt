def offline_guard(event, args):
    if event.startswith("socket.") or event in {"subprocess.Popen", "os.system", "os.exec", "os.spawn"}:
        raise RuntimeError("review CPU process is offline and cannot launch child processes")

import sys
sys.addaudithook(offline_guard)

def execute_snippet(source):
    capture = io.StringIO()
    namespace = {}
    with contextlib.redirect_stdout(capture):
        exec(compile(source, "course/chapters/01.md#1.8", "exec"), namespace)
    return {
        "code": source,
        "stdout": capture.getvalue(),
        "stderr": "",
        "result": "completed; original allclose assertion passed",
        "logits": namespace["logits"].tolist(),
        "logits_shape": list(namespace["logits"].shape),
        "logits_dtype": str(namespace["logits"].dtype),
        "target": namespace["target"].tolist(),
        "target_shape": list(namespace["target"].shape),
        "target_dtype": str(namespace["target"].dtype),
        "correct_probabilities": namespace["p"].tolist(),
        "negative_logs": (-namespace["p"].log()).tolist(),
        "distribution": namespace["probabilities"].tolist(),
        "manual": namespace["manual"].item(),
        "cross_entropy": namespace["loss"].item(),
        "manual_builtin_absolute_difference": abs(namespace["manual"].item() - namespace["loss"].item()),
        "device": str(namespace["logits"].device),
    }
