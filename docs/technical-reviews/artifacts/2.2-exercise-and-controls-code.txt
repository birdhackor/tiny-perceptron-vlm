from pathlib import Path
import json, os, sys, re
import torch
from torch import nn

torch.set_num_threads(1)
raw = Path("docs/technical-reviews/artifacts/2.2-section-raw.txt").read_bytes()
code = re.findall(rb"```python\n(.*?)```", raw, re.S)[0].decode()
modified = code.replace("ids = torch.tensor([[1, 2, 3], [3, 2, 1]])", "ids = torch.tensor([[2, 1, 3], [3, 2, 1]])")
assert modified != code
ns = {}
exec(compile(modified, "2.2-exercise-original-fence-with-ids-only-changed", "exec"), ns)
context = ns["context"]
expected = torch.tensor([[2.,20.,1.,10.,3.,30.], [3.,30.,2.,20.,1.,10.]])
assert torch.equal(context, expected)
assert ns["embedding"].weight.grad is None
exercise = {"ids":ns["ids"].tolist(), "shape":list(context.shape), "context":context.tolist(), "sums":context.sum(dim=1).tolist(), "grad_fn":type(context.grad_fn).__name__, "weight_grad_before_backward":str(ns["embedding"].weight.grad)}
# Supplemental local controls verify API behavior, not training quality.
emb = nn.Embedding(4,2)
original_values = torch.tensor([[0.,0.],[1.,10.],[2.,20.],[3.,30.]])
try:
    emb.weight.copy_(original_values)
except RuntimeError as error:
    omitted_no_grad = str(error)
else:
    raise AssertionError("copy_ of a requires_grad leaf should reject in default grad mode")

    emb.weight.copy_(original_values)
    no_grad_enabled_inside = torch.is_grad_enabled()
assert torch.equal(emb.weight, original_values)
assert emb.weight.is_leaf and emb.weight.requires_grad and emb.weight.grad_fn is None
x = emb(torch.tensor([[1,2,3],[3,2,1]]))
flat = x.reshape(2,-1)
assert torch.equal(flat, torch.cat(tuple(x.unbind(dim=1)), dim=1))
assert torch.equal(flat, torch.tensor([[1.,10.,2.,20.,3.,30.],[3.,30.,2.,20.,1.,10.]]))
assert flat.grad_fn is not None and emb.weight.grad is None
control_before = {"ids_shape":[2,3], "x_shape":list(x.shape), "context_shape":list(flat.shape), "x_numel":x.numel(), "x_contiguous":x.is_contiguous(), "same_storage":x.data_ptr()==flat.data_ptr(), "weight_requires_grad":emb.weight.requires_grad, "weight_grad_fn":str(emb.weight.grad_fn), "x_grad_fn":type(x.grad_fn).__name__, "context_grad_fn":type(flat.grad_fn).__name__, "no_grad_enabled_inside":no_grad_enabled_inside, "grad_enabled_after":torch.is_grad_enabled(), "weight_grad":str(emb.weight.grad), "sum_each":flat.sum(dim=1).tolist(), "equals_concatenation":True}
flat.sum().backward()

assert torch.equal(emb.weight.grad, expected_grad)
assert torch.equal(emb.weight, original_values)
# Distinct fixed-position weight slots can distinguish the order.

    linear.weight.copy_(torch.tensor([[1.,0.,0.,0.,0.,0.]]))
