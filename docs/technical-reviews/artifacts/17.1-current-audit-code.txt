def section(raw, identifier):
    match = re.search(rb"^## " + re.escape(identifier.encode()) + rb"(?:\s|\b)", raw, re.M)
    following = re.search(rb"^## ", raw[match.end():], re.M)
    end = match.end() + following.start() if following else len(raw)
    return raw[match.start():end]
primary = section(raw, "17.1")
assert primary == (A / "17.1-current-source-bbbd6a.txt").read_bytes()
assert sha(primary) == "bbbd6a41fae53426959e02f29db048d7cdd3e9bb0735b220abb803a5d7c94784"
assert raw.split(b"## ", 1)[0] == (A / "17.1-intro.txt").read_bytes()
assert re.findall(rb"```python\n(.*?)```", primary, re.S) == re.findall(rb"```python\n(.*?)```", (A / "17.1-source.txt").read_bytes(), re.S)

    support_checks.append({"source":item["source"],"sha256":sha(current),"unchanged":True})
