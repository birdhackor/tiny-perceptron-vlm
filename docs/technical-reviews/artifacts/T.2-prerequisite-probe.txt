def read_fence(path, identifier):
    raw = (ROOT / path).read_bytes()
    matches = list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+", raw))
    i = next(i for i, m in enumerate(matches) if m[1].decode() == identifier)
    end = matches[i + 1].start() if i + 1 < len(matches) else len(raw)
    body = raw[matches[i].start():end]
    fence = re.search(r"```python\n(.*?)```", body.decode(), re.S)[1]
    return fence, hashlib.sha256(body).hexdigest()

def execute(path, identifier, name, change=None, expected_error=None):
    code, sha = read_fence(path, identifier)
    if change:
        code = change(code)
    stream = io.StringIO()
    error = None
    with contextlib.redirect_stdout(stream):
        try:
            exec(compile(code, path + "#" + identifier + ":" + name, "exec"), {})
        except Exception as e:
            error = type(e).__name__ + ": " + str(e)
    assert (error is None) if expected_error is None else (expected_error in (error or "")), error
    records.append({"source": path + "#" + identifier, "source_sha256": sha, "name": name, "code": code, "stdout": stream.getvalue(), "expected_error": expected_error, "observed_error": error})

from tiny_perceptron.training import choose_device
from unittest.mock import patch
device_checks = []
for cuda, mps, expected in [(True, True, "cuda"), (False, True, "mps"), (False, False, "cpu")]:
    with patch("torch.cuda.is_available", return_value=cuda), patch("torch.backends.mps.is_available", return_value=mps):

        device_checks.append({"mock_cuda_available": cuda, "mock_mps_available": mps, "expected": expected, "observed": actual, "scope": "只核對選擇分支；未建立 GPU/MPS tensor 或執行其運算。"})

p = torch.nn.Parameter(torch.tensor([1.0, 2.0]))
p.grad = torch.tensor([3.0, 0.0])
before = p.detach().clone()
returned = torch.nn.utils.clip_grad_norm_([p], 1.0)
assert returned.item() == 3.0 and torch.equal(before, p.detach()) and p.grad[1].item() == 0.0
