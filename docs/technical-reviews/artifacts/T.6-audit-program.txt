def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def write(name, data):
    p = OUT / ('T.6-' + name + '.json')
    p.write_text(json.dumps(data, ensure_ascii=False, indent=2) + '\n')
    return str(p.relative_to(ROOT))

def command(args, timeout=40):
    p = subprocess.run([sys.executable, *args], cwd=ROOT, env={**os.environ, 'CUDA_VISIBLE_DEVICES': ''}, capture_output=True, text=True, timeout=timeout)
    result = {'command': [sys.executable, *args], 'returncode': p.returncode, 'stdout': p.stdout, 'stderr': p.stderr, 'environment': ENV}
    return result

# 親讀凍結原始配置/分母/逐題答案後，以CPU重算它們；不重新訓練。
frozen={}
def compact_training(t):
    d={k:v for k,v in t.items() if k!='history'}
    if t.get('history') and 'effective_targets' in t['history'][0]:
        d['sum_history_effective_targets']=sum(x['effective_targets'] for x in t['history'])
        assert d['sum_history_effective_targets']==t['effective_targets']
    return d
def compact_evaluation(e):
    d={k:v for k,v in e.items() if k!='samples'}
    field='exact_match' if 'examples' in e and 'correct' in e else 'correct' if 'count' in e else 'exact'
    d['recount_correct']=sum(s[field] for s in e['samples']); d['recount_samples']=len(e['samples'])
    assert d['recount_correct']==e.get('correct',e.get('matches'))
    d['rows']=[{k:v for k,v in s.items() if k!='generated_ids'} for s in e['samples']]
    return d

def edit(a,b):
    p=list(range(len(b)+1))
    for i,x in enumerate(a,1):
        c=[i]
        for j,y in enumerate(b,1):c.append(min(c[-1]+1,p[j]+1,p[j-1]+(x!=y)))
        p=c
    return p[-1]
