import json, torch
from tiny_perceptron.modern import MoEFFN

torch.manual_seed(123)
m = MoEFFN(1, experts=2, top_k=1, hidden=2).eval()
with torch.no_grad():
    m.router.weight.copy_(torch.tensor([[1.0], [-1.0]]))
x = torch.ones(6, 1)
seen = [0, 0]
handles = []
for i, e in enumerate(m.experts):
    def hook(module, args, result, i=i):
        seen[i] += len(args[0])
    handles.append(e.register_forward_hook(hook))
with torch.no_grad():
    out, auxiliary, chosen = m(x)
for h in handles:
    h.remove()
with torch.no_grad():
    expected = m.experts[0](x) * m.router(x).softmax(-1)[:, :1]
assert chosen.flatten().tolist() == [0] * 6
assert seen == [6, 0]
assert torch.allclose(out, expected, rtol=0, atol=0)
print(json.dumps({"chosen":chosen.flatten().tolist(),"expert_rows_processed":seen,"selected_assignments":chosen.numel(),"rows_processed":sum(seen),"output_equal_unrestricted_expert_contribution":True,"output_shape":list(out.shape),"parameter_updates":0,"device":str(x.device),"torch":torch.__version__,"torch_commit":torch.version.git_version}, ensure_ascii=False))
