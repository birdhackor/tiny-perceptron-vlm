import codecs
import json
import math
from types import SimpleNamespace
import torch
from tiny_perceptron.data import ByteTokenizer, render_chat
from tiny_perceptron.model import generate, loss_sum
from tiny_perceptron.tokenization import generation_report

tok = ByteTokenizer()
x, y = render_chat([{"role": "user", "content": "Q"}, {"role": "assistant", "content": "A"}])
assert x.tolist() == [1, 3, 89, 2, 4, 73]
assert y.tolist() == [-100, -100, -100, -100, 73, 2]
print("alignment", json.dumps({"x": x.tolist(), "y": y.tolist()}))

# No parameter update: directly check the two-position loss gradient.
logits = torch.zeros(1, 2, tok.vocab_size, dtype=torch.float64, requires_grad=True)
labels = torch.tensor([[-100, tok.eos_id]])
loss, count = loss_sum(logits, labels)
loss.backward()
assert count.item() == 1
assert abs(loss.item() - math.log(264)) < 1e-12
assert logits.grad[0, 0].abs().sum().item() == 0.0
assert abs(logits.grad[0, 1, 2].item() - (1 / 264 - 1)) < 1e-12
print("loss_gradient", json.dumps({"loss": loss.item(), "expected_log264": math.log(264), "effective_targets": count.item(), "ignored_gradient_sum": logits.grad[0, 0].abs().sum().item(), "eos_logit_gradient": logits.grad[0, 1, 2].item()}))

class ScriptedScores(torch.nn.Module):
    def __init__(self, scripted):
        super().__init__()
        self.config = SimpleNamespace(max_length=128)
        self.scripted = scripted
        self.calls = 0
    def forward(self, ids, cache=None):
        result = torch.full((*ids.shape, 264), -100.0)
        result[:, -1, self.scripted[self.calls]] = 100.0
        self.calls += 1
        return {"logits": result, "cache": None}

prompt = torch.tensor([[1, 3, 89, 2, 4]])  # Contains the user's historical EOS.
for name, predicted, limit, expected in [
    ("eos", [73, 2, 74], 3, [73, 2]),
    ("limit", tok.encode("ABCD"), 2, [73, 74]),
    ("punctuation", tok.encode(".B") + [2], 3, tok.encode(".B") + [2]),
]:
    model = ScriptedScores(predicted)
    model.train()
    result = generate(model, prompt, limit, eos_id=tok.eos_id)
    raw = result[0, prompt.shape[1]:].tolist()
    assert raw == expected
    assert model.training is True
    report = generation_report(tok, raw)
    assert report["generation_status"] == ("eos" if 2 in raw else "token_or_context_limit")
    print("generation_" + name, json.dumps({"raw_sequence": result.tolist(), "forward_calls": model.calls, "report": report}, ensure_ascii=False))

full = tok.encode("貓")
assert full == [240, 186, 155] and len(full) == 3
partial = bytes(i - 8 for i in full[:2])
assert partial.decode("utf-8", errors="replace") == "�"
decoder = codecs.getincrementaldecoder("utf-8")()
first = decoder.decode(partial, final=False)
second = decoder.decode(bytes([full[2] - 8]), final=True)
assert first == "" and second == "貓"
print("utf8", json.dumps({"cat_raw_bytes": list("貓".encode()), "cat_token_ids": full, "budget2_display": tok.decode(full[:2]), "incremental_prefix": first, "incremental_completed": second}, ensure_ascii=False))
assert tok.decode([73, 2]) == "A"
assert tok.decode([107, 113, 122, 109, 2]) == "cire"
assert tok.encode("circle") == [107, 113, 122, 107, 116, 109]
print("byte_ids", json.dumps({"A": tok.encode("A"), "B": tok.encode("B"), "cire": tok.decode([107, 113, 122, 109, 2]), "circle_ids": tok.encode("circle")}, ensure_ascii=False))
