from pathlib import Path
import contextlib
import datetime
import hashlib
import io
import json
import os
import sys
import torch

code_path = Path(sys.argv[1])
result_path = Path(sys.argv[2])
assert torch.version.cuda is None
assert not torch.cuda.is_available()
assert os.environ["OMP_NUM_THREADS"] == "1"
assert os.environ["MKL_NUM_THREADS"] == "1"
torch.set_default_device("cpu")
stdout = io.StringIO()
namespace = {"__name__": "__main__"}
with contextlib.redirect_stdout(stdout):
    exec(compile(code_path.read_bytes(), str(code_path), "exec"), namespace)
command = (
    "OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 CUDA_VISIBLE_DEVICES='' "
    "HF_HUB_OFFLINE=1 HF_DATASETS_OFFLINE=1 TRANSFORMERS_OFFLINE=1 "
    ".venv/bin/python docs/technical-reviews/artifacts/2.4-cpu-runner.txt "
    + str(code_path) + " " + str(result_path)
)
record = {
    "command": command,
    "executed_at": datetime.datetime.now(datetime.UTC).isoformat(),
    "code_path": str(code_path),
    "code_sha256": hashlib.sha256(code_path.read_bytes()).hexdigest(),
    "runner_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    "environment": {
        "python": sys.version,
        "torch": str(torch.__version__),
        "torch_git_version": str(torch.version.git_version),
        "device": "cpu",
        "OMP_NUM_THREADS": os.environ["OMP_NUM_THREADS"],
        "MKL_NUM_THREADS": os.environ["MKL_NUM_THREADS"],
        "cuda_build": str(torch.version.cuda),
        "cuda_available": str(torch.cuda.is_available()),
        "python_optimization": str(sys.flags.optimize),
    },
    "stdout": stdout.getvalue(),
    "stderr": "",
    "exit_code": 0,
}
result_path.write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n")
print(stdout.getvalue(), end="")
