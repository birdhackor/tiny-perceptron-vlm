def section(chapter, lesson):
    raw = (ROOT / f"course/chapters/{chapter}.md").read_bytes()
    headings = list(re.finditer(rb"(?m)^## [^\r\n]+", raw))
    index = next(i for i, h in enumerate(headings) if h[0].startswith(f"## {lesson} ".encode()))
    end = headings[index + 1].start() if index + 1 < len(headings) else len(raw)
    body = raw[headings[index].start():end]
    codes = re.findall(rb"(?ms)^```python\n(.*?)^```", body)
    return body, [code.decode() for code in codes]

cases = []

def add(chapter, lesson, label, edits=(), expected_error=False, append="", fence_count=1):
    body, fences = section(chapter, lesson)
    code = "\n".join(fences[:fence_count])
    for old, new in edits:
        assert old in code, (lesson, old)
        code = code.replace(old, new)
    code += append
    cases.append({"source": f"course/chapters/{chapter}.md#{lesson}", "source_sha256": hashlib.sha256(body).hexdigest(), "label": label, "code": code, "expected_assertion_error": expected_error})

results = []
for case in cases:
    stdout = io.StringIO()
    error = None
    with contextlib.redirect_stdout(stdout):
        try:
            exec(compile(case["code"], case["source"] + ":exercise", "exec"), {"__name__": "__main__"})
        except AssertionError:
            error = "AssertionError"
    matched = (error == "AssertionError") == case["expected_assertion_error"]
    assert matched, case["label"]
    results.append({**case, "stdout": stdout.getvalue(), "observed_error": error, "expectation_matched": matched})
