import json
import platform
import torch

scores = torch.tensor([[0.1, 0.2, 0.0]]).repeat(12, 1)
prob = scores.softmax(-1)
chosen = prob.topk(1, dim=-1).indices
counts = torch.bincount(chosen.flatten(), minlength=3)
assert chosen.shape == (12, 1)
assert chosen.flatten().tolist() == [1] * 12
assert counts.tolist() == [0, 12, 0]
print(json.dumps({
    "exercise": "15.8 第一項，每列 [0.1,0.2,0]",
    "environment": {"python": platform.python_version(), "torch": torch.__version__, "torch_commit": torch.version.git_version, "device": str(scores.device), "threads": str(torch.get_num_threads())},
    "scores_shape": list(scores.shape),
    "mean_probability": prob.mean(0).tolist(),
    "chosen": chosen.flatten().tolist(),
    "counts": counts.tolist(),
}, ensure_ascii=False, indent=2))
