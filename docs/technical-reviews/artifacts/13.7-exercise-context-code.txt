from tiny_perceptron.data import ByteTokenizer

tok = ByteTokenizer()
answers = ["4", "兩組各2塊積木合起來，共有4塊。"]
for answer in answers:
    body = tok.encode(answer)
    print("回答", answer, "正文位元組tokens", len(body), "含結束學習位置", len(body) + 1)
for prompt, judgments in [
    ("2+2=?只回數字", ["4符合只回數字；2+2=4", "解釋有分組與合併，但額外文字違反只回數字"]),
    ("2+2=?請用分組說明", ["4只給結論，缺少要求的分組說明", "兩組各2塊合成4塊，提供要求的分組與合併說明"]),
]:
    print("問題", prompt)
    for answer, judgment in zip(answers, judgments, strict=True):
        print("候選", answer, "人工條件核對", judgment)
assert 2 + 2 == 4
assert answers[0].isdigit() and not answers[1].isdigit()
