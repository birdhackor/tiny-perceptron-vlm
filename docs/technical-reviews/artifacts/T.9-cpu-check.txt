def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def section(path, identifier):
    raw = Path(path).read_bytes()
    heads = list(re.finditer(rb'(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+', raw))
    index = next(i for i, h in enumerate(heads) if h[1].decode() == identifier)
    return raw[heads[index].start():heads[index + 1].start() if index + 1 < len(heads) else len(raw)]

env = {'python': platform.python_version(), 'torch': str(torch.__version__),
       'device': 'cpu', 'torch_git_version': str(torch.version.git_version),
       'num_threads': str(torch.get_num_threads()), 'OMP_NUM_THREADS': os.getenv('OMP_NUM_THREADS'),
       'MKL_NUM_THREADS': os.getenv('MKL_NUM_THREADS'), 'network': 'offline execution; no downloads'}
records = []
prerequisites = []
for path, identifier in [('course/chapters/17.md','17.2'), ('course/chapters/17.md','17.8'),
                         ('course/chapters/17.md','17.9'), ('course/chapters/17.md','17.14'),
                         ('course/chapters/17.md','17.10'), ('course/chapters/04.md','4.1'),
                         ('course/training.md','T.4')]:
    raw = section(path, identifier)
    prerequisites.append({'source': path + '#' + identifier,
                          'raw_sha256': hashlib.sha256(raw).hexdigest(), 'raw_utf8': raw.decode()})
    if identifier == 'T.4':
        continue
    code = re.search(r'```python\n(.*?)\n```', raw.decode(), re.S)[1]
    exercise = code
    if identifier == '17.2': exercise = code.replace('scale = 0.5', 'scale = 0.25')
    if identifier == '17.8': exercise = code.replace('7, 2, 3]', '7, 2, 3, 4]')
    if identifier in ('17.9','17.10'): exercise = code.replace('bits=4', 'bits=8')
    if identifier == '17.14': exercise = code.replace('y.sum().backward()', 'y.square().sum().backward()')
    if identifier == '4.1':
        exercise = code.replace('ids = torch.tensor', '    p.weight.zero_()\nids = torch.tensor').replace('assert not torch.equal', 'assert torch.equal')
    for label, source in [('original',code), ('exercise',exercise)]:
        namespace = {}
        output = io.StringIO()
        with contextlib.redirect_stdout(output): exec(compile(source, identifier + '-' + label, 'exec'), namespace)
        if identifier == '17.2':
            assert namespace['q'].tolist() == ([-1,0,1,2] if label=='original' else [-3,1,3,5])
        if identifier == '17.8':
            assert namespace['packed'].tolist() == ([112,152,175,139] if label=='original' else [112,152,175,203])
        if identifier == '17.9':
            assert namespace['before'].shape == (1,3,264)
            assert torch.equal(namespace['before'], namespace['original_again'])
        if identifier == '17.14':
            target = torch.ones(3) if label=='original' else 2 * namespace['y'].detach()
            assert torch.equal(namespace['x'].grad, target)
            assert torch.equal(namespace['hard'].grad, torch.zeros(3))
        if identifier == '17.10':
            assert namespace['float_bytes'] == 8320
            assert namespace['compressed'].storage_bytes() == (1280 if label=='original' else 2304)
        records.append({'id':identifier + '-' + label, 'source':source, 'stdout':output.getvalue(), 'result':'assertions passed', 'environment':env})

raw = section('course/training.md','T.9')

    path=ROOT/'checkpoints/course/quantization'/filename
    entry=next(item for item in manifest['files'] if item['output']==filename)
    assert sha(path)==entry['sha256']
    assert path.stat().st_size==entry['bytes']
    model,payload=load_checkpoint(path,'cpu')
    validation=_evaluate(model,exercise_records,'cpu',tokens=24)
    test=_evaluate(model,tests,'cpu',tokens=24)
    for split, actual in [('validation',validation),('test',test)]:
        frozen=receipt['results']['runs'][variant][split]
        assert actual['correct']==frozen['correct']
        assert actual['supervised_tokens']==frozen['supervised_tokens']
        assert [s['generated_ids'] for s in actual['generated_samples']]==[s['generated_ids'] for s in frozen['generated_samples']]
        assert abs(actual['answer_nll']-frozen['answer_nll'])<1e-5
    storage=_storage(model,path)
    assert storage['tensor_bytes']==receipt['results']['runs'][variant]['storage']['tensor_bytes']
    weights.append({'variant':variant,'local_path':str(path.relative_to(ROOT)),'hydrated_sha256':sha(path),

                    'public_url':f'https://huggingface.co/{manifest["repo"]}/resolve/{public_weights_pin}/course/course-v1/quantization/{filename}',

                    'validation':validation,'test':test})
    models[variant]=model
