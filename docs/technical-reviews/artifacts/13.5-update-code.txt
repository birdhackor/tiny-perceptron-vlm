import math,json,torch
from tiny_perceptron.alignment import dpo_loss,sequence_log_probability
c=torch.tensor([-4.0],requires_grad=True);r=torch.tensor([-3.0],requires_grad=True)
rc=torch.tensor([-4.0],requires_grad=True);rr=torch.tensor([-3.0],requires_grad=True)
l=dpo_loss(c,r,rc,rr,beta=0.1);l.backward()
print(json.dumps({'loss':l.item(),'chosen_grad':c.grad.item(),'rejected_grad':r.grad.item(),'reference_grad':[rc.grad,rr.grad],'updated_chosen':c.item()-0.1*c.grad.item(),'updated_rejected':r.item()-0.1*r.grad.item(),'new_margin':(c.item()-0.1*c.grad.item())-(r.item()-0.1*r.grad.item())-(-1)},ensure_ascii=False))
for margin in [-1,0,1]:
 x=0.1*margin;print(json.dumps({'margin':margin,'sigmoid':1/(1+math.exp(-x)),'loss':math.log1p(math.exp(-x)),'dL_dc':-0.1/(1+math.exp(x)),'dL_dr':0.1/(1+math.exp(x))}))
z=torch.tensor([[[0.,2.,0.],[2.,0.,0.],[0.,0.,2.]]],requires_grad=True)
y=torch.tensor([[-100,0,2]])
c=sequence_log_probability(z,y)
r=torch.tensor([-3.],requires_grad=True)
l=dpo_loss(c,r,c.detach(),r.detach());l.backward()
print('masked_logits_grad',z.grad.tolist())
assert z.grad[0,0].abs().sum()==0 and z.grad[0,1:].abs().sum()>0
