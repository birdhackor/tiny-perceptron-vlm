from pathlib import Path
import hashlib
import json
import platform
from playwright.sync_api import sync_playwright

out = Path(__file__).resolve().parent
svg = (out / "practical_order.svg").read_text()
records = []
with sync_playwright() as p:
    browser = p.chromium.launch(executable_path='/usr/bin/chromium', headless=True, args=['--no-sandbox'])
    for width in (688, 311):
        page = browser.new_page(viewport={"width": width, "height": 1200}, device_scale_factor=1)
        page.set_content('<!doctype html><html><head><meta charset="UTF-8"><style>html,body{margin:0;padding:0;}svg{display:block;width:100%;height:auto;}</style></head><body>' + svg + '</body></html>')
        page.evaluate("document.fonts.ready")
        figure = page.locator("svg")
        bounds = figure.bounding_box()
        target = out / f"practical_order-{width}px.png"
        figure.screenshot(path=str(target))
        text_bounds = page.locator("svg text").evaluate_all("es => es.map(e => { const b=e.getBBox(); return {text:e.textContent,x:b.x,y:b.y,width:b.width,height:b.height}; })")
        records.append({"css_width": width, "figure_bounds": bounds, "path": str(target), "sha256": hashlib.sha256(target.read_bytes()).hexdigest(), "text_bounds": text_bounds})
        page.close()
    version = browser.version
    browser.close()
record = {"reviewer_task": "/root/natural_factual_mobile_11_14", "command": ".venv/bin/python docs/technical-reviews/artifacts/natural-mobile-11.14-second-factual/render-figure.source.txt", "browser": version, "python": platform.python_version(), "svg_sha256": hashlib.sha256((out / 'practical_order.svg').read_bytes()).hexdigest(), "renders": records}
(out / "render-proof.json").write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n")
print(json.dumps({"browser": version, "renders": [{k:v for k,v in r.items() if k!='text_bounds'} for r in records]}, indent=2))
