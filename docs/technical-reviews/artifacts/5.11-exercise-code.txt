def run_case(name, program):
    namespace = {}
    output = io.StringIO()
    error = None
    with contextlib.redirect_stdout(output):
        try:
            exec(compile(program, name, "exec"), namespace)
        except AssertionError:
            error = "AssertionError"
    hashes = namespace["hashes"]
    record = {"case": name, "stdout": output.getvalue(), "error": error, "normalized": namespace["normalized"], "distinct_hashes": len(set(hashes)), "hashes": hashes, "first_two_equal": hashes[0] == hashes[1]}
    print(json.dumps(record, ensure_ascii=False))
    return record

original = run_case("original", fence)
assert original["distinct_hashes"] == 2 and original["error"] is None
changed = fence.replace('"貓 看狗"', '"鳥 看狗"', 1)
stale = run_case("exercise_stale_assert", changed)
assert stale["distinct_hashes"] == 3 and stale["error"] == "AssertionError"
corrected = run_case("exercise_corrected_assert", changed.replace("assert hashes[0] == hashes[1]", "assert hashes[0] != hashes[1]"))
assert corrected["distinct_hashes"] == 3 and corrected["error"] is None
extra_space = run_case("exercise_extra_space", fence.replace('"貓 看狗"', '"貓   看狗"', 1))
assert extra_space["distinct_hashes"] == 2 and extra_space["error"] is None

whitespace = " \t貓\n\n看狗 \u3000"
normalized = " ".join(whitespace.split())
assert normalized == "貓 看狗" and fingerprint(whitespace) == fingerprint("貓 看狗")
assert fingerprint("貓看狗") != fingerprint("貓 看狗")
assert " ".join("貓，  看狗！".split()) == "貓， 看狗！"
print(json.dumps({"whitespace_case": repr(whitespace), "normalized": normalized, "no_space_vs_one_space_different": True, "punctuation_retained": True}, ensure_ascii=False))

left = fingerprint("貓 看狗")
right = fingerprint("鳥 看狗")
print(json.dumps({"sha256_digest_bytes": hashlib.sha256(b"").digest_size, "hex_characters": len(left), "alphabet": sorted(set(left)), "one_character_change": {"changed_hex_positions": sum(a != b for a, b in zip(left, right)), "changed_bits": (int(left, 16) ^ int(right, 16)).bit_count()}}, ensure_ascii=False))
assert len(left) == 64 and set(left) <= set("0123456789abcdef")
seen = {}
for number in range(100):
    text = f"文件 {number}"
    full = fingerprint(text)
    if full[:1] in seen:
        prior_text, prior_full = seen[full[:1]]
        assert text != prior_text and full != prior_full
        print(json.dumps({"short_prefix_collision": {"texts": [prior_text, text], "one_hex_prefix": full[:1], "full_digests_different": True}}, ensure_ascii=False))
        break
    seen[full[:1]] = text, full
else:
    raise AssertionError("pigeonhole bound was not met")
