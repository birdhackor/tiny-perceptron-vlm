import json
import platform
import sys
from datetime import datetime, timezone
from pathlib import Path
sys.path.insert(0,str(Path.cwd()))
import torch
from scripts.evaluate import evaluate
from tiny_perceptron.model import ModelConfig, TinyLM
torch.set_num_threads(1)
torch.manual_seed(42)
sft_model=TinyLM(ModelConfig(width=8,layers=1,heads=1,max_length=16))
sft=evaluate(sft_model,[{'question':'?','answer':'circle'},{'question':'x'*40,'answer':'circle'}],'sft',max_new_tokens=2)
assert sft['effective_tokens']==7 and sft['loss_evaluated_records']==1 and sft['generation_evaluated_records']==1
assert sft['skipped'][0]['phase']=='loss_and_generation'
text_model=TinyLM(ModelConfig(width=8,layers=1,heads=1,max_length=4))
text=evaluate(text_model,[{'text':'abcdef'},{'text':'xy'}],'text',max_new_tokens=2)
assert text['effective_tokens']==10 and text['loss_evaluated_records']==2 and text['generation_evaluated_records']==1
assert text['skipped'][0]['phase']=='generation'
print(json.dumps({'checked_at':datetime.now(timezone.utc).isoformat(),'environment':{'python':platform.python_version(),'torch':str(torch.__version__),'device':'cpu','threads':'1'},'sft':sft,'text':text,'scope':'Untrained bounded CPU models exercise both original skip branches; no training, benchmark or original 500-step score.'},ensure_ascii=False,indent=2))
