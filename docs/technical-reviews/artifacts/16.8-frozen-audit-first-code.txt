        assert p["flash_forward_operator_observed"] and p["flash_cuda_verified"] and p["cuda_kernel_names"]
        assert any("pytorch_flash::flash_fwd_kernel" in k for k in p["cuda_kernel_names"])
        if mode=="forward_backward":
            assert p["flash_backward_operator_observed"]
            assert any("pytorch_flash::flash_bwd" in k for k in p["cuda_kernel_names"])
