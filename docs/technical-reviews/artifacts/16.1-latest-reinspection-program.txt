from pathlib import Path
import ast, hashlib, json, platform, re

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

def section(raw, lesson):
    heads = list(re.finditer(rb'(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+', raw))
    pos = next(i for i, h in enumerate(heads) if h[1].decode() == lesson)
    return raw[heads[pos].start():heads[pos + 1].start() if pos + 1 < len(heads) else len(raw)]

root = Path('docs/technical-reviews/artifacts')
old = (root / '16.1-source-raw.txt').read_bytes()
middle = (root / '16.1-current-source-raw.txt').read_bytes()
whole = Path('course/chapters/16.md').read_bytes()
new = section(whole, '16.1')
assert sha(new) == '2cc9c51149a14143fd03708de1a5d98534acee6170de3308a850ad99dd394507'
assert new == (root / '16.1-latest-source-raw.txt').read_bytes()
fences = lambda raw: re.findall(rb'```python\n(.*?)```', raw, re.S)
assert fences(old) == fences(middle) == fences(new)
exercise = '練習把ids改成'.encode()
assert old[old.index(exercise):] == middle[middle.index(exercise):] == new[new.index(exercise):]

receipt_raw = Path('docs/course-experiments/results/efficiency.json').read_bytes()

branches = {**{k: v['training'] for k, v in results['models'].items()}, **{k: v['training'] for k, v in results['update_variants'].items()}, **{k: v for k, v in results['packing']['actual_updates'].items() if k in ('padded', 'packed')}}
coverage = {k: {'present': {f: v[f] for f in fields if f in v}, 'missing': [f for f in fields if f not in v], 'steps': v['steps'], 'effective_tokens': v['effective_tokens']} for k, v in branches.items()}
assert [k for k, v in coverage.items() if not v['missing']] == ['mha', 'gqa', 'ordinary', 'accumulated', 'activation_checkpoint', 'sdpa']

architecture = Path('/tmp/16.1-frozen-architecture.txt').read_bytes()
run_code = Path('/tmp/16.1-frozen-run.txt').read_bytes()
assert sha(architecture) == receipt['code_sha256']['scripts/course_experiments/architecture.py']
assert sha(run_code) == receipt['code_sha256']['scripts/course_experiments/run.py']
tree = ast.parse(architecture)
facts = {}
for fn in tree.body:
    if isinstance(fn, ast.FunctionDef) and fn.name in ('_train', '_packing_updates', '_compile_probe', '_compile_worker'):
        nodes = [x for x in ast.walk(fn) if isinstance(x, ast.Call)]
        calls = [ast.unparse(x.func) for x in nodes]
        facts[fn.name] = {'source_lines': [fn.lineno, fn.end_lineno], 'reset_peak_memory_stats_calls': calls.count('torch.cuda.reset_peak_memory_stats'), 'memory_allocated_calls': calls.count('torch.cuda.memory_allocated'), 'max_memory_allocated_calls': calls.count('torch.cuda.max_memory_allocated')}
        if fn.name == '_compile_worker':
            compile_call = next(x for x in nodes if ast.unparse(x.func) == 'torch.compile')
            facts[fn.name]['compile_keywords'] = {x.arg: ast.literal_eval(x.value) for x in compile_call.keywords}
            facts[fn.name]['executes_compiled_callable'] = 'compiled' in calls
        if fn.name == '_compile_probe':
            spawn = next(x for x in nodes if ast.unparse(x.func) == 'subprocess.run')
            facts[fn.name]['subprocess_command_ast'] = ast.unparse(spawn.args[0])
            facts[fn.name]['subprocess_run_calls'] = calls.count('subprocess.run')
assert all(facts['_train'][k] == 1 for k in ('reset_peak_memory_stats_calls', 'memory_allocated_calls', 'max_memory_allocated_calls'))
assert all(facts['_packing_updates'][k] == 0 for k in ('reset_peak_memory_stats_calls', 'memory_allocated_calls', 'max_memory_allocated_calls'))
assert facts['_compile_worker']['compile_keywords'] == {'backend': 'inductor', 'fullgraph': True, 'dynamic': False}
assert facts['_compile_worker']['executes_compiled_callable']
assert facts['_compile_probe']['subprocess_run_calls'] == 1
assert '--compile-worker' in facts['_compile_probe']['subprocess_command_ast']
compile_receipt = results['compile']
assert compile_receipt['status'] == 'completed' and compile_receipt['backend'] == 'inductor'
assert compile_receipt['inductor_generated_kernel_count'] > 0
