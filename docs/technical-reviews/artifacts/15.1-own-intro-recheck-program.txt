ROOT = Path.cwd()

def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def save_json(name, value):
    path = ART / name
    path.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
    return {'path': str(path.relative_to(ROOT)), 'sha256': sha(path.read_bytes())}


def sections(raw):
    heads = list(re.finditer(rb'(?m)^## [^\r\n]+', raw))
    return {h[0].decode().split()[1]: raw[h.start():heads[i + 1].start() if i + 1 < len(heads) else len(raw)] for i, h in enumerate(heads)}


def python_fences(raw):
    return re.findall(rb'(?ms)^```python\n(.*?)^```\s*$', raw)


def all_fences(raw):
    return re.findall(rb'(?ms)^```([^\r\n]*)\n(.*?)^```[ \t]*\r?$', raw)
