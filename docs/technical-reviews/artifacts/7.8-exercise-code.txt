import json
import sys
import torch

valid = torch.tensor([[False, False, True, True], [False, True, True, True]])
positions = torch.arange(4)[None].expand_as(valid)
last = positions.masked_fill(~valid, -1).amax(dim=-1)
logits = torch.arange(40, dtype=torch.float32).reshape(2, 4, 5)
selected = logits[torch.arange(2), last]
print("最後有效索引", last)
print("取出的候選分數", selected)
assert last.tolist() == [3, 3]
assert selected.tolist() == [[15, 16, 17, 18, 19], [35, 36, 37, 38, 39]]
print(json.dumps({"python": sys.version, "torch": str(torch.__version__),
    "torch_commit": str(torch.version.git_version), "device": str(logits.device),
    "valid_counts": valid.sum(-1).tolist(), "last": last.tolist(),
    "selected": selected.tolist(), "shape": list(selected.shape),
    "scope": "只改原例第一筆valid，並改assert核對；無參數更新"}, ensure_ascii=False))
