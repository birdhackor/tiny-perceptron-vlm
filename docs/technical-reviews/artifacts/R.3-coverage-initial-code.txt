from pathlib import Path
import sys, re, json, hashlib, platform
import torch
root = Path.cwd()
raw = Path("course/README.md").read_bytes()
heads = list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+", raw))
i = next(i for i, h in enumerate(heads) if h[1] == b"R.3")
section = raw[heads[i].start():heads[i+1].start() if i+1<len(heads) else len(raw)]
assert hashlib.sha256(section).hexdigest() == "395bfdbe0b0779817511c9d7b435a05c670a2c9e772904ae2d4aac0d7e434405"
mode=sys.argv[1]
result={"mode":mode,"environment":{"python":platform.python_version(),"torch":str(torch.__version__),"torch_source_commit":str(torch.version.git_version),"device":"cpu","OMP_NUM_THREADS":"1","MKL_NUM_THREADS":"1"},"section_sha256":hashlib.sha256(section).hexdigest()}
if mode == "links":
    links=re.findall(r"(?<!!)\[[^\]]+\]\(([^)]+)\)",section.decode())
    resolved=[]
    for target in links:
        relative, _, anchor=target.partition("#")
        p=(Path("course")/relative).resolve()
        assert p.is_relative_to(root) and p.is_file(), target
        resolved.append({"link":target,"path":str(p.relative_to(root)),"sha256":hashlib.sha256(p.read_bytes()).hexdigest()})
    assert len(links)==23
    expected=[f"chapters/{n:02}.md" for n in range(1,19)]+["chapters/0A.md","chapters/0B.md","chapters/0C.md","lessons.md","glossary.md"]
    assert links==expected
    section_text=section.decode()
    result.update({"all_23_links_resolve":True,"consecutive_numbered_chapters":list(range(1,19)),"appendices":["A","B","C"],"links":resolved,"python_fence_count":len(re.findall(r"(?m)^```python",section_text)),"any_fence_count":len(re.findall(r"(?m)^```",section_text)),"svg_count":len(re.findall(r"\.svg\)",section_text)),"execution_scope":"R.3 沒有原 Python、bash 或數值練習；此為本節原始路徑與目錄主張的有界離線核對。"})
elif mode == "navigation":
    question="回答第一個 token 在哪個位置被預測？"
    lessons=Path("course/lessons.md").read_text().splitlines()
    matches=[{"line":n,"raw":line} for n,line in enumerate(lessons,1) if question in line]
    assert len(matches)==1
    assert "7.4 " in matches[0]["raw"] and "../course/chapters/07.md#7.4" in matches[0]["raw"]
    chapter=Path("course/chapters/07.md").read_text()
    assert "## 7.4 "+question in chapter
    glossary=Path("course/glossary.md").read_text().splitlines()
    token_rows=[{"line":n,"raw":line} for n,line in enumerate(glossary,1) if line.startswith("| token、tokenizer |")]
    assert len(token_rows)==1
    chapter6=Path("course/chapters/06.md").read_text()
    assert "## 6.1 字元切詞有什麼成本？" in chapter6 and "## 6.7 一個 token 一定是一個完整字嗎？" in chapter6
    result.update({"exercise":"本人實做 R.3 的找題練習，從全部小節目錄找到指定問題，再查原始 7.4 標題。","matches":matches,"chapter_7_4_heading_exists":True,"token_glossary_rows":token_rows,"chapter_6_tokenizer_sections_exist":True,"execution_scope":"只核對本節找題與名詞導覽，不把前置 7.4 範例冒稱為本節已執行教學例。"})
elif mode == "coverage":
    snapshot=json.loads(Path("docs/technical-reviews/artifacts/R.3-directory-snapshot.json").read_text())
    checked=[]
    for entry in snapshot["sources"]:
        path=Path(entry["path"])
        raw_source=path.read_bytes()
        assert hashlib.sha256(raw_source).hexdigest()==entry["sha256"]
        lines=raw_source.decode().splitlines()
        selections=entry["inspected_heading_and_opening_prose"]+entry.get("additional_read_scope",[])
        for part in selections:
            observed="\n".join(lines[part["first_line"]-1:part["last_line"]])+"\n"
            assert observed==part["text"],(entry["path"],part["first_line"])
        checked.append({"path":entry["path"],"sha256":entry["sha256"],"checked_original_scopes":len(selections)})
    inventory={}
    for path in list(Path("course/chapters").glob("*.md"))+[Path("course")/name for name in ["first-steps.md","README.md","training.md","glossary.md"]]:
        for match in re.finditer(r"(?m)^## ([A-Z\d]+\.\d+) (.+)$",path.read_text()):
            inventory[match[1]]={"path":str(path),"title":match[2]}
    indexed={}
    for line in Path("course/lessons.md").read_text().splitlines():
        match=re.match(r"\| ([A-Z\d]+\.\d+) (.+?) \| \[閱讀\]\(([^)]+)\)",line)
        if match:
            indexed[match[1]]={"title":match[2],"link":match[3]}
    missing=sorted(set(inventory)-set(indexed))
    extra=sorted(set(indexed)-set(inventory))
    wrong=[]
    for sid,item in inventory.items():
        if sid in indexed:
            path,_,anchor=indexed[sid]["link"].partition("#")
            if (Path("course")/path).resolve()!=(root/item["path"]).resolve() or anchor!=sid or indexed[sid]["title"]!=item["title"]:
                wrong.append(sid)
    assert not missing and not extra and not wrong,(missing,extra,wrong)
    result.update({"snapshot_scopes_match_current_original":True,"checked_sources":checked,"original_numbered_heading_count":len(inventory),"indexed_row_count":len(indexed),"missing":missing,"extra":extra,"incorrect_title_or_path_or_anchor":wrong,"execution_scope":"只依編號標題核對全部小節目錄，未閱讀或判定未列為必要來源的正文。主題涵蓋語義由本人親讀所保存原始段落判定；程式只核對路徑、定位與 bytes。"})
else:
    raise ValueError(mode)
print(json.dumps(result,ensure_ascii=False,indent=2))
