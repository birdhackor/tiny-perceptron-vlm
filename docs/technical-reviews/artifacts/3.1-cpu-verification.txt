from pathlib import Path
import contextlib
import io
import json
import re
import sys
import torch

torch.set_num_threads(1)
root = Path(__file__).resolve().parents[3]
artifacts = root / 'docs/technical-reviews/artifacts'
raw = (artifacts / '3.1-section.raw.txt').read_text()
fence = re.findall(r'```python\n(.*?)```', raw, re.S)[0]

def execute(code):
    namespace = {}
    buffer = io.StringIO()
    with contextlib.redirect_stdout(buffer):
        exec(compile(code, '<course-verification>', 'exec'), namespace)
    return namespace, buffer.getvalue()

original, stdout = execute(fence)
assert stdout == 'tensor([1.8000, 0.4000])\n'
assert original['values'].shape == (2, 2)
assert original['weights'].shape == (2,)
assert original['output'].shape == (2,)
assert original['values'].dtype == torch.float32
exercise_code = fence.replace('weights = torch.tensor([0.9, 0.1])', 'weights = torch.tensor([0.1, 0.9])').replace('torch.tensor([1.8, 0.4])', 'torch.tensor([0.2, 3.6])')
exercise, exercise_stdout = execute(exercise_code)
assert torch.equal(exercise['values'], original['values'])
assert torch.allclose(exercise['output'], torch.tensor([0.2, 3.6]))
assert torch.linalg.vector_norm(exercise['output'] - exercise['values'][1]) < torch.linalg.vector_norm(exercise['output'] - exercise['values'][0])

endpoints = []
for weights, expected in [([0.5, 0.5], [1., 2.]), ([1., 0.], [2., 0.])]:
    out = torch.tensor(weights) @ original['values']
    assert torch.equal(out, torch.tensor(expected))
    endpoints.append({'weights': weights, 'output': out.tolist()})
comparison = torch.allclose(original['output'], torch.tensor([1.8, 0.4]))
bad_comparison = torch.allclose(original['output'], torch.tensor([2., 0.4]))
assert comparison is True and bad_comparison is False
failure_type = None
try:
    execute(fence.replace('torch.tensor([1.8, 0.4])', 'torch.tensor([2.0, 0.4])'))
except AssertionError as error:
    failure_type = type(error).__name__
assert failure_type == 'AssertionError'

prerequisites = {}
for id in ['2.2', '2.3', 'W.3']:
    pre = (artifacts / f'3.1-prerequisite-{id}.raw.txt').read_text()
    code = re.findall(r'```python\n(.*?)```', pre, re.S)[0]
    namespace, pre_stdout = execute(code)
    prerequisites[id] = {'stdout': pre_stdout}
    if id == '2.2':
        assert namespace['context'].tolist() == [[1.,10.,2.,20.,3.,30.],[3.,30.,2.,20.,1.,10.]]
    elif id == '2.3':
        assert namespace['y'].tolist() == [[10., 2.]]
        assert namespace['layer'].weight.tolist() == [[1.,0.,2.],[0.,-1.,1.]]
    else:
        scores = namespace['scores']
        assert scores.mean(dim=1).tolist() == [70.,90.]
        assert scores.mean(dim=0).tolist() == [70.,80.,90.]

# Equal-weight mixing is non-injective: opposite offsets cancel.
v = torch.tensor([[2.,0.],[0.,4.]])
alternate = torch.tensor([[3.,1.],[-1.,3.]])
half = torch.tensor([0.5,0.5])
assert torch.equal(half @ v, half @ alternate)

# A larger coefficient does not imply a larger weighted-vector magnitude.
counter_values = torch.tensor([[0.001,0.],[0.,100.]], dtype=torch.float64)
counter_weights = torch.tensor([0.9,0.1], dtype=torch.float64)
terms = counter_weights[:,None] * counter_values
norms = torch.linalg.vector_norm(terms,dim=1)
assert norms[0] < norms[1]

print(json.dumps({
    'environment': {'python': sys.version, 'torch': torch.__version__, 'torch_commit': torch.version.git_version, 'device': 'cpu', 'num_threads': str(torch.get_num_threads()), '__debug__': str(__debug__)},
    'original': {'stdout': stdout, 'values': original['values'].tolist(), 'weights': original['weights'].tolist(), 'output': original['output'].tolist(), 'shape': list(original['output'].shape), 'dtype': str(original['output'].dtype)},
    'exercise': {'stdout': exercise_stdout, 'output': exercise['output'].tolist(), 'values_unchanged': True},
    'endpoints': endpoints,
    'allclose': {'correct': comparison, 'incorrect': bad_comparison, 'rtol': 1e-5, 'atol': 1e-8, 'assert_failure': failure_type},
    'prerequisites': prerequisites,
    'non_unique_mixture': {'first_values': v.tolist(), 'alternate_values': alternate.tolist(), 'same_output': (half @ v).tolist()},
    'contribution_counterexample': {'weights': counter_weights.tolist(), 'values': counter_values.tolist(), 'weighted_terms': terms.tolist(), 'term_norms': norms.tolist(), 'output': (counter_weights @ counter_values).tolist()}
},ensure_ascii=False,indent=2))
