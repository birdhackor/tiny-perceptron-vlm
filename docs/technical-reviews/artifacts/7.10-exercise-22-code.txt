from tiny_perceptron.data import render_chat, pad_batch

example = render_chat(
    [
        {"role": "user", "content": "很長的問題"},
        {"role": "assistant", "content": "是"},
    ]
)
bx, by, valid = pad_batch([example], max_length=22)
print("保留長度", 22, "有效目標数", (by != -100).sum().item())
print("保留目標", by.tolist(), "EOS有效", bool((by == 2).any()))
assert (by != -100).sum().item() == 4
x, y = example
first = (y != -100).nonzero()[0].item()
print("首有效索引", first, "最低保留長度", first + 1, "完整長度", len(x))
