import json, os, platform, sys
import torch

def guard(event, args):
    if event in {"socket.connect", "socket.getaddrinfo", "socket.sendto", "subprocess.Popen", "os.system"}:
        raise RuntimeError("bounded offline CPU execution blocked " + event)
sys.addaudithook(guard)
torch.set_default_device("cpu")
scores = torch.tensor([[60.0, 70.0, 80.0], [80.0, 90.0, 100.0]])
before = scores.clone()
shape_tuple = tuple(scores.shape)
full_list = scores.tolist()
row_list = scores[0].tolist()
assert shape_tuple == (2, 3)
assert full_list == [[60.0, 70.0, 80.0], [80.0, 90.0, 100.0]]
assert row_list == [60.0, 70.0, 80.0]
assert torch.equal(scores, before)
full_list[0][0] = -999
assert torch.equal(scores, before)
negative_mean = scores.mean(dim=-1)
assert torch.equal(negative_mean, scores.mean(dim=1))
assert tuple(scores.mean(dim=1).shape) == (2,)
assert tuple(scores.mean(dim=0).shape) == (3,)
assert tuple(scores.mean(dim=1, keepdim=True).shape) == (2, 1)
flat = scores.reshape(6)
assert flat.tolist() == [60.0, 70.0, 80.0, 80.0, 90.0, 100.0]
assert torch.equal(flat.reshape(2, 3), scores)
assert torch.equal(scores, before)
errors = {}
for name, call in {"table_item":lambda: scores.item(),"eight_cell_reshape":lambda: scores.reshape(2, 4),"integer_mean":lambda: torch.tensor([[60,70,80],[80,90,100]]).mean(dim=1)}.items():
    try:
        call()
    except RuntimeError as e:
        errors[name] = {"type":type(e).__name__,"message":str(e)}
    else:
        raise AssertionError(name + " unexpectedly succeeded")
scalar = torch.tensor(5.0)
vector = torch.tensor([1.0,2.0,3.0])
assert (scalar.dim(), vector.dim(), scores.dim()) == (0,1,2)
positions = torch.arange(24, dtype=torch.float32).reshape(2,4,3)
assert tuple(positions[0].shape) == (4,3)
assert positions[0,3,2].item() == 11.0
assert positions[1,0,0].item() == 12.0
print(json.dumps({"mode":"supplemental API boundaries", "environment":{"python":platform.python_version(),"torch":str(torch.__version__),"torch_commit":torch.version.git_version,"device":str(scores.device),"OMP_NUM_THREADS":os.environ["OMP_NUM_THREADS"],"MKL_NUM_THREADS":os.environ["MKL_NUM_THREADS"],"threads":str(torch.get_num_threads())},"shape_tuple":shape_tuple,"dtype":str(scores.dtype),"original_tensor":scores.tolist(),"row_list":row_list,"conversions_preserve_tensor":torch.equal(scores,before),"mean_dim_negative1":negative_mean.tolist(),"mean_shapes":{"dim1":list(scores.mean(dim=1).shape),"dim0":list(scores.mean(dim=0).shape),"dim1_keepdim":list(scores.mean(dim=1,keepdim=True).shape)},"flat":flat.tolist(),"reshape_back":flat.reshape(2,3).tolist(),"ranks":{"scalar":scalar.dim(),"vector":vector.dim(),"matrix":scores.dim()},"BTD_shape":list(positions.shape),"BTD_first_index_shape":list(positions[0].shape),"BTD_first_index_values":positions[0].tolist(),"errors":errors,"assertions":"passed"},ensure_ascii=False,indent=2))
