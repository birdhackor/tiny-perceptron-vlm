import argparse
import json
import platform
import torch

parser = argparse.ArgumentParser()
parser.add_argument('--width', type=int, required=True)
parser.add_argument('--vocabs', type=int, nargs='+', required=True)
args = parser.parse_args()
assert torch.version.cuda is None
assert torch.get_num_threads() == 1
print(json.dumps({'python': platform.python_version(), 'torch': str(torch.__version__), 'torch_git_version': torch.version.git_version, 'device': 'cpu', 'width': args.width, 'vocabs': args.vocabs}))
width = args.width
for vocab in args.vocabs:
    embedding = vocab * width
    input_and_output = 2 * embedding
    assert torch.empty((vocab, width), dtype=torch.float32).element_size() == 4
    print('詞表', vocab, '輸入參數', embedding, '輸入FP32 bytes', embedding * 4, '不共享輸入輸出參數', input_and_output)
