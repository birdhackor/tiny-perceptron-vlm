import json
import os
import sys
from decimal import Decimal, ROUND_HALF_UP

value = 1.0625
inline_actual = round(1.0625, 3)
assert value.hex() == "0x1.1000000000000p+0"
assert inline_actual == 1.062
assert round(value, 3) == 1.062
assert str(Decimal("1.0625").quantize(Decimal("0.001"), rounding=ROUND_HALF_UP)) == "1.063"
