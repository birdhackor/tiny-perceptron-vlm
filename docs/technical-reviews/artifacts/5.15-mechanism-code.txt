import torch, random, json, hashlib
from decimal import Decimal
from tiny_perceptron.model import TinyLM, ModelConfig
from torch.utils.data import RandomSampler, BatchSampler

torch.manual_seed(1)
initial_state = torch.get_rng_state().clone()
a = TinyLM(ModelConfig(width=8))
advanced_state = torch.get_rng_state().clone()
assert not torch.equal(initial_state, advanced_state)
assert tuple(a.embedding.weight.shape) == (264, 8)
assert a.embedding.weight.requires_grad
lookup = a.embedding(torch.tensor([[1, 2, 1]]))
assert tuple(lookup.shape) == (1, 3, 8)
assert torch.equal(lookup[0, 0], lookup[0, 2])
assert torch.equal(lookup[0, 1], a.embedding.weight[2])
manual_std = ((a.embedding.weight.double() - a.embedding.weight.double().mean()).square().sum() / (a.embedding.weight.numel() - 1)).sqrt().item()
assert abs(manual_std - a.embedding.weight.std().item()) < 1e-6
seed = 1
b = TinyLM(ModelConfig(width=8))
assert not torch.equal(a.embedding.weight, b.embedding.weight)
torch.manual_seed(seed)
c = TinyLM(ModelConfig(width=8))
assert torch.equal(a.embedding.weight, c.embedding.weight)

orders = []
independent_orders = []
for width in [8, 16]:
    torch.manual_seed(7)
    model = TinyLM(ModelConfig(width=width))
    orders.append(list(RandomSampler(range(20))))
    data_generator = torch.Generator().manual_seed(23)
    independent_orders.append(list(RandomSampler(range(20), generator=data_generator)))
assert orders[0] != orders[1]
assert independent_orders[0] == independent_orders[1]
random.seed(10)
python_state = random.getstate()
torch.manual_seed(234)
assert random.getstate() == python_state

batches = list(BatchSampler(range(5), batch_size=2, drop_last=False))
assert batches == [[0, 1], [2, 3], [4]]
w = torch.tensor(0.0, requires_grad=True)
gradients = []
for x, y in [(1.0, 1.0), (2.0, 1.0)]:
    loss = (w * x - y).square()
    gradients.append(torch.autograd.grad(loss, w)[0].item())
assert gradients == [-2.0, -4.0]

A = [Decimal(x) for x in ['0.70', '0.72', '0.69']]
B = [Decimal(x) for x in ['0.71', '0.71', '0.73']]
differences = [b - a for a, b in zip(A, B)]
assert differences == [Decimal('0.01'), Decimal('-0.01'), Decimal('0.04')]
print(json.dumps({'embedding_shape': list(a.embedding.weight.shape), 'requires_grad': a.embedding.weight.requires_grad, 'lookup_shape': list(lookup.shape), 'std_correction_1_manual_float64': manual_std, 'std_default_float32': a.embedding.weight.std().item(), 'rng_state_changes_after_model': not torch.equal(initial_state, advanced_state), 'same_integer_variable_without_reseed_equal': torch.equal(a.embedding.weight, b.embedding.weight), 'explicit_reseed_equal': torch.equal(a.embedding.weight, c.embedding.weight), 'global_rng_data_orders': orders, 'independent_data_rng_orders': independent_orders, 'torch_seed_leaves_python_random_state_unchanged': random.getstate() == python_state, 'batches': batches, 'per_example_gradients_no_updates': gradients, 'AB_differences': [str(x) for x in differences], 'A_range': [str(min(A)), str(max(A))], 'B_range': [str(min(B)), str(max(B))], 'A_mean': str(sum(A)/3), 'B_mean': str(sum(B)/3)}, ensure_ascii=False, indent=2))
