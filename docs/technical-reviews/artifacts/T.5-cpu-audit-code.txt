def audit_report(report, where):
    if 'nll_sum' in report:
        assert abs(report['nll_sum'] / report['effective_tokens'] - report['nll']) < 1e-12, where
    if 'samples' not in report:
        return
    exact, ended, invalid = 0, 0, 0
    for sample in report['samples']:
        ids = sample['generated_ids']
        raw = ids[:-1] if ids and ids[-1] == tok.eos_id else ids
        actual = raw == tok.encode(sample['expected'])
        assert actual == sample['exact'], where
        exact += actual
        ended += ids.count(tok.eos_id) == 1 and ids[-1] == tok.eos_id
        invalid += bool(generation_report(tok, ids)['invalid_special_tokens'])
    assert exact == report['matches'], where
    assert len(report['samples']) == report['records'], where
    print('frozen', where, json.dumps({'records': report['records'], 'matches': exact, 'eos': ended, 'invalid_control_records': invalid, 'effective_tokens': report['effective_tokens'], 'rubric': report.get('rubric'), 'refusal_audit': report.get('refusal_audit')}, ensure_ascii=False))
    if where.endswith('/test') or where.endswith('/held_out_wording'):
        print('test_samples', where, json.dumps([{'prompt': s['messages'][0]['content'], 'expected': s['expected'], 'generated': s['generated'], 'exact': s['exact'], 'eos': s['eos']} for s in report['samples']], ensure_ascii=False))

def walk(value, where):
    if isinstance(value, dict):
        audit_report(value, where)
        for key, child in value.items():
            if key not in ('samples', 'history'):
                walk(child, where + '/' + key)
    elif isinstance(value, list):
        for index, child in enumerate(value):
            walk(child, where + '/' + str(index))

for experiment in ('style', 'safety', 'lora'):
    path = Path('docs/course-experiments/results') / (experiment + '.json')
    frozen = json.loads(path.read_text())
    r = frozen['results']
    print('receipt', experiment, json.dumps({'sha256': hashlib.sha256(path.read_bytes()).hexdigest(), 'revision': frozen['revision'], 'seed': frozen['seed'], 'device': frozen['device'], 'gpu': frozen['gpu'], 'evidence_status': frozen['evidence_status'], 'step_scale': frozen['step_scale']}))
    def training(value, where):
        if isinstance(value, dict):
            if 'steps' in value and 'effective_tokens' in value:
                print('training_config', where, json.dumps({k: value[k] for k in ('steps', 'effective_tokens', 'records', 'records_sha256', 'parameters', 'trainable_parameters') if k in value}))
            for k, v in value.items():
                if k not in ('samples', 'history'):
                    training(v, where + '/' + k)
    training(r, experiment)
    walk(r, experiment)
