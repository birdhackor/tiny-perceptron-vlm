import hashlib, json
from pathlib import Path
records=[]
for model in ("distillation", "multimodal_distillation"):
    source=Path("docs/course-experiments/results")/(model+".json")
    raw=source.read_bytes(); d=json.loads(raw)
    tasks=[]
    for task, data in d["results"]["tasks"].items():
        runs=[]
        for branch, r in data["runs"].items():
            if "training" not in r: continue
            tr=r["training"]
            train={k:v for k,v in tr.items() if k in ("steps","optimizer_updates","training_examples","training_sequence_chunks","effective_supervised_tokens","effective_answer_tokens","objective","alpha","temperature","temperature_squared_applied_once")}
            evaluations={split:{k:v for k,v in r.get(split,{}).items() if k in ("examples","records","supervised_tokens","effective_tokens","generation","max_new_tokens")} for split in ("validation","test")}
            runs.append({"branch":branch,"training":train,"evaluation_scope":evaluations})
        tasks.append({"task":task,"data":data["data"],"runs":runs})
    release=json.loads((Path("docs/course-experiments/public-releases")/(model+".json")).read_bytes())["release"]
    card=next(x for x in release["public_manifest"]["files"] if x["output"]=="README.md")
    local=Path("checkpoints/course")/model/"README.md"
    actual_sha=hashlib.sha256(local.read_bytes()).hexdigest()
    assert actual_sha==card["sha256"]
    records.append({"result_path":str(source),"result_sha256":hashlib.sha256(raw).hexdigest(),"training_revision":d["revision"],"reported_environment":{k:d[k] for k in ("seed","device","gpu","torch_version","python_version")},"tasks":tasks,"public_card":{"url":"https://huggingface.co/"+release["repo"]+"/resolve/"+release["revision"]+"/"+card["path"],"revision":release["revision"],"original_sha256":actual_sha,"bytes":len(local.read_bytes()),"verified_against_manifest":True}})
print(json.dumps({"inspection_scope":"只讀固定原始結果與原公開卡；沒有重訓、下载模型或執行GPU。設定只支持CE+KL軟目標組；CE/teacher-hard不使用KL。","records":records},ensure_ascii=False,indent=2))
