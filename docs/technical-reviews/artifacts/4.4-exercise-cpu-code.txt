import json
import math
import platform
import sys
from pathlib import Path

sys.path.insert(0, str(Path.cwd()))
import torch
from torch.nn import functional as F
from tiny_perceptron.modern import DenseFFN

assert torch.version.cuda is None
torch.set_default_device("cpu")
print("environment", json.dumps({"python": platform.python_version(), "torch": str(torch.__version__), "torch_git_version": str(torch.version.git_version), "device": "cpu", "dtype": "float32", "OMP_NUM_THREADS": "1", "MKL_NUM_THREADS": "1"}))

# 原例，與正文相同操作順序。
torch.manual_seed(42)
layer = DenseFFN(4)
x = torch.ones(1, 2, 4)
y = layer(x)
print("original_weight_shapes", list(layer.up.weight.shape), list(layer.down.weight.shape))
print("original_shapes", list(x.shape), list(layer.up(x).shape), list(y.shape))
print("original_y", y.detach().tolist())
print("original_two_positions_allclose", torch.allclose(y[:, 0], y[:, 1]))
assert torch.allclose(y[:, 0], y[:, 1])

# 本節練習：先保留y，只改第1位置一格。
x[0, 1, 0] = 2
changed = layer(x)
print("exercise_changed", changed.detach().tolist())
print("exercise_position0_allclose", torch.allclose(changed[:, 0], y[:, 0]))
print("exercise_position1_allclose", torch.allclose(changed[:, 1], y[:, 1]))
print("exercise_position0_max_abs_delta", (changed[:, 0] - y[:, 0]).abs().max().item())
print("exercise_position1_max_abs_delta", (changed[:, 1] - y[:, 1]).abs().max().item())
assert torch.allclose(changed[:, 0], y[:, 0])
assert not torch.allclose(changed[:, 1], y[:, 1])

# 固定本環境、初始化與操作順序，重新設定種子。
torch.manual_seed(42)
replay_layer = DenseFFN(4)
replay_y = replay_layer(torch.ones(1, 2, 4))
parameter_equal = all(torch.equal(layer.state_dict()[k], v) for k, v in replay_layer.state_dict().items())
print("same_environment_reseed_parameters_equal", parameter_equal)
print("same_environment_reseed_y_equal", torch.equal(y, replay_y))
assert parameter_equal and torch.equal(y, replay_y)
print("random_initialization_weight_bias_ranges", {k: [v.min().item(), v.max().item()] for k, v in layer.state_dict().items()})

# 最後軸的展開及殘差形狀。
manual = F.gelu(x @ layer.up.weight.T + layer.up.bias) @ layer.down.weight.T + layer.down.bias
print("manual_affine_gelu_max_abs_delta", (changed - manual).abs().max().item())
assert torch.allclose(changed, manual, atol=1e-7, rtol=1e-6)
print("residual_sum_shape", list((x + changed).shape))
print("default_activation_and_gate", layer.activation, layer.gate)
assert layer.activation == "gelu" and layer.gate is None

# GELU精確公式及ReLU，只作數值機制核對。
samples=torch.tensor([-3., -1., -.1, 0., .1, 1., 3.], dtype=torch.float64)
gelus=F.gelu(samples)
formula=.5*samples*(1+torch.erf(samples/math.sqrt(2)))
relus=F.relu(samples)
print("activation_input", samples.tolist())
print("gelu", gelus.tolist())
print("relu", relus.tolist())
print("gelu_formula_max_abs_delta", (gelus-formula).abs().max().item())
assert torch.allclose(gelus, formula, atol=1e-14, rtol=1e-14)
assert gelus[2] < 0 and relus[2] == 0
