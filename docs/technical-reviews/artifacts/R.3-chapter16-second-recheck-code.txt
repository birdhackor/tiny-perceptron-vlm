from pathlib import Path
import re,json,hashlib,platform
root=Path("docs/technical-reviews/artifacts")
sha=lambda b:hashlib.sha256(b).hexdigest()
old=(root/"R.3-chapter16-current-source.txt").read_bytes()
new=Path("course/chapters/16.md").read_bytes()
prior=json.loads((root/"R.3-before-chapter16-second-recheck-report.json").read_text())
assert sha(old)==next(s for s in prior["sources"] if s["id"]=="s_ch_16")["sha256"]=="bbeafa87829558c74543be6476f3210e0dd01d04fa3572bc82abd9b97707a016"
assert sha(new)=="05794da02b7d951f3ab9bd2d305e9f93497a277b5e4a971f857e22ff8dd8f6b5"
assert new==(root/"R.3-chapter16-second-current-source.txt").read_bytes()
def sections(raw):
    heads=list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+",raw))
    return {h[1].decode():{"raw":raw[h.start():heads[i+1].start() if i+1<len(heads) else len(raw)],"first_line":raw[:h.start()].count(b"\n")+1} for i,h in enumerate(heads)},raw[:heads[0].start()]
a,old_intro=sections(old);b,new_intro=sections(new)
assert a.keys()==b.keys()
changed=[sid for sid in a if a[sid]["raw"]!=b[sid]["raw"]]
assert changed==["16.1"]
assert old_intro==new_intro
assert re.findall(rb"```[^\n]*\n[\s\S]*?```",old)==re.findall(rb"```[^\n]*\n[\s\S]*?```",new)
assert re.findall(r"(?m)^練習[^\n]*",old.decode())==re.findall(r"(?m)^練習[^\n]*",new.decode())
assert re.findall(rb"!\[[^\]]*\]\(([^)]+\.svg)\)",old)==re.findall(rb"!\[[^\]]*\]\(([^)]+\.svg)\)",new)
previous=json.loads((root/"R.3-chapters15-16-recheck-output.json").read_text())
previous16=next(c for c in previous["chapter_identity_and_required_scope_checks"] if c["chapter"]=="16")
required=[]
for scope in previous16["required_ranges"]:
    text="\n".join(old.decode().splitlines()[scope["current_first_line"]-1:scope["current_last_line"]])+"\n"
    assert sha(text.encode())==scope["current_required_scope_sha256"]
    assert text.encode() in new
    start=new.index(text.encode())
    required.append({"old_first_line":scope["current_first_line"],"old_last_line":scope["current_last_line"],"current_first_line":new[:start].count(b"\n")+1,"current_last_line":new[:start].count(b"\n")+text.count("\n"),"scope_sha256_unchanged":sha(text.encode()),"required_support_bytes_identical":True})
for sid in ["16.3","16.9"]:assert a[sid]["raw"]==b[sid]["raw"]
assert sha(b["16.1"]["raw"])=="2cc9c51149a14143fd03708de1a5d98534acee6170de3308a850ad99dd394507"
primary=sections(Path("course/README.md").read_bytes())[0]["R.3"]["raw"]
assert sha(primary)==prior["source_sha256"] and primary==(root/"R.3-source.txt").read_bytes()
for artifact in prior["artifacts"]:assert sha(Path(artifact["path"]).read_bytes())==artifact["sha256"],artifact["id"]
for path,digest in prior["figure_sha256"].items():assert sha(Path(path).read_bytes())==digest,path
print(json.dumps({"environment":{"python":platform.python_version(),"device":"cpu","OMP_NUM_THREADS":"1","MKL_NUM_THREADS":"1","torch":"未使用；原文身份讀取"},"old_full_sha256":sha(old),"new_full_sha256":sha(new),"old_bytes":len(old),"new_bytes":len(new),"changed_sections":changed,"old_16_1_raw_sha256":sha(a["16.1"]["raw"]),"current_16_1_raw_sha256":sha(b["16.1"]["raw"]),"intro_raw_sha256_unchanged":sha(new_intro),"required_scopes":required,"section16_3_whole_raw_sha256_unchanged":sha(b["16.3"]["raw"]),"section16_9_whole_raw_sha256_unchanged":sha(b["16.9"]["raw"]),"primary_raw_sha256_unchanged":sha(primary),"all30_prior_artifact_sha256_unchanged":True,"all4_necessary_figure_sha256_unchanged":True,"all_code_fences_exercise_opening_lines_SVG_paths_unchanged":True,"scope":"只作此次真變更的16章來源身份／必要支持比較；不重做15章，不重跑既有例／圖，不為16.1其他primary下判詞。"},ensure_ascii=False,indent=2))
