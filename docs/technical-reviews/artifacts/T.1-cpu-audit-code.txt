def digest(data):
    return hashlib.sha256(data).hexdigest()

def save(name, value):
    (OUT / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")

def load(path):
    return json.loads(Path(path).read_text(encoding="utf-8"))

def rows(path):
    return [json.loads(line) for line in Path(path).read_text(encoding="utf-8").splitlines() if line]

def run(command):
    proc = subprocess.run(command, capture_output=True, text=True, timeout=40)
    assert proc.returncode == 0, proc.stderr
    return {"command": " ".join(command), "exit_code": proc.returncode,
            "stdout": proc.stdout, "stderr": proc.stderr, "environment": ENV}

manifest = load("assets/training/manifest.json")
asset_checks = []
for asset in manifest["assets"]:
    archive = Path(asset["archive"])
    assert archive.stat().st_size == asset["archive_bytes"]
    assert digest(archive.read_bytes()) == asset["archive_sha256"]
    expected = {r["path"]: r for r in asset["files"]}
    seen = set()
    with tarfile.open(archive, "r:gz") as package:
        for item in package:
            assert item.isfile() and item.name in expected and item.name not in seen
            data = package.extractfile(item).read()
            record = expected[item.name]
            assert len(data) == record["bytes"] and digest(data) == record["sha256"]
            seen.add(item.name)
    assert seen == set(expected)
    metadata = load(f"assets/training/sources/{asset['id']}.json")
    pins = {k: v for k, v in metadata.items() if k in
            ("source_revision", "source_sha", "source_repository_revision", "hf_revision", "revision", "license")}
    asset_checks.append({"asset": asset["id"], "archive_sha256": asset["archive_sha256"],
                         "files_verified": len(seen), "training_records": asset["training_records"],
                         "license": asset["license"], "metadata_sha256": digest(Path(f"assets/training/sources/{asset['id']}.json").read_bytes()),
                         "metadata_pins": pins})
result["asset_integrity"] = asset_checks

text_checks = {}
for asset, filename in (("tinystories", "tinystories-train-512.jsonl"),
                        ("chinese-poetry", "chinese-classical-train-365.jsonl")):
    path = Path("data/training/text-initial") / filename
    original_rows = rows(path)
    assert all(digest(r["text"].encode()) == r["text_sha256"] for r in original_rows)
    schema_keys = sorted({key for row in original_rows for key in row})
    paired_dialogues = sum("messages" in row or ("question" in row and "answer" in row) for row in original_rows)
    assert paired_dialogues == 0
    if asset == "tinystories":
        prefix = Path("data/training/text-initial/tinystories-train-prefix-complete.txt").read_text()
        pieces = prefix.split("<|endoftext|>")
        assert not pieces[-1].strip()
        complete = [piece.strip() for piece in pieces[:-1]]
        assert complete == [r["text"] for r in original_rows]
        assert len(complete) == 512 and all(r["source_split"] == "train" for r in original_rows)
    else:
        upstream = load("data/training/text-initial/tang300-source.json")
        texts = ["\n".join([r["title"], r["author"], *r["paragraphs"]]) for r in upstream]
        assert len(upstream) == 366 and len(set(texts)) == 365
        assert list(dict.fromkeys(texts)) == [r["text"] for r in original_rows]
        assert all(r["source_split"] == "unpartitioned" for r in original_rows)
    parts = split_records(_deduplicate_text(original_rows), seed=42)
    measured = {}
    for side, side_rows in parts.items():
        data = "".join(json.dumps(r, ensure_ascii=False) + "\n" for r in side_rows).encode()
        frozen = formal_text["results"]["runs"][asset]["data"][side]
        assert len(side_rows) == frozen["records"] and digest(data) == frozen["sha256"]
        measured[side] = {"records": len(side_rows), "sha256": digest(data)}
    text_checks[asset] = {"input_sha256": digest(path.read_bytes()), "source_records": len(original_rows),

                           "schema_keys_union": schema_keys, "paired_dialogue_records": paired_dialogues,
                           "split_reconstructed_without_training": measured}

formal_modal = load("docs/course-experiments/results/real_modal.json")
modal_checks = {}
tok = ByteTokenizer()
for asset, run_result in formal_modal["results"].items():
    checks = {}
    for side, split in run_result["data"]["splits"].items():
        families = [r["family"] for r in split["records"]]
        assert len(families) == split["count"]
        if asset == "fashion-mnist":
            counts = Counter(re.search(r'label-(\d+)', name)[1] for name in families)
            assert set(counts) == set(map(str, range(10)))
            assert set(counts.values()) == ({3} if side == "train" else {1})
        else:
            speaker = {name.split("_")[1] for name in families}
            assert speaker == {dict(train="jackson", validation="nicolas", test="theo")[side]}
            counts = Counter(name.split("_")[0] for name in families)
            assert set(counts) == set(map(str, range(10))) and set(counts.values()) == {2}
        checks[side] = {"records": len(families), "family_sha256": digest(json.dumps(families).encode()),
                        "per_label": dict(sorted(counts.items())), "speakers": sorted(speaker) if asset == "fsdd" else []}
    assert not any(set(run_result["data"]["splits"][a]["records"][i]["family"] for i in range(run_result["data"]["splits"][a]["count"])) &
                   set(r["family"] for r in run_result["data"]["splits"][b]["records"])
                   for a, b in (("train", "validation"), ("train", "test"), ("validation", "test")))
    for side in ("validation", "test"):
        report = run_result[side]
        comparisons = []
        for sample in report["samples"]:
            raw_ids = sample["generated_ids"]
            content = raw_ids[:raw_ids.index(tok.eos_id)] if tok.eos_id in raw_ids else raw_ids
            correct = content == tok.encode(sample["target"])
            assert correct == sample["exact_match"]
            comparisons.append({"row": sample["row"], "family": sample["family"], "target": sample["target"],
                                "generated": sample["generated"], "generated_ids": raw_ids,
                                "exact_match_recomputed": correct, "eos": sample["eos"]})
        assert len(comparisons) == report["examples"]
        assert sum(r["exact_match_recomputed"] for r in comparisons) == report["correct"]
        checks[side]["scoring"] = {"correct": report["correct"], "examples": report["examples"],
                                    "effective_tokens": report["effective_tokens"], "samples": comparisons}
    modal_checks[asset] = checks
result["frozen_modal_denominators"] = modal_checks

audio_checks = []
with tempfile.TemporaryDirectory(prefix="T.1-resample-") as temp:
    for conversion in formal_modal["results"]["fsdd"]["resampling"]:
        source = Path("data/training/fsdd-initial/recordings") / conversion["source"]
        info = sf.info(source)
        values, rate = sf.read(source, dtype="float32")
        assert digest(source.read_bytes()) == conversion["source_sha256"]
        assert rate == 8000 and info.channels == 1 and info.subtype == "PCM_16"
        output = _resample_8_to_16(values)
        destination = Path(temp) / source.name
        sf.write(destination, output, 16000, subtype="FLOAT")
        assert len(output) == 2 * len(values) == conversion["samples_after"]
        assert len(values) == conversion["samples_before"]
        assert len(values) / rate == len(output) / 16000
        audio_checks.append({"source": source.name, "source_sha256": conversion["source_sha256"],
                             "derived_sha256_recomputed": digest(destination.read_bytes()),
                             "frozen_derived_sha256": conversion["derived_sha256"],
                             "whole_file_sha_matches_frozen": digest(destination.read_bytes()) == conversion["derived_sha256"],
                             "source_rate": rate, "target_rate": 16000, "samples_before": len(values),
                             "samples_after": len(output), "duration_seconds": len(values) / rate})
    first = Path("data/training/fsdd-initial/recordings/0_jackson_5.wav")
    record = {"audio": str(first.resolve()), "question": "digit?", "answer": "0"}
    try:
        modal_example(record, ROOT, "audio")
    except ValueError as error:
        guard = str(error)
    else:
        raise AssertionError("8kHz should be rejected")
    record["audio"] = str(Path(temp) / first.name)
    _, _, _, wave = modal_example(record, ROOT, "audio")
    assert len(wave) == 9182 and wave.device.type == "cpu"
    result["audio_entry"] = {"8khz_error": guard, "16khz_accepted_points": len(wave), "device": str(wave.device)}
result["audio_resampling"] = {"count": len(audio_checks), "source_hashes_all_match": True,
       "replay_scope": "只驗證點數/時間/入口契約；重寫 FLOAT WAV 的整檔 SHA 未宣稱等同凍結產物，亦未重播 GPU 生成成績", "recordings": audio_checks}
