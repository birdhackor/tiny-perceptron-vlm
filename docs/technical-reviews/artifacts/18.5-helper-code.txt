import json, sys, torch
from tiny_perceptron.alignment import distillation_kl, distillation_loss
from torch.nn import functional as F
student = torch.tensor([[[1., 0., -1.], [0., 1., 2.]]], requires_grad=True)
teacher = torch.tensor([[[4., 1., 0.], [1., 2., 0.]]], requires_grad=True)
labels = torch.tensor([[-100, 1]])
T = 2.0
p = (teacher.detach() / T).softmax(-1)
log_q = (student / T).log_softmax(-1)
raw = F.kl_div(log_q, p, reduction="none").sum(-1)[labels != -100].mean()
helper = distillation_kl(student, teacher, labels, temperature=T)
combined = distillation_loss(student, teacher, labels, alpha=0.5, temperature=T)
ce = F.cross_entropy(student.reshape(-1,3), labels.reshape(-1), ignore_index=-100)
helper.backward()
print(json.dumps({"python":sys.version,"torch":str(torch.__version__),"torch_git":torch.version.git_version,"device":str(student.device),"raw_KL":raw.item(),"helper":helper.item(),"ratio":(helper/raw).item(),"combined":combined.item(),"expected_combined":(0.5*ce+0.5*helper).item(),"student_grad":student.grad.tolist(),"teacher_grad":str(teacher.grad)},ensure_ascii=False))
assert torch.equal(helper, raw*T*T)
assert torch.equal(combined, 0.5*ce+0.5*helper)
assert teacher.grad is None
