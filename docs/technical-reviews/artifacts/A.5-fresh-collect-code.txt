from pathlib import Path
import os,sys,subprocess,json,torch
base=Path("docs/technical-reviews/artifacts")
environment={"python":sys.version,"torch":str(torch.__version__),"torch_git_version":str(torch.version.git_version),"device":"cpu","OMP_NUM_THREADS":os.environ.get("OMP_NUM_THREADS",""),"MKL_NUM_THREADS":os.environ.get("MKL_NUM_THREADS","")}
for name in ("original","exercise"):
    rel=base / ("A.5-fresh-"+name+"-code.txt")
    command="OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 .venv/bin/python "+str(rel)
    run=subprocess.run([sys.executable,str(rel)],capture_output=True,text=True,timeout=10,env=dict(os.environ,OMP_NUM_THREADS="1",MKL_NUM_THREADS="1"))
    record={"command":command,"environment":environment,"exit_code":run.returncode,"stdout":run.stdout,"stderr":run.stderr,"source":str(rel),"boundary":"獨立新程序、CPU、10 秒 timeout；無網路、無權重載入與參數更新"}
    (base/("A.5-fresh-"+name+"-execution.json")).write_text(json.dumps(record,ensure_ascii=False,indent=2)+"\n")
    print(name,run.returncode,run.stdout)
    assert run.returncode==0
