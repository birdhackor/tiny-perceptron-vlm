import json, platform, os, torch
from tiny_perceptron.data import render_chat

torch.set_num_threads(1)
question = "2+2=?"
answers = {"簡短": "4", "生動": "4，共有四個"}
rows=[]
for style, answer in answers.items():
    inputs, labels = render_chat([
        {"role": "user", "content": question},
        {"role": "assistant", "content": answer},
    ])
    n=int((labels != -100).sum())
    print(style, "總位置", len(inputs), "學習位置", n)
    rows.append({"style":style,"answer":answer,"answer_utf8_bytes":list(answer.encode('utf-8')),"inputs":inputs.tolist(),"labels":labels.tolist(),"total_positions":len(inputs),"learning_positions":n,"label_dtype":str(labels.dtype)})
assert rows[0]['learning_positions'] == 2
assert rows[1]['learning_positions'] == 17
assert rows[1]['total_positions'] == 25
assert rows[1]['learning_positions'] < 38
print(json.dumps({"environment":{"python":platform.python_version(),"torch":torch.__version__,"torch_source":torch.version.git_version,"device":"cpu","threads":str(torch.get_num_threads()),"OMP_NUM_THREADS":os.environ['OMP_NUM_THREADS'],"MKL_NUM_THREADS":os.environ['MKL_NUM_THREADS']},"rows":rows,"predicted_change":"減少；37 bytes 的原答案改為16 bytes，含EOS由38減為17","weight_updates":0},ensure_ascii=False,indent=2))
