from pathlib import Path
import ast, hashlib, importlib.metadata, platform, re, sys, unicodedata
print("environment",sys.version,platform.machine(),"torch-installed",importlib.metadata.version("torch"),"device=cpu; standard-library computation; no torch execution")
print("Unicode data",unicodedata.unidata_version)
raw=Path("docs/technical-reviews/artifacts/6.2-recheck-section-raw.txt").read_bytes()
print("current section SHA",hashlib.sha256(raw).hexdigest())
source=re.findall(rb"```python\n(.*?)```",raw,re.S)[0]
old_source=Path("docs/technical-reviews/artifacts/6.2-original-code.txt").read_bytes()
print("original Python fence unchanged",source==old_source)
assert source==old_source
original=ast.parse(source)
function=next(n for n in original.body if isinstance(n,ast.FunctionDef) and n.name=="merge")
namespace={}
exec(compile(ast.Module(body=[function],type_ignores=[]),"current-6.2-merge","exec"),namespace)
raw_text="Ａ"
normalized=unicodedata.normalize("NFKC",raw_text)
print("NFKC example",repr(raw_text),[f"U+{ord(c):04X}" for c in raw_text],"->",repr(normalized),[f"U+{ord(c):04X}" for c in normalized])
assert normalized=="A" and raw_text!=normalized
assert unicodedata.normalize("NFKC","A")==normalized
print("fullwidth and ASCII inputs have equal normalized outputs",True)
row=list("ＡB")
result=namespace["merge"](row,("Ａ","B"))
print("current toy merge",repr(row),"->",repr(result),"joined preserved", "".join(result)=="".join(row))
assert result==["ＡB"] and "".join(result)=="".join(row)
print("new normalization example and toy/no-normalization boundary passed")
