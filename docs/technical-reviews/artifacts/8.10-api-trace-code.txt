import sys, os, json
import torch
assert torch.version.cuda is None
assert not torch.cuda.is_available()
assert os.environ['OMP_NUM_THREADS'] == '1'
assert os.environ['MKL_NUM_THREADS'] == '1'
torch.set_num_threads(1)
torch.set_default_device('cpu')
def guard(event, args):
    if event.startswith('socket.') or event in {'subprocess.Popen', 'os.system', 'os.exec', 'os.posix_spawn'}:
        raise RuntimeError('offline CPU audit guard: ' + event)
sys.addaudithook(guard)
def environment():
    return {'python': sys.version, 'python_executable': sys.executable, 'torch': str(torch.__version__), 'torch_git_version': str(torch.version.git_version), 'device': 'cpu', 'cuda_build': str(torch.version.cuda), 'cuda_available': str(torch.cuda.is_available()), 'OMP_NUM_THREADS': os.environ['OMP_NUM_THREADS'], 'MKL_NUM_THREADS': os.environ['MKL_NUM_THREADS']}
import re, io, contextlib, hashlib
from pathlib import Path
raw=Path('docs/technical-reviews/artifacts/8.10-section-raw.txt').read_bytes()
code=re.search(rb'```python\n(.*?)```',raw,re.S).group(1)
namespace={}
stdout=io.StringIO()
with contextlib.redirect_stdout(stdout):
    exec(compile(code,'course/chapters/08.md#8.10:original-fence','exec'),namespace)
human=namespace['human']; judge=namespace['recorded_judge']; agree=namespace['agree']
assert agree.tolist() == [True,False,True,True]
assert agree.float().tolist() == [1.0,0.0,1.0,1.0]
assert agree.float().mean().item() == 0.75
assert (~agree).tolist() == [False,True,False,False]
assert (~agree).nonzero().tolist() == [[1]]
assert (~agree).nonzero().flatten().tolist() == [1]
assert agree[3].item() is True
reordered=judge[torch.tensor([3,1,2,0])]
easy_human=torch.ones(4,dtype=torch.int64); always_one=torch.ones(4,dtype=torch.int64)
