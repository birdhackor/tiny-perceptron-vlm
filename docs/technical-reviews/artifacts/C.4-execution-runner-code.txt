import json
import os
import platform
import subprocess
from pathlib import Path

import torch

p = Path("docs/technical-reviews/artifacts")
env = {**os.environ, "OMP_NUM_THREADS": "1", "MKL_NUM_THREADS": "1", "PYTHONPATH": "."}
versions = {"python": platform.python_version(), "torch": str(torch.__version__), "device": "cpu",
            "OMP_NUM_THREADS": env["OMP_NUM_THREADS"], "MKL_NUM_THREADS": env["MKL_NUM_THREADS"]}
for label, filename in [("original", "C.4-original-code.txt"), ("exercise", "C.4-exercise-code.txt"), ("recompute", "C.4-recompute-code.txt")]:
    command = ["env", "OMP_NUM_THREADS=1", "MKL_NUM_THREADS=1", "PYTHONPATH=.", ".venv/bin/python", str(p / filename)]
    result = subprocess.run(command, env=env, capture_output=True, text=True, timeout=45)
    record = {"command": " ".join(command), "cwd": str(Path.cwd()), "code_path": str(p / filename),
              "stdout": result.stdout, "stderr": result.stderr, "exit_code": result.returncode,
              "environment": versions, "scope": "Bounded offline CPU; no training, cloud, GPU, or downloads."}
    (p / f"C.4-{label}-execution.json").write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
    print(json.dumps(record, ensure_ascii=False))
    if result.returncode:
        raise SystemExit(result.returncode)
