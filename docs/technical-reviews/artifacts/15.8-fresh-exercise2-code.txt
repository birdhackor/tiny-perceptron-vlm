import json
import platform
import torch

# 原文沒有指定兩組分數；各組只有一位嚴格最大者。
first = torch.tensor([[0.2, 0.1, 0.0]]).repeat(6, 1)
second = torch.tensor([[0.1, 0.2, 0.0]]).repeat(6, 1)
scores = torch.cat((first, second), dim=0)
prob = scores.softmax(-1)
chosen = prob.topk(1, dim=-1).indices
counts = torch.bincount(chosen.flatten(), minlength=3)
assert chosen.flatten().tolist() == [0] * 6 + [1] * 6
assert counts.tolist() == [6, 6, 0]
print(json.dumps({
    "exercise": "15.8 第二項，兩組不同分數各六列",
    "environment": {"python": platform.python_version(), "torch": torch.__version__, "torch_commit": torch.version.git_version, "device": str(scores.device), "threads": str(torch.get_num_threads())},
    "group_scores": [first[0].tolist(), second[0].tolist()],
    "mean_probability": prob.mean(0).tolist(),
    "chosen": chosen.flatten().tolist(),
    "counts": counts.tolist(),
}, ensure_ascii=False, indent=2))
