records, read_scope, codes = [], [], {}
for path, lesson in SPECS:
    raw = Path(path).read_bytes()
    headings = list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+", raw))
    index = next(i for i, h in enumerate(headings) if h[1].decode() == lesson)
    end = headings[index + 1].start() if index + 1 < len(headings) else len(raw)
    body = raw[headings[index].start():end]
    fences = re.findall(r"```python\n(.*?)```", body.decode(), re.S)
    assert len(fences) == 1, (lesson, len(fences))
    codes[lesson] = fences[0]
    read_scope.append({"source": path + "#" + lesson,
                       "raw_sha256": hashlib.sha256(body).hexdigest(),
                       "file_sha256": hashlib.sha256(raw).hexdigest(),
                       "svg": re.findall(r"\(([^)]+\.svg)\)", body.decode())})

def run(lesson, variant, code=None, expected_exception=None):
    code = codes[lesson] if code is None else code
    namespace = {}
    stdout = io.StringIO()
    record = {"source": lesson, "variant": variant,
              "code_sha256": hashlib.sha256(code.encode()).hexdigest()}
    if variant != "original":
        record["variant_code"] = code
    with contextlib.redirect_stdout(stdout):
        try:
            exec(compile(code, lesson + ":" + variant, "exec"), namespace)
        except Exception as error:
            record["exception"] = type(error).__name__
            record["exception_message"] = str(error)
            if expected_exception is None or not isinstance(error, expected_exception):
                raise
            record["result"] = "expected_exception"
        else:
            assert expected_exception is None, (lesson, variant, "expected exception absent")
            record["result"] = "completed"
    record["stdout"] = stdout.getvalue()
    if lesson == "5.7":
        record["post_update_max_logit_difference"] = float(
            (namespace["model"](namespace["x"])["logits"] -
             namespace["loaded"](namespace["x"])["logits"]).detach().abs().max())
    records.append(record)
    return namespace

hand = [math.exp(value) / sum(math.exp(t) for t in [2, 1, 0]) for value in [2, 1, 0]]
