import json
import platform
import torch
from tiny_perceptron.model import TinyLM, ModelConfig

print(json.dumps({"python": platform.python_version(), "torch": torch.__version__, "torch_git_version": torch.version.git_version, "device": "cpu", "threads": torch.get_num_threads()}))
plans = []
for width in [8, 16]:
    parameters = TinyLM(ModelConfig(width=width)).description()["parameters"]
    for records in [16, 64]:
        batch_size, answers_per_record, steps = 4, 8, 100
        if (width, records) == (8, 16):
            answers_per_record = 4
            steps = int(3200 / (batch_size * answers_per_record))
        plan = {"width": width, "不同記錄": records, "參數格數": parameters, "更新步數": steps, "有效答案預算": batch_size * answers_per_record * steps, "品質": "尚待訓練與量測"}
        plans.append(plan)
        print(json.dumps(plan, ensure_ascii=False))
assert plans[0]["更新步數"] == 200
assert all(plan["有效答案預算"] == 3200 for plan in plans)
unchanged_budget = 4 * 4 * 100
assert unchanged_budget == 1600
print(json.dumps({"不增加步數的答案數": unchanged_budget, "執行更新次數": 0}, ensure_ascii=False))
