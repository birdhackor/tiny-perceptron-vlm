import hashlib
import json
import math
import os
import platform
import random
import runpy
import statistics
from pathlib import Path
from types import SimpleNamespace

import torch

from scripts.prepare_data import generate_records
from scripts.course_experiments.common import records_sha256, split_records, text_examples
from tiny_perceptron.data import ByteTokenizer, IGNORE
from tiny_perceptron.model import ModelConfig, TinyLM

assert os.environ['OMP_NUM_THREADS'] == os.environ['MKL_NUM_THREADS'] == '1'
assert torch.get_num_threads() == 1
root = Path('.')
artifact = root / 'docs/technical-reviews/artifacts'
sha = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
environment = {
    'python': platform.python_version(),
    'torch': str(torch.__version__),
    'torch_git_version': str(torch.version.git_version),
    'device': 'cpu',
    'OMP_NUM_THREADS': os.environ['OMP_NUM_THREADS'],
    'MKL_NUM_THREADS': os.environ['MKL_NUM_THREADS'],
}
assert str(torch.__version__) == '2.14.1+cpu'
assert torch.version.git_version == '5c4886908584029761b579af026dcfb627c84070'
report = json.loads((root / 'docs/course-experiments/results/efficiency.json').read_text())
sft = json.loads((root / 'docs/course-experiments/results/sft.json').read_text())
data = split_records(generate_records('attributes-sft'), seed=42)
data_checks = {}
for split, rows in data.items():
    examples = text_examples(rows, mode='sft', max_length=128)
    count = sum(int((labels != IGNORE).sum()) for _, labels in examples)
    assert records_sha256(rows) == report['results']['dataset'][split]['sha256']
    data_checks[split] = {'records': len(rows), 'targets': count, 'semantic_sha256': records_sha256(rows)}
assert [data_checks[x]['targets'] for x in ('train', 'validation', 'test')] == [315, 36, 69]
sampler = random.Random(42)
examples = text_examples(data['train'], mode='sft', max_length=128)
sampled_targets = sum(
    sum(int((labels != IGNORE).sum()) for _, labels in sampler.choices(examples, k=16))
    for _ in range(100)
)
assert sampled_targets == 11278
tok = ByteTokenizer()
model_checks = {}
for name, expected_kv in [('mha', 4), ('gqa', 1)]:
    raw = report['results']['models'][name]
    config = ModelConfig(**raw['model']['config'])
    assert config.heads == 4 and config.kv_heads == expected_kv
    assert config.width == 64 and config.layers == 2 and config.backend == 'manual'
    model = TinyLM(config).eval()
    parameters = sum(p.numel() for p in model.parameters())
    assert parameters == raw['model']['parameters']
    prefix_ids = [tok.bos_id] + tok.encode('Once upon a time, a girl')
    assert len(prefix_ids) == 25
    with torch.no_grad():
        cache = model(torch.tensor([prefix_ids]))['cache']
    cache_bytes = sum(t.numel() * t.element_size() for pair in cache for t in pair)
    formula = 2 * config.layers * 1 * expected_kv * 25 * 16 * 4
    assert cache_bytes == formula == raw['cache']['prefill_cache_bytes']
    assert all(t.dtype == torch.float32 and t.element_size() == 4 for pair in cache for t in pair)
    heldout = {}
    for split, saved in raw['heldout'].items():
        denominator = sum(len(tok.encode(sample['expected'])) + 1 for sample in saved['samples'])
        matched = 0
        all_eos = True
        for sample in saved['samples']:
            ids = sample['generated_ids']
            assert 0 < len(ids) <= 32
            ended = tok.eos_id in ids
            content = ids[:ids.index(tok.eos_id)] if ended else ids
            exact = content == tok.encode(sample['expected'])
            assert sample['exact'] == exact and sample['eos'] == ended
            assert ids[-1] == tok.eos_id and ids.count(tok.eos_id) == 1
            matched += int(exact)
            all_eos = all_eos and ended
        mean = saved['nll_sum'] / denominator
        assert math.isclose(mean, saved['nll'], rel_tol=0, abs_tol=1e-15)
        assert denominator == saved['effective_tokens'] == data_checks[split]['targets']
        assert matched == saved['matches'] and len(saved['samples']) == saved['records']
        assert saved['eos_rate'] == 1.0 and all_eos
        heldout[split] = {
            'nll_sum': saved['nll_sum'], 'targets': denominator, 'mean_nll': mean,
            'rounded_nll': f'{mean:.5f}', 'matches': matched,
            'records': len(saved['samples']), 'all_normal_eos': all_eos,
        }
    training = raw['training']
    assert training['requested_steps'] == training['steps'] == training['optimizer_updates'] == 100
    assert training['skipped_updates'] == 0 and not training['budget_exhausted']
    assert training['effective_tokens'] == sampled_targets
    latency = raw['cache']['cached_decode']
    assert latency['warmup_calls'] == 3 and latency['measured_calls'] == 9
    assert latency['synchronized'] and len(latency['samples_seconds']) == 9
    median = statistics.median(latency['samples_seconds'])
    assert median == latency['median_seconds']
    model_checks[name] = {
        'parameters': parameters, 'prefill_k_shapes': [list(pair[0].shape) for pair in cache],
        'prefill_cache_bytes': cache_bytes, 'cache_formula': formula,
        'heldout': heldout, 'training_updates': 100, 'sampled_targets': sampled_targets,
        'cached_total_median_seconds': median, 'cached_total_ms_rounded': f'{1000 * median:.3f}',
    }
assert set(report['results']['models']) == {'mha', 'gqa'}
assert model_checks['mha']['parameters'] == 141568
assert model_checks['gqa']['parameters'] == 129280
assert model_checks['mha']['prefill_cache_bytes'] == 4 * model_checks['gqa']['prefill_cache_bytes']
assert model_checks['mha']['heldout']['test']['matches'] == model_checks['gqa']['heldout']['test']['matches'] == 6
assert model_checks['gqa']['heldout']['validation']['mean_nll'] < model_checks['mha']['heldout']['validation']['mean_nll']
assert model_checks['gqa']['heldout']['test']['mean_nll'] < model_checks['mha']['heldout']['test']['mean_nll']
source_checks = {}
for name, repo_path in [
    ('frozen-architecture', 'scripts/course_experiments/architecture.py'),
    ('frozen-common', 'scripts/course_experiments/common.py'),
]:
    retained = artifact / ('16.4-' + name + '.txt')
    downloaded = retained if retained.exists() else Path('/tmp/16.4-' + name + '.source')
    expected = report['code_sha256'][repo_path]
    assert sha(downloaded) == expected
    source_checks[repo_path] = {'revision': report['revision'], 'sha256': expected}
    (artifact / ('16.4-' + name + '.txt')).write_bytes(downloaded.read_bytes())
sft_retained = artifact / '16.4-frozen-sft-text.txt'
sft_text = sft_retained if sft_retained.exists() else Path('/tmp/16.4-frozen-sft-text.source')
assert sha(sft_text) == sft['code_sha256']['scripts/course_experiments/text.py']
(artifact / '16.4-frozen-sft-text.txt').write_bytes(sft_text.read_bytes())
for path in ['tiny_perceptron/model.py', 'tiny_perceptron/attention.py', 'tiny_perceptron/data.py']:
    assert sha(path) == report['code_sha256'][path]
    source_checks[path] = {'revision': report['revision'], 'sha256': sha(path)}
frozen = runpy.run_path(str(artifact / '16.4-frozen-architecture.txt'), run_name='review_pin')
initial = TinyLM(ModelConfig(width=64, layers=2, heads=1)).eval()
clone_checks = {}
for name, kv in [('mha', 4), ('gqa', 1)]:
    clone, copied = frozen['_clone_config'](initial, SimpleNamespace(seed=42, device='cpu'), heads=4, kv_heads=kv, experts=0, backend='manual')
    assert copied == report['results']['models'][name]['copied_initial_tables']
    for key in copied:
        assert torch.equal(initial.state_dict()[key], clone.state_dict()[key])
    clone_checks[name] = {'copied_tables': len(copied), 'k_projection_shapes': [list(b.attention.k.weight.shape) for b in clone.blocks]}
    if kv == 1:
        assert not any(key.endswith(('.attention.k.weight', '.attention.v.weight')) for key in copied)
        repeat, repeated = frozen['_clone_config'](initial, SimpleNamespace(seed=42, device='cpu'), heads=4, kv_heads=kv, experts=0, backend='manual')
        assert repeated == copied
        assert all(torch.equal(a.attention.k.weight, b.attention.k.weight) for a, b in zip(clone.blocks, repeat.blocks))
        assert all(torch.equal(a.attention.v.weight, b.attention.v.weight) for a, b in zip(clone.blocks, repeat.blocks))
        clone_checks[name]['new_kv_repeated_seed_identical'] = True
toy_cost = -math.log(0.5)
assert abs(toy_cost - 0.6931) < 0.00005
result = {
    'environment': environment, 'data': data_checks, 'models': model_checks,
    'p_half_nll': toy_cost, 'three_target_average': (toy_cost * 3) / 3,
    'source_checks': source_checks, 'clone_checks': clone_checks,
    'efficiency_result_sha256': sha('docs/course-experiments/results/efficiency.json'),
    'sft_result_sha256': sha('docs/course-experiments/results/sft.json'),
    'scope': '只建立未訓練 CPU 模型核對結構，並重算既有凍結 GPU 報告；沒有更新權重、沒有重測 GPU 耗時或品質。',
}
print(json.dumps(result, ensure_ascii=False, indent=2))
