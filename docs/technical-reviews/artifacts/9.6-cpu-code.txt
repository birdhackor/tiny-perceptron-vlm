import contextlib
import io
import json
import platform
import torch

torch.set_num_threads(1)
assert str(torch.__version__) == "2.14.1+cpu"
assert torch.version.cuda is None
should_refuse = torch.tensor([True, False, True, False])
cases = [
    ("原例：一律拒絕", torch.ones(4, dtype=torch.bool), [1.0, 0.0]),
    ("練習一：應有行為", torch.tensor([True, False, True, False]), [1.0, 1.0]),
    ("練習二：第一項改 False", torch.tensor([False, False, True, False]), [0.5, 1.0]),
    ("正文另一極端：一律不拒絕", torch.zeros(4, dtype=torch.bool), [0.0, 1.0]),
]
results = []
for name, actually_refuse, expected in cases:
    refusal_rate = actually_refuse[should_refuse].float().mean()
    completion_rate = (~actually_refuse[~should_refuse]).float().mean()
    observed = [refusal_rate.item(), completion_rate.item()]
    assert observed == expected
    results.append({"case": name, "should_refuse": should_refuse.tolist(),
                    "actually_refuse": actually_refuse.tolist(),
                    "refusal_selected": actually_refuse[should_refuse].tolist(),
                    "normal_nonrefusal_selected": (~actually_refuse[~should_refuse]).tolist(),
                    "denominators": [int(should_refuse.sum()), int((~should_refuse).sum())],
                    "expected": expected, "observed": observed})
assert torch.ones(4, dtype=torch.bool).tolist() == [True]*4
empty_group = torch.tensor([], dtype=torch.float32).mean()
assert torch.isnan(empty_group)
# Only a sampling calculation; does not claim course trials are independent.
lower_bounds = {str(n): 0.05 ** (1/n) for n in (2, 200)}
assert 2/2 == 200/200 == 1.0
print(json.dumps({"environment": {"python": platform.python_version(),
                   "torch": str(torch.__version__), "torch_source_commit": str(torch.version.git_version),
                   "device": "cpu", "threads": torch.get_num_threads()},
                  "cases": results, "empty_mean_is_nan": True,
                  "binomial_95percent_one_sided_all_success_lower_bound": lower_bounds}, ensure_ascii=False, indent=2))
