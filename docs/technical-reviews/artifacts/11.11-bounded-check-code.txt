def offline_guard(event, args):
    if event.startswith('socket.') or event in ('subprocess.Popen', 'os.system', 'os.exec', 'os.fork'):
        raise RuntimeError('offline bounded review forbids network and child processes')
sys.addaudithook(offline_guard)

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def accuracy(groups):
    return (sum(g['correct'] for g in groups.values()) / sum(g['count'] for g in groups.values()),
            sum(g['correct'] / g['count'] for g in groups.values()) / len(groups))

TOK = ByteTokenizer()
def checked_samples(e):
    values = []
    for s in e['samples']:
        ids = s['generated_ids']
        raw_ids = ids[:ids.index(TOK.eos_id)] if TOK.eos_id in ids else ids
        assert TOK.decode(raw_ids) == s['generated']
        exact = raw_ids == TOK.encode(s['target'])
        assert exact == s['exact_match']
        values.append(dict(s, recomputed_exact=exact))
    assert len(values) == e['examples']
    assert sum(s['recomputed_exact'] for s in values) == e['correct']
    return values
