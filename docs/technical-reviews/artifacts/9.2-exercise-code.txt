import contextlib
import hashlib
import io
import json
import os
import platform
import sys
from pathlib import Path

import torch

root = Path.cwd()
artifacts = root / "docs/technical-reviews/artifacts"
examples = [
    {"visible_count": 3, "hidden_count": 3},
    {"visible_count": None, "hidden_count": 3, "box_color": "紅"},
]
stream = io.StringIO()
with contextlib.redirect_stdout(stream):
    for row in examples:
        visible = row["visible_count"]
        target = str(visible) if visible is not None else "看不到球數，請提供數量或圖片。"
        print("可見", visible, "→", target)
expected = "可見 3 → 3\n可見 None → 看不到球數，請提供數量或圖片。\n"
assert stream.getvalue() == expected
# 額外有界核對：缺值sentinel並不把0當成None；隱藏真值改變不會改寫目標。
targets = []
for visible in [None, 0, 3]:
    per_hidden = []
    for hidden in [0, 3, 99]:
        row = {"visible_count": visible, "hidden_count": hidden, "box_color": "紅"}
        got = str(row["visible_count"]) if row["visible_count"] is not None else "看不到球數，請提供數量或圖片。"
        per_hidden.append(got)
    assert len(set(per_hidden)) == 1
    targets.append({"visible_count": visible, "hidden_count_values": [0, 3, 99], "target": per_hidden[0]})
record = {
    "command": "timeout 30s env OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 CUDA_VISIBLE_DEVICES= HF_HUB_OFFLINE=1 HF_DATASETS_OFFLINE=1 TRANSFORMERS_OFFLINE=1 .venv/bin/python docs/technical-reviews/artifacts/9.2-exercise-code.txt",
    "environment": {"python": platform.python_version(), "torch": torch.__version__, "device": "cpu", "OMP_NUM_THREADS": os.environ["OMP_NUM_THREADS"], "MKL_NUM_THREADS": os.environ["MKL_NUM_THREADS"]},
    "source_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    "exercise": "僅第二筆加入 box_color=紅，產生目標的條件保持原文。",
    "rows": examples,
    "stdout": stream.getvalue(),
    "additional_bounded_cases": targets,
    "result": "原練習兩行完全相符；第二筆仍要求球數。額外9個欄位不變性／0值案例全部相符。",
    "scope": "手寫標註規則的確定性；沒有模型推論或更新，不能外推自然語言能力。",
}
(artifacts / "9.2-exercise-execution.json").write_text(json.dumps(record, ensure_ascii=False, indent=2) + "\n")
print(record["result"])
