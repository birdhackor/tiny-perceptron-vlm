from pathlib import Path
import sys, io, contextlib, json, hashlib, platform, torch
mode = sys.argv[1]
assert mode in ("exercise", "mechanism")
base = Path("docs/technical-reviews/artifacts")
code_path = base / f"5.15-{mode}-code.txt"
output_path = base / f"5.15-{mode}-execution.json"
record = json.loads(output_path.read_text())
buffer = io.StringIO()
namespace = {}
with contextlib.redirect_stdout(buffer):
    exec(compile(code_path.read_text(), str(code_path), "exec"), namespace)
if mode == "exercise":
    assert not torch.equal(namespace["a"].embedding.weight, namespace["b"].embedding.weight)
    record["stdout"] = buffer.getvalue()
    record["result"] = "exit 0；只刪指定seed重設後字表torch.equal=False；assert通過；無訓練或參數更新。"
else:
    record["stdout"] = json.loads(buffer.getvalue())
    record["result"] = "exit 0；全部機制與Decimal算例assert通過；無參數更新或正式訓練。"
record["command"] = "OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 CUDA_VISIBLE_DEVICES='' HF_HUB_OFFLINE=1 HF_DATASETS_OFFLINE=1 TRANSFORMERS_OFFLINE=1 PYTHONDONTWRITEBYTECODE=1 timeout 45s .venv/bin/python -c 'from pathlib import Path; exec(compile(Path(\"docs/technical-reviews/artifacts/5.15-runner-code.txt\").read_text(), \"5.15-runner-code.txt\", \"exec\"))' " + mode
record["environment"] = {"python": platform.python_version(), "torch": str(torch.__version__), "torch_git_version": torch.version.git_version, "device": "cpu", "OMP_NUM_THREADS": "1", "MKL_NUM_THREADS": "1", "offline": "HF_HUB_OFFLINE=1 HF_DATASETS_OFFLINE=1 TRANSFORMERS_OFFLINE=1"}
