from pathlib import Path
import json, hashlib
root = Path.cwd()
path = root / "docs/course-experiments/results/moe.json"
raw = path.read_bytes()
r = json.loads(raw)
paths = ["tiny_perceptron/modern.py", "tiny_perceptron/model.py"]
matched = {p: hashlib.sha256((root / p).read_bytes()).hexdigest() == r["code_sha256"][p] for p in paths}
frozen = root / "outputs/technical-sources/15.5-fresh-own/formal-architecture.py"
matched["frozen architecture.py"] = hashlib.sha256(frozen.read_bytes()).hexdigest() == r["code_sha256"]["scripts/course_experiments/architecture.py"]
assert all(matched.values())
variants = {}
for name in ("top1_aux0", "top1_aux0.01", "top2_aux0", "top2_aux0.01"):
    v = r["results"]["variants"][name]
    t = v["training"]
    assert t["status"] == "completed" and t["optimizer_updates"] == t["requested_steps"] == 180 and t["skipped_updates"] == 0
    variants[name] = {"model_config":v["model"]["config"], "training":{k:t[k] for k in ("requested_steps","status","all_requested_attempts_completed","steps","optimizer_updates","skipped_updates","effective_tokens","batch_size","auxiliary_weight","learning_rate","scaler_enabled","all_parameters_finite","initial","final")}, "recorded_history_count":len(t["history"]), "recorded_history_all_finite":all(x["gradients_finite"] for x in t["history"])}
result = {"original_result_path":str(path.relative_to(root)),"original_result_sha256":hashlib.sha256(raw).hexdigest(),"original_code_revision":r["revision"],"original_run_environment":{k:r[k] for k in ("device","seed","torch_version","python_version","gpu")},"evidence_status":r["evidence_status"],"status":r["status"],"matched_original_source_hashes":matched,"original_source_hashes":{p:r["code_sha256"][p] for p in paths+["scripts/course_experiments/architecture.py"]},"variants":variants,"inspection_scope":"CPU僅讀取凍結原始結果與核對code hash，沒有重跑正式CUDA訓練。輸出保留每個MoE變體的原始記錄，不對新任務品質/速度外推。"}
print(json.dumps(result,ensure_ascii=False,indent=2))
