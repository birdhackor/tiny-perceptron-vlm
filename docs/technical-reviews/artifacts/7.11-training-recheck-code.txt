def extract(path, lesson):
    data = Path(path).read_bytes()
    start = re.search(rb"^## " + re.escape(lesson.encode()) + rb"\s", data, re.M).start()
    after = re.search(rb"^## ", data[start + 1:], re.M)
    end = len(data) if after is None else start + 1 + after.start()
    return data[start:end]
