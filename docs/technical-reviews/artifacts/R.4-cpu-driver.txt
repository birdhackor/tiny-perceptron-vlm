def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def section(path, lesson):
    raw = (ROOT / path).read_bytes()
    matches = list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+", raw))
    i = next(i for i, m in enumerate(matches) if m[1].decode() == lesson)
    end = matches[i + 1].start() if i + 1 < len(matches) else len(raw)
    return raw[matches[i].start():end]

    (ART / f"R.4-prerequisite-{lesson}-raw.txt").write_bytes(raw)
    code = re.findall(r"```python\n(.*?)```", raw.decode(), re.S)[0]
    variants = [("original", code)]

        variants.append(("exercise", code.replace("[[1, 2, 1]]", "[[1, 1, 1]]") + "\nassert torch.equal(x, torch.tensor([[[1.,0.,0.],[1.,0.,0.],[1.,0.,0.]]]))\n"))

        out = io.StringIO()
