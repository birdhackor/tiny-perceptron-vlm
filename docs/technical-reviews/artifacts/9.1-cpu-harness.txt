from pathlib import Path
import contextlib,io,json,hashlib,re,sys,tarfile,platform
import torch
from scripts.course_experiments.common import split_records
from scripts.course_experiments.text import _utf8_prefix,_digest
from scripts.course_experiments.behavior import _conversation
root=Path.cwd()
art=root/'docs/technical-reviews/artifacts'
sha=lambda b:hashlib.sha256(b).hexdigest()
def section(path,lesson):
    raw=(root/path).read_bytes()
    hs=list(re.finditer(rb'(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+',raw))
    n=next(i for i,h in enumerate(hs) if h[1].decode()==lesson)
    return raw[hs[n].start():hs[n+1].start() if n+1<len(hs) else len(raw)]
def run(path,lesson,variant='original',transform=lambda x:x):
    raw=section(path,lesson)
    blocks=re.findall(r'```python\n(.*?)```',raw.decode(),re.S)
    src=transform('\n\n'.join(blocks));ns={};stdout=io.StringIO()
    with contextlib.redirect_stdout(stdout):exec(compile(src,f'{path}#{lesson}:{variant}','exec'),ns)
    return {'source':path+'#'+lesson,'variant':variant,'raw_source_sha256':sha(raw),'executed_code':src,'stdout':stdout.getvalue(),'result':'exit 0; code completed'},ns
results=[]
r,ns=run('course/chapters/09.md','9.1');results.append(r)
assert ns['pair'][f"回答{ns['pair']['相對較安全']}安全"] is False
r,ns=run('course/chapters/09.md','9.1','exercise',lambda s:s.replace('"回答0安全": False','"回答0安全": True'));results.append(r)

    counts['both_safe' if a and b else 'both_unsafe' if not a and not b else 'mixed']+=1
    safer=int(row['safer_response_id'])
    if not row[f'is_response_{safer}_safe']:continue
    selected.append(_conversation(_utf8_prefix(row['prompt'],120),_utf8_prefix(row[f'response_{safer}'],120),_digest(row['prompt']),source_row=index,source_record_sha256=_digest(row),safer_response_id=safer,better_response_id=int(row['better_response_id'])))

fixed=json.loads((root/'docs/course-experiments/results/safety.json').read_text())['results']['pku_pilot']
checks={}
for split,records in parts.items():
    data=''.join(json.dumps(row,ensure_ascii=False)+'\n' for row in records).encode()
    hashes={row['family'] for row in records}
    checks[split]={'records':len(records),'families':len(hashes),'sha256':sha(data),'max_prompt_bytes':max(len(r['messages'][0]['content'].encode()) for r in records),'max_answer_bytes':max(len(r['messages'][1]['content'].encode()) for r in records),'all_legal_utf8':all(r['messages'][0]['content'].encode().decode()==r['messages'][0]['content'] for r in records)}
    assert checks[split]['records']==fixed['data'][split]['records']
    assert sha(data)==fixed['data'][split]['sha256']
assert not ({r['family'] for r in parts['train']} & {r['family'] for r in parts['validation']})
