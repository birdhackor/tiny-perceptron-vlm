import sys, platform, importlib.metadata, unicodedata
from pathlib import Path
print("environment", sys.version, platform.machine(), "torch", importlib.metadata.version("torch"), "device=cpu")
code=Path("docs/technical-reviews/artifacts/6.2-original-code.txt").read_bytes()
ns={}; exec(compile(code,"6.2-original-code.txt","exec"),ns)
merge=ns["merge"]
print("zip", list(zip(list("abab"),list("abab")[1:])))
print("tie", Counter("abba").most_common(1))
print("overlap", merge(list("aaa"),("a","a")))
print("empty row", merge([], ("a","b")))
print("last unmatched",merge(["a"],("a","b")))
assert Counter("abba").most_common(1)==[("a",2)]
assert merge(list("aaa"),("a","a"))==["aa","a"]
assert merge([], ("a","b"))==[]
assert merge(["a"],("a","b"))==["a"]
assert all("".join(merge(row,ns["pair"]))=="".join(row) for row in ns["corpus"])
