import json
import os
import platform
import subprocess
import sys
from pathlib import Path
import torch

base = Path('docs/technical-reviews/artifacts')
mode = sys.argv[1]
assert mode in ('original', 'exercise')
code = base / f'18.6-{mode}-code.txt'
command = ['.venv/bin/python', str(code)]
run = subprocess.run(command, capture_output=True, text=True, timeout=20)
record = {
    'command': 'OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 HF_HUB_OFFLINE=1 TRANSFORMERS_OFFLINE=1 ' + ' '.join(command),
    'exit_code': run.returncode,
    'stdout': run.stdout,
    'stderr': run.stderr,
    'environment': {'python': platform.python_version(), 'torch': torch.__version__, 'torch_git': torch.version.git_version, 'device': 'cpu', 'threads': str(torch.get_num_threads()), 'OMP_NUM_THREADS': os.environ['OMP_NUM_THREADS'], 'MKL_NUM_THREADS': os.environ['MKL_NUM_THREADS'], 'offline': 'HF_HUB_OFFLINE=1; TRANSFORMERS_OFFLINE=1'},
    'scope': '只執行18.6原例或指定練習，無模型訓練、資料下載或網路呼叫。',
}
(base / f'18.6-{mode}-execution.json').write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(record, ensure_ascii=False, indent=2))
sys.exit(run.returncode)
