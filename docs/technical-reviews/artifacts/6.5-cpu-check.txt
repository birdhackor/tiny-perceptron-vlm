def values(raw, costs):
    count = len(raw.encode("utf-8"))
    return [nll / (count * math.log(2)) for nll in costs]

baseline = values("貓🙂", [5.0, 4.0])
print("original", json.dumps({"raw": "貓🙂", "raw_bytes": list("貓🙂".encode("utf-8")),
                              "count": 7, "bpb": baseline}, ensure_ascii=False))
doubled_nll = values("貓🙂", [10.0, 8.0])
print("exercise_1", json.dumps({"raw": "貓🙂", "total_nll": [10.0, 8.0],
                                "bpb": doubled_nll, "ratios": [a / b for a, b in zip(doubled_nll, baseline)]}, ensure_ascii=False))
assert doubled_nll == [2 * x for x in baseline]
assert doubled_nll[0] > doubled_nll[1]
doubled_raw = values("貓🙂貓🙂", [5.0, 4.0])
print("exercise_2", json.dumps({"raw": "貓🙂貓🙂", "raw_byte_count": 14,
                                "total_nll": [5.0, 4.0], "bpb": doubled_raw,
                                "ratios": [a / b for a, b in zip(doubled_raw, baseline)]}, ensure_ascii=False))
assert doubled_raw == [x / 2 for x in baseline]
assert doubled_raw[0] > doubled_raw[1]
print("intro_average_example", {"A_total_nll": 10 * 0.6, "B_total_nll": 5 * 1.0})
product = 0.5 * 0.25
summed = -math.log(0.5) - math.log(0.25)
print("conditional_chain", {"joint_given_A": product, "negative_log_product": -math.log(product), "sum_nll": summed})
assert math.isclose(summed, -math.log(product), abs_tol=1e-14)

# Two valid next-token labels and one excluded position: no model update.
logits = torch.tensor([[[0.0, 1.0], [2.0, 0.0], [100.0, -100.0]]], dtype=torch.float64)
labels = torch.tensor([[1, 0, IGNORE]])
summed_tensor, count_tensor = loss_sum(logits, labels)
manual = -logits.log_softmax(-1)[0, 0, 1] - logits.log_softmax(-1)[0, 1, 0]
print("effective_target_mask", {"labels": labels.tolist(), "count": count_tensor.item(),
                               "total_nll": summed_tensor.item(), "manual_nll": manual.item()})
assert count_tensor.item() == 2
assert torch.allclose(summed_tensor, manual, atol=1e-14, rtol=0)

tok = ByteTokenizer()
content = tok.encode("貓🙂")
x, y = shifted([tok.bos_id] + content + [tok.eos_id])
print("boundary_encoding", {"bos_id": tok.bos_id, "eos_id": tok.eos_id,
                            "content_ids": content, "input": x.tolist(), "targets": y.tolist(),
                            "raw_byte_count": 7, "effective_target_count_including_eos": len(y)})
assert all(i >= 8 for i in content)
assert len(y) == 8 and y[-1].item() == tok.eos_id
assert tok.bos_id not in y

# Deterministic tiny module to test the repository generator's cap and EOS,
# without a training update or a claim about generated-text quality.
class ConstantNext(torch.nn.Module):
    def __init__(self, next_id):
        super().__init__()
        self.config = ModelConfig(vocab_size=3, max_length=384)
        self.next_id = next_id

    def forward(self, ids, cache=None):
        logits = torch.full((*ids.shape, 3), -10.0)
        logits[..., self.next_id] = 10.0
        return {"logits": logits, "cache": None}

prefix = torch.tensor([[1]], dtype=torch.long)
capped = generate(ConstantNext(0), prefix, max_new_tokens=32, eos_id=2)
stopped = generate(ConstantNext(2), prefix, max_new_tokens=32, eos_id=2)
print("repository_generate_contract", {"new_tokens_at_cap": capped.shape[1] - prefix.shape[1],
                                        "new_tokens_until_eos": stopped.shape[1] - prefix.shape[1]})
assert capped.shape[1] - prefix.shape[1] == 32
assert stopped.shape[1] - prefix.shape[1] == 1
