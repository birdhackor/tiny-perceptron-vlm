import json, platform, torch
values=torch.tensor([[2.0,0.0],[0.0,4.0]])
weights=torch.tensor([0.9,0.1])
original=weights@values
exercise=torch.tensor([0.1,0.9])@values
a=0.9*1
b=0.1*100
assert a==0.9 and b==10.0 and a<b
assert torch.allclose(original,torch.tensor([1.8,0.4]))
assert torch.allclose(exercise,torch.tensor([0.2,3.6]))
print(json.dumps({"environment":{"python":platform.python_version(),"torch":str(torch.__version__),"torch_commit":torch.version.git_version,"device":"cpu"},"new_example":{"0.9*1":a,"0.1*100":b},"existing_original":original.tolist(),"existing_exercise":exercise.tolist(),"result":"新數字及原有加權混合均相符；較小乘數可以因value大小而帶進較大數值"},ensure_ascii=False,indent=2))
