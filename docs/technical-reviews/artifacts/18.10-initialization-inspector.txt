import copy, json, os, platform
from pathlib import Path
import torch
from tiny_perceptron.model import TinyLM, ModelConfig
from tiny_perceptron.training import seed_everything
from scripts.course_experiments.compression import _parameter_hash
D=json.loads(Path("docs/course-experiments/results/distillation.json").read_bytes())
A=D["results"]["tasks"]["attributes"]
rows=[]
for width in [16,32]:
    seed_everything(42)
    initial=TinyLM(ModelConfig(width=width,layers=1,heads=2,max_length=128))
    value=_parameter_hash(initial)
    expected=A["runs"][f"w{width}_ce"]["training"]["initialization_sha256"]
    assert value==expected
    for method in ["ce","teacher_hard","ce_kl"]:
        assert _parameter_hash(copy.deepcopy(initial))==value==A["runs"][f"w{width}_{method}"]["training"]["initialization_sha256"]
    rows.append({"width":width,"initialization_sha256":value,"three_branches_equal":True,"parameter_count":sum(p.numel() for p in initial.parameters())})
print(json.dumps({"rows":rows,"environment":{"python":platform.python_version(),"torch":str(torch.__version__),"torch_git_version":str(torch.version.git_version),"device":"cpu","OMP_NUM_THREADS":os.environ["OMP_NUM_THREADS"],"MKL_NUM_THREADS":os.environ["MKL_NUM_THREADS"]},"exit_code":0,"scope":"建立原配置隨機初始化並 deepcopy；只比對初始 SHA，沒有更新任何參數或重訓學生。"},ensure_ascii=False,indent=2))
