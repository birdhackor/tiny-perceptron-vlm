import json

def assess(text):
    try:
        value = json.loads(text)
        valid = isinstance(value, dict) and set(value) == {"answer"}
        correct = valid and value["answer"] == 4
        return {"parsed": True, "type": type(value).__name__, "valid": valid, "correct": correct}
    except json.JSONDecodeError as error:
        return {"parsed": False, "error": type(error).__name__, "position": error.pos, "valid": False, "correct": False}

cases = [
    ('plain', '{"answer":4}', True, True),
    ('space', ' {"answer":4}', True, True),
    ('newline', '\n{"answer":4}', True, True),
    ('tab_cr_tail_lf', '\t\r{"answer":4}\n', True, True),
    ('prose_prefix', '答案是：{"answer":4}', False, False),
    ('friendly_prose_prefix', '當然！答案是：{"answer":4}', False, False),
    ('prose_suffix', '{"answer":4}完成', False, False),
    ('wrong_sum', '{"answer":5}', True, False),
    ('extra_note', '{"answer":4,"note":"完成"}', False, False),
]
for name, text, expected_valid, expected_correct in cases:
    result=assess(text)
    assert result['valid'] is expected_valid and result['correct'] is expected_correct
    print(json.dumps({'case':name,'input':text,'result':result},ensure_ascii=False))
raw = '答案是：{"answer":4}'
# A known-prefix demonstration is a different input transformation, not a general extractor.
transformed = raw[len('答案是：'):]
assert not assess(raw)['valid'] and assess(transformed)['valid']
print(json.dumps({'case':'changed_contract_by_deleting_known_prose_prefix','raw_input':raw,'raw_result':assess(raw),'transformed_input':transformed,'transformed_result':assess(transformed)},ensure_ascii=False))
