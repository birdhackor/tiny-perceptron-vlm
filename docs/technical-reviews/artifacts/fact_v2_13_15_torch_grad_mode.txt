   22: class no_grad(_NoParamDecoratorContextManager):
   23:     r"""Context-manager that disables gradient calculation.
   24: 
   25:     Disabling gradient calculation is useful for inference, when you are sure
   26:     that you will not call :meth:`Tensor.backward()`. It will reduce memory
   27:     consumption for computations that would otherwise have `requires_grad=True`.
   28: 
   29:     In this mode, the result of every computation will have
   30:     `requires_grad=False`, even when the inputs have `requires_grad=True`.
   31:     There is an exception! All factory functions, or functions that create
   32:     a new Tensor and take a requires_grad kwarg, will NOT be affected by
   33:     this mode.
   34: 
   35:     This context manager is thread local; it will not affect computation
   36:     in other threads.
   37: 
   38:     Also functions as a decorator.
   39: 
   40:     .. note::
   41:         No-grad is one of several mechanisms that can enable or
   42:         disable gradients locally see :ref:`locally-disable-grad-doc` for
   43:         more information on how they compare.
   44: 
   45:     .. note::
   46:         This API does not apply to :ref:`forward-mode AD <forward-mode-ad>`.
   47:         If you want to disable forward AD for a computation, you can unpack
   48:         your dual tensors.
   49: 
   50:     Example::
   51:         >>> # xdoctest: +SKIP
   52:         >>> x = torch.tensor([1.], requires_grad=True)
   53:         >>> with torch.no_grad():
   54:         ...     y = x * 2
   55:         >>> y.requires_grad
   56:         False
   57:         >>> @torch.no_grad()
   58:         ... def doubler(x):
   59:         ...     return x * 2
   60:         >>> z = doubler(x)
   61:         >>> z.requires_grad
   62:         False
   63:         >>> @torch.no_grad()
   64:         ... def tripler(x):
   65:         ...     return x * 3
   66:         >>> z = tripler(x)
   67:         >>> z.requires_grad
   68:         False
   69:         >>> # factory function exception
   70:         >>> with torch.no_grad():
   71:         ...     a = torch.nn.Parameter(torch.rand(10))
   72:         >>> a.requires_grad
   73:         True
   74:     """
   75: 
   76:     def __init__(self) -> None:
   77:         if not torch._jit_internal.is_scripting():
   78:             super().__init__()
   79:         self.prev = False
   80: 
   81:     def __enter__(self) -> None:
   82:         self.prev = torch.is_grad_enabled()
   83:         torch.set_grad_enabled(False)
   84: 
   85:     def __exit__(self, exc_type: Any, exc_value: Any, traceback: Any) -> None:
   86:         torch.set_grad_enabled(self.prev)
