import sys, json, os, codecs, platform, torch, tokenizers

def offline_guard(event, args):
    if event.startswith("socket.") or event in {"subprocess.Popen", "os.system", "os.posix_spawn", "os.posix_spawnp"}:
        raise RuntimeError("offline bounded review guard: " + event)
sys.addaudithook(offline_guard)
assert torch.version.cuda is None
print(json.dumps({"python":sys.version,"torch":torch.__version__,"tokenizers":tokenizers.__version__,"device":"cpu","OMP_NUM_THREADS":os.environ.get("OMP_NUM_THREADS"),"MKL_NUM_THREADS":os.environ.get("MKL_NUM_THREADS"),"guard":"network and child process calls rejected"},ensure_ascii=False))
from tokenizers import Tokenizer, models, pre_tokenizers, trainers

tok = Tokenizer(models.BPE(unk_token="[UNK]"))
tok.pre_tokenizer = pre_tokenizers.Whitespace()
tok.train_from_iterator(
    ["hello hello hello"],
    trainers.BpeTrainer(vocab_size=20, special_tokens=["[UNK]"], show_progress=False),
)
whole_exercise = tok.encode("hello hello").ids
space_chunks = tok.encode("hello ").ids + tok.encode("hello").ids
mid_chunks = tok.encode("hel").ids + tok.encode("lo").ids
whole_single = tok.encode("hello").ids
print(json.dumps({"whole_hello_hello":whole_exercise,"space_chunks":space_chunks,"equal":whole_exercise==space_chunks,"whole_single":whole_single,"mid_chunks":mid_chunks,"mid_different":whole_single!=mid_chunks},ensure_ascii=False))
assert whole_exercise == space_chunks
assert whole_single != mid_chunks
print("exercise assertions passed")
