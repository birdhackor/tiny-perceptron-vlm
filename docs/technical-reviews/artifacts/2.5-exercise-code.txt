import json
import platform
import sys
import torch

torch.set_num_threads(1)
examples = [("紅色的小小物體是", "圓"), ("藍色的小小物體是", "方")]
assert all(len(prefix) == 8 for prefix, _ in examples)
observed = {}
for length in [1, 3, 5, 8, 7]:
    visible = []
    for prefix, answer in examples:
        visible.append((prefix[-length:], answer))
    observed[str(length)] = visible
    print(length, visible)
assert observed["8"] == [("紅色的小小物體是", "圓"), ("藍色的小小物體是", "方")]
assert observed["7"] == [("色的小小物體是", "圓"), ("色的小小物體是", "方")]
assert all(observed[str(length)][0][0] == observed[str(length)][1][0] for length in [1, 3, 5, 7])
original = [("紅色物體是", "圓"), ("藍色物體是", "方")]
assert [prefix[-4:] for prefix, _ in original] == ["色物體是", "色物體是"]
print(json.dumps({"original_length_4": [prefix[-4:] for prefix, _ in original], "exercise_character_indices": [list(enumerate(prefix, 1)) for prefix, _ in examples], "result": "全部斷言通過；八字能區分，七字不能", "environment": {"python": sys.version, "torch": str(torch.__version__), "torch_git_version": str(torch.version.git_version), "device": "cpu", "platform": platform.platform(), "threads": str(torch.get_num_threads())}}, ensure_ascii=False))
