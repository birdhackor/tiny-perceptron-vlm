import json
import sys
from collections import Counter
import torch
from tiny_perceptron.retrieval import lexical_terms, retrieve
print("ENV", json.dumps({"python": sys.version.split()[0], "torch": str(torch.__version__), "device": "cpu", "threads": torch.get_num_threads()}, ensure_ascii=False))
query = "書店地址"
terms = Counter(lexical_terms(query))
for text in ["貓在窗邊", "書店地址尚未公布", "書店地址是青街8號", "書店地址尚未發布", "書店地址不是尚未公布；已公布青街8號"]:
    counts = Counter(lexical_terms(text))
    score = sum(min(counts[t], amount) for t, amount in terms.items())
    hit = retrieve(query, [{"id": "d2", "text": text}])
    diagnosis = "沒有取回相關來源，請補公告" if not hit else ("來源未給地址，目前無法確定" if "尚未公布" in hit[0]["text"] else "需進一步核對來源與答案")
    print(json.dumps({"text": text, "score": score, "ids": [r["id"] for r in hit], "decision": diagnosis}, ensure_ascii=False))
old = {"id": "a-old", "text": "書店地址是青街8號；日期2024"}
new = {"id": "z-new", "text": "書店地址是紅街9號；日期2026"}
print("old_vs_new", json.dumps(retrieve(query, [new, old], k=1), ensure_ascii=False))
assert retrieve(query, [new, old], k=1) == [old]
assert len(terms) == 4 and all(v == 1 for v in terms.values())
print("score_bound", sum(terms.values()))
