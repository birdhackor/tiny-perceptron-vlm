raw = Path('course/chapters/05.md').read_bytes()
headings = list(re.finditer(rb'(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+', raw))
def section(lesson):
    p = next(i for i, h in enumerate(headings) if h[1].decode() == lesson)
    end = headings[p+1].start() if p+1 < len(headings) else len(raw)
    return raw[headings[p].start():end]
def run_fence(lesson, replacement=None):
    code = re.findall(r'```python\n(.*?)```', section(lesson).decode(), re.S)[0]
    if replacement:
        code = code.replace(*replacement)
    output = io.StringIO()
    ns = {}
    error = None
    with contextlib.redirect_stdout(output):
        try:
            exec(compile(code, 'original-course-'+lesson, 'exec'), ns)
        except Exception as exc:
            error = {'type': type(exc).__name__, 'message': str(exc)}
    return ns, {'stdout': output.getvalue(), 'error': error}
