import torch
x = torch.tensor([[0.7, 0.2], [-0.7, 1.2]])
w = torch.tensor([[0.7, 0.2], [-0.7, 1.2]])
a, b = x @ w.T, x @ w.T
print("相同輸入的兩次中間特徵完全相同", torch.equal(a, b), a.tolist())
print("兩次特徵範圍", [a.min().item(), a.max().item()], [b.min().item(), b.max().item()])
zero_w = torch.zeros(2, 2)
c, d = torch.tensor([[1.0, 0.0]]) @ zero_w.T, torch.tensor([[0.0, 1.0]]) @ zero_w.T
print("不同輸入的中間特徵也可相同", torch.equal(c, d), c.tolist(), d.tolist())
