from pathlib import Path
import re, json, hashlib, platform
import torch
base=Path('docs/technical-reviews/artifacts')
checks={}
for identifier,path in [('1.1','course/chapters/01.md'),('W.1','course/first-steps.md'),('W.2','course/first-steps.md')]:
    raw=Path(path).read_bytes();heads=list(re.finditer(rb'(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+',raw));i=next(i for i,h in enumerate(heads) if h[1].decode()==identifier)
    part=raw[heads[i].start():heads[i+1].start() if i+1<len(heads) else len(raw)]
    checks[identifier]={'source':path+'#'+identifier,'sha256':hashlib.sha256(part).hexdigest(),'equals_personally_read_raw':part==(base/('R.1-prereq-'+identifier+'.txt')).read_bytes()}
raw=(base/'R.1-prereq-1.1.txt').read_text();code=re.findall(r'```python\n(.*?)```',raw,re.S)
n=json.loads(Path('notebooks/01/1.1.ipynb').read_bytes())
visible=[''.join(c['source']).strip() for c in n['cells'] if c['cell_type']=='code' and not c.get('metadata',{}).get('course_setup') and 'course_figure' not in c.get('metadata',{})]
checks['1.1_notebook_visible_code_equals_raw']=[s.strip() for s in code]==visible
checks['1.1_original_fence_no_import_or_transformer']=all('import ' not in c and 'Transformer' not in c for c in code)
chapter=Path('course/chapters/01.md').read_text()
checks['warm_link_metadata']={}
for identifier,warm in [('1.6','W.3'),('1.8','W.4'),('1.9','W.6')]:
    match=re.search(r'^## '+re.escape(identifier)+r' .*?(?=^## |\Z)',chapter,re.M|re.S)
    # Metadata-only inspection: record the href; no review of the other section's claims.
    checks['warm_link_metadata'][identifier]={'target':'../first-steps.md#'+warm,'exists_in_section':('../first-steps.md#'+warm) in match[0],'warm_heading_exists':re.search(r'^## '+re.escape(warm)+r' ',Path('course/first-steps.md').read_text(),re.M) is not None}
checks['W.1_operation_locations']={term:term in (base/'R.1-prereq-W.1.txt').read_text() for term in ['uv sync --frozen --extra cpu --group notebook','Shift+Enter','Restart Kernel and Run All Cells','鳥看狗，狗看鳥。']}
checks['W.2_topics']={term:term in (base/'R.1-prereq-W.2.txt').read_text() for term in ['清單（list）','字典（dictionary）','索引（index）']}
result={'scope':'R.1直接前置與第一章對暖身的href存在性；對1.9只讀link metadata，不作其他小節判定；未執行安裝、clone、Jupyter kernel或Colab','environment':{'python':platform.python_version(),'torch':str(torch.__version__),'device':'cpu'},'checks':checks}
print(json.dumps(result,ensure_ascii=False,indent=2));(base/'R.1-navigation-audit.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n')
assert all(checks[x]['equals_personally_read_raw'] for x in ['1.1','W.1','W.2'])
assert checks['1.1_notebook_visible_code_equals_raw'] and checks['1.1_original_fence_no_import_or_transformer']
assert all(v['exists_in_section'] and v['warm_heading_exists'] for v in checks['warm_link_metadata'].values())
assert all(checks['W.1_operation_locations'].values()) and all(checks['W.2_topics'].values())
