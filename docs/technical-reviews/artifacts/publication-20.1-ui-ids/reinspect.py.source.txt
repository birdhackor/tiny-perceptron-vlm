def digest(content):
    return hashlib.sha256(content).hexdigest()


def save(name, content):
    path = OUT / name
    path.write_bytes(content if isinstance(content, bytes) else content.encode("utf-8"))
    return str(path.relative_to(ROOT))


def section(path, identifier):
    raw = (ROOT / path).read_bytes()
    headings = list(re.finditer(rb"^## .+$", raw, re.M))
    for index, heading in enumerate(headings):
        if heading[0].startswith(b"## " + identifier.encode() + b" "):
            end = headings[index + 1].start() if index + 1 < len(headings) else len(raw)
            return raw[heading.start():end]
    raise AssertionError(identifier)

class CaptureCurrentTest:
    def __init__(self):
        self.current_tests = []
        self.call_reports = []

    @pytest.hookimpl(hookwrapper=True, tryfirst=True)
    def pytest_runtest_call(self, item):
        yield
        if item.originalname == FUNCTION:
            value = os.environ["PYTEST_CURRENT_TEST"]
            self.current_tests.append({"nodeid": item.nodeid, "current_test": value, "characters": len(value)})

    def pytest_runtest_logreport(self, report):
        if report.when == "call":
            self.call_reports.append({"nodeid": report.nodeid, "outcome": report.outcome})
