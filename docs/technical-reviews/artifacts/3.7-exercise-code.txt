import json
import platform
import torch
from tiny_perceptron.attention import CausalAttention

torch.manual_seed(42)
layer = CausalAttention(width=8, heads=4)
x = torch.randn(1, 4, 8)
out, cache = layer(x)
print("輸出", out.shape)
print("保存的K", cache[0].shape)
assert out.shape == x.shape
assert tuple(out.shape) == (1, 4, 8)
assert tuple(cache[0].shape) == (1, 4, 4, 2)
print(json.dumps({"python": platform.python_version(), "torch": torch.__version__, "torch_commit": torch.version.git_version, "device": str(x.device), "dtype": str(x.dtype)}, ensure_ascii=False))
