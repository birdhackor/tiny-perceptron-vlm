import json, torch, hashlib
from pathlib import Path
from scripts.course_experiments.common import text_examples
from tiny_perceptron.data import pad_batch, IGNORE
p=Path("docs/technical-reviews/artifacts/16.8-probe-input-snapshot.json")
s=json.loads(p.read_text())
e=json.loads(Path("docs/course-experiments/results/efficiency.json").read_text())
a=next(a for a in e["artifacts"] if a["path"]=="dataset.json")
assert s["original_sha256"]==a["sha256"] and s["original_bytes"]==a["bytes"]
for path in ["tiny_perceptron/data.py","scripts/course_experiments/common.py"]:
    assert hashlib.sha256(Path(path).read_bytes()).hexdigest()==e["code_sha256"][path]
examples=text_examples(s["train_first_four"],mode="sft",max_length=128)
x,y,valid=pad_batch(examples)
result={"scope":"既有同SHA prepared資料之固定前4筆，短CPU只轉ID／padding／count，沒有GPU或參數更新", "input_shape":list(x.shape),"unpad_lengths":[len(xx) for xx,yy in examples],"effective_targets_per_record":[int((yy!=IGNORE).sum()) for xx,yy in examples],"gradient_loss_denominator":int((y!=IGNORE).sum()),"valid_input_positions":int(valid.sum()),"padded_positions":int(x.numel()),"target_answer_and_eos_ids":[yy[yy!=IGNORE].tolist() for xx,yy in examples],"dataset_sha256":s["original_sha256"]}
assert result["input_shape"]==[4,53]
assert result["effective_targets_per_record"]==[7,7,6,4] and result["gradient_loss_denominator"]==24
expected_lengths=[4+len(row["messages"][0]["content"].encode("utf-8"))+len(row["messages"][1]["content"].encode("utf-8")) for row in s["train_first_four"]]
assert result["unpad_lengths"]==expected_lengths==[53,51,50,48]
assert result["valid_input_positions"]==sum(expected_lengths)==202 and result["padded_positions"]==212
print(json.dumps(result,ensure_ascii=False))
