import copy
import json
import platform
import torch
from tiny_perceptron.data import toy_conversations, render_chat, ByteTokenizer

torch.set_num_threads(1)
original = toy_conversations()[0]
edited = copy.deepcopy(original)
x0, y0 = render_chat(original)
edited[1]["content"] = "3"
x3, y3 = render_chat(edited)
active0, active3 = y0[y0 != -100].tolist(), y3[y3 != -100].tolist()
changed_x = (x0 != x3).nonzero().flatten().tolist()
changed_y = (y0 != y3).nonzero().flatten().tolist()
assert active0 == [56, 2] and active3 == [59, 2]
assert edited[0] == original[0] and [m["role"] for m in edited] == [m["role"] for m in original]
assert changed_x == [9] and changed_y == [8]
assert ByteTokenizer().encode("0") == [ord("0") + 8] == [56]
assert ByteTokenizer().encode("3") == [ord("3") + 8] == [59]
edited[1]["content"] = "0"
restored_x, restored_y = render_chat(edited)
assert torch.equal(restored_x, x0) and torch.equal(restored_y, y0)
assert toy_conversations()[0] == original
print(json.dumps({"environment": {"python": platform.python_version(), "torch": torch.__version__,
      "device": "cpu", "threads": str(torch.get_num_threads())}, "original": original,
      "effective_targets_before": active0, "effective_targets_changed": active3,
      "changed_x_positions": changed_x, "changed_y_positions": changed_y,
      "question_and_roles_unchanged": True, "restored_exactly": True,
      "official_toy_source_unchanged": True, "optimizer_steps": 0}, ensure_ascii=False, indent=2))
