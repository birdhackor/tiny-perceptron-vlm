import ast, pathlib, hashlib, json, torch
root = pathlib.Path("docs/technical-reviews/artifacts")
pin = json.loads((root / "17.11-formal-pinned-source.json").read_bytes())
assert pin["sha256"] == "28d8ce258369672d71a24f7efbcbf68f3e5e8114e073620aba91c4d807043a44"
assert hashlib.sha256(pin["raw_utf8_source"].encode()).hexdigest() == pin["sha256"]
tree = ast.parse(pin["raw_utf8_source"])
node = next(n for n in tree.body if isinstance(n, ast.ClassDef) and n.name == "_QATLinear")
forward = next(n for n in node.body if isinstance(n, ast.FunctionDef) and n.name == "forward")
assert not any(isinstance(n, ast.Assign) and any(isinstance(t,ast.Name) and t.id == "x" for t in n.targets) for n in ast.walk(forward))
assert ast.unparse(forward.body[-1].value) == "F.linear(x, simulated, self.bias)"
print("pinned_QAT_forward", ast.unparse(forward))
for n, stamp in [("qat", "20261002T221108297539Z"), ("quantization", "20261002T221059258895Z")]:

    b = p.read_bytes(); record = json.loads(b); r = record["results"]

    if n == "qat": assert r["activation_quantization"] is False
    release_path = pathlib.Path(f"docs/course-experiments/public-releases/{n}.json")

    card_path = pathlib.Path(f"checkpoints/course/{n}/README.md")
