def artifact(identifier, name, kind, description, command=None, result=None, environment=None):
    path = OUT / name
    item = dict(id=identifier, kind=kind, path=path.as_posix(), sha256=sha(path), description=description)

def evidence(identifier, locator, supports):
    return dict(source_id=identifier, locator=locator, supports=supports)


def concept(identifier, statement, location, scope, source, locator, support, derive=False):
    proof = [evidence(source, locator, support)]
    artifact_ids = ["a_sources"]
    if derive:
        proof.append(evidence("s_derivation", "W.4-fresh-derivation.txt", "逐步數學推導此性質及成立條件。"))
        artifact_ids.append("a_derivation")
    claims.append(dict(id=identifier, kind="concept", statement=statement, location=location, scope=scope, status="verified", evidence=proof, artifact_ids=artifact_ids))


def numeric(identifier, statement, location, expected, observed, details, artifact_id="a_verify", tolerance="三位近似絕對誤差0.0005；已顯示字串與所述精確值另作精確核對。", method="executed"):

    if method == "executed":
        proof.append(evidence(source, artifact_id, "親自 CPU 執行觀察值。"))
    claims.append(dict(
        id=identifier, kind="numeric", statement=statement, location=location,
        scope="只支持本節固定輸入的數學算例，不是模型訓練實測。", status="verified",
        evidence=proof, artifact_ids=[artifact_id, "a_derivation"],

    ))
