from tiny_perceptron.data import ByteTokenizer, render_chat, pad_batch
import torch
tok = ByteTokenizer()
x, y = render_chat([{"role": "user", "content": "2+2=?"}, {"role": "assistant", "content": "4"}])
_, padded, _ = pad_batch([(x, y), (x[:-1], y[:-1])])
print("4 byte-token IDs", tok.encode("4"))
print("4 response labels", y.tolist(), "effective", int((y != -100).sum()))
assert tok.encode("4") == [60] and int((y != -100).sum()) == 2
assert padded[1,-1].item() == -100
print("style positions", 30*2*100)
print("format positions", 30*2*10)
print("style/format token ratio", (30*2*100)/(30*2*10))
assert 30*2*100 == 6000 and 30*2*10 == 600

for value in (None, 0):
    print("None versus zero", repr(value), "尚未量測" if value is None else value)
