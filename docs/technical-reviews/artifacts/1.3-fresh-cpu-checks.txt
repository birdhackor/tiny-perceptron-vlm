"""Bounded, offline CPU review of section 1.3 and its direct prerequisites."""
import contextlib
import hashlib
import io
import json

import torch

ROOT = Path.cwd()
RECORDS = {}
HASHES = {}


def section(path, lesson):
    raw = (ROOT / path).read_bytes()
    headings = list(re.finditer(rb"(?m)^## [^\r\n]+", raw))
    index = next(i for i, h in enumerate(headings) if h[0].startswith(("## " + lesson + " ").encode()))
    end = headings[index + 1].start() if index + 1 < len(headings) else len(raw)
    body = raw[headings[index].start():end]
    HASHES[path + "#" + lesson] = hashlib.sha256(body).hexdigest()
    code = re.findall(r"```python\n(.*?)```", body.decode(), re.S)
    assert len(code) == 1
    return code[0]


def run(label, code, expected_exception=None):
    stream = io.StringIO()
    namespace = {"__name__": "__main__"}
    caught = None
    with contextlib.redirect_stdout(stream):
        try:
            exec(compile(code, label, "exec"), namespace)
        except Exception as error:
            if expected_exception is None or not isinstance(error, expected_exception):
                raise
            caught = type(error).__name__
    assert (expected_exception is None) == (caught is None)
    RECORDS[label] = {"stdout": stream.getvalue(), "expected_exception": caught}
    return namespace
