def guard(event, args):
    if event.startswith("socket.") or event in ("subprocess.Popen", "os.system"):
        raise RuntimeError("offline review: network and child processes disabled")


sys.addaudithook(guard)


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def section(path, identifier):
    raw = (ROOT / path).read_bytes()
    start = re.search(rb"(?m)^## " + re.escape(identifier.encode()) + rb" ", raw).start()
    following = re.search(rb"(?m)^## ", raw[start + 1:])
    end = start + 1 + following.start() if following else len(raw)
    return raw[start:end]


runs = []


def run_fence(path, identifier, label, old=None, new=None):
    raw = section(path, identifier)
    code = re.search(r"```python\n(.*?)\n```", raw.decode(), re.S)[1]
    if old is not None:
        assert code.count(old) == 1
        code = code.replace(old, new)
    namespace = {"__name__": "__main__"}
    output = io.StringIO()
    torch.manual_seed(42)
    with contextlib.redirect_stdout(output):
        exec(compile(code, label, "exec"), namespace)
    runs.append({"label": label, "source": path + "#" + identifier,
                 "source_sha256": hashlib.sha256(raw).hexdigest(),
                 "change": None if old is None else {"old": old, "new": new},
                 "stdout": output.getvalue(), "exit_code": 0})
    return namespace


main = run_fence("course/chapters/09.md", "9.9", "9.9 原例")
exercise = run_fence("course/chapters/09.md", "9.9", "9.9 練習", "correct_index = 1", "correct_index = 0")
assert main["scores"].shape == (1, 3)
assert main["scores"].equal(exercise["scores"])
diagnostics = []
for scale in (1.0, 10.0):
    probability = (main["scores"] * scale).softmax(-1)
    confidence, prediction = probability.max(-1)
    exact = 1 / (1 + math.exp(-scale) + math.exp(-2 * scale))
    assert abs(confidence.item() - exact) < 1e-7
    assert prediction.item() == 0
    assert confidence.numel() == prediction.numel() == 1
    assert isinstance(confidence.item(), float) and isinstance(prediction.item(), int)
    diagnostics.append({"scale": scale, "probabilities": probability.tolist(),
                        "confidence": confidence.item(), "prediction": prediction.item(),
                        "confidence_shape": list(confidence.shape), "prediction_shape": list(prediction.shape),
                        "analytic_confidence": exact, "rounded": round(confidence.item(), 4),
                        "below_one": confidence.item() < 1,
                        "correct_original": prediction.item() == 1,
                        "correct_exercise": prediction.item() == 0})
