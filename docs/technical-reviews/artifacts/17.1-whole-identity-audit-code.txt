import difflib
import hashlib
import importlib.metadata
import json
import re
import sys
from datetime import datetime, timezone
from pathlib import Path


def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def sections(raw):
    heads = list(re.finditer(rb'(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+', raw))
    return {
        m[1].decode(): raw[m.start():heads[i + 1].start() if i + 1 < len(heads) else len(raw)]
        for i, m in enumerate(heads)
    }, raw[:heads[0].start()]


base = Path('docs/technical-reviews/artifacts')
previous_report_path = base / '17.1-previous-genuine-pass-report-537551.json'
previous_report_bytes = previous_report_path.read_bytes()
assert sha(previous_report_bytes) == '5375512c4943d2e74ea1901dde387b3e03f85c8679af9a7fc09c4a9c08b2f3f8'
previous = json.loads(previous_report_bytes)
assert Path('docs/technical-reviews/17.1.json').read_bytes() == previous_report_bytes
old_path = base / '17.1-current-whole-source-7f4150.txt'
old = old_path.read_bytes()
new_path = Path('course/chapters/17.md')
new = new_path.read_bytes()
assert sha(old) == '7f415055f9086438eb8bab5b263f6ade8c967b4f6193a0dc9f39739809c36923'
assert sha(new) == '5ef17106a2c04cf901b24c5c914ea153d2a8bc65db356f912df6b8302d88c251'
assert (base / '17.1-latest-whole-source-5ef171.txt').read_bytes() == new
old_sections, old_intro = sections(old)
new_sections, new_intro = sections(new)
assert old_sections['17.1'] == new_sections['17.1'] == (base / '17.1-current-source-bbbd6a.txt').read_bytes()
assert sha(new_sections['17.1']) == previous['source_sha256']
assert old_intro == new_intro == (base / '17.1-intro.txt').read_bytes()
assert sha(new_intro) == previous['intro_sha256']
changed_sections = [key for key in old_sections if old_sections[key] != new_sections[key]]
assert changed_sections == ['17.9', '17.11', '17.12']
old_headings = [m[0].decode() for m in re.finditer(rb'(?m)^## [^\r\n]+', old)]
new_headings = [m[0].decode() for m in re.finditer(rb'(?m)^## [^\r\n]+', new)]
assert old_headings == new_headings
fence_pattern = rb'(?ms)^```[^\r\n]*\r?\n.*?^```[^\r\n]*\r?\n'
old_fences = [m[0] for m in re.finditer(fence_pattern, old)]
new_fences = [m[0] for m in re.finditer(fence_pattern, new)]
assert old_fences == new_fences
diff = json.loads((base / '17.1-whole-identity-lossless-diff.json').read_bytes())
actual_diff = ''.join(difflib.unified_diff(old.decode().splitlines(keepends=True), new.decode().splitlines(keepends=True), fromfile='previous7f4150/course/chapters/17.md', tofile='current5ef171/course/chapters/17.md'))
assert actual_diff == diff['raw_diff_utf8']
assert sha(actual_diff.encode()) == diff['raw_diff_sha256']
assert sha(old) == diff['old_raw_sha256'] and sha(new) == diff['new_raw_sha256']
supports = []
for item in json.loads((base / '17.1-supporting-book.json').read_bytes()):
    path, label = item['source'].split('#')
    part = sections(Path(path).read_bytes())[0][label]
    assert part == item['raw_utf8'].encode() and sha(part) == item['sha256']
    supports.append({'source': item['source'], 'sha256': sha(part), 'raw_bytes_unchanged': True})
repo_sources = []
for source in previous['sources']:
    if source['kind'] == 'repository_code' and source['id'] != 's_book':
        actual_sha = sha(Path(source['path']).read_bytes())
        assert actual_sha == source['sha256']
        repo_sources.append({'id': source['id'], 'path': source['path'], 'sha256': actual_sha, 'unchanged': True})
artifacts = []
for artifact in previous['artifacts']:
    actual_sha = sha(Path(artifact['path']).read_bytes())
    assert actual_sha == artifact['sha256']
    artifacts.append({'id': artifact['id'], 'path': artifact['path'], 'sha256': actual_sha, 'unchanged': True})
figures = []
for path, expected in previous['figure_sha256'].items():
    actual_sha = sha(Path(path).read_bytes())
    assert actual_sha == expected
    figures.append({'path': path, 'sha256': actual_sha, 'unchanged': True})
receipt_path = base / '17.1-consumed-537551-receipt.json'
receipt_bytes = receipt_path.read_bytes()
assert sha(receipt_bytes) == '8744cf87ed0357d5ffe20fda271239fa036cb86c684fb19961a197210459ae72'
receipt = json.loads(receipt_bytes)

assert receipt['entry']['source_sha256'] == previous['source_sha256']
assert receipt['entry']['reviewer_task'] == previous['reviewer_task']
assert receipt['entry']['verdict'] == 'pass'
