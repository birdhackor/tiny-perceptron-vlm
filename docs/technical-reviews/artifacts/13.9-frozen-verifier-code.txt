from pathlib import Path
import hashlib, json, random, sys, torch
from tiny_perceptron.data import ByteTokenizer, render_chat
from scripts.course_experiments.common import split_records

base = Path("docs/technical-reviews/artifacts")
H = lambda p: hashlib.sha256(Path(p).read_bytes()).hexdigest()
x = json.loads(Path("docs/course-experiments/results/dpo.json").read_text())
s = json.loads(Path("docs/course-experiments/results/style.json").read_text())
r = x["results"]
tok = ByteTokenizer()
checks = []
def generated(e):
    rows = []
    for z in e["samples"]:
        ids = z["generated_ids"]
        raw = ids[:ids.index(tok.eos_id)] if tok.eos_id in ids else ids
        good = raw == tok.encode(z["expected"])
        assert good == z["exact"]
        assert (tok.eos_id in ids) == z["eos"]
        rows.append({"prompt": z["messages"][0]["content"], "expected": z["expected"], "generated": z["generated"], "ids": ids, "exact": good})
    count = sum(z["exact"] for z in rows)
    assert count == e["matches"] and len(rows) == e["records"]
    return {"records": len(rows), "matches": count, "rows": rows}
def preference(e):
    rows = []
    for z in e["samples"]:
        margin = z["policy_chosen_logp"] - z["policy_rejected_logp"]
        relative = margin - z["reference_margin"]
        assert abs(margin-z["policy_margin"]) < 1e-5
        assert abs(relative-z["relative_margin"]) < 1e-5
        for side in ("chosen", "rejected"):
            _, y = render_chat([{"role":"user", "content":z["prompt"]},{"role":"assistant", "content":z[side]}])
            assert int((y != -100).sum()) == z[side+"_answer_tokens"]
        rows.append({k:z[k] for k in ["family", "policy_chosen_logp", "policy_rejected_logp", "reference_margin", "policy_margin", "relative_margin", "chosen_answer_tokens", "rejected_answer_tokens"]})
    chosen = sum(z["policy_margin"] > 0 for z in rows)
    improved = sum(z["relative_margin"] > 0 for z in rows)
    assert chosen == e["chosen_higher_absolute_probability"]
    assert improved == e["relative_preference_improved"] and len(rows) == e["records"]
    return {"records":len(rows), "chosen_higher":chosen, "relative_improved":improved, "rows":rows}
checks.append({"baseline_test": generated(s["results"]["content_evaluation"]["test"])})
for name,z in r["runs"].items():
    checks.append({"run":name,"beta":z["beta"],"training":{k:z["training"][k] for k in ["steps","effective_answer_tokens_both_sides"]},"arithmetic_test":generated(z["arithmetic"]["test"])})
checks.append({"format_test":preference(r["format_only"]["preference"]["test"]),"format_generation":generated(r["format_only"]["arithmetic"]["test"]),"format_steps":r["format_only"]["training"]["steps"]})

        for field in ("prompt","chosen","rejected"):
            assert len(z[field].encode("utf-8")) <= 120
    checks.append({"natural_split":split, "count":preference(e)})
parts = split_records(arithmetic_records(),seed=42)
assert {k:len(v) for k,v in parts.items()} == {"train":49,"validation":8,"test":7}
examples = []
for row in parts["train"]:
    lengths=[]
    for answer in (str(row["a"]+row["b"]),str(row["a"]+row["b"]+1)):
        _, y = render_chat([{"role":"user","content":row["messages"][0]["content"]},{"role":"assistant","content":answer}])
        lengths.append(int((y != -100).sum()))
    examples.append(sum(lengths))
sampler = random.Random(42)
effective = sum(sum(sampler.choices(examples,k=8)) for _ in range(250))
assert effective == 9448
