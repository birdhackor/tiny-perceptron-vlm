import json, platform
import torch
print(json.dumps({"python":platform.python_version(),"torch":torch.__version__,"device":"cpu","threads":torch.get_num_threads()},ensure_ascii=False))
requests = [
    {"task": "確認會議日期與時間", "date": None, "time": None},
    {"task": "確認會議日期與時間", "date": "2026-10-03", "time": "10:00"},
]
def answer(request):
    missing = [label for key, label in [("date", "日期"), ("time", "時間")] if request[key] is None]
    if missing:
        return "請提供" + "與".join(missing) + "。"
    return "已確認日期：" + request["date"] + "；時間：" + request["time"]
for request in requests:
    print(json.dumps({"request":request,"answer":answer(request)},ensure_ascii=False))
expected = ["請提供日期與時間。", "請提供日期。", "請提供時間。", "已確認日期：2026-10-03；時間：10:00"]
actual = []
for date,time in [(None,None),(None,"10:00"),("2026-10-03",None),("2026-10-03","10:00")]:
    request = {"task":"確認會議日期與時間", "date":date, "time":time}
    actual.append(answer(request))
    print(json.dumps({"request":request,"answer":actual[-1]},ensure_ascii=False))
assert actual == expected
print("四種 date/time 有無情況：4/4；兩項齊才確認，缺項只問缺項。")
