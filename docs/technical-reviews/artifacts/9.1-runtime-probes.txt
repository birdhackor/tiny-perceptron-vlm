class Constant(torch.nn.Module):
    def __init__(self,token):super().__init__();self.token=token;self.config=SimpleNamespace(max_length=256)
    def forward(self,ids,cache=None):
        logits=torch.zeros((*ids.shape,264));logits[...,self.token]=1
        return {'logits':logits,'cache':None}
prompt=torch.full((1,124),8,dtype=torch.long)
no_eos=generate(Constant(8),prompt,max_new_tokens=128,eos_id=2)
with_eos=generate(Constant(2),prompt,max_new_tokens=128,eos_id=2)
assert no_eos.shape==(1,252) and no_eos[0,124:].tolist()==[8]*128
assert with_eos.shape==(1,125) and with_eos[0,-1].item()==2
