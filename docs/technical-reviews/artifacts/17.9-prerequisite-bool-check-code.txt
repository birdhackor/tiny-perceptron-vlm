import json
import platform
import sys

import torch


evaluation = torch.tensor([-0.2, 0.4, 3.0])
rows = []
for maximum in (torch.tensor(1.0), torch.tensor(3.0)):
    mask = evaluation.abs() > maximum
    count = mask.sum()
    rows.append({
        "maximum": maximum.item(),
        "mask": mask.tolist(),
        "mask_dtype": str(mask.dtype),
        "sum": count.item(),
        "sum_dtype": str(count.dtype),
        "item_python_type": type(count.item()).__name__,
    })
assert rows[0]["mask"] == [False, False, True]
assert rows[0]["sum"] == 1
assert rows[1]["mask"] == [False, False, False]
assert rows[1]["sum"] == 0
assert all(row["mask_dtype"] == "torch.bool" and row["item_python_type"] == "int" for row in rows)
print(json.dumps({
    "environment": {
        "python": sys.version,
        "torch": torch.__version__,
        "torch_commit": torch.version.git_version,
        "device": "cpu",
        "threads": str(torch.get_num_threads()),
        "platform": platform.platform(),
    },
    "scope_zh": "只核必要17.12新增的三項布林比較/count/item解說，不執行simulate、不重跑17.9原例/練習或原seed證據、不更新參數。",
    "rows": rows,
}, ensure_ascii=False, indent=2))
