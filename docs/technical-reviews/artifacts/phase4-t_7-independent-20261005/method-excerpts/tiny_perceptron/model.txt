def generate(model, ids, max_new_tokens=16, temperature=0.0, eos_id=2, use_cache=False):
    """先提供無 padding 版本；遇到長度上限停止，位置不會悄悄回捲。"""
    was_training = model.training
    model.eval()
    sequence, state = ids.clone(), None
    try:
        for _ in range(max_new_tokens):
            if sequence.shape[1] >= model.config.max_length:
                break
            current = sequence if state is None else sequence[:, -1:]
            result = model(current, cache=state if use_cache else None)
            scores = result["logits"][:, -1]
            if temperature <= 0:
                next_id = scores.argmax(-1, keepdim=True)
            else:
                next_id = torch.multinomial((scores / temperature).softmax(-1), 1)
            sequence = torch.cat((sequence, next_id), 1)
            state = result["cache"] if use_cache else None
            if sequence.shape[0] == 1 and next_id.item() == eos_id:
                break
    finally:
        model.train(was_training)
    return sequence
