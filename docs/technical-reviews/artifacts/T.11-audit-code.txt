def digest(p):
    return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def emit(value):
    print(json.dumps(value, ensure_ascii=False, indent=2, allow_nan=False))
def compact(v):
    if isinstance(v, dict):
        return {k: compact(x) for k, x in v.items() if k not in ('samples', 'continuations', 'history', 'reward_history', 'icl_samples_same_model_weights', 'one_step_cap_samples')}
    if isinstance(v, list):
        return [compact(x) for x in v]
    return v
def token_budget(rows, steps, batch):
    examples = text_examples(rows, 'sft', 10000)
    counts = [int((y != IGNORE).sum()) for x, y in examples]
    sampler = random.Random(42)
    return sum(sum(sampler.choices(counts, k=batch)) for _ in range(steps))

    torch.manual_seed(42)
    model=TinyLM(ModelConfig(width=8,layers=1,heads=1))
    with tempfile.TemporaryDirectory(prefix='T.11-quantization-') as folder:
        source=Path(folder)/'float.pt'
        save_checkpoint(source,model,metadata={'review_scope':'untrained CPU example'})
        commands=[]
        observations={}
        for bits in (8,4):
            target=Path(folder)/f'bits{bits}.pt'
            command=[sys.executable,'scripts/quantize.py',str(source),'--bits',str(bits),'--output',str(target)]
            completed=subprocess.run(command,capture_output=True,text=True,timeout=30)
            assert completed.returncode==0,completed.stderr
            loaded,payload=load_checkpoint(target,'cpu')
            state=payload['model']
            assert torch.equal(model.embedding.weight,state['embedding.weight']) and state['embedding.weight'].dtype==torch.float32
            commands.append({'command':command,'returncode':completed.returncode,'stdout':completed.stdout,'stderr':completed.stderr})
            observations[str(bits)]={'report':json.loads(completed.stdout),'model_tensor_storage_bytes':sum(v.numel()*v.element_size() for v in state.values()),'embedding_dtype':str(state['embedding.weight'].dtype),'embedding_equal_original':True,'packed_values':{k:{'shape':list(v.shape),'dtype':str(v.dtype),'bytes':v.numel()*v.element_size()} for k,v in state.items() if k.endswith('.values')}}
