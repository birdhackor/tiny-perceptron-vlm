import sys
import json
import os
import hashlib
from pathlib import Path

import torch
from tiny_perceptron.data import ByteTokenizer, render_chat
from tiny_perceptron.model import TinyLM, ModelConfig, masked_loss
from tiny_perceptron.attention import attention_mask

torch.set_num_threads(1)
torch.manual_seed(42)

def offline_guard(event, args):
    if event in ("socket.connect", "socket.connect_ex", "socket.getaddrinfo", "subprocess.Popen", "os.system", "os.posix_spawn"):
        raise RuntimeError("bounded offline CPU probe blocked: " + event)

sys.addaudithook(offline_guard)
tok = ByteTokenizer()
original = [{"role": "user", "content": "1+1=?"}, {"role": "assistant", "content": "2"}]
exercise = [{"role": "user", "content": "2+1=?"}, {"role": "assistant", "content": "2"}]
x0, y0 = render_chat(original, tok)
x, y = render_chat(exercise, tok)
assert x.tolist() == [1, 3, 58, 51, 57, 69, 71, 2, 4, 58]
assert y.tolist() == [-100] * 8 + [58, 2]
assert len(x) == len(y) == 10
assert torch.equal(y, y0)
changed = (x != x0).nonzero().flatten().tolist()
assert changed == [2]
print(json.dumps({"test": "原練習：只改問題，答案固定2", "x": x.tolist(), "y": y.tolist(), "effective_targets": y[y != -100].tolist(), "lengths": [len(x), len(y)], "changed_x_positions": changed}, ensure_ascii=False))

corrected = [{"role": "user", "content": "2+1=?"}, {"role": "assistant", "content": "3"}]
xc, yc = render_chat(corrected, tok)
assert yc[yc != -100].tolist() == [59, 2]
print(json.dumps({"test": "另行修正算術示範的回答", "answer": 2 + 1, "effective_targets": yc[yc != -100].tolist()}, ensure_ascii=False))

literal = "<assistant>"
assert tok.assistant_id not in tok.encode(literal)
literal_x, literal_y = render_chat([{"role": "user", "content": literal}, {"role": "assistant", "content": "2"}], tok)
assert literal_x.tolist().count(tok.assistant_id) == 1
assert tok.decode(tok.encode(literal)) == literal
multi = [{"role": "user", "content": "a"}, {"role": "assistant", "content": "b"}, {"role": "user", "content": "c"}, {"role": "assistant", "content": "d"}]
mx, my = render_chat(multi, tok)
expected_ids = [1, 3, 105, 2, 4, 106, 2, 3, 107, 2, 4, 108, 2]
assert mx.tolist() == expected_ids[:-1]
assert my[my != -100].tolist() == [106, 2, 108, 2]
print(json.dumps({"test": "結構與字面內容分開；多輪依清單順序保留邊界", "literal_content_ids": tok.encode(literal), "literal_structural_assistant_count": literal_x.tolist().count(tok.assistant_id), "multiturn_full_ids": expected_ids, "multiturn_effective_targets": my[my != -100].tolist()}, ensure_ascii=False))

encoding = {text: {"bytes": list(text.encode("utf-8")), "token_ids": tok.encode(text)} for text in ["2", "中", "🙂"]}
assert encoding["2"]["bytes"] == [50]
assert encoding["2"]["token_ids"] == [58]
assert len(encoding["中"]["bytes"]) == 3
assert len(encoding["🙂"]["bytes"]) == 4
assert min(tok.encode(literal)) >= 8
print(json.dumps({"test": "每個UTF-8 byte加8；8位元共有256值", "byte_range": [0, 2**8 - 1], "examples": encoding}, ensure_ascii=False))

logits = torch.randn(1, len(y0), tok.vocab_size, requires_grad=True)
loss = masked_loss(logits, y0[None])
loss.backward()
ignored_grad_max = logits.grad[0, :8].abs().max().item()
active_grad_max = logits.grad[0, 8:].abs().max().item()
assert ignored_grad_max == 0.0 and active_grad_max > 0.0
with torch.no_grad():
    changed_logits = logits.detach().clone()
    changed_logits[:, :8] += torch.arange(tok.vocab_size)
    loss_delta = abs((masked_loss(changed_logits, y0[None]) - loss.detach()).item())
assert loss_delta == 0.0
allowed = attention_mask(torch.arange(10), torch.arange(10))
assert allowed[0, 0, 8].nonzero().flatten().tolist() == list(range(9))
model = TinyLM(ModelConfig(width=8)).eval()
with torch.no_grad():
    old_scores = model(x0[None])["logits"]
    new_scores = model(x[None])["logits"]
context_delta = (old_scores[0, 8] - new_scores[0, 8]).abs().max().item()
assert context_delta > 0.0
print(json.dumps({"test": "-100只忽略該位置直接交叉熵；問題仍可供後面讀取", "effective_target_count": int((y0 != -100).sum()), "ignored_logit_grad_max": ignored_grad_max, "active_logit_grad_max": active_grad_max, "ignored_logit_perturbation_loss_delta": loss_delta, "assistant_position_visible_key_positions": list(range(9)), "question_change_assistant_logits_max_delta": context_delta}, ensure_ascii=False))
print(json.dumps({"environment": {"python": sys.version, "torch": torch.__version__, "torch_source_sha": torch.version.git_version, "device": "cpu", "torch_threads": str(torch.get_num_threads()), "OMP_NUM_THREADS": os.environ.get("OMP_NUM_THREADS"), "MKL_NUM_THREADS": os.environ.get("MKL_NUM_THREADS"), "source_code_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest()}, "result": "全部bounded CPU核對通過；沒有參數更新、下載或雲端呼叫"}, ensure_ascii=False))
