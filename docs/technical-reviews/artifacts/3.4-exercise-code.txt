import torch

q = torch.tensor([[1.0, 0.0]])
k = torch.tensor([[1.0, 0.0], [0.0, 1.0]])
v = torch.tensor([[20.0, 40.0, 60.0], [-1.0, -2.0, -3.0]])
weights = (q @ k.T).softmax(dim=-1)
output = weights @ v
print(weights)
print(output, output.shape)

print("environment", __import__("sys").version, torch.__version__, torch.version.git_version, "device", str(q.device))
print("q", q.tolist(), "k", k.tolist(), "v", v.tolist())
print("scores", (q @ k.T).tolist(), "weights", weights.tolist(), "weight_sums", weights.sum(dim=-1).tolist())
print("exercise output", output.tolist(), "shapes", list(q.shape), list(k.shape), list(v.shape), list(weights.shape), list(output.shape))
reference_weights = (q @ k.T).softmax(dim=-1)
assert torch.equal(weights, reference_weights)
assert torch.equal(weights, (q @ k.T).softmax(dim=1))
assert torch.allclose(output, torch.tensor([[14.352230, 28.704460, 43.056690]]), atol=2e-5, rtol=0)
assert output.shape == (1,3)
print("exercise checks passed: fixed Q/K; weights unchanged; dim=-1 matches dim=1; expected output and shape")
