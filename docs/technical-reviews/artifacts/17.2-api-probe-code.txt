import json,sys,platform,os
from fractions import Fraction
import torch

def values(t):
    return t.tolist()
x=torch.tensor([-0.7,0.2,0.7,1.2])
q=(x/0.5).round().clamp(-127,127).to(torch.int8)
f=q.float()
rounded=(x-f*0.5).round(decimals=4)
before=q.clone()
pointer=q.data_ptr()
plain=q.tolist()
plain[0]=99
assert torch.equal(q,before) and q.data_ptr()==pointer
assert q.dtype==torch.int8 and q.element_size()==1
assert f.dtype==torch.float32 and values(f)==[-1.0,0.0,1.0,2.0]
info=torch.iinfo(torch.int8)
assert (info.bits,info.min,info.max)==(8,-128,127)
ties=torch.tensor([-2.5,-1.5,-0.5,0.5,1.5,2.5]).round()
assert values(ties)==[-2.0,-2.0,-0.0,0.0,2.0,2.0]
extreme=torch.tensor([-100.0,100.0])
saturated=(extreme/0.5).round().clamp(-127,127).to(torch.int8)
extreme_error=(extreme-saturated.float()*0.5).abs()
assert values(saturated)==[-127,127] and values(extreme_error)==[36.5,36.5]
assert torch.equal(x.to(torch.int8),torch.tensor([0,0,0,1],dtype=torch.int8))
record={
    'environment':{'python':sys.version,'python_executable':sys.executable,'torch':torch.__version__,'torch_git':torch.version.git_version,'platform':platform.platform(),'device':str(x.device),'threads':str(torch.get_num_threads()),'OMP_NUM_THREADS':os.environ['OMP_NUM_THREADS'],'MKL_NUM_THREADS':os.environ['MKL_NUM_THREADS'],'default_dtype':str(torch.get_default_dtype())},
    'input':values(x),'division':values(x/0.5),'rounded_integer_float':values((x/0.5).round()),'clamped_integer_float':values((x/0.5).round().clamp(-127,127)),
    'q_dtype':str(q.dtype),'q_element_size':q.element_size(),'q_storage_bytes':q.untyped_storage().nbytes(),'x_storage_bytes':x.untyped_storage().nbytes(),
    'int8':{'bits':info.bits,'min':info.min,'max':info.max},'float_dtype':str(f.dtype),'float_values':values(f),
    'tolist_did_not_mutate_tensor':bool(torch.equal(q,before)),'tolist_types':[type(v).__name__ for v in q.tolist()],
    'round_dtype':str(rounded.dtype),'rounded_difference':values(rounded),'binary32_point2_as_exact_fraction':str(Fraction(rounded[1].item())),
    'binary32_point2_minus_exact_decimal_point2':str(Fraction(rounded[1].item())-Fraction(1,5)),
    'ties_input':[-2.5,-1.5,-0.5,0.5,1.5,2.5],'ties_output':values(ties),
    'saturation_example':{'input':values(extreme),'q':values(saturated),'restored':values(saturated.float()*0.5),'absolute_error':values(extreme_error),'half_scale':0.25},
    'direct_float_to_int8':values(x.to(torch.int8)),
    'max_represented_value':{'scale_0.5':127*0.5,'scale_0.25':127*0.25},
    'assertions':'全部通過；補充核對 dtype、bytes、清單不改 tensor、最近偶數、端點飽和與十進位尾數。'
}
print(json.dumps(record,ensure_ascii=False,indent=2))
