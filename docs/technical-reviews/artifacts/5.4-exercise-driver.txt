import contextlib
import hashlib
import io
import json
import platform
import runpy
from pathlib import Path
import torch

code = Path("docs/technical-reviews/artifacts/5.4-exercise-code.txt")
stream = io.StringIO()
with contextlib.redirect_stdout(stream):
    ns = runpy.run_path(str(code))
expected = {"g": [1.0, -100.0], "m": [0.1, -10.0], "v": [0.001, 10.0],
            "m_hat": [1.0, -100.0], "v_hat": [1.0, 10000.0], "update": [0.001, -0.001]}
for key, values in expected.items():
    torch.testing.assert_close(ns[key], torch.tensor(values), rtol=1e-6, atol=1e-9)
assert ns["update"][1].item() < 0
result = {
    "command": "OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 CUDA_VISIBLE_DEVICES= .venv/bin/python docs/technical-reviews/artifacts/5.4-exercise-driver.txt",
    "environment": {"python": platform.python_version(), "torch": torch.__version__, "torch_git_version": torch.version.git_version,
                    "device": str(ns["g"].device), "dtype": str(ns["g"].dtype), "threads": str(torch.get_num_threads())},
    "code_sha256": hashlib.sha256(code.read_bytes()).hexdigest(), "stdout": stream.getvalue(),
    "values": {key: ns[key].tolist() for key in expected}, "expected": expected,
    "assertions": "所有六組值 assert_close(rtol=1e-6, atol=1e-9) 通過；更新第二格為負。",
    "scope": "只把原例第二個梯度 100 改為 -100；同樣零歷史、第一個更新步。"
}
print(json.dumps(result, ensure_ascii=False, indent=2))
