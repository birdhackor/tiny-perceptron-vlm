import torch
import torch.nn.functional as F
content = torch.tensor([2.0, 2.0, 2.0])
gate = torch.tensor([0.0, 0.0, 0.0])
factor = F.silu(gate)
print("gate", gate.tolist())
print("factor", factor.tolist())
print("output", (content * factor).tolist())
assert torch.allclose(content * factor, torch.tensor([0.0, 0.0, 0.0]), rtol=0, atol=1e-6)
print("assertion passed")
