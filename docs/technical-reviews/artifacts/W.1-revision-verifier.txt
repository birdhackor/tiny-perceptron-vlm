import datetime
import hashlib
import importlib.metadata
import json
import os
import platform

ROOT = Path('/workspace/tiny-perceptron-vlm')
ART = ROOT / 'docs/technical-reviews/artifacts'
sha = lambda b: hashlib.sha256(b).hexdigest()
now = lambda: datetime.datetime.now(datetime.timezone.utc).isoformat()
started = now()
assert os.environ['OMP_NUM_THREADS'] == os.environ['MKL_NUM_THREADS'] == '1'
import torch

assert torch.version.cuda is None
ENV = {
    'python': platform.python_version(), 'torch': torch.__version__,
    'torch_source_commit': torch.version.git_version, 'device': 'cpu',
    'platform': platform.platform(), 'OMP_NUM_THREADS': '1', 'MKL_NUM_THREADS': '1',
    'ipykernel': importlib.metadata.version('ipykernel'),
    'nbclient': importlib.metadata.version('nbclient'),

def write(name, value):
    target = ART / ('W.1-' + name + '.json')
    assert not target.exists(), target
    target.write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')
    return {'path': str(target.relative_to(ROOT)), 'sha256': sha(target.read_bytes())}

def section(raw, lesson):
    heads = list(re.finditer(rb'(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+', raw))
    for i, h in enumerate(heads):
        if h[1].decode() == lesson:
            return raw[h.start():heads[i + 1].start() if i + 1 < len(heads) else len(raw)]
    raise ValueError(lesson)

# This is a read-only inventory association, not an execution or a review of 222 lessons.
original_path = ROOT / 'docs/validation-artifacts/reader-round-notebook-kernels.json'
original_bytes = original_path.read_bytes()
original = json.loads(original_bytes)
assert original_bytes == (ART / 'W.1-frozen-kernel-receipt.json').read_bytes()
assert original['result']['total'] == original['result']['passed'] == 222
assert original['result']['failed'] == 0
assert len(original['records']) == 222
revisions = {}
receipt_copies = []
for lesson in ('2.1', '2.4', '3.1', '3.5', '5.11', '5.16', '5.6', '6.2', '7.13'):
    path = ROOT / f'docs/validation-artifacts/technical-revision-{lesson}-kernel.json'
    raw = path.read_bytes()
    receipt = json.loads(raw)
    assert receipt['result']['total'] == receipt['result']['passed'] == 1
    assert receipt['result']['failed'] == 0
    target = ART / f'W.1-cpu-{lesson}-receipt.json'
    assert not target.exists()
    target.write_bytes(raw)
    saved = {'path': str(target.relative_to(ROOT)), 'sha256': sha(raw),
             'original_path': str(path.relative_to(ROOT)), 'lesson_id': lesson}
    receipt_copies.append(saved)
    revisions[receipt['notebook']] = (receipt, saved)

inventory = []
programs = {}
chapter_cache = {}
for historical in original['records']:
    path = ROOT / historical['notebook']
    input_raw = path.read_bytes()
    is_changed = sha(input_raw) != historical['input_sha256']
    if is_changed:
        receipt, copied = revisions[path.relative_to(ROOT).as_posix()]
        entry = receipt
        receipt_ref = copied
    else:
        assert historical['status'] == 'passed'
        entry = historical
        receipt_ref = {'path': 'docs/technical-reviews/artifacts/W.1-frozen-kernel-receipt.json',
                       'sha256': sha(original_bytes)}
    assert sha(input_raw) == entry['input_sha256'], path
    executed_path = ROOT / 'outputs/notebooks-reader-round' / path.relative_to(ROOT / 'notebooks')
    executed_raw = executed_path.read_bytes()
    assert sha(executed_raw) == entry['executed_sha256'], executed_path
    notebook = json.loads(input_raw)
    executed = json.loads(executed_raw)
    assert len(notebook['cells']) == len(executed['cells'])
    assert notebook['metadata']['kernelspec']['name'] == 'tiny-perceptron'
    assert notebook['metadata']['lesson_id'] == path.stem
    cell_inspections = []
    counts = 0
    output_blocks = 0
    demonstration_sources = []
    for i, (input_cell, executed_cell) in enumerate(zip(notebook['cells'], executed['cells'])):
        assert input_cell['cell_type'] == executed_cell['cell_type']
        assert input_cell['source'] == executed_cell['source']
        if input_cell['cell_type'] != 'code':
            continue
        counts += 1
        source = ''.join(input_cell['source'])
        source_sha = sha(source.encode())
        programs[source_sha] = source
        assert isinstance(executed_cell.get('execution_count'), int)
        outputs = executed_cell.get('outputs', [])
        assert all(o.get('output_type') != 'error' for o in outputs)
        output_blocks += len(outputs)
        output_inspections = []
        for o in outputs:
            reduced = {'output_type': o['output_type']}
            if o['output_type'] == 'stream':
                reduced.update(name=o.get('name'), text=o.get('text'))
            if 'data' in o:
                reduced['data'] = {}
                for mime, value in o['data'].items():
                    value_raw = json.dumps(value, ensure_ascii=False, separators=(',', ':')).encode()
                    reduced['data'][mime] = {'json_sha256': sha(value_raw), 'json_bytes': len(value_raw)}
                    if mime == 'text/plain':
                        reduced['data'][mime]['value'] = value
            output_inspections.append(reduced)
        cell_inspections.append({'cell_index': i, 'source_sha256': source_sha,
                                 'execution_count': executed_cell['execution_count'],
                                 'outputs': output_inspections, 'error_outputs': 0})
        metadata = input_cell.get('metadata', {})
        if not metadata.get('course_setup') and not metadata.get('course_figure'):
            demonstration_sources.append(source)
    if 'code_cells' in entry:
        assert counts == entry['code_cells']
    if 'output_blocks' in entry:
        assert output_blocks == entry['output_blocks']
    chapter_path = ROOT / 'course/chapters' / (path.parent.name + '.md')
    chapter_key = str(chapter_path.relative_to(ROOT))
    if chapter_key not in chapter_cache:
        chapter_cache[chapter_key] = chapter_path.read_bytes()
    current_section = section(chapter_cache[chapter_key], path.stem)
    # Match only Python programs. This does not inspect unrelated prose or SVG contents.
    fences = re.findall(rb'\x60\x60\x60python\n(.*?)\x60\x60\x60', current_section, re.S)
    authored_sources = [b.decode().strip().rstrip('\n') for b in fences]
    assert demonstration_sources == authored_sources, path
    inventory.append({
        'lesson_id': path.stem, 'notebook': str(path.relative_to(ROOT)),
        'source': chapter_key + '#' + path.stem, 'section_sha256': sha(current_section),
        'chapter_file_sha256': sha(chapter_cache[chapter_key]),
        'input_sha256': sha(input_raw), 'executed_original_path': str(executed_path.relative_to(ROOT)),
        'executed_sha256': sha(executed_raw), 'receipt': receipt_ref,
        'receipt_kind': 'specific_revision' if is_changed else 'original_222',
        'all_cell_sources_equal': True, 'demonstration_code_equal_to_current_python_fences': True,
        'demonstration_cells': len(demonstration_sources), 'code_cells': counts,
        'output_blocks': output_blocks, 'code_cell_inspections': cell_inspections,
        'error_outputs': 0,
    })
assert len({r['notebook'] for r in inventory}) == 222
assert set(r['notebook'] for r in inventory if r['receipt_kind'] == 'specific_revision') == set(revisions)
assert {str(p.relative_to(ROOT)) for p in (ROOT / 'notebooks').rglob('*.ipynb')} == {r['notebook'] for r in inventory}

guard = '''import sys
def guard(event, args):
    if event.startswith('socket.') or event in ('subprocess.Popen', 'os.system'):
        raise RuntimeError('W.1 bounded offline CPU guard denied ' + event)
sys.addaudithook(guard)
'''
cases = []
for record in guide['records']:
    program = record['program']
    assert program == expected_programs[record['case']]
    assert sha(program.encode()) == record['program_sha256']
    guarded = guard + 'exec(compile(' + repr(program) + ', "W.1:' + record['case'] + '", "exec"))\n'
    command = [str(ROOT / '.venv/bin/python'), '-c', guarded]
    run = subprocess.run(command, cwd=ROOT, env={**os.environ, 'PYTHONDONTWRITEBYTECODE': '1'},
                         capture_output=True, text=True, timeout=60)
    assert run.returncode == 0, (record['case'], run.stderr)
    assert run.stdout == record['stdout'], (record['case'], run.stdout)
    assert run.stderr == record['stderr'] == ''
    cases.append({'case': record['case'], 'program': program, 'program_sha256': sha(program.encode()),
                  'executed_program': guarded, 'executed_program_sha256': sha(guarded.encode()),
                  'command': command, 'exit_code': run.returncode, 'stdout': run.stdout,
                  'stderr': run.stderr, 'matches_existing_receipt': True,
                  'offline_guard': 'Python audit hook rejects all socket events and subprocess.Popen/os.system in child; no rejected event occurred.'})
assert len(cases) == 3
assert section((ROOT / 'course/first-steps.md').read_bytes(), 'W.1') == body
