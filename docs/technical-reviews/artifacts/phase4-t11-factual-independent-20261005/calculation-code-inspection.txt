171: def _rag_records(seed_value):
172:     rng = random.Random(seed_value)
173:     records = []
174:     # 同一店名的全部地址、來源及提示變形共用 family，避免逐筆切分洩漏。
175:     for index in range(120):
176:         key = f"K{index:03d}"
177:         for variant in range(8):
178:             address = f"{rng.choice('ABCD')}{rng.randrange(10)}"
179:             source = f"D{rng.randrange(10)}"
180:             correct = {"id": source, "text": f"{key} address={address}"}
181:             distractor = {
182:                 "id": f"D{(int(source[1:]) + 1) % 10}",
183:                 "text": f"Z{index:03d} address={rng.choice('ABCD')}{rng.randrange(10)}",
184:             }
185:             documents = [correct] if variant < 2 else [correct, distractor]
186:             if variant == 6:
187:                 documents = []
188:             elif variant == 7:
189:                 documents = [distractor]
190:             elif variant % 2:
191:                 documents.reverse()
192:             answer = f"{address}[{source}]" if variant < 6 else "UNKNOWN"
193:             records.append(
194:                 {
195:                     "family": key,
196:                     "key": key,
197:                     "address": address,
198:                     "source": source,
199:                     "documents": documents,
200:                     "variant": variant,
201:                     "messages": _conversation(_rag_question(key, documents), answer),
202:                 }
203:             )
204:     return records

707: def _verify_reasoning(sample, row, mode):
708:     text = sample["generated"].strip()
709:     clean = not sample["invalid_special_tokens"]
710:     if mode == "direct":
711:         parsed = int(text) if clean and re.fullmatch(r"-?[0-9]+", text) else None
712:         return {
713:             "final_answer": parsed,
714:             "parsed": parsed is not None,
715:             "final_correct": parsed == row["truth"],
716:             "fully_verified": parsed == row["truth"],
717:         }
718:     matched = re.fullmatch(
719:         r"(-?[0-9]+)\+(-?[0-9]+)=(-?[0-9]+);(-?[0-9]+)\+(-?[0-9]+)=(-?[0-9]+);answer=(-?[0-9]+)", text
720:     )
721:     if not matched or not clean:
722:         # 結尾格式可獨立解析，讓「最終對但過程錯」保留在資料中。
723:         final_match = re.search(r";answer=(-?[0-9]+)$", text) if clean else None
724:         parsed = int(final_match[1]) if final_match else None
725:         return {
726:             "final_answer": parsed,
727:             "parsed": False,
728:             "final_correct": parsed == row["truth"],
729:             "equations_valid": False,
730:             "linked": False,
731:             "task_operands_valid": False,
732:             "fully_verified": False,
733:         }
734:     a, b, subtotal, previous, c, total, final = map(int, matched.groups())
735:     equations = a + b == subtotal and previous + c == total
736:     linked = previous == subtotal and total == final
737:     task = (a, b, c) == (row["a"], row["b"], row["c"])
738:     return {
739:         "final_answer": final,
740:         "parsed": True,
741:         "steps": [[a, b, subtotal], [previous, c, total]],
742:         "final_correct": final == row["truth"],
743:         "equations_valid": equations,
744:         "linked": linked,
745:         "task_operands_valid": task,
746:         "fully_verified": equations and linked and task and final == row["truth"],
747:     }
