42: def _sample(model, messages, ctx, count=1, tokens=32, temperature=0.0):
43:     """實際 batch 抽樣，保留 EOS、非法特殊 token 與測得的生成成本。"""
44:     tokenizer = ByteTokenizer()
45:     prompt = _prompt_ids(messages, tokenizer)
46:     if len(prompt) + tokens > model.config.max_length:
47:         raise ValueError(f"輸入 {len(prompt)} + 答案預留 {tokens} > {model.config.max_length}")
48:     was_training = model.training
49:     model.eval()
50:     current = torch.tensor([prompt] * count, device=ctx.device)
51:     active = torch.ones(count, dtype=torch.bool, device=ctx.device)
52:     outputs = [[] for _ in range(count)]
53:     state = None
54:     forward_tokens = 0
55:     _sync(ctx)
56:     started = time.perf_counter()
57:     try:
58:         for _ in range(tokens):
59:             result = model(current, cache=state)
60:             forward_tokens += current.numel()
61:             scores = result["logits"][:, -1]
62:             chosen = (
63:                 scores.argmax(-1)
64:                 if temperature <= 0
65:                 else torch.multinomial((scores / temperature).softmax(-1), 1).squeeze(-1)
66:             )
67:             was_active = active.tolist()
68:             for row, (keep, token) in enumerate(zip(was_active, chosen.tolist(), strict=True)):
69:                 if keep:
70:                     outputs[row].append(token)
71:             active = active & (chosen != tokenizer.eos_id)
72:             if not active.any():
73:                 break
74:             current = chosen.masked_fill(~active, tokenizer.pad_id).unsqueeze(1)
75:             state = result["cache"]
76:     finally:
77:         model.train(was_training)
78:     _sync(ctx)
79:     seconds = time.perf_counter() - started
80:     samples = []
81:     for output in outputs:
82:         raw = output[:-1] if output and output[-1] == tokenizer.eos_id else output
83:         samples.append(
84:             {
85:                 "generated": tokenizer.decode(raw),
86:                 "generated_ids": output,
87:                 "eos": bool(output and output[-1] == tokenizer.eos_id),
88:                 "invalid_special_tokens": [token for token in raw if token < 8],
89:                 "generated_tokens": len(output),
90:             }
91:         )
92:     return {
93:         "messages": messages,
94:         "input_ids": prompt,
95:         "input_tokens": len(prompt),
96:         "samples": samples,
97:         "candidate_count": count,
98:         "temperature": temperature,

550: def parse_action(trace):
551:     if not trace["eos"]:
552:         return {"status": "invalid", "reason": "unterminated_generation"}
553:     raw = trace["raw"]
554:     if raw.startswith("DIRECT:") and len(raw) > 7:
555:         return {"status": "direct", "content": raw[7:]}
556:     if raw.startswith("ASK:") and len(raw) > 4:
557:         return {"status": "ask", "content": raw[4:]}
558:     match = re.fullmatch(r"TOOL:([a-z_]+):([0-9]{1,3})\+([0-9]{1,3})", raw)
559:     if match:
560:         return {"status": "tool", "name": match[1], "a": int(match[2]), "b": int(match[3])}
561:     return {"status": "invalid", "reason": "malformed_action"}
