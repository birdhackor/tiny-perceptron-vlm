import torch

logits = torch.tensor([8.0, 2.0, 0.0])
for temperature in (1.0, 2.0, 4.0):
    probability = (logits / temperature).softmax(0)
    print(
        "T",
        temperature,
        "機率",
        probability.round(decimals=4).tolist(),
        "第一比第二",
        round((probability[0] / probability[1]).item(), 4),
    )

base = torch.tensor([4.0, 1.0, 0.0])
for new_t, base_t in ((2.0, 1.0), (4.0, 2.0)):
    new_probability = (logits / new_t).softmax(0)
    base_probability = (base / base_t).softmax(0)
    print("exact_equal", new_t, base_t, torch.equal(new_probability, base_probability))
    assert torch.equal(new_probability, base_probability)
