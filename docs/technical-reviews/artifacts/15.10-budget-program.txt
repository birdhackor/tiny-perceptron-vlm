import json
from pathlib import Path

import torch

from tiny_perceptron.model import ModelConfig, TinyLM
from scripts.course_experiments.architecture import _parameter_budget

torch.set_num_threads(1)
frozen = json.loads(Path('docs/course-experiments/results/moe.json').read_text())
for name in ('top1_aux0', 'top1_aux001', 'top2_aux0', 'top2_aux001'):
    if name not in frozen['results']['variants']:
        continue
    variant = frozen['results']['variants'][name]
    model = TinyLM(ModelConfig(**variant['model']['config'])).cpu()
    observed = _parameter_budget(model)
    assert observed == variant['budget'], (name, observed, variant['budget'])
    nonexpert_nonrouter = (
        observed['total_parameters'] - observed['expert_parameters'] - observed['router_parameters']
    )
    elements = {key: parameter.numel() for key, parameter in model.named_parameters()}
    assert all(parameter.element_size() == 4 for parameter in model.parameters())
    assert nonexpert_nonrouter == 75392
    print(json.dumps({
        'variant': name,
        'config': variant['model']['config'],
        'budget': observed,
        'other_parameters': nonexpert_nonrouter,
        'parameter_bytes': sum(parameter.numel() * parameter.element_size() for parameter in model.parameters()),
        'named_parameter_elements': elements,
    }, ensure_ascii=False))
