from pathlib import Path
import hashlib,json,re
p=Path("docs/course-experiments/results/style.json")
d=json.loads(p.read_text());t=d["results"]["after"]["test"]
print("source_sha256",hashlib.sha256(p.read_bytes()).hexdigest())
counts={"records":0,"parse_success":0,"only_answer_int":0,"content_correct":0,"valid_raw_byte_tokens":0}
for s in t["samples"]:
    if s.get("style") != "json": continue
    prompt=s["messages"][0]["content"]
    a,b=map(int,re.search(r"(\d+)\+(\d+)=",prompt).groups())
    value=json.loads(s["generated"])
    valid=isinstance(value,dict) and set(value)=={"answer"} and type(value["answer"]) is int
    correct=valid and value["answer"]==a+b
    ids=s["generated_ids"]
    raw=ids[:ids.index(2)] if 2 in ids else ids
    valid_raw=all(8<=i<264 for i in raw) and bytes(i-8 for i in raw).decode("utf-8")==s["generated"]
    counts["records"]+=1;counts["parse_success"]+=1;counts["only_answer_int"]+=int(valid);counts["content_correct"]+=int(correct);counts["valid_raw_byte_tokens"]+=int(valid_raw)
    print(json.dumps({"question":prompt,"generated":s["generated"],"target":a+b,"parsed_type":type(value["answer"]).__name__,"valid":valid,"correct":correct,"raw_tokens_valid":valid_raw},ensure_ascii=False))
print(json.dumps(counts,ensure_ascii=False,sort_keys=True))
assert counts=={"records":7,"parse_success":7,"only_answer_int":7,"content_correct":0,"valid_raw_byte_tokens":7}
assert t["rubric"]["json"]=={"records":7,"content_correct":0,"style_correct":7,"json_valid":7}
