def collect(client, code):
    msg_id = client.execute(code, store_history=False)
    streams = []
    errors = []
    deadline = time.monotonic() + 20
    while True:
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            raise TimeoutError("W.2 local CPU kernel cell timeout")
        msg = client.get_iopub_msg(timeout=remaining)
        if msg.get("parent_header", {}).get("msg_id") != msg_id:
            continue
        kind = msg["header"]["msg_type"]
        if kind == "stream":
            streams.append(msg["content"]["text"])
        elif kind == "error":
            errors.append({k: msg["content"][k] for k in ("ename", "evalue")})
        elif kind == "status" and msg["content"]["execution_state"] == "idle":
            break
    while True:
        reply = client.get_shell_msg(timeout=20)
        if reply.get("parent_header", {}).get("msg_id") == msg_id:
            break
    assert not errors, errors
    assert reply["content"]["status"] == "ok", reply["content"]
    return {"stdout": "".join(streams), "errors": errors, "execute_reply_status": reply["content"]["status"]}
