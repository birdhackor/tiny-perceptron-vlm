import contextlib
import hashlib
import io
import json
import os
import platform
import sys
from pathlib import Path

import torch

def deny_network(event, args):
    if event in {"socket.connect", "socket.getaddrinfo", "socket.bind", "socket.sendto", "subprocess.Popen", "os.system"}:
        raise RuntimeError("offline reviewer evidence prohibits " + event)
sys.addaudithook(deny_network)
path = Path("docs/technical-reviews/artifacts/8.2-exercise-code.txt")
code = path.read_bytes()
buf = io.StringIO()
with contextlib.redirect_stdout(buf):
    exec(compile(code, str(path), "exec"), {})
actual = buf.getvalue()
expected = "用一句話回答：2+2=?\n只回一個數字：2+2=?\n兩次都使用同一份權重與同一個選字方式\n"
assert actual == expected
print(json.dumps({"code_path": str(path), "code_sha256": hashlib.sha256(code).hexdigest(), "stdout": actual, "expected": expected, "exact_match": actual == expected, "exit_code": 0, "environment": {"python": platform.python_version(), "torch": torch.__version__, "torch_git_version": torch.version.git_version, "device": "cpu", "OMP_NUM_THREADS": os.environ["OMP_NUM_THREADS"], "MKL_NUM_THREADS": os.environ["MKL_NUM_THREADS"], "offline": "HF flags and audit hook block sockets and subprocesses"}}, ensure_ascii=False, indent=2))
