import torch

w = torch.tensor([[0.7, 0.2], [-0.7, 1.2]])
scale = 0.5
restored = (w / scale).round() * scale
x = torch.tensor([[1.0, 1.0], [1.0, -1.0]])
original_output = x @ w.T
compressed_output = x @ restored.T
print("權重平均絕對差", round((w - restored).abs().mean().item(), 4))
print("原輸出", original_output.round(decimals=4).tolist())
print("還原權重輸出", compressed_output.round(decimals=4).tolist())
print("逐項輸出差", (original_output - compressed_output).round(decimals=4).tolist())

# 本人追加的有界 CPU 核對；上方保留原文例子。
import json,sys,platform

print("DETAIL",json.dumps({"w":w.tolist(),"restored":restored.tolist(),"x":x.tolist(),"weight_mae_raw":(w-restored).abs().mean().item(),"output_mae_raw":(original_output-compressed_output).abs().mean().item(),"w_dtype":str(w.dtype),"restored_dtype":str(restored.dtype),"original_output_shape":list(original_output.shape),"restored_element_size":restored.element_size(),"same_diff_formula":torch.allclose(original_output-compressed_output,x@(w-restored).T)}))
assert torch.allclose(original_output-compressed_output,x@(w-restored).T,atol=2e-6,rtol=1e-5)
assert abs((w-restored).abs().mean().item()-0.2)<1e-6
assert abs((original_output-compressed_output).abs().mean().item()-0.2)<1e-6
