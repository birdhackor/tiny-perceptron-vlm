def cli(args):
    command = [sys.executable] + args
    completed = subprocess.run(command,cwd=ROOT,capture_output=True,text=True,timeout=30,check=True)
    report["cli"].append({"command":command,"exit_code":completed.returncode,"stdout":completed.stdout,"stderr":completed.stderr})
    return completed.stdout


def public_model(group,name):
    folder = ROOT / "checkpoints/course" / group
    manifest = json.loads((folder/"download-manifest.json").read_text())
    expected = next(x for x in manifest["files"] if x["output"] == name)
    path = folder/name
    digest = hashlib.sha256(path.read_bytes()).hexdigest()
    assert digest == expected["sha256"] and path.stat().st_size == expected["bytes"]
    model,payload = load_checkpoint(path,"cpu")
    report["public_inputs"].append({"local_input":str(path.relative_to(ROOT)),"sha256":digest,"bytes":path.stat().st_size,
        "immutable_url":f"https://huggingface.co/{manifest['repo']}/resolve/{manifest['revision']}/{expected['path']}",
        "revision":manifest["revision"],"format_version":payload["format_version"],"config":payload["config"],
        "payload_keys":sorted(payload),"optimizer_or_rng_present":any("rng" in k or k == "optimizer" for k in payload),
        "modal_config":payload.get("modal_config"),"storage":_storage(model,path),
        "evidence_class":"ignored public input inspected directly; this own persistent metadata records pin/config, not tracked repository code"})
    return model

               "offset":int(components[2]),"family":s["family"],"question":s["question"],"answer":s["target"]}

        model = public_model("multimodal_distillation",f"{task}-{method}.pt")
        for ablation in (["none","blank_image","blank_audio"] if task=="joint" and method=="ce_kl" else ["none"]):
            e = evaluate_modal(model,rows,ctx,ablation)
            expected = t["runs"][method]["test" if ablation=="none" else f"{ablation}_test"]
            assert [s["generated_ids"] for s in e["samples"]] == [s["generated_ids"] for s in expected["samples"]]
            assert abs(e["mean_token_nll"]-expected["mean_token_nll"]) < 1e-5
            report["modal_replay"][f"{task}-{method}-{ablation}"] = e
    teacher = public_model("multimodal_distillation",f"{task}-teacher.pt")
    student = public_model("multimodal_distillation",f"{task}-ce_kl.pt")
    probe = {"modality":"vision","color":"red","shape":"square","offset":0,"question":"color?","answer":"red","family":"probe"} if task=="vqa" else {
        "modality":"joint","color":"blue","shape":"square","offset":0,"frequency":180,"question":"joint?","answer":"square,low","family":"probe"}
    ids,labels,image,waveform,_ = modal_inputs(probe,ctx)
    with torch.no_grad():
        outputs = [m(ids,labels,image=image,waveform=waveform) for m in (teacher,student)]
    selected = [o["labels"][0]!=-100 for o in outputs]
    target_ids = [o["labels"][0,s].tolist() for o,s in zip(outputs,selected,strict=True)]
    assert target_ids[0] == target_ids[1]
    prediction_rows = [s.nonzero().flatten().tolist() for s in selected]
    expected_rows = [list(range(25,29)),list(range(13,17))] if task=="vqa" else [list(range(36,47)),list(range(24,35))]

teacher = torch.tensor([[[2.,0.,-1.],[0.,1.,2.]]])
student = torch.tensor([[[0.,1.,2.],[2.,1.,0.]]],requires_grad=True)
labels = torch.tensor([[1,-100]])
loss = distillation_kl(student,teacher,labels,temperature=2)
p = (teacher[0,0]/2).softmax(-1)
q = (student[0,0]/2).softmax(-1)
manual = 4*(p*(p.log()-q.log())).sum()
assert torch.allclose(loss,manual,atol=1e-6)
loss.backward()
assert not student.grad[0,1].any()
report["api_conditions"]["temperature_mask_probe"] = {"actual":loss.item(),"manual_T_squared_once":manual.item(),"ignored_row_gradient":student.grad[0,1].tolist()}
