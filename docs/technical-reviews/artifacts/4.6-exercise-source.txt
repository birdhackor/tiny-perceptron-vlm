import json
import torch
from tiny_perceptron.model import TinyLM, ModelConfig
import torch.nn.functional as F

# 原節練習：只把 vocab_size=20 改為 25，輸入 ID 保持不動。
torch.manual_seed(42)
model = TinyLM(ModelConfig(vocab_size=25, width=8))
ids = torch.tensor([[1, 2, 3]])
logits = model(ids)["logits"]
print(logits.shape)
print("第一位置的前五個候選分數", logits[0, 0, :5])
print("各位置最高分ID", logits.argmax(dim=-1))
assert logits.shape == (1, 3, 25)
target = torch.tensor([[2, 3, 4]])
scores_for_questions = logits.reshape(-1, logits.shape[-1])
answers_for_questions = target.reshape(-1)
print("三題平均代價", F.cross_entropy(scores_for_questions, answers_for_questions).item())
print("25項字表參數", model.description()["parameters"])

torch.manual_seed(42)
baseline = TinyLM(ModelConfig(vocab_size=20, width=8))
print("20項字表參數", baseline.description()["parameters"])
print(json.dumps({
    "exercise_shape": list(logits.shape),
    "reshape_shape": list(scores_for_questions.shape),
    "target_shape": list(answers_for_questions.shape),
    "argmax_shape": list(logits.argmax(-1).shape),
    "parameters_20": baseline.description()["parameters"],
    "parameters_25": model.description()["parameters"],
    "delta_parameters": model.description()["parameters"] - baseline.description()["parameters"],
    "candidate_axis_delta": logits.shape[-1] - baseline(ids)["logits"].shape[-1],
    "argmax_20": baseline(ids)["logits"].argmax(-1).tolist(),
    "argmax_25": logits.argmax(-1).tolist(),
    "same_input_ids": ids.tolist(),
    "updates": 0,
}, ensure_ascii=False))
assert model.description()["parameters"] - baseline.description()["parameters"] == 2 * 5 * 8
