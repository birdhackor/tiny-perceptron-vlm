import json,hashlib
from pathlib import Path
from tiny_perceptron.data import ByteTokenizer
r=json.loads(Path('docs/course-experiments/results/multimodal_distillation.json').read_text())
tok=ByteTokenizer()
for task,t in r['results']['tasks'].items():
    for branch,e in [('teacher',t['teacher_test'])]+[(b,t['runs'][b]['test']) for b in ['ce','ce_kl']]:
        samples=e['samples']
        raw=[s['generated_ids'][:s['generated_ids'].index(2)] if 2 in s['generated_ids'] else s['generated_ids'] for s in samples]
        observed=sum(ids==tok.encode(s['target']) for ids,s in zip(raw,samples,strict=True))
        tokens=sum(len(tok.encode(s['target']))+1 for s in samples)
        assert observed==e['correct'] and tokens==e['effective_tokens']
        assert len(samples)==e['examples']==12 and all(2 in s['generated_ids'] for s in samples)
        assert all(not s['generation_error'] and not s['invalid_special_tokens'] for s in samples) and not e['skipped']
        print(json.dumps(dict(task=task,branch=branch,N=len(samples),correct_recount=observed,targets_recount=tokens,frozen_nll=e['mean_token_nll'],nll_round4=round(e['mean_token_nll'],4),EOS=12),ensure_ascii=False))
    ce,kl=t['runs']['ce'],t['runs']['ce_kl']
    assert ce['training']['initialization_sha256']==kl['training']['initialization_sha256']
    assert ce['training']['batch_plan_sha256']==kl['training']['batch_plan_sha256']
    print(json.dumps(dict(task=task,teacher_frozen=t['teacher_frozen_and_unchanged'],same_initialization=True,same_batch_plan=True,blank_image_correct={b:t['runs'][b]['blank_image_test']['correct'] for b in ['ce','ce_kl']}),ensure_ascii=False))
    for s in kl['test']['samples']:
        if task=='vqa' and s['family']=='blue:square:2' and s['question']=='color?':print('VQA blue square',s['target'],s['generated'])
    if task=='joint':
        for b in ['ce','ce_kl']:
            print('joint blank audio',b,t['runs'][b]['blank_audio_test']['correct'])
            s=t['runs'][b]['test']['samples'][2];print('joint red square260',b,s['target'],s['generated'])
print('four_updates_sum',sum(b['training']['steps'] for t in r['results']['tasks'].values() for b in t['runs'].values()))
print('elapsed',r['elapsed_seconds'],'seconds','GPU',r['gpu'],'scope',r['timing_scope'])
print('No training schedule rerun. These are independent recounts of frozen original L4 records, not new quality measurements.')
