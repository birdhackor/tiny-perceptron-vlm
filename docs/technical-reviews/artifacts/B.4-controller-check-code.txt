import json
from tiny_perceptron.retrieval import tool_loop

request = {"name": "add", "arguments": {"a": 2, "b": 3}}
wrong_done = tool_loop([request, {"done": True, "answer": 6}], max_steps=3)
bad_name = tool_loop([{"name": "unknown", "arguments": {"a": 2, "b": 3}}])
bad_parameter = tool_loop([{"name": "add", "arguments": {"a": True, "b": 3}}])
exhausted = tool_loop([request], max_steps=3)
two_steps = tool_loop([request, {"done": True, "answer": 5}], max_steps=2)
assert wrong_done["status"] == "done" and wrong_done["answer"] == 6
assert wrong_done["trace"][0]["result"] == 5
assert bad_name["status"] == "invalid_request" and bad_name["error"] and not bad_name["trace"]
assert bad_parameter["status"] == "invalid_request" and bad_parameter["error"] and not bad_parameter["trace"]
assert exhausted == {"status": "needs_more_model_output", "trace": [{"request": request, "result": 5}]}
assert two_steps["status"] == "done" and two_steps["answer"] == 5 and len(two_steps["trace"]) == 1
print(json.dumps({"wrong_done": wrong_done, "bad_name": bad_name, "bad_parameter": bad_parameter,
                  "exhausted": exhausted, "two_steps": two_steps}, ensure_ascii=False, indent=2))
