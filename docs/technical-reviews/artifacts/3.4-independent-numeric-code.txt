import json, math, platform, torch
w0=math.exp(1)/(math.exp(1)+math.exp(0))
w1=math.exp(0)/(math.exp(1)+math.exp(0))
v0=[10.0,20.0,30.0]
v1=[-1.0,-2.0,-3.0]
original=[w0*x+w1*y for x,y in zip(v0,v1)]
exercise=[w0*(2*x)+w1*y for x,y in zip(v0,v1)]
swapped=[w0*y+w1*x for x,y in zip(v0,v1)]
weights=torch.tensor([[w0,w1]],dtype=torch.float64)
values=torch.tensor([v0,v1],dtype=torch.float64)
delta=torch.tensor([1.0,-2.0,3.0],dtype=torch.float64)
changed=values.clone()
changed[0]+=delta
changed[1]-=(w0/w1)*delta
collision_error=(weights@changed-weights@values).abs().max().item()
assert collision_error<1e-12
assert original!=swapped
assert sum([w0,w1])==1.0
print(json.dumps({"environment":{"python":platform.python_version(),"torch":str(torch.__version__),"torch_commit":torch.version.git_version,"device":"cpu"},"exp1":math.exp(1),"exp0":math.exp(0),"denominator":math.exp(1)+math.exp(0),"weights":[w0,w1],"original_output":original,"exercise_output":exercise,"unmatched_v_order_output":swapped,"different_values_same_output":changed.tolist(),"collision_max_abs_error":collision_error},ensure_ascii=False,indent=2))
