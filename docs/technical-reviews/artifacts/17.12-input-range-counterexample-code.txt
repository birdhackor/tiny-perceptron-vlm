import hashlib
import json
import os
import platform
import sys
from pathlib import Path
import torch

assert torch.get_num_threads() == 1
assert torch.version.git_version == '5c4886908584029761b579af026dcfb627c84070'
assert not torch.cuda.is_available()

def offline_guard(event, arguments):
    if event in ('socket.connect', 'socket.getaddrinfo', 'socket.bind', 'socket.sendto', 'subprocess.Popen', 'os.system'):
        raise RuntimeError('Offline bounded CPU review blocks ' + event)

sys.addaudithook(offline_guard)
weights = torch.tensor([[1.0, 1.0], [2.0, 2.0]])
first_input = torch.tensor([1.0, 0.0])
second_input = torch.tensor([0.0, 1.0])
with torch.no_grad():
    first_features = weights @ first_input
    repeated_features = weights @ first_input.clone()
    second_features = weights @ second_input
assert not torch.equal(first_input, second_input)
assert torch.equal(first_features, repeated_features)
assert torch.equal(first_features, second_features)
assert first_features.tolist() == [1.0, 2.0]
assert first_features.abs().max().item() == second_features.abs().max().item() == 2.0
record = {

    'code_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),

    'first_input': first_input.tolist(),
