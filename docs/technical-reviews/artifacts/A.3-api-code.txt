from collections import Counter
from tiny_perceptron.retrieval import lexical_terms, retrieve
import json

def score(query, text):
    q, d = Counter(lexical_terms(query)), Counter(lexical_terms(text))
    return sum(min(d[t], n) for t, n in q.items())

checks = {
    "book_shop": lexical_terms("Book shop"),
    "bookstore": lexical_terms("書店"),
    "mixed_ascii_contiguous": lexical_terms("K017 C2"),
    "d1_score": score("書店地址", "貓喜歡曬太陽"),
    "d2_score": score("書店地址", "書店地址是青街8號"),
    "repeated_book_score": score("書", "書" * 10),
    "not_published_score": score("書店地址", "書店地址尚未公布"),
    "alias_terms": lexical_terms("營業處在哪"),
    "zero_result": retrieve("無", [{"id": "a", "text": "書店"}], k=1),
    "tie_ids": [d["id"] for d in retrieve("書店", [{"id": "b", "text": "書店地址未知"}, {"id": "a", "text": "書店"}], k=2)],
}
assert checks["book_shop"] == ["book", "shop"]
assert checks["bookstore"] == ["書", "店"]
assert checks["mixed_ascii_contiguous"] == ["k017", "c2"]
assert [checks[k] for k in ["d1_score", "d2_score", "repeated_book_score", "not_published_score"]] == [0, 4, 1, 4]
assert checks["zero_result"] == [] and checks["tie_ids"] == ["a", "b"]
print(json.dumps(checks, ensure_ascii=False, indent=2))
