import json
from pathlib import Path
from scripts.infer_modal import load_modal_checkpoint
from scripts.course_experiments.modalities import evaluate_modal
from scripts.course_experiments.common import Context
import torch
r=json.loads(Path('docs/course-experiments/results/multimodal_distillation.json').read_text())['results']['tasks']['joint']['runs']['ce_kl']
a,b=r['test']['samples'],r['blank_image_test']['samples']
records=[]
for x,y in zip(a,b,strict=True):
    assert x['family']==y['family'] and x['generated_ids']==y['generated_ids']
    c,s,o,f=x['family'].split(':')
    records.append(dict(modality='joint',color=c,shape=s,offset=int(o),frequency=int(f),question='joint?',answer=x['target'],family=x['family']))
    print(json.dumps(dict(family=x['family'],target=x['target'],normal=x['generated_ids'],blank_image=y['generated_ids'],shape_correct=x['generated'].split(',')[0]==s,pitch_correct=x['generated'].split(',')[1]==x['target'].split(',')[1]),ensure_ascii=False))
squares=[x for x in a if x['target'].startswith('square,')]
assert len(squares)==6 and all(x['generated'].startswith('circle,') for x in squares)
assert all(x['generated'].split(',')[1]==x['target'].split(',')[1] for x in a)
model,_,_=load_modal_checkpoint('checkpoints/course/multimodal_distillation/joint-ce_kl.pt','cpu')
ctx=Context('cpu',Path('/tmp/18.13-unused'),Path('/tmp/18.13-unused'),Path('assets/training'))
with torch.no_grad():
    normal=evaluate_modal(model,records,ctx)
    blank=evaluate_modal(model,records,ctx,'blank_image')
assert all(x['generated_ids']==y['generated_ids']==z['generated_ids'] for x,y,z in zip(a,normal['samples'],blank['samples'],strict=True))
print('frozen 與本次 pinned public CPU重播的12組ID全部一致；六道方形題音高正確但形狀circle錯，所以整題不正確。')
print(json.dumps(dict(normal_correct=normal['correct'],blank_correct=blank['correct'],normal_nll=normal['mean_token_nll'],blank_nll=blank['mean_token_nll'],shape_correct=normal['shape_correct'],pitch_correct=normal['pitch_correct']),ensure_ascii=False))
