def read(name):
    path = ROOT / f"docs/course-experiments/results/{name}.json"
    raw = path.read_bytes()
    data = json.loads(raw)
    assert data["evidence_status"] == "complete_run"
    return data, {"path": str(path.relative_to(ROOT)), "sha256": hashlib.sha256(raw).hexdigest(),
                  "revision": data["revision"], "device": data["device"], "gpu": data["gpu"],
                  "torch": data["torch_version"], "python": data["python_version"], "seed": data["seed"],
                  "elapsed_seconds": data["elapsed_seconds"], "timing_scope": data["timing_scope"]}


def compact(value):
    if isinstance(value, dict):
        return {k:compact(v) for k,v in value.items() if k not in ("history","loss_trace")}
    if isinstance(value, list):
        return [compact(v) for v in value]
    return value


def text_eval(e, samples=False):
    rows = e["generated_samples"]
    assert len(rows) == e["examples"]
    assert sum(r["exact"] for r in rows) == e["correct"]
    assert sum(r["ended_with_eos"] for r in rows) == e["eos_count"]
    assert abs(e["nll_sum"] / e["supervised_tokens"] - e["answer_nll"]) < 1e-12
    for r in rows:
        ids = r["generated_ids"]
        raw = ids[:ids.index(2)] if 2 in ids else ids
        assert raw == [b + 8 for b in r["expected"].encode()] if r["exact"] else raw != [b + 8 for b in r["expected"].encode()]
    keep = {k: v for k, v in e.items() if k != "generated_samples"}
    keep["invalid_control_tokens_recounted"] = sum(sum(i < 8 for i in r["generated_ids"] if i != 2) for r in rows)
    if samples:
        keep["generated_samples"] = rows
    return keep


def modal_eval(e, samples=False):
    rows = e["samples"]
    assert len(rows) == e["examples"]
    assert sum(r["exact_match"] for r in rows) == e["correct"]
    assert sum(r["eos"] for r in rows) / len(rows) == e["eos_rate"]
    assert sum(r["invalid_special_tokens"] for r in rows) == e["invalid_special_tokens"]
    keep = {k: v for k, v in e.items() if k != "samples"}
    if samples:
        keep["samples"] = rows
    return keep
