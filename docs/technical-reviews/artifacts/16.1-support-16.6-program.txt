sha = lambda raw: hashlib.sha256(raw).hexdigest()

def means(groups):
    groups = [[Fraction(x) for x in g] for g in groups]
    total_count = sum(map(len, groups))
    overall = sum(sum(g) for g in groups) / total_count
    equal_groups = sum(sum(g) / len(g) for g in groups) / len(groups)
    common_denominator = sum(sum(g) / total_count for g in groups)
    return {'group_counts': list(map(len, groups)), 'total_count': total_count,
            'overall_mean_exact': str(overall), 'average_of_group_means_exact': str(equal_groups),
            'sum_over_common_denominator_exact': str(common_denominator),
            'overall_matches_average_of_group_means': overall == equal_groups,
            'overall_matches_common_denominator': overall == common_denominator}

cases = {
    'unequal_counts_all_losses_one': means([[1], [1] * 10]),
    'unequal_counts_changed_losses': means([[1], [2] * 10]),
    'equal_counts_different_losses': means([[1, 1], [2, 2]]),
}
assert cases['unequal_counts_all_losses_one']['overall_mean_exact'] == '1'
assert cases['unequal_counts_all_losses_one']['average_of_group_means_exact'] == '1'
assert cases['unequal_counts_changed_losses']['overall_mean_exact'] == '21/11'
assert cases['unequal_counts_changed_losses']['average_of_group_means_exact'] == '3/2'
assert all(c['overall_matches_common_denominator'] for c in cases.values())
assert cases['equal_counts_different_losses']['overall_matches_average_of_group_means']

# 新增自有短CPU反例：在w=0每個loss都是1，並不推出兩種目標的梯度相同。
# 第一組1個(w+1)^2，第二組10個(2w+1)^2；沒有參數更新或訓練排程。
def losses(w):
    return (w + 1).square().reshape(1), (2 * w + 1).square().expand(10)

gradient_facts = {}
for mode in ('whole', 'common_denominator', 'equal_group_means'):
    w = torch.tensor(0.0, dtype=torch.float64, device='cpu', requires_grad=True)
    first, second = losses(w)
    if mode == 'whole':
        objective = torch.cat([first, second]).mean()
        objective.backward()
    elif mode == 'common_denominator':
        objective = (first.sum() + second.sum()) / 11
        (first.sum() / 11).backward()
        (second.sum() / 11).backward()
    else:
        objective = (first.mean() + second.mean()) / 2
        objective.backward()
    gradient_facts[mode] = {'initial_losses_all_one': True, 'objective': objective.item(), 'gradient': w.grad.item()}
assert abs(gradient_facts['whole']['gradient'] - 42 / 11) < 1e-12
assert abs(gradient_facts['common_denominator']['gradient'] - 42 / 11) < 1e-12
assert gradient_facts['equal_group_means']['gradient'] == 3

def section(raw, lesson):
    hs = list(re.finditer(rb'(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+', raw))
    pos = next(i for i, h in enumerate(hs) if h[1].decode() == lesson)
    return raw[hs[pos].start():hs[pos + 1].start() if pos + 1 < len(hs) else len(raw)]
