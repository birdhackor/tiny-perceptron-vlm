import ast
import hashlib
import json
import platform
from pathlib import Path

import torch

root = Path.cwd()
sha = lambda path: hashlib.sha256(Path(path).read_bytes()).hexdigest()
source_path = root / "scripts/course_experiments/applications.py"
source = source_path.read_text()
result_path = root / "docs/course-experiments/results/reasoning.json"
original = json.loads(result_path.read_text())
assert sha(source_path) == original["code_sha256"]["scripts/course_experiments/applications.py"]
assert sha(root / "scripts/course_experiments/common.py") == original["code_sha256"]["scripts/course_experiments/common.py"]
assert original["revision"] == "910aebc6419c9fc6217279a27fde9851c5cfad30"
assert original["evidence_status"] == "complete_run" and original["status"] == "completed"
assert original["step_scale"] == 1.0 and original["seed"] == 42
tree = ast.parse(source)
functions = {n.name: ast.get_source_segment(source, n) for n in tree.body if isinstance(n, (ast.FunctionDef, ast.ClassDef))}
raw = functions["_reinforce"]
assert "actions = distribution.sample()" in raw
assert "advantage = reward - baseline" in raw
assert raw.index("advantage = reward - baseline") < raw.index("baseline = 0.95 * baseline")
assert "advantage.detach() * distribution.log_prob(actions)" in raw
assert "0.005 * distribution.entropy().mean()" in raw
assert "steps=1200" in raw and "lr=0.003" in raw
assert "k=64" in raw
assert "nn.Linear(18, 48)" in functions["_FinitePolicy"] and "nn.Linear(48, 17)" in functions["_FinitePolicy"]
assert "re.fullmatch" in functions["_policy_reward"]
branches = {}
for name in ("strict", "weak_proxy"):
    training = original["results"]["reinforce"][name]["training"]
    assert training["steps"] == training["planned_steps"] == 1200
    assert training["step_scale"] == 1.0
    assert training["effective_tokens"] == 0
    assert training["sampled_actions"] == 1200 * 64 == 76800
    assert training["history"][-1]["step"] == 1200
    checkpoint = next(a for a in original["artifacts"] if a["path"] == training["checkpoint"])
    branches[name] = {"training": {k: training[k] for k in ("steps", "planned_steps", "step_scale", "sampled_actions", "effective_tokens", "checkpoint")}, "original_checkpoint_record": checkpoint}

    "formal_result": {"path": "docs/course-experiments/results/reasoning.json", "sha256": sha(result_path), "revision": original["revision"], "evidence_status": original["evidence_status"], "device": original["device"], "seed": original["seed"], "scope": "僅T.7新增C.7小策略訓練存在性及方法來源，沒有重審C.7品質/速度/全章。"},
