import json
from tiny_perceptron.retrieval import retrieve

docs = [
    {'id': 'd1', 'text': '貓喜歡曬太陽'},
    {'id': 'd2', 'text': '書店地址是青街8號'},
]
hits = retrieve('書店地址', docs, k=1)
assert hits == [docs[1]]
assert hits[0] is docs[1]
print(json.dumps({'hits': hits, 'same_document_object': hits[0] is docs[1], 'keys_retained': sorted(hits[0])}, ensure_ascii=False, indent=2))
