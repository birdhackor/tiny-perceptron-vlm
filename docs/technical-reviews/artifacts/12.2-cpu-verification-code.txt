def sha(value):
    return hashlib.sha256(value).hexdigest()

def save(name, value):
    (ART / name).write_text(json.dumps(value, ensure_ascii=False, indent=2) + '\n')

def chunks(value):
    assert value[:4] == b'RIFF' and value[8:12] == b'WAVE'
    offset = 12
    result = {}
    while offset + 8 <= len(value):
        size = struct.unpack_from('<I', value, offset + 4)[0]
        result[value[offset:offset+4].decode()] = (offset + 8, size)
        offset += 8 + size + size % 2
    return result

def peak(w):
    power = torch.stft(w, 400, 160, window=torch.hann_window(400), return_complex=True).abs().square()
    return power, int(power.mean(-1).argmax())
