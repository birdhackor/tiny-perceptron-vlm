import sys, json, itertools, hashlib
from pathlib import Path
import torch
from tiny_perceptron.data import split_documents
assert torch.version.cuda is None
assert not torch.cuda.is_available()
docs = ["貓看狗。", "狗看貓。", "鳥看魚。", "魚看鳥。", "貓看狗。"]
outputs = {}
for name, data in [("original", docs), ("append_duplicate", docs+["貓看狗。"]), ("append_new", docs+["鳥看貓。"])]:
    parts=split_documents(data, seed=42)
    sets={k:set(v) for k,v in parts.items()}
    overlaps={a+"/"+b:sorted(sets[a]&sets[b]) for a,b in itertools.combinations(parts,2)}
    count=sum(map(len,parts.values()))
    assert all(not x for x in overlaps.values())
    assert count==len(set(data))
    assert parts==split_documents(data,seed=42)
    outputs[name]={"input_count":len(data),"unique_count":len(set(data)),"parts":parts,"counts":{k:len(v) for k,v in parts.items()},"total":count,"pairwise_intersections":overlaps,"same_seed_repeat_equal":True,"items_pairs":[[k,v] for k,v in parts.items()]}
assert outputs["original"]["counts"]=={"train":2,"validation":1,"test":1}
assert outputs["append_duplicate"]["parts"]==outputs["original"]["parts"]
assert outputs["append_duplicate"]["total"]==4
assert outputs["append_new"]["total"]==5
raw="貓看狗。狗看貓。"
a,b=raw[:4],raw[1:5]
assert len(a)==len(b)==4
assert a[1:]==b[:3]=="看狗。"
outputs["overlapping_windows"]={"source":raw,"source_length":len(raw),"first":a,"second":b,"overlap":a[1:],"overlap_length":len(a[1:]),"first_source_positions":[0,1,2,3],"second_source_positions":[1,2,3,4]}
outputs["set_semantics"]={"deduplicated":sorted(set(["貓看狗。","貓看狗。"])),"empty_intersection":sorted(set(["貓看狗。"])&set(["狗看貓。"])),"not_empty_intersection":not(set(["貓看狗。"])&set(["狗看貓。"])),"exact_paraphrase_intersection":sorted(set(["貓望著狗。"])&set(["貓看狗。"]))}
try:
    split_documents(["甲","乙","甲"],seed=42)
except ValueError as exc:
    outputs["fewer_than_three_unique"]={"type":type(exc).__name__,"message":str(exc)}
else:
    raise AssertionError("expected documented small-input guard")
text="貓看狗，狗看貓。"
chars=sorted(set(text))
to_id={char:i for i,char in enumerate(chars)}
ids=[to_id[char] for char in text]
restored="".join(chars[i] for i in ids)
assert restored==text
assert chars==["。","狗","看","貓","，"]
assert ids==[3,2,1,4,1,2,3,0]
outputs["prerequisite_1.1_encoding"]={"text":text,"chars":chars,"ids":ids,"restored":restored,"assert_restored":True}
print(json.dumps({"command":"OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 HF_HUB_OFFLINE=1 HF_DATASETS_OFFLINE=1 TRANSFORMERS_OFFLINE=1 PYTHONDONTWRITEBYTECODE=1 .venv/bin/python docs/technical-reviews/artifacts/1.2-exercises.txt", "environment":{"python":sys.version,"torch":str(torch.__version__),"device":"cpu (no CUDA build)","OMP_NUM_THREADS":"1","MKL_NUM_THREADS":"1"},"repository_source":{"path":"tiny_perceptron/data.py","sha256":hashlib.sha256(Path("tiny_perceptron/data.py").read_bytes()).hexdigest()},"result":"all assertions passed; no training, network or GPU work", "outputs":outputs},ensure_ascii=False,indent=2))
