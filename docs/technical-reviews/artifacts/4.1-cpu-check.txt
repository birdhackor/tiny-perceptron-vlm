def section(path, lesson):
    raw = (ROOT / path).read_bytes()
    heads = list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+", raw))
    i = next(i for i, h in enumerate(heads) if h[1].decode() == lesson)
    end = heads[i + 1].start() if i + 1 < len(heads) else len(raw)
    return raw[:heads[0].start()], raw[heads[i].start():end]

def code_of(raw):
    return re.search(rb"```python\n(.*?)```", raw, re.S)[1].decode()

def run(name, code, expected_error=None):
    ns = {}
    buf = io.StringIO()
    error = None
    with contextlib.redirect_stdout(buf):
        try:
            exec(compile(code, name, "exec"), ns)
        except Exception as exc:
            error = {"type": type(exc).__name__, "message": str(exc)}
    if expected_error is None:
        assert error is None, error
    else:
        assert error and error["type"] == expected_error, error
    return ns, {"name": name, "stdout": buf.getvalue(), "error": error}
