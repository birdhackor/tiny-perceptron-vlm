import contextlib
import io
import json
import os
import platform
import sys
from pathlib import Path
import torch

path = Path(sys.argv[1])
text = path.read_text(encoding="utf-8")
stdout = io.StringIO()
with contextlib.redirect_stdout(stdout):
    namespace = {}
    exec(compile(text, str(path), "exec"), namespace)
expected = [0.75, 1.0]
observed = [row["correct"] / row["total"] for row in namespace["runs"]]
assert observed == expected
assert [row["student_params"] for row in namespace["runs"]] == [6104, 6104]
seconds = [row["student_seconds"] + row["teacher_seconds"] for row in namespace["runs"]]
assert seconds == ([10, 10] if "exercise" in path.name else [10, 50])
print(json.dumps({"source": str(path), "stdout": stdout.getvalue(), "accuracy": observed, "total_seconds": seconds, "environment": {"python": platform.python_version(), "torch": str(torch.__version__), "torch_git_version": str(torch.version.git_version), "device": "cpu", "OMP_NUM_THREADS": os.environ.get("OMP_NUM_THREADS", ""), "MKL_NUM_THREADS": os.environ.get("MKL_NUM_THREADS", ""), "threads": str(torch.get_num_threads())}, "scope": "原文人工記帳例／僅改教師秒數的練習；未建立或訓練模型。", "exit_code": 0}, ensure_ascii=False, indent=2))
