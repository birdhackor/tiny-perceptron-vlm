def raw_section(path, identifier):
    raw = (ROOT / path).read_bytes()
    start = re.search(rb"^## " + re.escape(identifier.encode()) + rb"(?:\s|\b)", raw, re.M)
    following = re.search(rb"^## ", raw[start.end():], re.M)
    end = start.end() + following.start() if following else len(raw)
    return raw[start.start():end]

def example(path, identifier):
    raw = raw_section(path, identifier)
    fences = re.findall(r"```python\n(.*?)```", raw.decode(), re.S)
    return raw, "\n\n".join(fences)
