import json, torch
import torch.nn.functional as F
torch.manual_seed(0)
q,k,v=[torch.randn(1,1,4,3) for _ in range(3)]
allowed=torch.ones(4,4,dtype=torch.bool).tril()[None,None]
scores=q@k.transpose(-2,-1)/(3**0.5)
weights=scores.masked_fill(~allowed,float("-inf")).softmax(-1)
manual=weights@v
out=F.scaled_dot_product_attention(q,k,v,attn_mask=allowed,dropout_p=0.0)
print(json.dumps({"qkv_shape":list(q.shape),"k_transpose_shape":list(k.transpose(-2,-1).shape),"scores_shape":list(scores.shape),"weights_shape":list(weights.shape),"mask_shape":list(allowed.shape),"output_shape":list(out.shape),"allowed_rows":allowed[0,0].tolist(),"weights_rows":weights[0,0].tolist(),"forbidden_weights_exact_zero":bool((weights[~allowed]==0).all()),"allclose":bool(torch.allclose(manual,out,atol=1e-6)),"double_scaled_query_max_error":float((manual-F.scaled_dot_product_attention(q/(3**0.5),k,v,attn_mask=allowed,dropout_p=0.0)).abs().max()),"dtype_bits_and_bytes":{str(d):{"bits":torch.finfo(d).bits,"bytes":torch.tensor(1,dtype=d).element_size()} for d in [torch.float32,torch.float16,torch.bfloat16]}},ensure_ascii=False))
