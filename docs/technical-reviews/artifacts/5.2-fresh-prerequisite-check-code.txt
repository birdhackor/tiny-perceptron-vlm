import hashlib
import json
import sys
from pathlib import Path
import torch
from scripts.prepare_data import generate_records
from scripts.course_experiments.common import split_records, text_examples
x=json.loads(Path("docs/course-experiments/results/text_foundation.json").read_bytes())
r=x["results"]
parts=split_records(generate_records("toy-text"), seed=x["seed"])
data={}
for name,records in parts.items():
    raw="".join(json.dumps(row,ensure_ascii=False)+"\n" for row in records).encode()
    digest=hashlib.sha256(raw).hexdigest()
    assert digest==r["data"][name]["sha256"]
    examples=text_examples(records,mode="text",max_length=128)
    count=sum(int((label!=-100).sum()) for _,label in examples)
    data[name]={"records":len(records),"effective_tokens_for_full_split_evaluation":count,"sha256":digest}
assert data["validation"]["effective_tokens_for_full_split_evaluation"]==35
assert data["test"]["effective_tokens_for_full_split_evaluation"]==70
assert len(parts["train"])+len(parts["validation"])+len(parts["test"])==12
print(json.dumps({"environment":{"python":sys.version,"torch":torch.__version__,"device":"cpu","threads":str(torch.get_num_threads())},"scope":"親核所讀5.1與4.7前置的固定資料分母；只在記憶體重建12份規則短文，不訓練、不下載、不重跑正式實驗。","formal_result_sha256":hashlib.sha256(Path("docs/course-experiments/results/text_foundation.json").read_bytes()).hexdigest(),"fixed_run":{"revision":x["revision"],"seed":x["seed"],"gpu":x["gpu"],"torch_version":x["torch_version"],"model_width":64,"model_layers":2,"max_length":128,"optimizer":"AdamW","learning_rate":0.003,"batch_size":16,"steps":r["training"]["steps"],"sampled_training_effective_tokens":r["training"]["effective_tokens"],"initial_train_nll":r["training"]["initial_loss"],"final_train_nll":r["training"]["final_loss"],"validation_nll_sum":r["after"]["validation"]["nll_sum"],"validation_effective_tokens":r["after"]["validation"]["effective_tokens"],"validation_nll":r["after"]["validation"]["nll"],"causal_probe_before":[1,10,11,12,13],"causal_probe_after":[1,10,11,12,14],"causal_probe_compared_positions":4,"causal_max_difference":r["causal_max_difference"]},"reconstructed_data":data},ensure_ascii=False,indent=2))
