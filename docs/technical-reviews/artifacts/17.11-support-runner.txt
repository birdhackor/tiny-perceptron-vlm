import sys, os, json, pathlib, hashlib, torch
sys.path.insert(0, str(pathlib.Path.cwd()))
p = pathlib.Path(sys.argv[1])
print("ENVIRONMENT", json.dumps({"python": sys.version, "torch": torch.__version__, "torch_git": torch.version.git_version, "device": "cpu", "cuda_available": torch.cuda.is_available(), "omp_threads": os.environ.get("OMP_NUM_THREADS"), "mkl_threads": os.environ.get("MKL_NUM_THREADS"), "torch_threads": torch.get_num_threads(), "code_sha256": hashlib.sha256(p.read_bytes()).hexdigest()}, ensure_ascii=False))
def guard(event, args):
    if event in ("socket.connect", "socket.getaddrinfo", "subprocess.Popen", "os.system"):
        raise RuntimeError("Offline bounded CPU guard: " + event)
sys.addaudithook(guard)
exec(compile(p.read_text(), str(p), "exec"), {"__name__": "__main__"})
