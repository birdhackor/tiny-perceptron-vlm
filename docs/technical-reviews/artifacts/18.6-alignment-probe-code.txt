import json
import platform
import sys
import time
from pathlib import Path
from types import SimpleNamespace
import torch
sys.path.insert(0, str(Path.cwd()))
from tiny_perceptron.data import ByteTokenizer, IGNORE, SPECIALS, pad_batch, render_chat
from tiny_perceptron.alignment import distillation_kl
from scripts.train import prepare_examples

base = Path('docs/technical-reviews/artifacts')
namespace = {'torch': torch, 'time': time, 'ByteTokenizer': ByteTokenizer, 'IGNORE': IGNORE, 'pad_batch': pad_batch, 'render_chat': render_chat}
exec(compile((base / '18.6-formal-functions-code.txt').read_text(), '18.6-formal-functions-code.txt', 'exec'), namespace)

class PrefixProbe(torch.nn.Module):
    def __init__(self):
        super().__init__()
        self.config = SimpleNamespace(max_length=128)
    def forward(self, x, valid=None):
        # 決定性前文訊號，不含可訓練參數；每個候選是同一列的明確函數。
        prefix = x.cumsum(-1).float().remainder(19) / 19
        candidates = torch.arange(264).float() / 264
        return {'logits': prefix[..., None] * candidates}

records = [{'question': 'a?', 'answer': 'x'}, {'question': 'bb?', 'answer': 'yz'}]
teacher = PrefixProbe()
cache, seconds = namespace['_cache_text'](teacher, records, 'cpu')
indices = [1, 0, 1]
examples = namespace['_examples'](records, 128)
x, y, valid = pad_batch([examples[i] for i in indices])
logits = teacher(x, valid=valid)['logits']
selected = logits[y != IGNORE][None]
targets = torch.cat([cache[i] for i in indices])[None]
labels = y[y != IGNORE][None]
assert torch.equal(selected, targets)
assert len(cache) == 2 and [v.shape for v in cache] == [torch.Size([2,264]), torch.Size([3,264])]
assert selected.shape == (1,8,264)
loss = distillation_kl(selected, targets, labels, temperature=2)
assert abs(loss.item()) < 1e-6
mismatch_error = None
try:
    distillation_kl(selected, targets[:, :-1], labels, temperature=2)
except ValueError as e:
    mismatch_error = str(e)
assert mismatch_error
hard_record = {'question': 'a?', 'answer': 'x', '_hard_ids': [ByteTokenizer().encode('y')[0], 2], '_hard_eos': True}
_, hard_y = namespace['_example'](hard_record,128)
assert hard_y[hard_y != IGNORE].tolist() == hard_record['_hard_ids']
entry_examples = prepare_examples(SimpleNamespace(task="distill", data=None), ByteTokenizer())
assert len(entry_examples) == 16
assert entry_examples[0][1][entry_examples[0][1] != IGNORE].tolist() == ByteTokenizer().encode("0") + [2]
byte1, byte2 = ByteTokenizer(), ByteTokenizer()
assert byte1.state() == byte2.state()
assert byte1.vocab_size == 256 + len(SPECIALS) == 264
assert byte1.encode('小貓追') == byte2.encode('小貓追')
wrong_p = torch.tensor([0.8,0.2]); wrong_q = torch.tensor([0.2,0.8])
wrong_only_vocab_order = [['貓','狗'].index(t) for t in ['貓','狗']]
assert wrong_only_vocab_order == [0,1]
assert round((wrong_p-wrong_q[wrong_only_vocab_order]).abs().mean().item(),4)==0.6
item_error = None
try:
    wrong_q.item()
except RuntimeError as e:
    item_error = str(e)
assert item_error
