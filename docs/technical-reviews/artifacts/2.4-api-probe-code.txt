import torch
from torch import nn
from torch.nn import functional as F

a = nn.Linear(1, 2, bias=False)
b = nn.Linear(2, 1, bias=False)
with torch.no_grad():
    a.weight.copy_(torch.tensor([[1.0], [-1.0]]))
    b.weight.copy_(torch.tensor([[1.0, 1.0]]))
x = torch.tensor([[-2.0], [0.0], [2.0]])
hidden = a(x)
linear = b(hidden)
curved = b(F.relu(hidden))
combined = x @ (b.weight @ a.weight).T
print(hidden)
print("純線性", linear.squeeze(-1))
print("加ReLU", curved.squeeze(-1))
assert torch.allclose(linear, combined)

import json, sys, contextlib, io, math
print("additional",json.dumps({"x":x.tolist(),"hidden":hidden.tolist(),"linear":linear.tolist(),"curved":curved.tolist(),"combined_weight":(b.weight@a.weight).tolist(),"allclose":torch.allclose(linear,combined),"bias_none":[a.bias is None,b.bias is None],"dtype":str(x.dtype),"device":str(x.device),"grad_enabled_after":torch.is_grad_enabled(),"weights_require_grad":[a.weight.requires_grad,b.weight.requires_grad],"gradients_none":[a.weight.grad is None,b.weight.grad is None],"hidden_grad_fn":type(hidden.grad_fn).__name__,"squeezed_grad_fn":type(linear.squeeze(-1).grad_fn).__name__,"original_shape":list(linear.shape),"squeezed_shape":list(linear.squeeze(-1).shape),"signed_zero_equals_zero":bool(torch.tensor(-0.0)==torch.tensor(0.0)),"python_optimization":sys.flags.optimize},ensure_ascii=False))

with torch.no_grad():
    grad_disabled_inside = not torch.is_grad_enabled()
    copied = a.weight.copy_(torch.tensor([[1.0], [-1.0]]))
assert grad_disabled_inside and torch.is_grad_enabled()
assert copied is a.weight
try:
    a.weight.copy_(torch.tensor([[1.0], [-1.0]]))
except RuntimeError as exc:
    print("copy_without_no_grad", type(exc).__name__, str(exc))
else:
    raise AssertionError("leaf parameter copy should fail in default grad mode")
try:
    assert torch.allclose(linear, curved)
except AssertionError:
    print("assert_mismatch", "AssertionError")
else:
    raise AssertionError("intentional differing tensors unexpectedly close")
print("relu_hidden", F.relu(hidden).tolist())
assert torch.allclose(F.relu(hidden), torch.tensor([[0.0,2.0],[0.0,0.0],[2.0,0.0]]))
assert math.copysign(1.0, hidden[1,1].item()) == -1.0
assert a.weight.grad is None and b.weight.grad is None
