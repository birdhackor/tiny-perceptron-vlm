import json, ast, torch
import torch.nn.functional as F
from pathlib import Path
from torch.nn.attention import SDPBackend, sdpa_kernel

torch.manual_seed(0)
q,k,v=[torch.randn(1,1,4,3) for _ in range(3)]
allowed=torch.ones(4,4,dtype=torch.bool).tril()[None,None]
a=F.scaled_dot_product_attention(q,k,v,attn_mask=allowed,dropout_p=0.0)
b=F.scaled_dot_product_attention(q,k,v,attn_mask=allowed,dropout_p=0.0,scale=3**-0.5)
c=F.scaled_dot_product_attention(q,k,v,dropout_p=0.0,is_causal=True)
with sdpa_kernel(backends=[SDPBackend.MATH]):
    math_out=F.scaled_dot_product_attention(q,k,v,attn_mask=allowed,dropout_p=0.0)
class DirectSDPA(torch.nn.Module):
    def forward(self, p):
        return F.scaled_dot_product_attention(q,k,v,attn_mask=allowed,dropout_p=p)
mod=DirectSDPA().eval()
torch.manual_seed(42)
drop=mod(0.5)
layer=torch.nn.Linear(2,1).eval()
x=torch.ones(1,2)
y=layer(x)
with torch.no_grad():
    y_ng=layer(x)
torch.manual_seed(1)
seed1=[torch.randn(1,1,4,3) for _ in range(3)]
torch.manual_seed(1)
seed1_replay=[torch.randn(1,1,4,3) for _ in range(3)]
text=Path("scripts/course_experiments/architecture.py").read_text()
lines=text.splitlines(keepends=True)
node=next(n for n in ast.parse(text).body if isinstance(n,ast.FunctionDef) and n.name=="_flash_manual_attention")
namespace={"torch":torch,"F":F}
exec("".join(lines[node.lineno-1:node.end_lineno]),namespace)
low=[]
for dtype in [torch.float16,torch.bfloat16]:
    inputs=[t.to(dtype).requires_grad_() for t in [q,k,v]]
    prod=inputs[0]@inputs[1].transpose(-2,-1)
    score=prod.float()*(3**-0.5)
    prob=score.masked_fill(~allowed,float("-inf")).softmax(-1)
    out=namespace["_flash_manual_attention"](*inputs)
    grads=torch.autograd.grad(out,inputs,torch.ones_like(out)*0.125)
    low.append({"input_dtype":str(dtype),"QK_output_dtype":str(prod.dtype),"score_dtype":str(score.dtype),"softmax_dtype":str(prob.dtype),"PV_output_dtype":str(out.dtype),"finite_qkv_gradients":all(bool(torch.isfinite(g).all()) for g in grads)})
result={"scope":"CPU短API／dtype語意核查，沒有執行CUDA或替代GPU速度／品質實報", "default_explicit_scale_max_error":float((a-b).abs().max()),"boolmask_causal_max_error":float((a-c).abs().max()),"CPU_math_vs_default_max_error":float((a-math_out).abs().max()),"eval_module_training":mod.training,"SDPA_eval_p05_vs_p0_max_difference":float((drop-a).abs().max()),"linear_eval_training":layer.training,"linear_eval_output_requires_grad":y.requires_grad,"no_grad_output_requires_grad":y_ng.requires_grad,"original_example_output_requires_grad":a.requires_grad,"seed1_qkv_repeat_bitwise_equal":all(torch.equal(u,w) for u,w in zip(seed1,seed1_replay)),"low_precision_manual_small_cpu":low}
assert torch.allclose(a,b,atol=1e-6) and torch.allclose(a,c,atol=1e-6)
assert not mod.training and result["SDPA_eval_p05_vs_p0_max_difference"]>0
assert y.requires_grad and not y_ng.requires_grad and not a.requires_grad
assert result["seed1_qkv_repeat_bitwise_equal"]
print(json.dumps(result,ensure_ascii=False))
