def section(path,ident):
    data=(root/path).read_bytes()
    m=re.search(rb'(?m)^## '+re.escape(ident.encode())+rb' [^\r\n]+',data)
    next_m=re.search(rb'(?m)^## ',data[m.end():])
    end=m.end()+next_m.start() if next_m else len(data)
    return data[m.start():end]
