import json
import os
import platform
import sys
from pathlib import Path

import torch
from torch import nn


def offline_guard(event, arguments):
    if event.startswith("socket.") or event in {
        "subprocess.Popen", "os.system", "os.exec", "os.posix_spawn"
    }:
        raise RuntimeError("bounded offline CPU review prohibits network and child processes")


sys.addaudithook(offline_guard)
assert torch.version.cuda is None
torch.set_num_threads(1)
torch.manual_seed(17)  # reviewer harness only, fixes the random initialization
layer = nn.Linear(2, 1)
parameter_ids_before = [id(parameter) for parameter in layer.parameters()]
for parameter in layer.parameters():
    parameter.requires_grad_(False)
parameter_ids_after = [id(parameter) for parameter in layer.parameters()]
x = torch.ones(1, 2, requires_grad=True)
layer.eval()
weight_before = layer.weight.detach().clone()
bias_before = layer.bias.detach().clone()
output = layer(x)
output.sum().backward()
assert output.requires_grad
assert output.grad_fn is not None
assert torch.equal(x.grad, layer.weight)
assert tuple(x.grad.shape) == (1, 2)
assert not any(parameter.requires_grad for parameter in layer.parameters())
assert all(parameter.grad is None for parameter in layer.parameters())
assert parameter_ids_before == parameter_ids_after
assert torch.equal(weight_before, layer.weight)
assert torch.equal(bias_before, layer.bias)
