usage: train.py [-h] [--task {text,sft,dpo,distill,vision,audio,joint}]
                [--train] [--data DATA] [--checkpoint CHECKPOINT] [--resume]
                [--teacher TEACHER] [--freeze {none,projector,partial}]
                [--vision-encoder VISION_ENCODER]
                [--audio-encoder AUDIO_ENCODER] [--steps STEPS]
                [--stop-after STOP_AFTER] [--save-every SAVE_EVERY]
                [--batch-size BATCH_SIZE] [--width WIDTH] [--layers LAYERS]
                [--heads HEADS] [--max-length MAX_LENGTH] [--experts EXPERTS]
                [--top-k TOP_K] [--rotary] [--norm {layer,rms}]
                [--activation {gelu,relu2,swiglu}] [--backend {manual,sdpa}]
                [--tied] [--kv-heads KV_HEADS] [--device DEVICE] [--lr LR]
                [--seed SEED] [--beta BETA] [--alpha ALPHA]
                [--temperature TEMPERATURE] [--output OUTPUT]

GPU 訓練入口；不加 --train 時只做 forward/backward，不更新權重。

options:
  -h, --help            show this help message and exit
  --task {text,sft,dpo,distill,vision,audio,joint}
  --train               明確開啟權重更新；預設只檢查梯度
  --data DATA           text/messages 或 chosen/rejected JSONL；預設使用課程合成規則
  --checkpoint CHECKPOINT
                        載入文字或完整多模態模型；--resume 同時恢復 optimizer/RNG
  --resume
  --teacher TEACHER
  --freeze {none,projector,partial}
  --vision-encoder VISION_ENCODER
  --audio-encoder AUDIO_ENCODER
  --steps STEPS
  --stop-after STOP_AFTER
                        提前在這個絕對步數停止；不改變 --steps 的 LR 排程
  --save-every SAVE_EVERY
                        完整 checkpoint 的週期存檔間隔
  --batch-size BATCH_SIZE
  --width WIDTH
  --layers LAYERS
  --heads HEADS
  --max-length MAX_LENGTH
  --experts EXPERTS
  --top-k TOP_K
  --rotary
