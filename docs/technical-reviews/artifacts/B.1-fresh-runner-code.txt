import contextlib
import hashlib
import io
import json
import os
import platform
import subprocess
import sys
from pathlib import Path

import torch

art = Path('docs/technical-reviews/artifacts')
environment = {'python': platform.python_version(), 'torch': str(torch.__version__),
               'device': 'cpu', 'OMP_NUM_THREADS': os.environ['OMP_NUM_THREADS'],
               'MKL_NUM_THREADS': os.environ['MKL_NUM_THREADS'], 'torch_threads': str(torch.get_num_threads())}
records = []
for name in ['original', 'exercise-question', 'exercise-aligned', 'intro-B.2', 'intro-B.3', 'intro-B.4']:
    path = art / ('B.1-fresh-' + name + '-code.txt')
    command = 'OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 .venv/bin/python < ' + str(path)
    result = subprocess.run([sys.executable], input=path.read_bytes(), capture_output=True, timeout=30,
                            env={**os.environ, 'OMP_NUM_THREADS': '1', 'MKL_NUM_THREADS': '1'})
    namespace = {}
    with contextlib.redirect_stdout(io.StringIO()):
        exec(compile(path.read_bytes(), str(path), 'exec'), namespace)
    row = {'case': name, 'command': command, 'exit_code': result.returncode,
           'code_sha256': hashlib.sha256(path.read_bytes()).hexdigest(),
           'stdout': result.stdout.decode(), 'stderr': result.stderr.decode(), 'environment': environment}
    if name in ['original', 'exercise-question', 'exercise-aligned']:
        training = namespace['training_example']
        requested = namespace['request']
        row.update({'question': training['user'], 'request': requested,
                    'json_roundtrip_equal': json.loads(namespace['text']) == requested,
                    'model_loaded': False, 'parameter_updates': 0, 'tool_calls': 0})
    records.append(row)
    (art / ('B.1-fresh-' + name + '-execution.json')).write_text(
        json.dumps(row, ensure_ascii=False, indent=2) + '\n', encoding='utf-8')
    print(json.dumps(row, ensure_ascii=False))
assert all(row['exit_code'] == 0 for row in records)
