import json, sys, torch
from torch import nn
captures = []
def capture_backend(gm, example_inputs):
    captures.append(str(gm.graph))
    return gm.forward
torch.manual_seed(0)
layer = nn.Linear(4, 3).eval()
compiled = torch.compile(layer, backend=capture_backend, dynamic=False)
before = len(captures)
with torch.no_grad():
    x = torch.ones(2, 4)
    expected, actual = layer(x), compiled(x)
    first = len(captures)
    compiled(x)
    repeated = len(captures)
    compiled(torch.ones(3, 4))
    changed = len(captures)
print(json.dumps({"environment": {"python": sys.version, "torch": str(torch.__version__), "torch_git_version": str(torch.version.git_version), "device": "cpu", "threads": str(torch.get_num_threads())}, "backend_calls_before_first": before, "backend_calls_after_first": first, "after_repeated_same_shape": repeated, "after_changed_shape_dynamic_false": changed, "shape": list(actual.shape), "max_difference": (expected-actual).abs().max().item(), "graphs": captures, "scope": "有界CPU捕捉後仍用gm.forward；驗證延後捕捉與固定shape重編譯，不是Inductor加速測試。"}, ensure_ascii=False, indent=2))
assert (before, first, repeated, changed) == (0, 1, 1, 2)
assert actual.shape == (2, 3)
assert torch.equal(expected, actual)
