from pathlib import Path
import json
import math
import xml.etree.ElementTree as ET
import torch
from torch import nn
from torch.nn import functional as F

a = nn.Linear(1, 2, bias=False)
b = nn.Linear(2, 1, bias=False)
with torch.no_grad():
    a.weight.copy_(torch.tensor([[1.0], [-1.0]]))
    b.weight.copy_(torch.tensor([[1.0, 1.0]]))
    sample_x = torch.tensor([[-2.0], [0.0], [2.0]])
    direct = b(a(sample_x)).squeeze(-1)
    rectified = b(F.relu(a(sample_x))).squeeze(-1)
    dense_x = torch.tensor([[-2.0], [-1.0], [0.0], [1.0], [2.0]])
    dense_direct = b(a(dense_x)).squeeze(-1)
    dense_rectified = b(F.relu(a(dense_x))).squeeze(-1)

assert direct.tolist() == [0.0, 0.0, 0.0]
assert rectified.tolist() == [2.0, 0.0, 2.0]
assert dense_direct.tolist() == [0.0] * 5
assert dense_rectified.tolist() == [2.0, 1.0, 0.0, 1.0, 2.0]
svg = ET.fromstring(Path("course/figures/foundations_nonlinearity.svg").read_bytes())
ns = {"s": "http://www.w3.org/2000/svg"}
circles = svg.findall("s:circle", ns)
assert len(circles) == 6
panels = []
for panel, origin, values in [("left", 205.0, direct.tolist()), ("right", 595.0, rectified.tolist())]:
    selected = circles[:3] if panel == "left" else circles[3:]
    recovered = [[(float(c.attrib["cx"]) - origin) / 52.5, (380.0 - float(c.attrib["cy"])) / 85.0] for c in selected]
    expected = [[x, y] for x, y in zip([-2.0, 0.0, 2.0], values)]
    assert recovered == expected
    panels.append({"panel": panel, "screen_origin_x": origin, "screen_zero_y": 380.0, "x_pixels_per_unit": 52.5, "y_pixels_per_unit": 85.0, "recovered_data_points": recovered})
colored = [p for p in svg.findall("s:path", ns) if p.attrib.get("stroke") in ["#2563b8", "#b6530f"]]
assert [p.attrib["d"] for p in colored] == ["M100 380H310", "M490 210L595 380L700 210"]
left_slope = (0.0 - 2.0) / (0.0 - (-2.0))
right_slope = (2.0 - 0.0) / (2.0 - 0.0)
assert [left_slope, right_slope] == [-1.0, 1.0]
print(json.dumps({"same_weights": {"a": a.weight.tolist(), "b": b.weight.tolist()}, "sample_x": sample_x.squeeze(-1).tolist(), "direct": direct.tolist(), "relu": rectified.tolist(), "dense_x": dense_x.squeeze(-1).tolist(), "dense_direct": dense_direct.tolist(), "dense_relu": dense_rectified.tolist(), "panels": panels, "line_paths": [p.attrib["d"] for p in colored], "data_slopes": [left_slope, right_slope], "note": "XML數值核對補充已實際render+view_image的視覺檢查；兩個軸的像素比例不同，資料斜率仍為±1。"}, ensure_ascii=False))
