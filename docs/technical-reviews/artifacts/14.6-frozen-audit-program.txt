import json
import math
from pathlib import Path
p=json.loads(Path("docs/technical-reviews/artifacts/14.6-frozen-modern-evidence.json").read_text())
v=p["selected_variants"]
assert v["baseline"]["model"]["parameters"] - v["tied"]["model"]["parameters"] == 264*64 == 16896
expected={"baseline": ["5.76007", "2.26331", "2.29163"], "tied": ["43.87647", "2.36276", "2.38422"]}
for name in ["baseline", "tied"]:
    obj=v[name]
    entries=[obj["training"]["initial"],obj["heldout"]["validation"],obj["heldout"]["test"]]
    for label, item, rounded in zip(["initial_train", "validation", "test"],entries,expected[name]):
        quotient=item["nll_sum"] / item["effective_tokens"]
        assert math.isclose(quotient,item["nll"],rel_tol=0,abs_tol=1e-12)
        assert format(quotient,".5f")==rounded
        print(name,label,"sum",item["nll_sum"],"tokens",item["effective_tokens"],"nll",quotient,"rounded",rounded)
    assert obj["training"]["steps"] == obj["training"]["optimizer_updates"] == 240
    assert obj["training"]["effective_tokens"] == 452102
    print(name,"updates",obj["training"]["optimizer_updates"],"sampled_train_tokens",obj["training"]["effective_tokens"])
assert v["tied"]["heldout"]["validation"]["nll"] > v["baseline"]["heldout"]["validation"]["nll"]
assert v["tied"]["heldout"]["test"]["nll"] > v["baseline"]["heldout"]["test"]["nll"]
assert set(v["baseline"]["copied_initial_tables"]) - set(v["tied"]["copied_initial_tables"]) == {"output.weight"}
print("counts",v["baseline"]["model"]["parameters"],v["tied"]["model"]["parameters"],"difference",16896)
print("equal copied table names except output.weight; no formal training was rerun")
