import json
import platform
import torch
from tiny_perceptron.data import ByteTokenizer, render_chat

torch.set_num_threads(1)
tok = ByteTokenizer()
print(json.dumps({"python": platform.python_version(), "torch": torch.__version__, "torch_git_version": torch.version.git_version, "device": "cpu", "threads": torch.get_num_threads()}, ensure_ascii=False))
for text, expected_byte, expected_id in [("Q", 81, 89), ("A", 65, 73), ("B", 66, 74)]:
    raw = list(text.encode("utf-8"))
    ids = tok.encode(text)
    assert raw == [expected_byte] and ids == [expected_id]
    print(json.dumps({"case": "byte與ID算術", "text": text, "utf8_bytes": raw, "offset": 8, "token_ids": ids}, ensure_ascii=False))
for name, question, answer, expected_first, expected_target in [
    ("練習一：Q改QQ", "QQ", "A", 5, 73),
    ("練習二：只把A改B", "QQ", "B", 5, 74),
]:
    x, y = render_chat([{"role": "user", "content": question}, {"role": "assistant", "content": answer}])
    first = (y != -100).nonzero()[0].item()
    print(json.dumps({"case": name, "question": question, "answer": answer, "x": x.tolist(), "y": y.tolist(), "first": first, "input_at_first": x[first].item(), "target_at_first": y[first].item(), "valid_count": (y != -100).sum().item()}, ensure_ascii=False))
    assert first == expected_first
    assert x[first].item() == tok.assistant_id == 4
    assert y[first].item() == expected_target
    assert (y != -100).sum().item() == 2
print("兩項練習全部通過；沒有參數更新。")
