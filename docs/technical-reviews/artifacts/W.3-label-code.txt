import json, os, platform, keyword
B, T, D = 2, 4, 3
value = [B, T, D]
assert value == [2, 4, 3]
assert not any(keyword.iskeyword(name) for name in ["B", "T", "D"])
try:
    eval("[B,T,D]", {}, {})
except NameError as error:
    unbound_error = str(error)
else:
    raise AssertionError("Names must be defined to evaluate a Python list expression")
print(json.dumps({"environment":{"python":platform.python_version(),"device":"cpu; Python only","OMP_NUM_THREADS":os.environ["OMP_NUM_THREADS"],"MKL_NUM_THREADS":os.environ["MKL_NUM_THREADS"]},"ordinary_identifiers":["B","T","D"],"defined_value":value,"unbound_error":unbound_error,"assertions":"passed"}, ensure_ascii=False, indent=2))
