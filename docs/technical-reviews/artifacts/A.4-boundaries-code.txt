import json
from scripts.course_experiments.applications import _grounding

cases = [
    ('第二份改d9', {'id': 'd2', 'text': '書店地址是青街8號'}, {'address': '紅街9號', 'citation': 'd9'}),
    ('否定仍含相同字串', {'id': 'd2', 'text': '地址不是青街8號'}, {'address': '青街8號', 'citation': 'd2'}),
    ('八與8同義表示但不逐字相同', {'id': 'd2', 'text': '書店地址是青街八號'}, {'address': '青街8號', 'citation': 'd2'}),
]
out = []
for name, source, answer in cases:
    valid = answer['citation'] == source['id']
    supports = valid and answer['address'] in source['text']
    out.append({'case': name, 'source': source, 'answer': answer, 'valid': valid, 'substring_support': supports})
assert [(x['valid'], x['substring_support']) for x in out] == [(False, False), (True, True), (True, False)]
formal_cases = []
sample = {'generated': 'D2[D9]', 'invalid_special_tokens': []}
for text in ['K003 address=D2', 'K003 address=D2 is false', 'K003 address=D2 asof=2000', 'K013 address=D2']:
    score = _grounding(sample, [{'id': 'D9', 'text': text}], 'D2[D9]')
    assert score['citation_valid'] is True
    assert score['supported_by_cited_source'] is True
    formal_cases.append({'text': text, 'grounding': score})
print(json.dumps({'literal_example_cases': out, 'formal_field_only_cases': formal_cases}, ensure_ascii=False, indent=2))
