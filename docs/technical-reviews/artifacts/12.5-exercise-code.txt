import contextlib, io, json, os, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parents[3]))
import torch
assert torch.version.cuda is None and not torch.cuda.is_available()
torch.set_default_device("cpu")
torch.set_num_threads(1)
code = 'import torch\nfrom tiny_perceptron.multimodal import tone\n\nwave = tone(880.0, seconds=0.2)\nspectrum = torch.stft(wave, 400, 160, window=torch.hann_window(400), return_complex=True)\npower = spectrum.abs().square()\npeak = power.mean(-1).argmax().item()\nprint("頻率格、時間框", tuple(power.shape))\nprint("最高格編號", peak, "對應Hz", peak * 16000 / 400)\n'
namespace = {"__name__": "__main__"}
stdout = io.StringIO()
with contextlib.redirect_stdout(stdout):
    exec(compile(code, "course/chapters/12.md#12.5:880Hz-exercise", "exec"), namespace)
wave, power = namespace["wave"], namespace["power"]
assert tuple(power.shape) == (201, 21)
assert namespace["peak"] == 22
assert wave.numel() == 3200
result = {
    "exercise": "只把440Hz改880Hz，其他原例設定完全相同",
    "code": code, "stdout": stdout.getvalue(),
    "observed": {"samples": wave.numel(), "shape": list(power.shape), "peak_bin": namespace["peak"], "peak_hz": namespace["peak"]*16000/400, "wave_dtype": str(wave.dtype), "power_dtype": str(power.dtype)},
    "environment": {"python": sys.version, "torch": str(torch.__version__), "torch_git_version": str(torch.version.git_version), "device": str(wave.device), "OMP_NUM_THREADS": os.environ.get("OMP_NUM_THREADS"), "MKL_NUM_THREADS": os.environ.get("MKL_NUM_THREADS")},
    "result": "assertions通過；形狀(201,21)，峰22與880.0Hz",
}
print(json.dumps(result, ensure_ascii=False, indent=2))
