"""Index the independently executed supplement without changing its main review."""

import hashlib
import json
from datetime import datetime, timezone
from pathlib import Path

root = Path.cwd()
out = root / "docs/technical-reviews/artifacts/publication-20.7-ui-ids-supplement"
execution = json.loads((out / "reinspection-execution.json").read_bytes())
audit = json.loads((out / "source-and-evidence-audit.json").read_bytes())
cases = json.loads((out / "case-comparison.json").read_bytes())
checker = json.loads((out / "technical-checker.execution.json").read_bytes())
report = root / "docs/technical-reviews/20.7.json"
assert report.read_bytes() == (out / "prior-main-report.json").read_bytes()
assert checker["exit_code"] == execution["exit_code"] == 0
assert all(item["matches"] for item in audit["registered_hash_checks"])
assert not cases["actual_cpu_failed_or_skipped_phases"]
files = []
for path in sorted(out.iterdir()):
    if not path.is_file() or path.name == "supplement.json":
        continue
    content = path.read_bytes()
    files.append({"path": str(path.relative_to(root)), "bytes": len(content),
                  "sha256": hashlib.sha256(content).hexdigest()})
summary = {
    "schema_version": 1,
    "record_type": "publication_dependency_supplement",
    "lesson_id": "20.7",
    "reviewer_task": execution["reviewer_task"],
    "recorded_at": datetime.now(timezone.utc).isoformat(),
    "scope": execution["scope"],
    "main_report": {
        "path": "docs/technical-reviews/20.7.json",
        "sha256": execution["main_report_sha256"],
        "owner": execution["main_report_original_owner"],
        "unchanged_byte_exact": True,
        "original_full_review_and_GPU_evidence_preserved": True,
    },
    "baseline_revision": audit["baseline_revision"],
    "lesson_source_sha256": audit["lesson_source_sha256"],
    "old_test_sha256": audit["baseline_test_sha256"],
    "current_test_sha256": audit["current_test_sha256"],
    "only_ids_changed": True,
    "source_and_assertions_unchanged_after_removing_ids": True,
    "independently_collected_upload_case_values_equal": True,
    "short_upload_ids": cases["current_ids"],
    "overlong_upload_seconds_unchanged": cases["overlong_audio_seconds"],
    "baseline_copied_file_setup_label_characters": cases["original_overlong_audio_setup_current_test_characters"],
    "current_setup_label_characters": cases["current_overlong_audio_setup_current_test_characters"],
    "baseline_length_note": "Actual original-value pytest collection used a temporary byte-exact file copy, whose filename prefix differs from Windows CI.",
    "actual_CPU_tests_passed": cases["actual_cpu_call_phases_passed"],
    "actual_CPU_failed_or_skipped_phases": [],
    "paired_frozen_context_cases_checked": 12,
    "registered_source_artifact_figure_SHA_pairs_checked": audit["registered_hash_check_count"],
    "technical_checker_lesson_20_7_exit_code": checker["exit_code"],
    "environment": execution["environment"],
    "limits": {
        "new_whole_claims_factual_review": False,
        "new_external_primary_source_verification": False,
        "new_semantic_answer_grading": False,
        "Windows_execution": False,
        "training_or_model_inference": False,
        "GPU_or_paid_compute": False,
        "audio_playback": False,
        "historical_serialized_prompt_replay": False,
    },
    "files": files,
}
(out / "supplement.json").write_text(json.dumps(summary, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")
print(json.dumps({"supplement": str((out / "supplement.json").relative_to(root)),
                  "indexed_files": len(files), "main_report_unchanged": True}))
