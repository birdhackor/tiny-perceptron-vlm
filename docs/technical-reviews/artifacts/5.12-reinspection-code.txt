base = Path("docs/technical-reviews/artifacts")
raw = Path("course/chapters/05.md").read_bytes()
headers = list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+", raw))

def extract(identifier):
    index = next(i for i, h in enumerate(headers) if h[1].decode() == identifier)
    end = headers[index + 1].start() if index + 1 < len(headers) else len(raw)
    return raw[headers[index].start():end]

def digest(content):
    return hashlib.sha256(content).hexdigest()

previous = (base / "5.12-prerequisite-5.11.txt").read_bytes()
current = extract("5.11")
body = extract("5.12")
assert digest(previous) == "d0d51110437dd1bd23e05182f8b38c24210fa5f95648d22f43b86bd9f16b8042"
assert digest(current) == "b48199a023ef77bab327a8ad1c23e4de1606f0060084ddbcf7622082da17bd34"
assert body == (base / "5.12-section.txt").read_bytes()
fence = re.search(rb"```python\n(.*?)```", body, flags=re.S)[1]
assert fence == (base / "5.12-original-code.txt").read_bytes()
assert (base / "5.12-exercise-code.txt").read_bytes() == fence.replace("grams(\"紅色圓形在左邊\")".encode(), "grams(\"紅色圓形在左邊\", n=3)".encode()).replace("grams(\"紅色圓形在右邊\")".encode(), "grams(\"紅色圓形在右邊\", n=3)".encode())

unchanged_evidence = {}
for artifact in report["artifacts"]:
    if artifact["id"] in ("a3", "a6", "a8", "a15"):
        actual = digest(Path(artifact["path"]).read_bytes())
        assert actual == artifact["sha256"]
        unchanged_evidence[artifact["id"]] = {"path": artifact["path"], "sha256": actual, "matches_previous_report": True}
code_identity = {}
for source in report["sources"]:

        actual = digest(Path(source["path"]).read_bytes())
        assert actual == source["sha256"]
        code_identity[source["path"]] = {"sha256": actual, "unchanged": True}
formal = json.loads(Path("docs/course-experiments/results/real_text.json").read_bytes())
asset_identity = {}
for asset in formal["assets"]:
    actual = digest(Path(asset["archive"]).read_bytes())
    assert actual == asset["archive_sha256"]
    asset_identity[asset["id"]] = {"archive": asset["archive"], "sha256": actual, "matches_frozen_report": True}
