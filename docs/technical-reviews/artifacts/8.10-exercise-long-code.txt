import sys, os, json
import torch
assert torch.version.cuda is None
assert not torch.cuda.is_available()
assert os.environ['OMP_NUM_THREADS'] == '1'
assert os.environ['MKL_NUM_THREADS'] == '1'
torch.set_num_threads(1)
torch.set_default_device('cpu')
def guard(event, args):
    if event.startswith('socket.') or event in {'subprocess.Popen', 'os.system', 'os.exec', 'os.posix_spawn'}:
        raise RuntimeError('offline CPU audit guard: ' + event)
sys.addaudithook(guard)
def environment():
    return {'python': sys.version, 'python_executable': sys.executable, 'torch': str(torch.__version__), 'torch_git_version': str(torch.version.git_version), 'device': 'cpu', 'cuda_build': str(torch.version.cuda), 'cuda_available': str(torch.cuda.is_available()), 'OMP_NUM_THREADS': os.environ['OMP_NUM_THREADS'], 'MKL_NUM_THREADS': os.environ['MKL_NUM_THREADS']}
question='用生活比喻解釋2+2'
long_answer='4。左盤放兩顆蘋果，右盤也放兩顆，倒到同一盤後數出四顆'
repeated_answer='4。'*20
# 以下是本獨立technical reviewer親讀文字後明列的對應；程式不會理解中文比喻。
reviewer_annotation={'long':{'answer':4,'groups':[2,2],'merge_explained':True,'groups_correspond_to_question':True},'repeated':{'answer':4,'groups':[],'merge_explained':False,'groups_correspond_to_question':False}}
labels={}
for name,row in reviewer_annotation.items():
    correct=row['answer']==2+2
    analogy=row['groups']==[2,2] and row['merge_explained'] and row['groups_correspond_to_question']
    labels[name]=int(correct and analogy)
assert labels=={'long':1,'repeated':0}
print(json.dumps({'environment':environment(),'exercise':'同題較長生活比喻與單純重複答案','question':question,'long_answer':long_answer,'repeated_answer':repeated_answer,'reviewer_annotation':reviewer_annotation,'labels':labels,'reason_long':'兩盤各兩顆，合併後總數2+2=4；同時符合算對與具體分組合併對應。','reason_repeated':'重複4仍算對，但沒有兩組各兩個與合併的生活對應。','scope':'本獨立代理依明定rubric作技術核對，並真實執行布林合取與算術；沒有宣稱真人同伴已完成讀者測試，也沒有自動判讀語意。'},ensure_ascii=False,indent=2))
