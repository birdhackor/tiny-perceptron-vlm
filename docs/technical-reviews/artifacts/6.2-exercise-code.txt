import sys, platform, importlib.metadata
print("environment",sys.version,platform.machine(),"torch",importlib.metadata.version("torch"),"device=cpu; standard-library computation")
from collections import Counter

corpus = [list("abab"), list("abac"), list("acacacac")]
pairs = Counter()
for row in corpus:
    for left, right in zip(row, row[1:]):
        pairs[(left, right)] += 1
pair = pairs.most_common(1)[0][0]


def merge(row, pair):
    out = []
    i = 0
    while i < len(row):
        if i + 1 < len(row) and tuple(row[i : i + 2]) == pair:
            out.append("".join(pair))
            i += 2
        else:
            out.append(row[i])
            i += 1
    return out


print(pairs)
print("選擇", pair, "結果", [merge(row, pair) for row in corpus])

assert pairs == Counter({("a", "c"): 5, ("a", "b"): 3, ("b", "a"): 2, ("c", "a"): 3})
assert pair == ("a", "c")
result = [merge(row, pair) for row in corpus]
assert result == [["a", "b", "a", "b"], ["a", "b", "ac"], ["ac", "ac", "ac", "ac"]]
assert list(map(len, result)) == [4, 3, 4]
print("exercise counts/selection/results/lengths verified", list(map(len, result)))
