import contextlib, io, json, sys
from pathlib import Path
import torch

def offline_guard(event, args):
    if event.startswith('socket.') or event in ('subprocess.Popen','os.system','os.posix_spawn'):
        raise RuntimeError('offline CPU review: network and child processes forbidden')
sys.addaudithook(offline_guard)
namespace={'__name__':'__main__'}
output=io.StringIO()
code_path=Path('docs/technical-reviews/artifacts/7.5-exercise-code.txt')
with contextlib.redirect_stdout(output):
    exec(compile(code_path.read_bytes(),str(code_path),'exec'),namespace)
x,y,model,logits=(namespace[k] for k in ('x','y','model','logits'))
first=float(logits.grad[0,0].norm())
row=float(model.embedding.weight.grad[x[2]].norm())
assert first==0 and row>0 and x[2].item()==98 and y.tolist()==[-100,-100,-100,-100,73,2]
assert not torch.cuda.is_available() and torch.version.cuda is None
print(json.dumps({'mode':'原例只將user Q換成Z','stdout':output.getvalue(),'x':x.tolist(),'y':y.tolist(),'input_question_id':x[2].item(),'valid_positions':(y!=-100).nonzero().flatten().tolist(),'valid_targets':int((y!=-100).sum()),'first_logits_norm':first,'Z_embedding_norm':row,'logits_shape':list(logits.shape),'embedding_grad_shape':list(model.embedding.weight.grad.shape),'config':model.description()['config'],'environment':{'python':sys.version,'torch':str(torch.__version__),'torch_git_version':str(torch.version.git_version),'device':'cpu','OMP_NUM_THREADS':'1','MKL_NUM_THREADS':'1','network':'audit guard blocks sockets and subprocesses'},'exit_code':0},ensure_ascii=False,indent=2))
