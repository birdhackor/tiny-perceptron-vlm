from pathlib import Path
import hashlib,json,re,platform
sha=lambda b:hashlib.sha256(b).hexdigest()
oldpath=Path("/tmp/root-full-course-source-history/glossary-5335ea60c11591373a78c5f72f2b9d6029e4eb701f3168cd21a5fd7607e11974.md")
old=oldpath.read_bytes()
new=Path("course/glossary.md").read_bytes()
prior=json.loads(Path("docs/technical-reviews/artifacts/R.3-before-glossary-recheck-report.json").read_text())
assert sha(old)=="5335ea60c11591373a78c5f72f2b9d6029e4eb701f3168cd21a5fd7607e11974"
assert sha(new)=="6ef03140f67322036645816da7f774da44554f01c41309edaf700df6f9ec78a9"
raw=Path("course/README.md").read_bytes()
heads=list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+",raw))
i=next(i for i,h in enumerate(heads) if h[1]==b"R.3")
primary=raw[heads[i].start():heads[i+1].start() if i+1<len(heads) else len(raw)]
assert sha(primary)==prior["source_sha256"]
assert primary==Path("docs/technical-reviews/artifacts/R.3-source.txt").read_bytes()
oldlines=old.decode().splitlines(keepends=True)
newlines=new.decode().splitlines(keepends=True)
changed=[n+1 for n,(a,b) in enumerate(zip(oldlines,newlines)) if a!=b]
assert len(oldlines)==len(newlines) and changed==[19]
assert oldlines[11]==newlines[11]
assert oldlines[16]==newlines[16]
assert b".svg" not in new.split(b"## G.1",1)[1].split(b"## G.2",1)[0]
unchanged_artifacts=[]
for artifact in prior["artifacts"]:
    observed=sha(Path(artifact["path"]).read_bytes())
    assert observed==artifact["sha256"],artifact["path"]
    unchanged_artifacts.append({"id":artifact["id"],"path":artifact["path"],"sha256":observed})
unchanged_figures=[]
for path,digest in prior["figure_sha256"].items():
    observed=sha(Path(path).read_bytes())
    assert observed==digest,path
    unchanged_figures.append({"path":path,"sha256":observed})
print(json.dumps({"environment":{"python":platform.python_version(),"device":"cpu","OMP_NUM_THREADS":"1","MKL_NUM_THREADS":"1","torch":"未使用；本次為原文差異／SHA讀取"},"old_full_sha256":sha(old),"new_full_sha256":sha(new),"old_bytes":len(old),"new_bytes":len(new),"changed_line_numbers":changed,"primary_sha256_unchanged":sha(primary),"token_definition_line12_identical":True,"token_granularity_paragraph_line17_identical":True,"line12_raw_current":newlines[11],"line17_raw_current":newlines[16],"line19_raw_old":oldlines[18],"line19_raw_current":newlines[18],"all_prior_artifact_sha256_unchanged":True,"prior_artifacts":unchanged_artifacts,"necessary_figures_sha256_unchanged":unchanged_figures,"scope":"本次短CPU只核對來源身份、完整差異及R.3必要支持字節；未重跑未變教學/導航例、未重render圖、未重新審讀21章或為G.1下判詞。"},ensure_ascii=False,indent=2))
