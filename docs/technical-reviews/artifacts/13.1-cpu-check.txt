def sha(raw):
    return hashlib.sha256(raw).hexdigest()


def section(path, lesson):
    raw = (ROOT / path).read_bytes()
    headings = list(re.finditer(rb"(?m)^## ([A-Z\d]+\.\d+) [^\r\n]+", raw))
    i = next(i for i, heading in enumerate(headings) if heading[1].decode() == lesson)
    end = headings[i + 1].start() if i + 1 < len(headings) else len(raw)
    body = raw[headings[i].start():end]
    first = raw[:headings[i].start()].count(b"\n") + 1
    return body, raw[:headings[0].start()], first


def code(path, lesson):
    raw, _, first = section(path, lesson)
    fences = re.findall(rb"(?ms)^```python\n(.*?)^```", raw)
    assert len(fences) == 1
    return fences[0].decode(), {"source": path + "#" + lesson, "source_sha256": sha(raw), "first_line": first}


def run(name, source, provenance, modify=None):
    actual = modify(source) if modify else source
    out = io.StringIO()
    namespace = {"__name__": "__review_example__"}
    torch.manual_seed(42)
    with contextlib.redirect_stdout(out):
        exec(compile(actual, name, "exec"), namespace)
    row = {"name": name, **provenance, "code": actual, "code_sha256": sha(actual.encode()), "stdout": out.getvalue(), "exit_code": 0}
    checks.append(row)
    return namespace, row
