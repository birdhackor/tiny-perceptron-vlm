def split(raw):
    matches = list(re.finditer(r'^## (C\.\d+) ', raw, re.M))
    sections = {}
    for i, m in enumerate(matches):
        text = raw[m.start():matches[i+1].start() if i+1 < len(matches) else len(raw)]
        sections[m.group(1)] = {'raw': text, 'sha256': sha(text.encode())}
    return {'encoding':'UTF8', 'normalization':'none; EOF空行保留', 'intro':raw[:matches[0].start()],
            'intro_sha256':sha(raw[:matches[0].start()].encode()), 'whole_chapter_sha256':sha(raw.encode()),
            'sections':sections}
