import json, math, os, sys
from pathlib import Path
sys.path.insert(0, str(Path(__file__).resolve().parents[3]))
import torch
from tiny_perceptron.multimodal import tone
assert torch.version.cuda is None and not torch.cuda.is_available()
torch.set_default_device("cpu")
torch.set_num_threads(1)
z=torch.tensor(3+4j)
assert z.abs().item()==5.0 and z.abs().square().item()==25.0
wave=tone(440.0,seconds=0.2)
window=torch.hann_window(400)
s=torch.stft(wave,400,160,window=window,return_complex=True)
power=s.abs().square()
padded=torch.nn.functional.pad(wave[None,None],(200,200),mode="reflect")[0,0]
manual=torch.fft.rfft(padded.unfold(0,400,160)*window,dim=-1).T
max_error=(manual-s).abs().max().item()
assert torch.allclose(s,manual,atol=2e-5,rtol=2e-5)
assert wave.numel()==3200 and tuple(wave.unfold(0,400,160).shape)==(18,400)
assert tuple(power.mean(-1).shape)==(201,)
assert torch.allclose(power.mean(-1),power.sum(-1)/21)
t450=tone(450.0,seconds=0.2)
p450=torch.stft(t450,400,160,window=window,return_complex=True).abs().square().mean(-1)
peak450=p450.argmax().item()
assert peak450==11
frame=wave[200:600].double()*torch.hann_window(400,dtype=torch.float64)
fft400=torch.fft.rfft(frame,n=400)
fft3200=torch.fft.rfft(frame,n=3200)
zero_pad_error=(fft3200[::8]-fft400).abs().max().item()
assert torch.allclose(fft3200[::8],fft400,rtol=1e-12,atol=1e-12)
