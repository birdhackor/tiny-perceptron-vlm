import json
import platform
import torch

def score(reply):
    parsed = json.loads(reply)
    return parsed, {
        "格式符合": isinstance(parsed, dict) and set(parsed) == {"answer"} and isinstance(parsed["answer"], int),
        "1+1內容正確": parsed.get("answer") == 2,
    }

results = []
for name, reply in [('original', '{"answer": 3}'), ('exercise_2', '{"answer": 2}'), ('exercise_result', '{"result": 2}'), ('boundary_true', '{"answer": true}'), ('boundary_false', '{"answer": false}'), ('boundary_float', '{"answer": 2.0}')]:
    parsed, scores = score(reply)
    old_assert_failed = False
    try:
        assert scores["格式符合"] and not scores["1+1內容正確"]
    except AssertionError:
        old_assert_failed = True
    if name == 'exercise_2':
        assert old_assert_failed
        assert scores["格式符合"] and scores["1+1內容正確"]
    if name == 'exercise_result':
        assert scores == {"格式符合": False, "1+1內容正確": False}
        assert parsed.get('answer') is None
    if name.startswith('boundary_'):
        assert not (type(parsed.get('answer')) is int)
    results.append({'case': name, 'reply': reply, 'parsed': parsed, 'value_type': type(next(iter(parsed.values()))).__name__, 'scores': scores, 'old_assert_failed': old_assert_failed, 'strict_integer_format': set(parsed) == {'answer'} and type(parsed.get('answer')) is int})
assert isinstance(True, int) and isinstance(False, int)
print(json.dumps({'environment': {'python':platform.python_version(), 'torch':torch.__version__, 'device':'cpu', 'threads':str(torch.get_num_threads())}, 'results':results}, ensure_ascii=False, indent=2))
