records = [
    {"image": "red_circle", "question": "什麼顏色？", "answer": "紅"},
    {"image": "red_circle", "question": "什麼形狀？", "answer": "圓形"},
]
for row in records:
    print(row["image"], row["question"], "→", row["answer"])
print("同圖", records[0]["image"] == records[1]["image"])
print("答案不同", records[0]["answer"] != records[1]["answer"])

records.append({"image": "red_circle", "question": "請同時給顏色與形狀", "answer": "紅色圓形"})
print("第三筆", records[2])
print("三筆image都相同", len({r["image"] for r in records}) == 1)
print("三筆question都不同", len({r["question"] for r in records}) == 3)
print("三筆answer都不同", len({r["answer"] for r in records}) == 3)
assert records[2]["image"] == records[0]["image"] == records[1]["image"]
assert records[2]["question"] not in [r["question"] for r in records[:2]]
assert records[2]["answer"] == "紅色圓形"
