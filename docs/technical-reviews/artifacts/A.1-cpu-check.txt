"""A.1 獨立技術審閱的有界 CPU 檢查；不訓練模型。"""
import contextlib
import hashlib
import io
import json
import os
import pathlib
import platform
import re
import sys

ROOT = pathlib.Path.cwd()
sys.path.insert(0, str(ROOT))
raw = (ROOT / "course/chapters/0A.md").read_bytes()
text = raw.decode("utf-8")
sections = {}
marks = list(re.finditer(r"^## (A\.\d+) .*\n", text, re.M))
for index, match in enumerate(marks):
    stop = marks[index + 1].start() if index + 1 < len(marks) else len(text)
    sections[match[1]] = text[match.start():stop]
primary = sections["A.1"]
code = re.search(r"```python\n(.*?)```", primary, re.S)[1]
mode = sys.argv[1]
records = []
if mode in ("original", "exercise"):
    if mode == "exercise":
        code = code.replace('examples = ["1→2", "2→4"]', 'examples = ["2→4", "1→2"]')
    stream = io.StringIO()
    namespace = {}
    with contextlib.redirect_stdout(stream):
        exec(compile(code, "A.1-" + mode, "exec"), namespace)
    assert namespace["question"] == "請依本次規則回答：3→?"
    assert namespace["zero"] == namespace["question"]
    assert namespace["few"].endswith("。\n" + namespace["question"])
    assert namespace["changed"] == "示例：1→3；2→6。\n" + namespace["question"]
    assert namespace["examples"] == (["1→2", "2→4"] if mode == "original" else ["2→4", "1→2"])
    records.append({"scope": "A.1 " + mode, "code": code, "stdout": stream.getvalue(), "assertions": "全部成立；未載入模型"})
elif mode == "intro-fences":
    for lesson, section in sections.items():
        for index, match in enumerate(re.finditer(r"```python\n(.*?)```", section, re.S)):
            stream = io.StringIO()
            with contextlib.redirect_stdout(stream):
                exec(compile(match[1], lesson + "-intro-support", "exec"), {})
            records.append({"scope": lesson + " Python fence " + str(index + 1), "code": match[1], "stdout": stream.getvalue(), "note": "新 namespace、專案根目錄、既有 .venv/retrieval 依賴"})
    assert len(records) == 7
    records.append({"scope": "A.2 bash fence", "executed": False, "reason": "正式 1000 次訓練排程，依規約只親讀 runner、凍結結果與中途資料清單，不重跑"})
elif mode == "frozen-icl":
    import torch
    from scripts.course_experiments.applications import _icl_records, _icl_question, _prompt_ids, _sample
    from scripts.course_experiments.common import Context, split_records, records_sha256
    from tiny_perceptron.data import ByteTokenizer
    from tiny_perceptron.training import load_checkpoint
    torch.set_num_threads(1)
    result = json.loads((ROOT / "docs/course-experiments/results/rag.json").read_text())
    split = split_records(_icl_records(), 42)
    for name, rows in split.items():
        assert len(rows) == result["results"]["icl_split"][name]["records"]
        assert records_sha256(rows) == result["results"]["icl_split"][name]["sha256"]
    records.append({"scope": "ICL 資料家族", "splits": {name: {"families": sorted({row["family"] for row in rows}), "records": len(rows), "sha256": records_sha256(rows)} for name, rows in split.items()}})
    path = ROOT / "checkpoints/course/rag/model.pt"
    weight_sha = hashlib.sha256(path.read_bytes()).hexdigest()
    assert weight_sha == "37f4a570e878a85949d7670d5da38148779811dcc4cc5bb3f2df0e5fb7e5b194"
    model, checkpoint = load_checkpoint(path, "cpu")
    ctx = Context("cpu", ROOT / "docs/technical-reviews/artifacts", ROOT, ROOT, seed=42)
    tensor_sha_before = hashlib.sha256(b"".join(t.detach().contiguous().numpy().tobytes() for t in model.state_dict().values())).hexdigest()
    tok = ByteTokenizer()
    for row in result["results"]["icl_samples_same_model_weights"]:
        generation = row["generation"]
        assert _prompt_ids(generation["messages"], tok) == generation["input_ids"]
        actual = _sample(model, generation["messages"], ctx, tokens=16, temperature=0.0)
        assert actual["input_ids"] == generation["input_ids"]
        assert actual["samples"][0]["generated_ids"] == generation["samples"][0]["generated_ids"]
        records.append({"scope": row["mode"], "expected": row["expected"], "frozen_generation": generation, "own_cpu_generation": actual})
    tensor_sha_after = hashlib.sha256(b"".join(t.detach().contiguous().numpy().tobytes() for t in model.state_dict().values())).hexdigest()
    assert tensor_sha_before == tensor_sha_after
    records.append({"scope": "既有公開權重與張量", "weight_sha256": weight_sha, "tensor_sha256_before": tensor_sha_before, "tensor_sha256_after": tensor_sha_after, "parameters": sum(t.numel() for t in model.parameters()), "config": vars(model.config), "checkpoint_keys": list(checkpoint), "no_update": True})
else:
    raise ValueError(mode)
try:
    import torch
    torch_version = torch.__version__
except ImportError:
    torch_version = "未匯入"
print(json.dumps({"mode": mode, "command": "OMP_NUM_THREADS=1 MKL_NUM_THREADS=1 .venv/bin/python docs/technical-reviews/artifacts/A.1-cpu-check.txt " + mode, "environment": {"python": platform.python_version(), "torch": str(torch_version), "device": "cpu", "platform": platform.platform(), "OMP_NUM_THREADS": os.environ.get("OMP_NUM_THREADS", ""), "MKL_NUM_THREADS": os.environ.get("MKL_NUM_THREADS", "")}, "records": records}, ensure_ascii=False, indent=2))
