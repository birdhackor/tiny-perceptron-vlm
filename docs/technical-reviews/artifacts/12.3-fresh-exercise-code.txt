import json
import os
import platform
from pathlib import Path

import torch

from tiny_perceptron.multimodal import tone

torch.set_num_threads(1)
assert torch.version.cuda is None
assert not torch.cuda.is_available()
wave = tone()
frames = wave.unfold(0, 400, 400)
print("框形狀", tuple(frames.shape))
print("框秒數", 400 / 16000, "移動秒數", 400 / 16000)
print("第一框末索引", 399, "第二框首索引", 400)
indices = torch.arange(wave.numel()).unfold(0, 400, 400)
result = {
    "exercise": "教材唯一練習：hop 改為 400、同步修改移動秒數與索引比較",
    "shape": list(frames.shape),
    "frame_seconds": 400 / 16000,
    "hop_seconds": 400 / 16000,
    "index_ranges": [[int(row[0]), int(row[-1])] for row in indices],
    "adjacent_shared_indices": len(set(indices[0].tolist()) & set(indices[1].tolist())),
    "last_end": int(indices[-1, -1]),
    "original_samples": wave.numel(),
    "environment": {
        "python": platform.python_version(),
        "torch": str(torch.__version__),
        "torch_source_commit": str(torch.version.git_version),
        "device": str(wave.device),
        "dtype": str(wave.dtype),
        "OMP_NUM_THREADS": os.environ["OMP_NUM_THREADS"],
        "MKL_NUM_THREADS": os.environ["MKL_NUM_THREADS"],
    },
}
assert result["shape"] == [4, 400]
assert result["adjacent_shared_indices"] == 0
assert result["last_end"] == 1599
Path("docs/technical-reviews/artifacts/12.3-fresh-exercise-result.json").write_text(
    json.dumps(result, ensure_ascii=False, indent=2) + "\n", encoding="utf-8"
)
