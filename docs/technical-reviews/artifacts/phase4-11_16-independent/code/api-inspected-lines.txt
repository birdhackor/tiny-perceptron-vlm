47: def edit_distance(reference, prediction):
48:     """Levenshtein distance on Python Unicode code points, not byte tokens."""
49:     previous = list(range(len(prediction) + 1))
50:     for i, original in enumerate(reference, 1):
51:         current = [i]
52:         for j, guessed in enumerate(prediction, 1):
53:             current.append(min(current[-1] + 1, previous[j] + 1, previous[j - 1] + (original != guessed)))
54:         previous = current
55:     return previous[-1]
