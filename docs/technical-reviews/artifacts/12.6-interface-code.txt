import torch, json, sys
from tiny_perceptron.multimodal import AudioEncoder, mel_filter_bank, tone
print(json.dumps({"python":sys.version.split()[0],"torch":torch.__version__,"torch_git_version":torch.version.git_version,"device":"cpu","torch_threads":torch.get_num_threads()}))
b=mel_filter_bank(bands=16)
print("bank_requires_grad",b.requires_grad)
assert b.requires_grad is False
p=torch.zeros(201,1);p[0]=1
print("DC_input_power_total",p.sum().item(),"output_power_total",(b@p).sum().item())
assert p.sum().item()==1 and (b@p).sum().item()==0
with torch.no_grad():
 y=AudioEncoder(bands=16,width=16)(tone(440.,seconds=.2))
print("AudioEncoder_output",list(y.shape),"finite",bool(torch.isfinite(y).all()))
assert y.shape==(21,16) and bool(torch.isfinite(y).all())
