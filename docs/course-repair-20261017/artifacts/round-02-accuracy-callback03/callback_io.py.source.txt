from pathlib import Path
import json, hashlib, datetime, re
OUT=Path('/workspace/work/textbook-repair-20261017/round-02-accuracy-callback03')
PACK=OUT.parent/'input-freeze03'
ORDER=json.loads((PACK/'reading-order.json').read_text())
TASK='/root/astra_repair_02_accuracy'
def sha(p):return hashlib.sha256(Path(p).read_bytes()).hexdigest()
def add(name,value):
    value={'reviewer_task':TASK,'created_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'collection_role':'technical','wave':'textbook-repair-20261017',**value}
    with (OUT/name).open('a') as f:f.write(json.dumps(value,ensure_ascii=False)+'\n')
def page(page_id,start=None,end=None,outputs=True):
    x=next(x for x in ORDER if x['page_id']==page_id);p=Path(x['frozen_path']);assert sha(p)==x['source_sha256']
    lines=p.read_text().splitlines();a=start or 1;b=end or len(lines)
    print(f'PAGE {page_id} SOURCE {x["source_sha256"]} LINES {a}-{b}/{len(lines)} DETAILS {x["details_blocks"]}')
    for i in range(a,b+1):print(f'{i:3d} {lines[i-1]}')
    add('read-events.jsonl',{'page_id':page_id,'source_sha256':x['source_sha256'],'emitted_lines':[a,b],'note':'Emission only; semantic note written after actual read and any recovery.'})
    if x.get('notebook') and outputs:
        n=OUT.parent/'final-executed03'/x['notebook'].removeprefix('notebooks/');nb=json.loads(n.read_text());cells=[c for c in nb['cells'] if c['cell_type']=='code'];code=[''.join(c['source']).strip() for c in cells]
        expected=re.findall(r'```python\n(.*?)```',p.read_text(),re.S);matches=[s.strip() in code for s in expected]
        print('ORIGINAL ROOT NOTEBOOK OUTPUTS, NOT REVIEWER RERUN',n)
        for i,c in enumerate(cells):
            for o in c.get('outputs',[]):
                if o.get('output_type')=='error':print('ERROR',o.get('ename'),o.get('evalue'))
                text=''.join(o.get('text',o.get('data',{}).get('text/plain',[])))
                if text:print('cell',i,text)
        add('notebook-output-receipts.jsonl',{'page_id':page_id,'path':str(n),'sha256':sha(n),'canonical_python_fenced_blocks':len(expected),'exact_block_matches':matches,'execution_by':'root initial kernel and recorded delta combination; not reviewer rerun'})
def note(value):
    x=next(x for x in ORDER if x['page_id']==value['page_id']);raw=Path(x['frozen_path']).read_text()
    assert value['quote'] in raw
    add('reading-log.jsonl',{'source':x['source'],'source_sha256':x['source_sha256'],'figures_sha256':x['figures_sha256'],'details_blocks':x['details_blocks'],'read_range':f'全檔1-{len(raw.splitlines())}行正文/code/練習/全部details','canonical_order':x['ordinal'],**value})
if __name__=='__main__':
    import sys
    if sys.argv[1]=='read':page(sys.argv[2],int(sys.argv[3]) if len(sys.argv)>3 else None,int(sys.argv[4]) if len(sys.argv)>4 else None)
    elif sys.argv[1]=='note':note(json.load(sys.stdin))
