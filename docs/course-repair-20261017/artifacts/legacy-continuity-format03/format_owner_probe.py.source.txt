def now():
    return datetime.datetime.now(datetime.timezone.utc).isoformat()


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def ref(path):
    path = Path(path)
    return {'path': path.relative_to(REPO).as_posix(), 'sha256': sha(path)}


def syntax_probe():
    results = []
    errors = []
    for name in INPUTS[:4]:
        source = (REPO / name).read_text(encoding='utf-8')
        blocks = []
        active = None
        for number, line in enumerate(source.splitlines(keepends=True), 1):
            if active is None:
                match = re.match(r'^\s*(`{3,}|~{3,})[ \t]*(\S*)', line)
                if match:
                    fence, language = match.groups()
                    active = {'char': fence[0], 'length': len(fence), 'python': language.lower() in {'python', 'python3', 'py'}, 'start_line': number + 1, 'lines': []}
            elif re.match(r'^\s*' + re.escape(active['char']) + '{' + str(active['length']) + r',}[ \t]*\r?\n?$', line):
                if active['python']:
                    blocks.append(active)
                active = None
            elif active['python']:
                active['lines'].append(line)
        if active and active['python']:
            errors.append({'path': name, 'start_line': active['start_line'], 'error': 'unclosed Python fence'})
        fence_records = []
        for index, block in enumerate(blocks, 1):
            code = ''.join(block['lines'])
            try:
                ast.parse(code, filename=f"{name}:fence-{index}:line-{block['start_line']}")
                parsed = True
            except SyntaxError as error:
                parsed = False
                errors.append({'path': name, 'fence': index, 'start_line': block['start_line'], 'error': str(error)})
            fence_records.append({'index': index, 'start_line': block['start_line'], 'python_bytes_sha256': hashlib.sha256(code.encode()).hexdigest(), 'parsed': parsed})
        results.append({'path': name, 'sha256': sha(REPO / name), 'python_fence_count': len(blocks), 'fences': fence_records})
    result = {'python_executable': sys.executable, 'python_version': sys.version, 'files': results, 'total_python_fences': sum(item['python_fence_count'] for item in results), 'errors': errors, 'scope': 'All currently marked python/python3/py fences; ast.parse only. No code execution or semantic/AST-equivalence claim.'}
    print(json.dumps(result, ensure_ascii=False, indent=2))
    return bool(errors)
