try:masked_loss(logits,torch.full_like(labels,-100))
except ValueError as e:print('empty target rejected',str(e))
else:raise AssertionError('empty target accepted')

w=torch.nn.Parameter(torch.tensor(1.));opt=torch.optim.SGD([w],lr=.1);loss=(w-3).square();opt.zero_grad();loss.backward();opt.step();print('one scalar optimizer update',w.item(),(w-3).square().item());assert abs(w.item()-1.4)<1e-6
