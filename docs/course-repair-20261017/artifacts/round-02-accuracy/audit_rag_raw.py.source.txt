from pathlib import Path
import json,hashlib,re,datetime
W=Path(__file__).parent;P=Path('/workspace/tiny-perceptron-vlm/docs/course-experiments/results/rag.json')
x=json.loads(P.read_text())['results']; rows=x['samples'];mismatch=[];results={};parsed=[]
for r in rows:
 s=r['generation']['samples'][0];text=s['generated'];clean=not s['invalid_special_tokens'];f=r['context_fact'];m=re.fullmatch(r'([A-E][0-9])\[(D[0-9])\]',text)
 vals={'exact_match':clean and text==r['expected'], 'fact_answer_correct':clean and text==f"{f['address']}[{f['source']}]", 'unknown':clean and text=='UNKNOWN',
 'citation_valid':bool(clean and m and m[2] in {d['id'] for d in r['documents']}),
 'supported_by_cited_source':bool(clean and m and any(d['id']==m[2] and f'address={m[1]}' in d['text'] for d in r['documents']))}
 if any(vals[k]!=r[k] for k in vals):mismatch.append((r['family'],r['mode']))
 parsed.append({**r,**vals})
for mode in sorted({r['mode'] for r in rows}):
 rr=[r for r in parsed if r['mode']==mode];results[mode]={'denominator':len(rr),**{k:sum(r[k] for r in rr) for k in ['exact_match','fact_answer_correct','citation_valid','supported_by_cited_source','unknown']}}
base={r['family']:r for r in parsed if r['mode']=='correct_context'}
paired={mode:sum(r['fact_answer_correct'] and base[r['family']]['fact_answer_correct'] for r in parsed if r['mode']==mode) for mode in ['changed_address_context','changed_source_context']}
examples=[{'family':r['family'],'mode':r['mode'],'documents':r['documents'],'generated':r['generation']['samples'][0]['generated'],'expected':r['expected']} for r in rows if r['family'] in ['K017','K028'] and r['mode'] in ['correct_context','changed_address_context','changed_source_context']]
out={'reviewer_task':'/root/astra_repair_02_accuracy','time':datetime.datetime.now(datetime.timezone.utc).isoformat(),'source':str(P),'source_sha256':hashlib.sha256(P.read_bytes()).hexdigest(),'method':'independent regex and count checks over old raw generations; no model run','sample_count':len(rows),'scoring_mismatches':mismatch,'recomputed_metrics':results,'paired_both_correct':paired,'retrieval_hits':sum(r['retrieval_hit'] for r in rows if r['mode']=='retrieved_context'),'examples':examples}
(W/'rag-raw-audit.json').write_text(json.dumps(out,ensure_ascii=False,indent=2)+'\n');print(json.dumps(out,ensure_ascii=False,indent=2))
