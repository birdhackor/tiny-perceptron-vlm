"""Read-only CPU arithmetic over original natural-v4 producer outputs; no model run."""
from pathlib import Path
import json, hashlib, datetime, math
from collections import Counter
from fractions import Fraction
W = Path(__file__).parent
R = Path('/workspace/tiny-perceptron-vlm')
B = R/'outputs/natural-v4/modal-runs'
def digest(p): return hashlib.sha256(p.read_bytes()).hexdigest()
tp = B/'train-37217452291/natural-natural-v4-train-37217452291-1/review/training.json'
t = json.loads(tp.read_text())
sp = R/'docs/natural-assistant/evidence/v4-runtime/validation-37219466611/blind-review/scored/scores.json'
s = json.loads(sp.read_text())
initial, final = t['initial_adapter_tensors'], t['final_adapter_tensors']
names = t['trainable_parameter_names']
training = {
 'path': str(tp), 'sha256': digest(tp),
 'parameters_from_shapes': sum(math.prod(v['shape']) for v in final.values()),
 'tensor_count': len(final),
 'changed_hash_count': sum(initial[k]['sha256_values'] != final[k]['sha256_values'] for k in final),
 'same_tensor_names': set(initial) == set(final),
 'optimizer_equals_trainable_names': set(t['optimizer_parameter_names']) == set(names) == set(final),
 'all_qv_lora_names': all('.self_attn.' in n and ('.q_proj.' in n or '.v_proj.' in n) and '.lora_' in n for n in names),
 'all_adapter_fp32': all(v['dtype'] == 'torch.float32' for v in final.values()),
 'frozen_sample_tensors': len(t['frozen_parameter_samples_initial']),
 'frozen_sample_values': sum(len(v['flat_indices']) for v in t['frozen_parameter_samples_initial'].values()),
 'frozen_samples_equal': t['frozen_parameter_samples_initial'] == t['frozen_parameter_samples_final'],
 'completed_steps': t['completed_steps'], 'history_rows': len(t['history']),
 'history_steps_contiguous': [r['step'] for r in t['history']] == list(range(1,2078)),
 'trained_row_occurrences': sum(len(r['row_ids']) for r in t['history']),
 'unique_trained_rows': len({i for r in t['history'] for i in r['row_ids']}),
 'row_occurrence_multiplicities': dict(Counter(Counter(i for r in t['history'] for i in r['row_ids']).values())),
 'elapsed_seconds': t['elapsed_seconds'], 'peak_cuda_memory_allocated_bytes': t['peak_cuda_memory_allocated_bytes'],
 'archived_adapter_files': [{ 'step': c['completed_steps'], **next(f for f in c['files'] if f['path']=='adapter_model.safetensors')} for c in t['archived_checkpoints']],
}
validation = {}
for entry in s['artifact_binding']['generation_files']:
 p = B/'validation-37219466611/natural-natural-v4-validation-37219466611-1/review'/entry['name']
 rows = json.loads(p.read_text()); v = entry['variant']; score=s['variants'][v]
 incomplete = [r for r in rows if not (r['stop_reason']=='eos' and r['ended_with_eos'] and r['generated_token_ids'][-1] in r['eos_token_ids'] and not r['truncated'] and not r['completion_unknown'])]
 num = sum(score['correct_counts'][k]*w for k,w in s['integer_weights'].items())
 macro = ( (Fraction(score['correct_counts']['photo_summary'],28)+Fraction(score['correct_counts']['photo_fact'],56))/2
  + Fraction(score['correct_counts']['text_presence'],18)+Fraction(score['correct_counts']['single_ocr'],10)
  + Fraction(score['correct_counts']['ordered_ocr'],3)
  + (Fraction(score['correct_counts']['text_chat'],9)+Fraction(score['correct_counts']['voice_typed_reference_chat'],4)+Fraction(score['correct_counts']['voice_actual_asr_chat'],4))/3 )/5
 validation[v] = {'path':str(p),'sha256':digest(p),'matches_bound_sha':digest(p)==entry['sha256'],
  'rows':len(rows), 'tasks':dict(Counter(r['task'] for r in rows)),
  'incomplete':[{'id':r['id'],'generated_tokens':r['generated_tokens'],'stop_reason':r['stop_reason'],'truncated':r['truncated'],'ended_with_eos':r['ended_with_eos']} for r in incomplete],
  'all_generated_token_counts_match':all(r['generated_tokens']==len(r['generated_token_ids']) for r in rows),
  'scores_source_sha256':digest(sp),'correct_counts_from_fixed_original_grades':score['correct_counts'],
  'recomputed_primary_numerator':num,'matches_recorded_numerator':num==score['primary_numerator'],
  'macro_fraction':str(macro),'macro_equals_integer':macro==Fraction(num,75600),
  'semantic_grades_independently_regraded':False}
result = {'reviewer_task':'/root/astra_repair_02_accuracy','time':datetime.datetime.now(datetime.timezone.utc).isoformat(),
 'method':'Python standard-library read-only SHA/shape/count/fraction checks on original producer outputs; no model loaded, generated, or trained; original semantic grades are fixed observations, not independent truth',
 'training':training,'validation':validation}
(W/'natural-raw-audit.json').write_text(json.dumps(result,ensure_ascii=False,indent=2)+'\n')
print(json.dumps(result,ensure_ascii=False,indent=2))
