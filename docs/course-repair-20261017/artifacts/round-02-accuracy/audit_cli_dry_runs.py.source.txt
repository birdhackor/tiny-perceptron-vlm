from pathlib import Path
import subprocess,json,datetime,hashlib,os
R=Path('/workspace/tiny-perceptron-vlm');W=Path(__file__).parent
results=[]
for task in ['text','sft','vision']:
 cmd=[str(R/'.venv/bin/python'),'scripts/train.py','--task',task,'--device','cpu']
 run=subprocess.run(cmd,cwd=R,capture_output=True,env={**os.environ,'OMP_NUM_THREADS':'1','MKL_NUM_THREADS':'1'},timeout=45)
 (W/f'cli-dry-run-{task}.stdout').write_bytes(run.stdout);(W/f'cli-dry-run-{task}.stderr').write_bytes(run.stderr)
 output=[json.loads(s) for s in run.stdout.decode().splitlines() if s.startswith('{')]
 results.append({'task':task,'command_argv':cmd,'cwd':str(R),'returncode':run.returncode,'stdout':str(W/f'cli-dry-run-{task}.stdout'),'stderr':str(W/f'cli-dry-run-{task}.stderr'),'stdout_sha256':hashlib.sha256(run.stdout).hexdigest(),'output':output})
out={'reviewer_task':'/root/astra_repair_02_accuracy','time':datetime.datetime.now(datetime.timezone.utc).isoformat(),'method':'three default CLI dry runs: one forward/backward each, no --train, no optimizer.step or checkpoint save','environment':'existing fixed torch2.14.1+cpu venv; OMP/MKL threads1; no dependency sync','results':results}
(W/'cli-dry-run-audit.json').write_text(json.dumps(out,ensure_ascii=False,indent=2)+'\n');print(json.dumps(out,ensure_ascii=False,indent=2))
