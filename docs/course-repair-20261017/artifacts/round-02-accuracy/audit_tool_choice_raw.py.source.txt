"""Rebuild small routing records and recount saved predictions; no model creation/run."""
from pathlib import Path
import json, hashlib, sys, datetime, platform
from collections import Counter
REPO = Path('/workspace/tiny-perceptron-vlm')
WORK = Path('/workspace/work/textbook-repair-20261017/round-02-accuracy')
sys.path.insert(0, str(REPO))
from scripts.course_experiments.tool_choice import build_records, split_records
from scripts.course_experiments.common import records_sha256
TASK = '/root/astra_repair_02_accuracy'
def sha(path): return hashlib.sha256(path.read_bytes()).hexdigest()
source = REPO/'docs/course-experiments/results/tool_choice.json'
raw = json.loads(source.read_text())
result = raw['results']
partitions = split_records(build_records())
test_families = {r['family'] for r in partitions['test']}
partitions['paraphrase_diagnostic'] = [r for r in build_records(paraphrase=True) if r['family'] in test_families]
def score(rows, actions):
    pairs = list(zip(rows, actions, strict=True))
    def rate(predicate, subset):
        return {'numerator': sum(predicate(r, a) for r,a in subset), 'denominator': len(subset)}
    needed = [(r,a) for r,a in pairs if r['expected_action']=='TOOL']
    not_needed = [(r,a) for r,a in pairs if r['expected_action']!='TOOL']
    unavailable = [(r,a) for r,a in pairs if not r['calculator_available']]
    return {
      'accuracy': rate(lambda r,a:a==r['expected_action'], pairs),
      'valid_action':rate(lambda r,a:a in ('DIRECT','TOOL','ASK'), pairs),
      'needed_tool_selected':rate(lambda r,a:a=='TOOL',needed),
      'needed_tool_missed':rate(lambda r,a:a!='TOOL',needed),
      'unnecessary_tool_selected':rate(lambda r,a:a=='TOOL',not_needed),
      'unavailable_tool_selected':rate(lambda r,a:a=='TOOL',unavailable),
      'confusion':{e:dict(Counter(a or 'INVALID' for r,a in pairs if r['expected_action']==e)) for e in ('DIRECT','TOOL','ASK')}
    }
out = {'reviewer_task':TASK, 'time':datetime.datetime.now(datetime.timezone.utc).isoformat(),
 'environment':{'python':platform.python_version(),'device':'CPU; metadata only'},
 'scope':'Own bounded metadata and saved-output recount, no model creation, training, generation, downloads or inference; complements earlier inline check.',
 'source':str(source),'source_sha256':sha(source),
 'producer_script':str(REPO/'scripts/course_experiments/tool_choice.py'),
 'producer_script_sha256':sha(REPO/'scripts/course_experiments/tool_choice.py'),
 'partitions':{}, 'evaluation':{}}
assert all(not ({r['family'] for r in partitions[a]} & {r['family'] for r in partitions[b]}) for a,b in [('train','validation'),('train','test'),('validation','test')])
for name, rows in partitions.items():
    counts = {'records':len(rows), 'families':sorted({r['family'] for r in rows}), 'action_counts':dict(Counter(r['expected_action'] for r in rows)), 'sha256':records_sha256(rows)}
    assert counts == result['data'][name], name
    out['partitions'][name]={'records':len(rows),'families':len(counts['families']),'records_sha256':counts['sha256'],'matches_saved_data':True}
    if name=='train': continue
    old = result[name]
    assert [x['record'] for x in old['samples']] == rows
    actions=[]
    for x in old['samples']:
        generations=x['generation']['samples']; assert len(generations)==1
        s=generations[0]
        action=s['generated'] if s['eos'] and not s['invalid_special_tokens'] and s['generated'] in ('DIRECT','TOOL','ASK') else None
        assert action==x['chosen_action']
        actions.append(action)
    measured=score(rows,actions)
    for k,v in measured.items():
        if k=='confusion': assert v==old['metrics'][k]
        else:
            assert v['numerator']==old['metrics'][k]['numerator'] and v['denominator']==old['metrics'][k]['denominator']
            assert v['numerator']/v['denominator']==old['metrics'][k]['rate']
    baseline={}
    for key,action in [('always_direct','DIRECT'),('always_tool','TOOL')]:
        baseline[key]=score(rows,[action]*len(rows))
        for k,v in baseline[key].items():
            if k=='confusion': assert v==old['baselines'][key][k]
            else:
                assert v['numerator']==old['baselines'][key][k]['numerator'] and v['denominator']==old['baselines'][key][k]['denominator']
    out['evaluation'][name]={'recount':measured,'baseline_accuracy':{k:v['accuracy'] for k,v in baseline.items()}, 'saved_records_and_actions_exact':True, 'all_requested_metric_comparisons_equal':True}
text=json.dumps(out,ensure_ascii=False,indent=2)+'\n'
(WORK/'tool-choice-raw-audit.json').write_text(text)
print(text,end='')
