def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def norm(s,remove=False):
 s=unicodedata.normalize('NFKC',s).strip()
 return ''.join(s.split()) if remove else s
def dist(a,b):
 row=list(range(len(b)+1))
 for i,c in enumerate(a,1):
  new=[i]
  for j,d in enumerate(b,1):new.append(min(new[-1]+1,row[j]+1,row[j-1]+(c!=d)))
  row=new
 return row[-1]
g=json.loads((B/'generations-base.json').read_text());t=json.loads((B/'transcripts.json').read_text());s=json.loads(S.read_text())
bad=[x for x in g if not (x['ended_with_eos'] and x['stop_reason']=='eos' and x['generated_token_ids'][-1] in x['eos_token_ids'] and not x['truncated'] and not x['completion_unknown'])]
asr=[]
for x in t:
 a,b=x['reference_transcript'],x['transcript']; an,bn=norm(a,True),norm(b,True)
 row={'id':x['id'],'task':x['task'],'raw_errors':dist(a,b),'raw_reference_characters':len(a),'normalized_errors':dist(an,bn),'normalized_reference_characters':len(an)}
 row['matches_recorded']=row['raw_errors']==x['raw_errors'] and row['raw_reference_characters']==x['raw_reference_characters'] and row['normalized_errors']==x['errors'] and row['normalized_reference_characters']==x['reference_characters']
 asr.append(row)
groups={}
for task in ['all','speech_chat','speech_transcription']:
 rows=[x for x in asr if task=='all' or x['task']==task]
 groups[task]={'count':len(rows),**{k:sum(x[k] for x in rows) for k in ['raw_errors','raw_reference_characters','normalized_errors','normalized_reference_characters']}}
exact={}
for task in ['text_presence','ocr','ocr_order']:
 rows=[x for x in g if x['task']==task]
 exact[task]={'correct':sum(norm(x['reference_answer'],task=='ocr')==norm(x['prediction'],task=='ocr') for x in rows),'count':len(rows),'normalization':'NFKC outerstrip; remove all whitespace only for ocr'}
paired=[]
for a in [x for x in g if x['task']=='typed_chat']:
 b=next(x for x in g if x['task']=='speech_chat' and x['id']==a['id'])
 paired.append({'id':a['id'],'same_input_text':a['user']==b['user'],'same_prediction':a['prediction']==b['prediction']})
out={'reviewer_task':'/root/astra_repair_02_accuracy','time':datetime.datetime.now(datetime.timezone.utc).isoformat(),
 'method':'read-only standard-library independent Levenshtein/count/normalization checks on old producer records; no model run; semantic grades not regraded',

 'responses':len(g),'task_counts':dict(Counter(x['task'] for x in g)),

 'all_token_counts_match':all(x['generated_tokens']==len(x['generated_token_ids']) for x in g),

 'asr_all_raw_eos':all(x['ended_with_eos'] and x['stop_reason']=='eos' and x['raw_token_ids'][-1] in x['eos_token_ids'] for x in t),
