"""Seal this one review's original evidence; copy bytes without rewriting notes."""
from pathlib import Path
import json, hashlib, datetime, shutil
W=Path('/workspace/work/textbook-repair-20261017/round-02-accuracy')
R=Path('/workspace/tiny-perceptron-vlm')
D=R/'docs/course-repair-20261017'
A=D/'artifacts/round-02-accuracy'
F=D/'reviews/wholebook-accuracy-original-01.json'
TASK='/root/astra_repair_02_accuracy'
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def rec(p):return {'path':str(p),'bytes':p.stat().st_size,'sha256':sha(p)}
assert not (W/'seal-source-first.json').exists()
assert not (W/'transport-map.json').exists()
assert not F.exists(), 'Do not overwrite an existing review'
report=json.loads((W/'report-source-first.json').read_text())
coverage=json.loads((W/'coverage.json').read_text())
assert report['reviewer_task']==TASK and coverage['reviewer_task']==TASK
assert report['counts']==coverage['counts']
assert report['coverage']['sha256']==sha(W/'coverage.json')
assert report['required_findings']==report['optional_findings']==[]
for p in W.iterdir():
 if p.is_file() and p.suffix=='.json':assert json.loads(p.read_text())['reviewer_task']==TASK,p
 if p.is_file() and p.suffix=='.jsonl':
  for i,line in enumerate(p.read_text().splitlines(),1):assert json.loads(line)['reviewer_task']==TASK,(p,i)
for artifact in report['evidence_index']:
 p=Path(artifact['path']);assert sha(p)==artifact['sha256'] and p.stat().st_size==artifact['bytes']
assert (W/'report.txt').read_bytes()==(W/'report-source-first.txt').read_bytes()
paths=[p for p in sorted(W.rglob('*')) if p.is_file() and '__pycache__' not in p.parts]
seal={'reviewer_task':TASK,'sealed_at':datetime.datetime.now(datetime.timezone.utc).isoformat(),'kind':'Initial source-first review original content SHA256 seal','source_freeze':report['source_freeze'],'review_report':rec(W/'report-source-first.json'),'coverage':rec(W/'coverage.json'),'completed_scope':{'canonical_pages':350,'lesson_pages':312,'details_blocks':277,'unique_figures_actually_viewed':115},'independence':'No peer textbook findings, historical textbook review narratives, author-private/coordinator, git diff/log or revised03 content read before this seal. Original raw-data metadata and premature link-follow/figure observations disclosed in report. Shared filesystem, not physical isolation.','policy':'Original listed content must not be rewritten to incorporate later findings. Subsequent callback work uses new files. SHA seal records bytes and time; it is not an externally trusted timestamp.','files':[{**rec(p),'relative_path':str(p.relative_to(W)),'persistent_copy_expected':p.parent==W} for p in paths]}
sealpath=W/'seal-source-first.json';sealpath.write_text(json.dumps(seal,ensure_ascii=False,indent=2)+'\n')
A.mkdir(parents=True,exist_ok=True);F.parent.mkdir(parents=True,exist_ok=True)
entries=[]
for p in sorted(W.iterdir()):
 if p.is_file():
  dest=A/p.name
  with dest.open('xb') as target:target.write(p.read_bytes())
  assert dest.read_bytes()==p.read_bytes()
  entries.append({'original':str(p),'copy':str(dest),'bytes':p.stat().st_size,'sha256':sha(p),'byte_exact':True})
with F.open('xb') as target:target.write((W/'report-source-first.json').read_bytes())
assert F.read_bytes()==(W/'report-source-first.json').read_bytes()
entries.append({'original':str(W/'report-source-first.json'),'copy':str(F),'bytes':F.stat().st_size,'sha256':sha(F),'byte_exact':True})
transport={'reviewer_task':TASK,'time':datetime.datetime.now(datetime.timezone.utc).isoformat(),'method':'binary read/write exact bytes; no JSON normalization or semantic rewriting','seal_sha256':sha(sealpath),'entries':entries,'omitted_from_repository':['primary/ contains third-party fulltexts and primary downloaded files; remains in WORK','figures/ contains PNG renders, including retained cropped initial image; remains in WORK','__pycache__/ is runtime cache, excluded from seal and transport'],'self_hash_note':'Transport map intentionally excludes itself; work/repository copies are byte identical and its hash is returned by the seal operation. Seal itself excludes this later transport map, avoiding a self-reference cycle.'}
transportpath=W/'transport-map.json';transportpath.write_text(json.dumps(transport,ensure_ascii=False,indent=2)+'\n')
with (A/'transport-map.json').open('xb') as target:target.write(transportpath.read_bytes())
for x in seal['files']:
 p=Path(x['path']);assert p.stat().st_size==x['bytes'] and sha(p)==x['sha256']
for x in transport['entries']:
 p=Path(x['original']);q=Path(x['copy']);assert p.read_bytes()==q.read_bytes() and sha(q)==x['sha256']
assert (A/'transport-map.json').read_bytes()==transportpath.read_bytes()
print(json.dumps({'reviewer_task':TASK,'sealed_at':seal['sealed_at'],'sealed_files':len(seal['files']),'byte_exact_transport_entries':len(entries),'report':rec(F),'work_report':rec(W/'report-source-first.json'),'coverage':rec(W/'coverage.json'),'seal':rec(sealpath),'transport_map':rec(transportpath),'all_sealed_files_hash_verified_after_copy':True,'all_copies_byte_exact':True,'source_first_initial_complete':True},ensure_ascii=False,indent=2))
