def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def ref(p):
    return {'path': str(p.relative_to(R)), 'sha256': sha(p)}


def normalized(s, punctuation=False):
    # Return a normalized string and its original-character positions. This
    # locates a raw literal span without silently rewriting the sealed quote.
    keep = [True] * len(s)
    for m in re.finditer(r'\[([^\]]+)\]\([^)]*\)', s):
        for i in range(m.start(), m.start(1)):
            keep[i] = False
        for i in range(m.end(1), m.end()):
            keep[i] = False
    for i, c in enumerate(s):
        if c.isspace() or c in '`*~' or (punctuation and c in '「」『』“”‘’'):
            keep[i] = False
    ix = [i for i, yes in enumerate(keep) if yes]
    return ''.join(s[i] for i in ix), ix

for row, note in enumerate(rows, 1):
    pid, quote = note['page_id'], note['quote']
    main = (W / 'input-freeze03/main' / f'{pid}.md').read_text()
    if quote in main:
        raw_literal_rows.append(row)
        continue
    assert pid in by_id and by_id[pid]['quote'] == quote
    page = pages[pid]
    snapshot = R / page['snapshot']
    assert sha(snapshot) == page['source_sha256']
    lines = snapshot.read_text().splitlines()
    entry = {
        'page_id': pid,
        'original_note': {**ref(ORIG / 'reading-log.jsonl'), 'jsonl_line_1_based': row, 'ordinal_1_based': note['ordinal'], 'recorded_at': note['recorded_at'], 'field': 'quote'},
        'coordinator_diagnostic_record': by_id[pid]['record'],
        'original_quote_unchanged': quote,
        'authored_source': page['source'],
        'source_sha256': page['source_sha256'],
        'new_literal_anchors': [],
        'meaning_verdict': 'unchanged; no curriculum issue',
    }
    if pid == '14.9':
        fig = R / 'course/figures/rewrite-14-9-nearby-angles.svg'
        anchor = '卡片仍鄰接，新座標只差 0.5'
        flines = fig.read_text().splitlines()
        lineno = next(i for i, line in enumerate(flines, 1) if anchor in line)
        assert normalized(anchor)[0] == normalized(quote)[0]
        entry.update(source_class='svg_text_whitespace_normalized', explanation='原quote取自當頁原SVG底行，省略0.5前空白；非main Markdown直接子字串。原獨立圖筆記第69列已實看，peer階段另實看同PNG。')
        entry['new_literal_anchors'].append({**ref(fig), 'line_1_based': lineno, 'literal': anchor, 'source_class': 'original_svg_text'})
    elif pid in special:
        kind, anchors, explanation = special[pid]
        entry.update(source_class=kind, explanation=explanation)
        for anchor in anchors:
            lineno = next(i for i, line in enumerate(lines, 1) if anchor in line)
            entry['new_literal_anchors'].append({**ref(snapshot), 'line_1_based': lineno, 'literal': anchor, 'source_class': 'frozen_main_text'})
    else:
        found = False
        for punctuation in [False, True]:
            q = normalized(quote, punctuation)[0]
            for lineno, line in enumerate(lines, 1):
                norm, positions = normalized(line, punctuation)
                start = norm.find(q)
                if start < 0:
                    continue
                literal = line[positions[start]:positions[start + len(q) - 1] + 1]
                entry.update(source_class='main_markdown_whitespace_punctuation_normalized' if punctuation else 'main_markdown_whitespace_normalized', explanation='原quote省略Markdown標記／空白' + ('及語句內引號。' if punctuation else '。') + '下列anchor為本次補定位的原始Markdown連續子字串，不回寫原筆記。')
                entry['new_literal_anchors'].append({**ref(snapshot), 'line_1_based': lineno, 'literal': literal, 'source_class': 'frozen_main_text'})
                found = True
                break
            if found:
                break
        assert found, (pid, quote)
    entries.append(entry)
