def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def read(p):
    return json.loads(p.read_text())


def ref(p):
    return {'path': str(p.relative_to(R)), 'sha256': sha(p), 'bytes': p.stat().st_size}


def pointer(obj, ptr):
    if not ptr:
        return obj
    for s in ptr.strip('/').split('/'):
        k = s.replace('~1', '/').replace('~0', '~')
        obj = obj[int(k)] if isinstance(obj, list) else obj[k]
    return obj

seal_path = A / 'new-peer-seal.json'
assert not seal_path.exists()
refs_checked = set()
pointer_checks = 0


def walk(obj):
    global pointer_checks
    if isinstance(obj, dict):
        if isinstance(obj.get('path'), str) and isinstance(obj.get('sha256'), str):
            assert not pathlib.Path(obj['path']).is_absolute(), obj['path']
            p = R / obj['path']
            assert p.is_file() and sha(p) == obj['sha256'], obj
            refs_checked.add((obj['path'], obj['sha256']))
            if 'json_pointer' in obj:
                pointer(read(p), obj['json_pointer'])
                pointer_checks += 1
        for v in obj.values():
            walk(v)
    elif isinstance(obj, list):
        for v in obj:
            walk(v)

    walk(data)
