from pathlib import Path
import hashlib
import json
import os
import sys

os.environ['CUDA_VISIBLE_DEVICES'] = ''
import torch
from torch.nn import functional as F
from tiny_perceptron.natural_concepts import thin_stroke_report

torch.set_num_threads(1)
first = torch.zeros(1, 1, 32, 32, device='cpu')
first[:, :, :, 15] = 1
second = torch.zeros_like(first)
second[:, :, :, 8] = 1
small_first = F.interpolate(first, (4, 4), mode='area')
small_second = F.interpolate(second, (4, 4), mode='area')
assert first.device.type == second.device.type == small_first.device.type == 'cpu'
assert not torch.equal(first, second)
assert torch.equal(small_first, small_second)
assert float(small_first.max()) == 1 / 8 == 0.125
assert int((small_first > 0).sum()) == 4
assert int((small_first >= 0.5).sum()) == 0
assert thin_stroke_report() == {
    'original_shape': [32, 32], 'small_shape': [4, 4],
    'original_brightest': 1.0, 'small_brightest': 0.125,
    'pixels_at_least_half_bright_before': 32,
    'pixels_at_least_half_bright_after': 0,
}
root = Path('/workspace/tiny-perceptron-vlm')
proof = {
    'reviewer_task': '/root/natural_reader_mobile_11_15',
    'python_version': sys.version,
    'python_executable': sys.executable,
    'python_prefix': sys.prefix,
    'torch_version': torch.__version__,
    'torch_default_device': str(torch.get_default_device()),
    'tensor_devices': [str(x.device) for x in [first, second, small_first, small_second]],
    'CUDA_VISIBLE_DEVICES': os.environ['CUDA_VISIBLE_DEVICES'],
    'model_loaded': False,
    'network_or_model_downloads': False,
    'input_line_columns': [15, 8],
    'input_equal': bool(torch.equal(first, second)),
    'pooled_equal': bool(torch.equal(small_first, small_second)),
    'pooled_numbers': small_first[0, 0].tolist(),
    'pooled_nonzero_count': int((small_first > 0).sum()),
    'pooled_half_bright_count': int((small_first >= 0.5).sum()),
    'arithmetic': {'bright_values_in_one_block': 8, 'values_in_one_block': 64,
                   'average': 8 / 64, 'one_divided_by_eight': 1 / 8},
    'natural_concepts_sha256': hashlib.sha256((root/'tiny_perceptron/natural_concepts.py').read_bytes()).hexdigest(),
    'source_sha256': hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
    'all_assertions_passed': True,
}
(root/'docs/reader-reviews/artifacts/natural-mobile-11.15-proof.json').write_text(
    json.dumps(proof, ensure_ascii=False, indent=2) + '\n')
print(json.dumps(proof, ensure_ascii=False, indent=2))
