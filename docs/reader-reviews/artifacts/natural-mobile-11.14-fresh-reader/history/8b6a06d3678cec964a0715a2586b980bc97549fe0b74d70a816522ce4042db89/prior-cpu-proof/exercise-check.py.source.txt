# Numerical check of the paper exercise, under an explicit 4-column strip assumption.
COLORS = {"R": (1, 0, 0), "B": (0, 0, 1), ".": (0, 0, 0)}
def rgb_average(rows):
    return [sum(COLORS[c][channel] for row in rows for c in row) / 64 for channel in range(3)]
before = [["RRRRBBBB"] * 8, ["........"] * 8]
after = [["....BBBB"] * 8, ["RRRR...."] * 8]
print("Before, each 8x8 cell RGB mean:", [rgb_average(cell) for cell in before])
print("After moving red into second cell:", [rgb_average(cell) for cell in after])
print("Each cell remains unchanged:", [rgb_average(a) == rgb_average(b) for a, b in zip(before, after)])
