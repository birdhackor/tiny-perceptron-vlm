# 11.14 exact reader-facing snippet.
report = picture_order_report()
print("不同的像素數", report["different_pixels"])
print("4乘4平均摘要相同", report["same_4_by_4_summary"])

# Independent paper prediction and exercise verification.
first, second = swapped_picture()
assert first.device.type == second.device.type == "cpu"
changed_colour_positions = int((first != second).any(dim=0).sum())
assert changed_colour_positions == 64
before = F.adaptive_avg_pool2d(first[None], (4,4))
across = first.clone()
across[0,8:16,0:4] = 0
across[0,8:16,8:12] = 1
later = F.adaptive_avg_pool2d(across[None],(4,4))
assert before[0,:,1,0].tolist() == [0.5,0.0,0.5]
assert before[0,:,1,1].tolist() == [0.0,0.0,0.0]
assert later[0,:,1,0].tolist() == [0.0,0.0,0.5]
assert later[0,:,1,1].tolist() == [0.5,0.0,0.0]
assert not torch.equal(before,later)

# Read and ran the direct patchify prerequisite and its patch-size-8 exercise.
image = scene()[None]
patches = patchify(image, 4)
restored = unpatchify(patches, 3, 16, 16, 4)
print("小塊序列",tuple(patches.shape))
print("拼回相同",torch.equal(image,restored))
assert tuple(patches.shape) == (1,16,48)
assert torch.equal(image,restored)
patches8 = patchify(image,8)
restored8 = unpatchify(patches8,3,16,16,8)
assert tuple(patches8.shape) == (1,4,192)
assert torch.equal(image,restored8)

# Check the stated row order directly, independent of round-trip compatibility.

for row in range(4):
    for col in range(4):
        known[:,:,row*4:(row+1)*4,col*4:(col+1)*4] = row*4+col
ordered = patchify(known,4)
assert ordered[0,:,0].tolist() == list(range(16))
