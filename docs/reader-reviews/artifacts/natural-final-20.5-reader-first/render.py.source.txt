from pathlib import Path
import base64, hashlib, json, platform
import markdown
from playwright.sync_api import sync_playwright
root=Path(__file__).resolve().parents[4]
out=Path(__file__).resolve().parent
figures=['natural_photo_evidence','natural_actual_bike','natural_actual_glass']
section=(out/'cpu/section.md').read_text()
html=markdown.markdown(section,extensions=['tables','fenced_code'])
for name in figures:
 svg=(out/f'cpu/figures/course/figures/{name}.svg').read_bytes()
 html=html.replace(f'../figures/{name}.svg',f'data:image/svg+xml;base64,{base64.b64encode(svg).decode()}" data-name="{name}')
(out/'section-render.html').write_text('<!DOCTYPE html><html lang="zh-Hant"><meta charset="utf-8"><style>body{margin:0;font:18px/1.8 "Noto Sans CJK TC",sans-serif;color:#172b4d}main{width:100%;max-width:720px;margin:auto}p,h2{padding:0 12px}img{display:block;width:100%;height:auto}pre{overflow-x:auto;padding:12px;background:#f3f5f8}table{border-collapse:collapse;font-size:16px}th,td{border:1px solid #ddd;padding:4px}a{color:#07599b}</style><main>'+html+'</main></html>')
items=[]
with sync_playwright() as p:
 browser=p.chromium.launch(executable_path='/usr/bin/chromium',headless=True,args=['--no-sandbox'])
 for width in [720,358]:
  page=browser.new_page(viewport={'width':width,'height':900},device_scale_factor=1)
  page.set_content((out/'section-render.html').read_text());page.evaluate('document.fonts.ready');page.wait_for_function('Array.from(document.images).every(i=>i.complete&&i.naturalWidth>0)')
  for name in figures:
   img=page.locator(f'img[data-name="{name}"]'); path=out/f'{name}-{width}.png'; img.screenshot(path=str(path))
   box=img.bounding_box()
   items.append({'figure':name,'viewport_width':width,'viewport_height':900,'device_scale_factor':1,'rendered_box':box,'png':path.relative_to(root).as_posix(),'png_sha256':hashlib.sha256(path.read_bytes()).hexdigest(),'actual_browser_render':True,'all_images_loaded':True})
  page.close()
 version=browser.version;browser.close()
(out/'render-metadata.json').write_text(json.dumps({'browser':f'Chromium {version}','renderer':'Playwright synchronous Chromium screenshots of image elements in the full section HTML','python':platform.python_version(),'html_source_sha256':hashlib.sha256((out/'section-render.html').read_bytes()).hexdigest(),'responsive_rule':'img width:100%; height:auto; max section width720; aspect ratio retained; no crop CSS','items':items},ensure_ascii=False,indent=2)+'\n')
print(json.dumps(items,ensure_ascii=False,indent=2))
